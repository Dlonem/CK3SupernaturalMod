# .1000 -- The Council's Business. The Grand Magus, from the meeting's second part (upstream I1), or from the debug event.
# Saved scopes from Witchcraft's meeting (amage, bmage, loremaster, admin) travel with it.
spn_order.1000 = {
	type = character_event
	title = spn_order.1000.t
	desc = {
		desc = spn_order.1000.desc
		triggered_desc = {
			trigger = { exists = scope:spn_rm_ruler }
			desc = spn_order.1000.desc_work
		}
	}
	theme = diplomacy
	override_background = { reference = council_chamber }
	left_portrait = { character = root animation = personality_rational }
	trigger = {
		is_alive = yes
		spn_order_on_trigger = yes
		global_var:current_archmage_global ?= root
		spn_is_order_member_trigger = yes
	}
	# Left the chair the same day: the meeting simply goes on, as Witchcraft's own first option would take it.
	on_trigger_fail = {
		if = {
			limit = { has_variable = spn_imp_in_meeting }
			remove_variable = spn_imp_in_meeting
		}
		scope:loremaster ?= { trigger_event = magic_ev10.86 }
	}
	immediate = {
		save_scope_as = spn_imp_gm
		clear_saved_scope = spn_rm_ruler
		if = {
			limit = { spn_council_work_due_trigger = yes }
			global_var:spn_respondeo_magna_target = { save_scope_as = spn_rm_ruler }
		}
	}
	# a. Imperium.
	option = {
		name = spn_order.1000.a
		trigger = { spn_imperium_askable_trigger = yes }
		custom_tooltip = spn_order.1000.a.tt
		hidden_effect = { spn_imperium_open_effect = { CRISIS = flag:none MOTION = flag:imperium } }
		ai_chance = {
			base = 10
		}
	}
	# b. The Great Respondeo (F13), when one is due. The target is checked again now (one realm, once).
	option = {
		name = spn_order.1000.b
		trigger = {
			exists = scope:spn_rm_ruler
			spn_imperium_has_voter_trigger = yes
			custom_tooltip = {
				text = spn_rm_target_ok_tt
				spn_respondeo_magna_target_ok_trigger = { RULER = scope:spn_rm_ruler }
			}
			custom_tooltip = {
				text = spn_rm_gm_pays_tt
				can_spend_huge_ps_trigger = yes
			}
		}
		show_as_unavailable = { exists = scope:spn_rm_ruler }
		custom_tooltip = spn_order.1000.b.tt
		hidden_effect = { spn_imperium_open_effect = { CRISIS = flag:none MOTION = flag:respondeo } }
		ai_chance = {
			base = 30
		}
	}
	# c. Then let us hear the others: exactly what Witchcraft's ev10.85.a does, so the meeting goes on.
	option = {
		name = spn_order.1000.c
		hidden_effect = {
			if = {
				limit = { has_variable = spn_imp_in_meeting }
				remove_variable = spn_imp_in_meeting
			}
			scope:loremaster ?= { trigger_event = magic_ev10.86 }
		}
		ai_chance = {
			base = 1
		}
	}
}

# .1005 -- The Ground Splits. The councillor who voted Nay, 30-90 days after the stager chose it. Re-checks everything
# when it fires (red team M6): if the stager is gone, out of the chair, already holds Imperium, or the refusal is no longer
# on record (a natural crisis reconsidered it first), nothing happens and no secret is recorded. Otherwise the crime is
# recorded FIRST, then the councillor answers. No answer kills anyone outright (D-I3): the danger is Witchcraft's own
# escaped demon, which hunts him as A Rift Opens does. Then the emergency session opens.
spn_order.1005 = {
	type = character_event
	title = spn_order.1005.t
	desc = {
		desc = spn_order.1005.desc
		triggered_desc = {
			trigger = { exists = scope:spn_imp_sensed }
			desc = spn_order.1005.desc_sensed
		}
	}
	theme = dread
	left_portrait = { character = root animation = fear }
	trigger = {
		is_alive = yes
		scope:spn_imp_stager ?= {
			is_alive = yes
			global_var:current_archmage_global ?= this
			spn_is_order_member_trigger = yes
			NOT = { has_character_modifier = mage_council_support_modifier }
		}
		global_var:spn_imperium_refused ?= scope:spn_imp_stager
	}
	immediate = {
		hidden_effect = {
			save_scope_as = spn_imp_struck
			# The bank first: the crime is recorded the moment it happens.
			scope:spn_imp_stager = {
				add_secret = {
					type = spn_secret_staged_rift
					target = scope:spn_imp_struck
				}
			}
			# Witchcraft's demon_escapes_effect aims at scope:this0 when one exists: make it him, whatever came before.
			save_scope_as = this0
			# Magical Senses know the hand behind it (deterministic; no odds).
			if = {
				limit = { spn_senses_magic_trigger = yes }
				scope:spn_imp_stager = {
					random_secret = {
						limit = {
							secret_type = spn_secret_staged_rift
							secret_target ?= scope:spn_imp_struck
							NOT = { is_known_by = scope:spn_imp_struck }
						}
						reveal_to = scope:spn_imp_struck
					}
				}
				save_scope_value_as = { name = spn_imp_sensed value = yes }
			}
			spn_flavor_mark_effect = yes
			scope:spn_imp_stager = {
				if = {
					limit = { is_ai = no }
					send_interface_toast = {
						type = event_toast_effect_neutral
						title = spn_imperium_toast_staged
						left_icon = scope:spn_imp_struck
					}
				}
			}
		}
	}
	# a. A Mage closes it at the cost of power and blood. Nothing comes through.
	option = {
		name = spn_order.1005.a
		trigger = {
			has_trait = archmage
			can_spend_standard_ps_trigger = yes
		}
		spend_standard_ps_effect = yes
		increase_wounds_no_death_effect = { REASON = fight }
		custom_tooltip = spn_order.1005.a.tt
		ai_chance = {
			base = 0
			modifier = {
				add = 100
				has_trait = archmage
			}
		}
	}
	# b. A creature drives it back: a hidden roll on his strength (never shown).
	option = {
		name = spn_order.1005.b
		trigger = { spn_has_supernatural_trait = yes }
		custom_tooltip = spn_order.1005.b.tt
		hidden_effect = {
			random = {
				chance = spn_imperium_drive_back_chance_value
				save_scope_value_as = { name = spn_imp_held value = yes }
			}
			if = {
				limit = { exists = scope:spn_imp_held }
				send_interface_toast = {
					type = event_toast_effect_good
					title = spn_imperium_toast_rift_held
					left_icon = root
				}
			}
			else = {
				increase_wounds_no_death_effect = { REASON = fight }
				demon_escapes_effect = yes
				send_interface_toast = {
					type = event_toast_effect_bad
					title = spn_imperium_toast_rift_failed
					left_icon = root
				}
			}
		}
		ai_chance = {
			base = 0
			modifier = {
				add = 40
				has_trait = brave
			}
			modifier = {
				add = 10
				spn_has_supernatural_trait = yes
			}
		}
	}
	# c. Get everyone out.
	option = {
		name = spn_order.1005.c
		custom_tooltip = spn_order.1005.c.tt
		hidden_effect = { demon_escapes_effect = yes }
		ai_chance = {
			base = 10
			modifier = {
				add = 30
				has_trait = brave
			}
			modifier = {
				add = 10
				has_trait = compassionate
			}
		}
	}
	# d. Run.
	option = {
		name = spn_order.1005.d
		add_stress = medium_stress_gain
		custom_tooltip = spn_order.1005.c.tt
		hidden_effect = { demon_escapes_effect = yes }
		ai_chance = {
			base = 10
			modifier = {
				add = 40
				has_trait = craven
			}
		}
	}
	# Then the emergency session, with him as the councillor it struck (+25 for Aye if he is AI).
	after = {
		hidden_effect = { spn_imperium_crisis_effect = { CRISIS = flag:rift } }
	}
}
