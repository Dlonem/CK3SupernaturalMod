spn_gain_power.4010 = {
	type = character_event
	title = spn_gain_power.4010.t
	desc = spn_gain_power.4010.desc
	theme = secret
	
	right_portrait = {
		character = root
		animation = anger
	}
	
	left_portrait = {
		character = scope:courtier_from_realm
		animation = shock
	}
	
	immediate = { 
		# 2.32 -- routed through the shared selector: the local holder's people first, then a
		# neighbouring county, and only then your own court. The old NOR excluded vampires and
		# werewolves but not supernatural_hybrid and not supernatural_hunter, so your court could
		# hand you a hybrid or a monster hunter as prey, and it never touched the species gate.
		vf_select_victim_effect = { SCOPE_NAME = courtier_from_realm GENDER_FEMALE_CHANCE = 50 }
	}


	option = {
		name = spn_gain_power.4010.a
		custom_tooltip = spn_gain_power.4010.a.tt

		random_list = {
			70 = { #Chance of No Complications
			    custom_tooltip =  spn_gain_power.4010.ab.t
				trigger_event = spn_gain_power.4011 #Success Event
				modifier = {
					add = {
						value = prowess
					}
				}
				modifier = { add = 5 has_trait = content }
				modifier = { add = 5 has_trait = ambitious }
				modifier = { add = 5 has_trait_xp = { trait = supernatural_vampire track = vampirism value >= 15 } }
				modifier = { add = 10 has_trait_xp = { trait = supernatural_vampire track = vampirism value >= 35 } }
				modifier = { add = 15 has_trait_xp = { trait = supernatural_vampire track = vampirism value >= 55 } }
				
				modifier = { add = -15 has_trait = sadistic }
				modifier = { add = -15 has_trait = gluttonous }
				modifier = { add = -10 has_trait = impatient }
				modifier = { add = -15 has_trait = lazy }
			}
			20 = { #Chance of the Courtier Dying
			    custom_tooltip =   spn_gain_power.4010.ad.t
				trigger_event = spn_gain_power.4013 #Discovered Event
				modifier = { add = 10 has_trait = sadistic }
				modifier = { add = 15 has_trait = gluttonous }
				modifier = { add = 5 has_trait = comfort_eater }
				
				modifier = { add = -5 has_trait_xp = { trait = supernatural_vampire track = vampirism value >= 15 } }
				modifier = { add = -10 has_trait_xp = { trait = supernatural_vampire track = vampirism value >= 35 } }
				modifier = { add = -15 has_trait_xp = { trait = supernatural_vampire track = vampirism value >= 55 } }
				modifier = { add = -5 has_trait = inappetetic }
				modifier = { add = -5 has_trait = content }
			}
			10 = { #Chance of Becoming Known Vampire
			    custom_tooltip =   spn_gain_power.4010.ac.t
				trigger_event = spn_gain_power.4012 #Discovered Event

				modifier = { add = 10 has_trait = impatient }
				modifier = { add = 15 has_trait = lazy }
				
				modifier = { add = -2 has_trait_xp = { trait = supernatural_vampire track = vampirism value >= 15 } }
				modifier = { add = -4 has_trait_xp = { trait = supernatural_vampire track = vampirism value >= 35 } }
				modifier = { add = -6 has_trait_xp = { trait = supernatural_vampire track = vampirism value >= 55 } }
				modifier = { add = -5 has_trait = paranoid }
			}
		}
		

	}
	option = {
		name = spn_gain_power.4010.b
		custom_tooltip = spn_gain_power.4010.b.tt

		# 2.32 -- this option used to cost stress and grant NOTHING, and its own tooltip admitted it:
		# "no power gain possible." A decline strictly worse than not clicking the decision is not a
		# choice. It is now the safe floor -- less blood, no risk -- and it is landless-native by
		# construction, because the traveller is created on the spot rather than drawn from a court.
		ai_chance = {
			base = 30
			modifier = { add = 25 has_trait = temperate }
			modifier = { add = 20 has_trait = craven }
			modifier = { add = 20 has_trait = paranoid }
			modifier = { add = 15 has_trait = compassionate }
			modifier = { add = -20 has_trait = gluttonous }
			modifier = { add = -15 has_trait = sadistic }
		}
		trigger_event = spn_gain_power.4015
	}

	option = {
		name = spn_gain_power.4010.c
		custom_tooltip = spn_gain_power.4010.c.tt

		# 2.32 -- blood you paid for. The risk is a wasted cost, never a death roll.
		# minor_gold_value is vanilla's own income-scaled amount (about three months' income,
		# clamped and rounded, and it already handles the landless and vassal cases) -- a better
		# choice than the hand-rolled formula in the commented-out hunt decision, and Paradox
		# maintains it.
		trigger = { gold >= minor_gold_value }
		ai_chance = {
			base = 25
			modifier = { add = 25 has_trait = greedy }
			modifier = { add = 20 has_trait = paranoid }
			modifier = { add = 15 has_trait = calm }
			modifier = { add = 15 has_trait = shrewd }
			modifier = { add = -25 has_trait = arrogant }
			modifier = { add = -15 has_trait = wrathful }
		}
		remove_short_term_gold = minor_gold_value
		trigger_event = spn_gain_power.4016
	}
}

		trigger_event = { id = spn_gain_power.4025 }
	}
	
}

	type = character_event
	title = spn_gain_power.4024.t
	desc = spn_gain_power.4024.desc
	theme = skull
	override_background = {
		reference = wilderness_forest_pine
	}
	
	immediate = { 
		spn_create_skilled_hunter_effect = { WHO = root }
		scope:spn_created_hunter = { save_scope_as = spn_new_created_hunter }
	}
	
	right_portrait = {
		character = root
		animation = shock
	}
	
	left_portrait = {
		character = scope:spn_new_created_hunter
		animation = anger
	}
	
	option = {
		name = spn_gain_power.4024.a
		custom_tooltip = spn_gain_power.4024.a.tt

		# 2.32 -- HYBRID GUARD. Hybrids are immortal and this killed them anyway, so ~1% of every
		# hybrid satiate click was instant death (50% wolf route x 2% here) with nothing warning of
		# it. A hybrid now survives the arrow -- and the hunter who loosed it walks away KNOWING,
		# which is a cost that feeds the suspicion chain instead of ending the playthrough.
		# Standing rule: the irreversible step goes last, after everything else has been banked.
		if = {
			limit = { has_trait = supernatural_hybrid }
			increase_wounds_no_death_effect = { REASON = spn_death_silver_arrow_by_hunter_as_werewolf }
			add_stress = major_stress_impact_gain
			if = {
				limit = { exists = scope:spn_new_created_hunter }
				spn_give_knowledge_of_hybrid_secret = { CHARACTER = scope:spn_new_created_hunter }
			}
		}
		else = {
			death = {
				death_reason = spn_death_silver_arrow_by_hunter_as_werewolf
			}
		}
	}
	
}

	desc = spn_gain_power.4015.desc
	theme = intrigue
	override_background = { reference = alley_night }

	immediate = {
		hidden_effect = {
			create_character = {
