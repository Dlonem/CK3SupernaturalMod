# THE ONE WHO FOLLOWED YOU HOME. Root = the creature. scope:spn_mo_admirer is made here (vanilla's alluring suitor
# template: adult, a stranger, of the sex the creature is drawn to). scope:spn_mo_partner, if the picker found one, is
# the mortal lover or spouse he already has.
spn_moment.0014 = {
	type = character_event
	title = spn_moment.0014.t
	desc = spn_moment.0014.desc
	theme = seduction
	override_background = { reference = corridor_night }
	left_portrait = { character = root animation = interested }
	right_portrait = { character = scope:spn_mo_admirer animation = flirtation }
	trigger = {
		spn_mo_root_ok_trigger = yes
		is_imprisoned = no
		exists = location
	}
	immediate = {
		spn_flavor_mark_effect = yes
		add_character_flag = { flag = spn_mo_cd_admirer years = 20 }
		# Now and then the stranger is not a stranger to the dark at all: one of his own kind, far older, who has been
		# deciding whether he is worth knowing (the mod's own elder vampire). Nothing here shows it; .0023 does, later.
		random_list = {
			85 = { }
			15 = {
				trigger = { exists = capital_province }
				spn_create_elder_skilled_attractive_vampire_effect = { WHO = root }
				scope:spn_created_vampire = {
					# 2.42 review: the template alone is a MORTAL of 130-345 years. These are the steps the mod's own elder
					# takes (10_spn_embrace.txt): made a vampire, young to look at, with an elder's Vampirism.
					add_character_flag = {
						flag = supernatural_embrace_vampire
					}
					trigger_event = spn_initiate.0001
					set_immortal_age = { 20 32 }
					set_variable = {
						name = spn_age_aquired_immortality
						value = 20
					}
					hidden_effect = {
						set_variable = { name = spn_initiate_xp value = { integer_range = { min = 20 max = 65 } } days = 1 }
					}
					add_character_flag = spn_mo_elder
					save_scope_as = spn_mo_admirer
				}
			}
		}
		if = {
			limit = { NOT = { exists = scope:spn_mo_admirer } }
			create_character = {
				template = alluring_suitor_character
				location = root.location
				gender_female_chance = root_attraction_based_female_chance
				save_scope_as = spn_mo_admirer
			}
		}
		# 2.42 review: whoever stands in the doorway came for him. (A stranger drawn the other way could not be taken
		# as a lover, and the story had already said they were.)
		scope:spn_mo_admirer ?= { # 2.43: logged an invalid scope three times on 22 Nov
			if = {
				limit = { NOT = { is_attracted_to_gender_of = root } }
				set_sexuality = bisexual
			}
		}
	}
	# Come in out of the cold.
	option = {
		name = spn_moment.0014.a
		add_courtier = scope:spn_mo_admirer
		if = {
			limit = { spn_romance_ok_trigger = { CHARACTER = scope:spn_mo_admirer } }
			set_relation_lover = scope:spn_mo_admirer
		}
		add_stress = minor_stress_impact_loss
		if = {
			limit = { scope:spn_mo_admirer = { has_character_flag = spn_mo_elder } }
			trigger_event = { id = spn_moment.0023 days = { 60 120 } }
		}
	}
	# Come in. Neither of us is sleeping tonight.
	option = {
		name = spn_moment.0014.b
		trait = lustful
		trigger = { has_trait = lustful }
		add_courtier = scope:spn_mo_admirer
		if = {
			limit = { spn_romance_ok_trigger = { CHARACTER = scope:spn_mo_admirer } }
			set_relation_lover = scope:spn_mo_admirer
		}
		add_stress = major_stress_impact_loss
		if = {
			limit = { scope:spn_mo_admirer = { has_character_flag = spn_mo_elder } }
			trigger_event = { id = spn_moment.0023 days = { 60 120 } }
		}
		# There was already someone. She will find out; what she does about it is the next event.
		if = {
			limit = {
				exists = scope:spn_mo_partner
				scope:spn_mo_partner = { is_alive = yes }
			}
			trigger_event = { id = spn_moment.0015 days = { 20 60 } }
		}
	}
	# You did not come for my bed. You came for what I am.
	option = {
		name = spn_moment.0014.c
		trigger = { scope:spn_mo_admirer = { spn_event_turn_ok_trigger = yes } }
		custom_tooltip = spn_moment.0014.c.tt
		add_courtier = scope:spn_mo_admirer
		scope:spn_mo_admirer = {
			spn_give_vampire_blood_effect = { DONOR = root }
			spn_turn_by_sire_effect = { SIRE = root }
		}
	}
	# You look half frozen. And I have not eaten.
	option = {
		name = spn_moment.0014.d
		trait = gluttonous
		trigger = {
			has_trait = gluttonous
			scope:spn_mo_admirer = { NOT = { has_character_flag = spn_mo_elder } } # one does not dine on an elder
		}
		vf_mark_fed_on_effect = { VICTIM = scope:spn_mo_admirer }
		spn_moment_xp_effect = { XP = 8 }
		add_stress = medium_stress_impact_loss
	}
	# Who sent you?
	option = {
		name = spn_moment.0014.e
		trait = paranoid
		trigger = { has_trait = paranoid }
		hidden_effect = {
			random_list = {
				80 = {
					send_interface_toast = {
						type = event_toast_effect_neutral
						title = spn_moment.0014.e.nobody
						left_icon = scope:spn_mo_admirer
					}
				}
				20 = {
					add_character_flag = { flag = spn_forewarned years = 1 }
					send_interface_toast = {
						type = event_toast_effect_good
						title = spn_moment.0014.e.spy
						left_icon = scope:spn_mo_admirer
					}
				}
			}
		}
		add_stress = minor_stress_impact_loss
	}
	# Go home. Whatever you think you saw, you are mistaken.
	option = {
		name = spn_moment.0014.f
		add_stress = minor_stress_impact_gain
	}
