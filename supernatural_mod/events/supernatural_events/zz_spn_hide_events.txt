# Supernatural 2.37 -- "The Six Hides": the hide's events (WORK-ORDER-2.37 §19). Event titles in comments; loc in
# localization/english/event_localization/spn_hide_l_english.yml. The voice is the mod's own
# (RULE-supernatural-event-writing-voice.md) and the six approved samples (design §11.1) are verbatim.
#
# Every delayed event re-checks the world in its trigger: a delay can outlive what it was about. Saved scopes go first,
# `exists = scope:<x>`, because an artifact can be destroyed in between.

# ==================================================================================================
# .0001 -- (hidden) the weekly hide check (§5.5). One chain per holder; the only timer besides the pack's year.
# ==================================================================================================
spn_hide.0001 = {
	hidden = yes
	trigger = {
		is_alive = yes
		spn_hides_on_trigger = yes
		OR = {
			spn_owns_hide_trigger = yes
			spn_is_champion_trigger = yes
		}
	}
	immediate = {
		# The chain re-fires itself, and a triggered event inherits every saved scope of the one that fired it. Clear
		# what an earlier pass, the Choosing or a death left behind, or one old offer would block every later case.
		if = {
			limit = { exists = scope:spn_offered_hide }
			clear_saved_scope = spn_offered_hide
		}
		if = {
			limit = { exists = scope:spn_off_hide }
			clear_saved_scope = spn_off_hide
		}
		if = {
			limit = { exists = scope:spn_in_circle }
			clear_saved_scope = spn_in_circle
		}
		if = {
			limit = { exists = scope:spn_exit_hide }
			clear_saved_scope = spn_exit_hide
		}
		save_scope_as = spn_wk
		add_character_flag = { flag = spn_hide_check_pending days = 8 }
		# First, for a champion: still a wolf? Still holding his hide?
		if = {
			limit = { spn_is_champion_trigger = yes }
			if = {
				limit = { NOT = { exists = var:spn_champion_of } }
				spn_hide_heal_champion_var_effect = yes
			}
			if = {
				limit = { spn_can_bear_hide_trigger = no }
				spn_hide_exit_effect = { REASON = flag:lost_blood }
			}
			else_if = {
				limit = {
					NOT = { any_character_artifact = { var:spn_hide ?= scope:spn_wk.var:spn_champion_of } }
				}
				if = {
					limit = {
						NOT = { any_artifact = { var:spn_hide ?= scope:spn_wk.var:spn_champion_of } }
					}
					spn_hide_exit_effect = { REASON = flag:lost }
				}
				else = {
					spn_hide_exit_effect = { REASON = flag:lost_blood }
				}
			}
		}
		every_character_artifact = {
			limit = { has_variable = spn_hide }
			save_scope_as = spn_wk_hide
			scope:spn_wk = { spn_hide_weekly_case_effect = yes }
		}
		if = {
			limit = {
				OR = {
					spn_owns_hide_trigger = yes
					spn_is_champion_trigger = yes
				}
			}
			trigger_event = { id = spn_hide.0001 days = 7 }
		}
		else = {
			remove_character_flag = spn_hide_check_pending
		}
	}
}

# ==================================================================================================
# .0002 -- (hidden, no root) seeding at game start, and once for an older save (§5.12). By the game rule.
# ==================================================================================================
spn_hide.0002 = {
	scope = none
	hidden = yes
	trigger = {
		NOT = { exists = global_var:spn_hides_seeded }
	}
	immediate = {
		# 2.37o: the holy-site hook runs at the end of this immediate, after spn_hides_seeded below is set. This keeps
		# it from telling every player wolf where every shrine is on day one (the review of 4 Oct, N3).
		set_global_variable = { name = spn_hides_seeding value = yes days = 1 }
		if = {
			limit = { has_game_rule = spn_hides_off }
		}
		else_if = {
			limit = { has_game_rule = spn_hides_lost }
			spn_hide_record_lost_effect = { HIDE = silverback }
			spn_hide_record_lost_effect = { HIDE = timber }
			spn_hide_record_lost_effect = { HIDE = tundra }
			spn_hide_record_lost_effect = { HIDE = graybeard }
			spn_hide_record_lost_effect = { HIDE = midnight }
			spn_hide_record_lost_effect = { HIDE = alpha }
		}
		else_if = {
			limit = { has_game_rule = spn_hides_awakened }
			spn_hide_seed_one_effect = { HIDE = silverback CHAMPION = yes SHRINE = yes }
			spn_hide_seed_one_effect = { HIDE = timber CHAMPION = yes SHRINE = yes }
			spn_hide_seed_one_effect = { HIDE = tundra CHAMPION = yes SHRINE = yes }
			spn_hide_seed_one_effect = { HIDE = graybeard CHAMPION = yes SHRINE = yes }
			spn_hide_seed_one_effect = { HIDE = midnight CHAMPION = yes SHRINE = yes }
			spn_hide_record_lost_effect = { HIDE = alpha }
		}
		else = {
			# Scattered (default): Alpha lost; ONE of the other five to the best-fitting AI wolf if one exists; the
			# rest in shrines (design T8).
			random_list = {
				1 = { save_scope_value_as = { name = spn_seed_wolf_hide value = flag:silverback } }
				1 = { save_scope_value_as = { name = spn_seed_wolf_hide value = flag:timber } }
				1 = { save_scope_value_as = { name = spn_seed_wolf_hide value = flag:tundra } }
				1 = { save_scope_value_as = { name = spn_seed_wolf_hide value = flag:graybeard } }
				1 = { save_scope_value_as = { name = spn_seed_wolf_hide value = flag:midnight } }
			}
			if = {
				limit = { scope:spn_seed_wolf_hide = flag:silverback }
				spn_hide_seed_one_effect = { HIDE = silverback CHAMPION = yes SHRINE = yes }
			}
			else = {
				spn_hide_seed_one_effect = { HIDE = silverback CHAMPION = no SHRINE = yes }
			}
			if = {
				limit = { scope:spn_seed_wolf_hide = flag:timber }
				spn_hide_seed_one_effect = { HIDE = timber CHAMPION = yes SHRINE = yes }
			}
			else = {
				spn_hide_seed_one_effect = { HIDE = timber CHAMPION = no SHRINE = yes }
			}
			if = {
				limit = { scope:spn_seed_wolf_hide = flag:tundra }
				spn_hide_seed_one_effect = { HIDE = tundra CHAMPION = yes SHRINE = yes }
			}
			else = {
				spn_hide_seed_one_effect = { HIDE = tundra CHAMPION = no SHRINE = yes }
			}
			if = {
				limit = { scope:spn_seed_wolf_hide = flag:graybeard }
				spn_hide_seed_one_effect = { HIDE = graybeard CHAMPION = yes SHRINE = yes }
			}
			else = {
				spn_hide_seed_one_effect = { HIDE = graybeard CHAMPION = no SHRINE = yes }
			}
			if = {
				limit = { scope:spn_seed_wolf_hide = flag:midnight }
				spn_hide_seed_one_effect = { HIDE = midnight CHAMPION = yes SHRINE = yes }
			}
			else = {
				spn_hide_seed_one_effect = { HIDE = midnight CHAMPION = no SHRINE = yes }
			}
			spn_hide_record_lost_effect = { HIDE = alpha }
		}
		set_global_variable = { name = spn_hides_seeded value = yes }
	}
}

# .0003 -- (hidden, artifact scope) a hide broken through to nothing (§8.2), from on_artifact_broken_through_effect and
# _decay. Root = the artifact; scope:owner. 2.37i: an event rather than the on_action's own effect (see the on_action).
# ==================================================================================================
spn_hide.0003 = {
	scope = artifact
	hidden = yes
	trigger = {
		has_variable = spn_hide
	}
	immediate = {
		save_scope_as = spn_broken_hide
		spn_hide_save_champion_effect = yes
		if = {
			limit = { exists = scope:spn_hide_champ }
			if = {
				limit = { scope:owner ?= scope:spn_hide_champ }
				scope:spn_hide_champ = { spn_hide_exit_effect = { REASON = flag:lost_blood } }
			}
		}
		spn_hide_shrine_left_effect = yes
		if = {
			limit = { var:spn_hide = flag:silverback }
			spn_hide_record_lost_effect = { HIDE = silverback }
		}
		if = {
			limit = { var:spn_hide = flag:timber }
			spn_hide_record_lost_effect = { HIDE = timber }
		}
		if = {
			limit = { var:spn_hide = flag:tundra }
			spn_hide_record_lost_effect = { HIDE = tundra }
		}
		if = {
			limit = { var:spn_hide = flag:graybeard }
			spn_hide_record_lost_effect = { HIDE = graybeard }
		}
		if = {
			limit = { var:spn_hide = flag:midnight }
			spn_hide_record_lost_effect = { HIDE = midnight }
		}
		if = {
			limit = { var:spn_hide = flag:alpha }
			spn_hide_record_lost_effect = { HIDE = alpha }
		}
		if = {
			limit = { NOT = { scope:owner ?= { is_alive = yes } } }
			destroy_artifact = this
		}
	}
}

# .0009 -- (hidden) the hide decides (§5.8). The one place a hide rolls fit. From §5.5 case 3 and the fever (§10).
# ==================================================================================================
spn_hide.0009 = {
	hidden = yes
	trigger = {
		is_alive = yes
		spn_can_bear_hide_trigger = yes
		spn_is_champion_trigger = no
		OR = {
			exists = scope:spn_offered_hide
			exists = var:spn_hide_fever_hide
		}
	}
	immediate = {
		if = {
			limit = {
				NOT = { exists = scope:spn_offered_hide }
				exists = var:spn_hide_fever_hide
			}
			random_character_artifact = {
				limit = { var:spn_hide ?= root.var:spn_hide_fever_hide }
				save_scope_as = spn_offered_hide
			}
		}
		if = {
			limit = {
				exists = scope:spn_offered_hide
				scope:spn_offered_hide = {
					artifact_owner ?= root
					NOT = { spn_hide_refused_trigger = { WHO = root } }
				}
				OR = {
					scope:spn_offered_hide = { is_equipped = yes }
					has_character_flag = spn_hide_fever_bonus
				}
			}
			spn_hide_save_flag_effect = { NAME = spn_fit_hide FROM = scope:spn_offered_hide.var:spn_hide }
			if = {
				limit = { spn_hide_accepts_wolf_trigger = yes }
				trigger_event = spn_hide.0010
			}
			else = {
				# Recorded here, where it decides: the button on It Will Not Have You does it too, but a window closed
				# any other way must not leave the hide trying him again next week (3 Oct save: no flag after a refusal).
				spn_hide_mark_refused_effect = { FROM = scope:spn_offered_hide.var:spn_hide YEARS = 5 }
				trigger_event = spn_hide.0011
			}
		}
		if = {
			limit = { has_character_flag = spn_hide_takes_hold_pending }
			remove_character_flag = spn_hide_takes_hold_pending
		}
		if = {
			limit = { exists = var:spn_hide_fever_hide }
			remove_variable = spn_hide_fever_hide
		}
	}
}
