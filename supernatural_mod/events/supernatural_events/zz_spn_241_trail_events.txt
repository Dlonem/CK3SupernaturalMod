##################################################
# The Trail Leads Home
# Root = the employer. scope:spn_trailsman, scope:spn_menace_county. scope:spn_tlh_where = yes (WHERE) / no (WHO).
##################################################
spn_hunt.0402 = {
	type = character_event
	title = spn_hunt.0402.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:spn_tlh_where = yes }
				desc = spn_hunt.0402.desc_where
			}
			desc = spn_hunt.0402.desc_who
		}
	}
	theme = dread
	override_background = { reference = corridor_night }
	left_portrait = {
		character = root
		animation = worry
	}
	right_portrait = {
		character = scope:spn_trailsman
		animation = hunting_knife_start
	}

	trigger = {
		is_alive = yes
		is_imprisoned = no
		exists = scope:spn_trailsman
		scope:spn_trailsman = { is_alive = yes }
		exists = scope:spn_menace_county
	}

	immediate = {
		# Somebody else on or beside the county who could carry it instead: a creature or a witch, not you, not him.
		scope:spn_menace_county = {
			holder ?= {
				if = {
					limit = {
						NOT = { this = root }
						NOT = { this = scope:spn_trailsman }
						spn_menace_heir_candidate_trigger = yes
					}
					save_scope_as = spn_tlh_neighbour
				}
				else = {
					random_courtier_or_guest = {
						limit = {
							NOT = { this = root }
							NOT = { this = scope:spn_trailsman }
							spn_menace_heir_candidate_trigger = yes
						}
						save_scope_as = spn_tlh_neighbour
					}
				}
			}
		}
		# A rival, nemesis or grudge of yours who is a creature you know to be one.
		random_relation = {
			type = rival
			limit = { spn_cold_trail_rival_creature_trigger = { KNOWER = root } }
			save_scope_as = spn_tlh_rival
		}
		if = {
			limit = { NOT = { exists = scope:spn_tlh_rival } }
			random_relation = {
				type = nemesis
				limit = { spn_cold_trail_rival_creature_trigger = { KNOWER = root } }
				save_scope_as = spn_tlh_rival
			}
		}
		if = {
			limit = { NOT = { exists = scope:spn_tlh_rival } }
			random_relation = {
				type = grudge
				limit = { spn_cold_trail_rival_creature_trigger = { KNOWER = root } }
				save_scope_as = spn_tlh_rival
			}
		}
		# A mortal rival you could put it on instead.
		random_relation = {
			type = rival
			limit = {
				is_alive = yes
				spn_has_supernatural_trait = no
				spn_is_valid_supernatural_target = yes
			}
			save_scope_as = spn_tlh_mortal
		}
		if = {
			limit = { NOT = { exists = scope:spn_tlh_mortal } }
			random_relation = {
				type = grudge
				limit = {
					is_alive = yes
					spn_has_supernatural_trait = no
					spn_is_valid_supernatural_target = yes
				}
				save_scope_as = spn_tlh_mortal
			}
		}
	}

	# Then let them know what I am.
	option = {
		name = spn_hunt.0402.a
		trigger = {
			OR = {
				spn_hunt_is_creature_trigger = yes
				any_secret = { secret_type = secret_witch }
			}
		}
		custom_tooltip = spn_hunt.0402.a.tt
		every_secret = {
			limit = {
				OR = {
					secret_type = spn_secret_vampire
					secret_type = spn_secret_werewolf
					secret_type = spn_secret_hybrid
					secret_type = secret_witch
				}
			}
			expose_secret = root
		}
		# A creature with no secret to expose (a rite that does not condemn his kind) is made known outright.
		if = {
			limit = { has_trait = supernatural_hybrid  NOT = { has_trait = spn_known_hybrid } }
			add_trait = spn_known_hybrid
		}
		else_if = {
			limit = { has_trait = supernatural_vampire  NOT = { has_trait = spn_known_vampire } }
			add_trait = spn_known_vampire
		}
		else_if = {
			limit = { has_trait = supernatural_werewolf  NOT = { has_trait = spn_known_werewolf } }
			add_trait = spn_known_werewolf
		}
		stress_and_fulfillment_impact = {
			honest = minor_stress_impact_loss
			arrogant = minor_stress_impact_loss
			brave = minor_stress_impact_loss
			deceitful = minor_stress_impact_gain
			paranoid = medium_stress_impact_gain
		}
		ai_chance = {
			base = 5
			modifier = { add = 25 has_trait = honest }
			modifier = { add = 20 has_trait = arrogant }
			modifier = { add = -20 has_trait = paranoid }
			modifier = { add = -20 has_trait = deceitful }
		}
	}

	# Forget what you found.
	option = {
		name = spn_hunt.0402.b
		trigger = {
			OR = {
				has_trait = supernatural_vampire
				has_trait = supernatural_hybrid
			}
			spn_compel_can_pay_trigger = yes
		}
		custom_tooltip = spn_hunt.0402.b.tt
		spn_compel_pay_effect = yes
		scope:spn_menace_county = {
			set_variable = { name = spn_trailsman_named value = yes years = 5 }
			set_variable = { name = spn_trailsman_reported value = yes years = 5 }
		}
		ai_chance = {
			base = 25
			modifier = { add = 20 has_trait = paranoid }
			modifier = { add = 15 has_trait = deceitful }
		}
	}

	# It was them. Look again. (somebody in or beside the county)
	option = {
		name = spn_hunt.0402.c
		trigger = { exists = scope:spn_tlh_neighbour }
		custom_tooltip = spn_hunt.0402.c.tt
		scope:spn_menace_county = { spn_ledger_move_heat_effect = { FROM = root TO = scope:spn_tlh_neighbour } }
		stress_and_fulfillment_impact = {
			honest = medium_stress_impact_gain
			compassionate = minor_stress_impact_gain
			just = minor_stress_impact_gain
		}
		ai_chance = {
			base = 20
			modifier = { add = 20 has_trait = deceitful }
			modifier = { add = 10 has_trait = callous }
			modifier = { add = -20 has_trait = honest }
			modifier = { factor = 0 scope:spn_tlh_neighbour = { is_ai = no } }
		}
	}

	# Ask yourself where my rival was that night. (a rival creature)
	option = {
		name = spn_hunt.0402.d
		trigger = {
			scope:spn_tlh_where = no
			exists = scope:spn_tlh_rival
		}
		custom_tooltip = spn_hunt.0402.d.tt
		duel = {
			skill = intrigue
			target = scope:spn_trailsman
			localization = spn_hunt_redirect_duel # 2.43: a named contest, as in 10_spn_hunt_events.txt
			60 = {
				desc = spn_hunt.0402.d.success
				compare_modifier = { value = scope:duel_value multiplier = 3 min = -40 }
				scope:spn_menace_county = { spn_ledger_move_heat_effect = { FROM = root TO = scope:spn_tlh_rival } }
				scope:spn_tlh_rival = { trigger_event = { id = spn_hunt.0717 days = { 5 15 } } }
			}
			40 = {
				desc = spn_hunt.0402.d.failure
				compare_modifier = { value = scope:duel_value multiplier = -3 min = -30 }
				# He knows you lied. He asks again next year.
				scope:spn_menace_county = { remove_variable = spn_trailsman_named }
				scope:spn_trailsman = { add_opinion = { target = root modifier = suspicion_opinion opinion = -20 } }
			}
		}
		ai_chance = {
			base = 25
			modifier = { add = 25 has_trait = deceitful }
			modifier = { add = 20 has_trait = vengeful }
			modifier = { add = -25 has_trait = honest }
			modifier = { factor = 0 scope:spn_tlh_rival = { is_ai = no } }
		}
	}

	# They have a monster's habits. (a mortal rival, framed)
	option = {
		name = spn_hunt.0402.e
		trigger = {
			scope:spn_tlh_where = no
			exists = scope:spn_tlh_mortal
		}
		custom_tooltip = spn_hunt.0402.e.tt
		duel = {
			skill = intrigue
			target = scope:spn_trailsman
			localization = spn_hunt_redirect_duel # 2.43
			45 = {
				desc = spn_hunt.0402.e.success
				compare_modifier = { value = scope:duel_value multiplier = 3 min = -40 }
				scope:spn_menace_county = { spn_ledger_move_heat_effect = { FROM = root TO = scope:spn_tlh_mortal } }
			}
			55 = {
				desc = spn_hunt.0402.e.failure
				compare_modifier = { value = scope:duel_value multiplier = -3 min = -30 }
				scope:spn_menace_county = { remove_variable = spn_trailsman_named }
				scope:spn_trailsman = { add_opinion = { target = root modifier = suspicion_opinion opinion = -30 } }
			}
		}
		stress_and_fulfillment_impact = {
			honest = major_stress_impact_gain
			just = medium_stress_impact_gain
			compassionate = medium_stress_impact_gain
		}
		ai_chance = {
			base = 10
			modifier = { add = 20 has_trait = deceitful }
			modifier = { add = 15 has_trait = callous }
			modifier = { add = 15 has_trait = sadistic }
			modifier = { add = -30 has_trait = honest }
			modifier = { factor = 0 scope:spn_tlh_mortal = { is_ai = no } }
		}
	}

	# Keep it to yourself, and this purse with it.
	option = {
		name = spn_hunt.0402.f
		trigger = { gold >= medium_gold_value }
		custom_tooltip = spn_hunt.0402.f.tt
		pay_short_term_gold = {
			target = scope:spn_trailsman
			gold = medium_gold_value
		}
		scope:spn_menace_county = { spn_menace_buy_silence_effect = { PAYER = root } }
		every_secret = {
			limit = {
				OR = {
					secret_type = spn_secret_vampire
					secret_type = spn_secret_werewolf
					secret_type = spn_secret_hybrid
					secret_type = secret_witch
					secret_type = secret_murder
				}
				NOT = { is_known_by = scope:spn_trailsman }
			}
			reveal_to = scope:spn_trailsman
		}
		ai_chance = {
			base = 30
			modifier = { add = 20 has_trait = compassionate }
			modifier = { add = 15 has_trait = paranoid }
			modifier = { add = -25 has_trait = greedy }
		}
	}

	# I will clean up after myself.
	option = {
		name = spn_hunt.0402.g
		custom_tooltip = spn_hunt.0402.g.tt
		scope:spn_menace_county = {
			spn_menace_clear_one_effect = yes
			spn_ledger_cool_culprit_effect = { WHO = root } # 2.41 boot 6: his own heat here, too
		}
		add_character_modifier = { modifier = spn_gone_to_ground_modifier years = 1 }
		ai_chance = {
			base = 20
			modifier = { add = 20 has_trait = patient }
			modifier = { add = 15 has_trait = craven }
		}
	}

	# You have done your job too well.
	option = {
		name = spn_hunt.0402.h
		trigger = { scope:spn_tlh_where = no }
		custom_tooltip = spn_hunt.0402.h.tt
		add_dread = medium_dread_gain
		stress_and_fulfillment_impact = {
			compassionate = major_stress_impact_gain
			just = medium_stress_impact_gain
			honest = minor_stress_impact_gain
		}
		# Last: everything above has read what it needs; the death goes after it (the standing ordering rule).
		scope:spn_trailsman = {
			death = {
				death_reason = death_murder
				killer = root
			}
		}
		ai_chance = {
			base = 10
			modifier = { add = 25 has_trait = sadistic }
			modifier = { add = 20 has_trait = callous }
			modifier = { add = 15 has_trait = wrathful }
			modifier = { add = 15 has_trait = paranoid }
			modifier = { add = -25 has_trait = compassionate }
		}
	}

	# Find someone else to follow.
	option = {
		name = spn_hunt.0402.i
		custom_tooltip = spn_hunt.0402.i.tt
		if = {
			limit = { employs_court_position = spn_trailsman_court_position }
			revoke_court_position = {
				recipient = scope:spn_trailsman
				court_position = spn_trailsman_court_position
			}
		}
		else_if = {
			limit = { employs_court_position = spn_trailsman_camp_officer }
			revoke_court_position = {
				recipient = scope:spn_trailsman
				court_position = spn_trailsman_camp_officer
			}
		}
		ai_chance = {
			base = 10
			modifier = { add = 15 has_trait = arrogant }
		}
	}

	# WHERE only: Then I will be more careful.
	option = {
		name = spn_hunt.0402.j
		trigger = { scope:spn_tlh_where = yes }
		custom_tooltip = spn_hunt.0402.j.tt
		ai_chance = {
			base = 30
			modifier = { add = 20 has_trait = patient }
			modifier = { add = 15 has_trait = content }
		}
	}
}

# A Man, Not a Monster
# Root = the hunter (host). Hunt scopes bound: spn_hunt_activity, spn_hunt_county, spn_culprit.
##################################################
spn_hunt.0700 = {
	type = activity_event
	title = spn_hunt.0700.t
	desc = spn_hunt.0700.desc
	theme = hunt_activity
	override_background = { reference = alley_night }
	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		character = scope:spn_culprit
		animation = nervous
	}

	trigger = {
		is_alive = yes
		exists = scope:spn_culprit
		scope:spn_culprit = { is_alive = yes }
	}

	immediate = {
		spn_hunt_bind_scopes_effect = yes
	}

	# I don't hunt people.
	option = {
		name = spn_hunt.0700.a
		custom_tooltip = spn_hunt.0700.a.tt
		spn_hunt_set_outcome_effect = { OUTCOME = human_spared }
		if = {
			limit = {
				exists = scope:spn_hunt_county.holder
				NOT = { scope:spn_hunt_county.holder = scope:spn_culprit }
			}
			scope:spn_culprit = { spn_hunt_reveal_culprit_effect = { LORD = scope:spn_hunt_county.holder } }
		}
		stress_and_fulfillment_impact = {
			compassionate = minor_stress_impact_loss
			forgiving = minor_stress_impact_loss
			content = minor_stress_impact_loss
			wrathful = minor_stress_impact_gain
			vengeful = minor_stress_impact_gain
			sadistic = minor_stress_impact_gain
		}
		ai_chance = {
			base = 35
			modifier = { add = 30 has_trait = compassionate }
			modifier = { add = 20 has_trait = forgiving }
			modifier = { add = 10 has_trait = content }
			modifier = { add = -25 has_trait = wrathful }
			modifier = { add = -25 has_trait = vengeful }
			modifier = { add = -20 has_trait = sadistic }
		}
	}

	# A monster is a monster.
	option = {
		name = spn_hunt.0700.b
		# 2.41 audit -- not against a player: this kills with no roll and no answer.
		trigger = { scope:spn_culprit = { is_ai = yes } }
		custom_tooltip = spn_hunt.0700.b.tt
		spn_hunt_set_outcome_effect = { OUTCOME = human_slain }
		stress_and_fulfillment_impact = {
			diligent = minor_stress_impact_loss
			zealous = minor_stress_impact_loss
			just = minor_stress_impact_loss
			wrathful = minor_stress_impact_loss
			vengeful = minor_stress_impact_loss
			compassionate = medium_stress_impact_gain
			forgiving = minor_stress_impact_gain
		}
		# Last: the outcome is banked; the death goes after it.
		scope:spn_culprit = {
			death = {
				death_reason = spn_death_killed_as_culprit # 2.42 review: not a creature; this killing stays public (zz_spn_242_deaths.txt)
				killer = root
			}
		}
		ai_chance = {
			base = 35
			modifier = { add = 25 has_trait = diligent }
			modifier = { add = 20 has_trait = zealous }
			modifier = { add = 15 has_trait = just }
			modifier = { add = 20 has_trait = wrathful }
			modifier = { add = 20 has_trait = vengeful }
			modifier = { add = -30 has_trait = compassionate }
			modifier = { add = -15 has_trait = forgiving }
		}
	}

	# He answers to his lord.
	option = {
		name = spn_hunt.0700.c
		trigger = {
			exists = scope:spn_hunt_county.holder
			NOT = { scope:spn_hunt_county.holder = scope:spn_culprit }
			scope:spn_hunt_county.holder = {
				basic_allowed_to_imprison_character_trigger = { CHARACTER = scope:spn_culprit }
			}
		}
		custom_tooltip = spn_hunt.0700.c.tt
		spn_hunt_set_outcome_effect = { OUTCOME = human_jailed }
		scope:spn_hunt_county.holder = {
			imprison = {
				target = scope:spn_culprit
				type = dungeon
			}
		}
		stress_and_fulfillment_impact = {
			just = minor_stress_impact_loss
			honest = minor_stress_impact_loss
			deceitful = minor_stress_impact_gain
		}
		ai_chance = {
			base = 30
			modifier = { add = 25 has_trait = just }
			modifier = { add = 15 has_trait = honest }
			modifier = { add = -15 has_trait = deceitful }
		}
	}
}

##################################################
# There Is Another One
# Root = the hunter. scope:spn_ct_next = the next creature standing in the county. Hunt scopes bound (conclude).
##################################################
spn_hunt.0705 = {
	type = character_event
	title = spn_hunt.0705.t
	desc = spn_hunt.0705.desc
	theme = hunt_activity
	override_background = { reference = wilderness_forest }
	left_portrait = {
		character = root
		animation = marshal
	}
	right_portrait = {
		character = scope:spn_ct_next
		animation = menacing
	}

	trigger = {
		is_alive = yes
		is_imprisoned = no
		exists = scope:spn_ct_next
		scope:spn_ct_next = {
			is_alive = yes
			is_imprisoned = no
		}
	}

	# I go after it now.
	option = {
		name = spn_hunt.0705.a
		custom_tooltip = spn_hunt.0705.a.tt
		save_scope_as = spn_hunter
		duel = {
			skill = prowess
			target = scope:spn_ct_next
			localization = spn_hunt_fight_duel # 2.43
			50 = {
				desc = spn_hunt.0705.a.success
				compare_modifier = { value = scope:duel_value multiplier = 3 min = -40 }
				modifier = { add = 15 spn_hunter_is_expert_trigger = yes }
				add_prestige = minor_prestige_gain
				add_trait_xp = { trait = supernatural_hunter track = monster_hunting value = { 10 25 } }
				scope:spn_hunt_county ?= {
					if = {
						limit = { spn_ledger_has_other_living_culprit_trigger = { WHO = scope:spn_ct_next } }
						spn_ledger_forget_culprit_effect = { WHO = scope:spn_ct_next }
						spn_menace_recompute_lead_effect = yes
						hidden_effect = { spn_trails_map_sync_effect = yes }
					}
					else = {
						spn_menace_clear_county_effect = yes
					}
				}
				# Last: everything above is banked.
				scope:spn_ct_next = { spn_hunt_perform_the_kill_effect = yes }
			}
			50 = {
				desc = spn_hunt.0705.a.failure
				compare_modifier = { value = scope:duel_value multiplier = -3 min = -30 }
				increase_wounds_no_death_effect = { REASON = fight }
				add_stress = minor_stress_impact_gain
			}
		}
		ai_chance = {
			base = 50
			modifier = { add = 25 has_trait = brave }
			modifier = { add = 15 has_trait = diligent }
			modifier = { add = -40 has_trait = craven }
			modifier = { add = -20 has_trait_xp = { trait = supernatural_hunter track = monster_hunting value < 75 } }
		}
	}

	# Not today.
	option = {
		name = spn_hunt.0705.b
		ai_chance = {
