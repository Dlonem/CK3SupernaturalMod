# .0001 -- The Masks Must Meet. A player asked to call the Conclave: the planner opens from here.
spn_conclave.0001 = {
	type = character_event
	title = spn_conclave.0001.t
	desc = {
		first_valid = {
			# 2.40 (WORK-ORDER-2.40 A5): the text says why this temple was asked.
			triggered_desc = {
				trigger = {
					var:spn_conclave_reason ?= flag:chair
					has_variable = spn_conclave_passed
				}
				desc = spn_conclave.0001.desc_chair_passed
			}
			triggered_desc = {
				trigger = {
					var:spn_conclave_reason ?= flag:prerogative
					has_variable = spn_conclave_passed
				}
				desc = spn_conclave.0001.desc_prerogative_passed
			}
			triggered_desc = {
				trigger = {
					var:spn_conclave_reason ?= flag:prerogative
					NOT = { global_var:current_archmage_global ?= root }
				}
				desc = spn_conclave.0001.desc_prerogative_why
			}
			triggered_desc = {
				trigger = { var:spn_conclave_reason ?= flag:chair }
				desc = spn_conclave.0001.desc_chair
			}
			triggered_desc = {
				trigger = { var:spn_conclave_reason ?= flag:prerogative }
				desc = spn_conclave.0001.desc_prerogative
			}
			# 2.39 (WORK-ORDER-2.39 F1): Invocare.
			triggered_desc = {
				trigger = {
					var:spn_conclave_reason ?= flag:invocare
					var:spn_conclave_iv_caller ?= root
				}
				desc = spn_conclave.0001.desc_invocare_self
			}
			triggered_desc = {
				trigger = { var:spn_conclave_reason ?= flag:invocare }
				desc = spn_conclave.0001.desc_invocare
			}
			desc = spn_conclave.0001.desc_will
		}
		triggered_desc = {
			trigger = { exists = scope:spn_cv_second }
			desc = spn_conclave.0001.desc_second
		}
	}
	theme = learning
	override_background = { reference = magic_temple }
	left_portrait = { character = root animation = personality_rational }
	# 2.38c: your second, the highest member of your temple after you (Thomas: "so it doesn't feel like it's just me").
	right_portrait = {
		character = scope:spn_cv_second
		trigger = { exists = scope:spn_cv_second }
		animation = personality_bold
	}
	trigger = {
		is_alive = yes
		has_character_flag = spn_conclave_may_call
		NOT = { involved_activity ?= { has_activity_type = spn_conclave } } # 2.38c: already planning or travelling to it
		# 2.40 (WORK-ORDER-2.40 A3/A5): a call the Order hands him waits for a free player and 30 days after other Order
		# business. A Grand Magus calling at will asked for it now.
		trigger_if = {
			limit = { NOT = { var:spn_conclave_reason ?= flag:will } }
			spn_order_player_free_trigger = yes
			spn_order_business_ready_trigger = yes
		}
		# 2.39 (WORK-ORDER-2.39 B12 d): the chair may have been filled while the note waited.
		trigger_if = {
			limit = { var:spn_conclave_reason ?= flag:chair }
			NOT = { exists = global_var:current_archmage_global }
		}
		# 2.39 (WORK-ORDER-2.39 F1, G1): the Grand Magus the motion names is still in the chair.
		trigger_if = {
			limit = { var:spn_conclave_reason ?= flag:invocare }
			exists = var:spn_conclave_iv_magus
			global_var:current_archmage_global ?= var:spn_conclave_iv_magus
		}
	}
	on_trigger_fail = {
		# 2.39 (WORK-ORDER-2.39 B12 d): the chair was filled meanwhile: the call is spent. Only this failure clears it.
		if = {
			limit = {
				var:spn_conclave_reason ?= flag:chair
				exists = global_var:current_archmage_global
			}
			if = {
				limit = { has_character_flag = spn_conclave_may_call }
				remove_character_flag = spn_conclave_may_call
			}
			remove_variable = spn_conclave_reason
		}
		# 2.39 (review L-3): the Grand Magus it named has left the chair, and the chair's own call reached this host while the
		# motion waited: the call becomes the chair's (one of the two notes is still on its way; failing that, the activity
		# list serves), and the motion lapses unheard. Nobody pays.
		else_if = {
			limit = {
				var:spn_conclave_reason ?= flag:invocare
				NOT = { exists = global_var:current_archmage_global }
				exists = var:spn_conclave_chair_after
				has_character_flag = spn_conclave_may_call
			}
			set_variable = { name = spn_conclave_reason value = flag:chair years = 2 }
			remove_variable = spn_conclave_chair_after
			if = {
				limit = { has_variable = spn_conclave_iv_magus }
				remove_variable = spn_conclave_iv_magus
			}
			if = {
				limit = { has_variable = spn_conclave_iv_caller }
				remove_variable = spn_conclave_iv_caller
			}
			if = {
				limit = { has_global_variable = spn_invocare_called }
				remove_global_variable = spn_invocare_called
			}
		}
		# 2.39 (WORK-ORDER-2.39 F1, G1): the Grand Magus it named has left the chair: the motion lapses, and nobody pays.
		else_if = {
			limit = {
				var:spn_conclave_reason ?= flag:invocare
				NOT = {
					AND = {
						exists = var:spn_conclave_iv_magus
						global_var:current_archmage_global ?= var:spn_conclave_iv_magus
					}
				}
			}
			spn_invocare_clear_call_effect = yes
		}
		# 2.40 (WORK-ORDER-2.40 A3/A5): the call still stands and only he stood in the way. The spacing alone: ask again
		# in 31 days. Busy: the masks pass the call to another temple, and tell him.
		else_if = {
			limit = {
				is_alive = yes
				has_character_flag = spn_conclave_may_call
				exists = var:spn_conclave_reason
				NOT = { var:spn_conclave_reason ?= flag:will }
				NOT = { involved_activity ?= { has_activity_type = spn_conclave } }
			}
			if = {
				limit = { spn_order_player_free_trigger = yes }
				trigger_event = { id = spn_conclave.0001 days = 31 }
			}
			else = {
				spn_conclave_pass_on_effect = yes
				if = {
					limit = { exists = scope:spn_cl_host }
					send_interface_toast = {
						type = event_toast_text_neutral
						title = spn_conclave_toast_passed
						desc = spn_conclave_toast_passed_desc
					}
				}
				else = {
					send_interface_toast = {
						type = event_toast_text_neutral
						title = spn_conclave_toast_passed
						desc = spn_conclave_toast_passed_none_desc
					}
				}
			}
		}
	}
	immediate = {
		spn_order_business_mark_effect = yes # 2.40 (WORK-ORDER-2.40 A3)
		# The named heir first, as the succession does (spn_order_succession_effect); else the highest member.
		var:spn_order_chapter.var:spn_order_heir ?= {
			if = {
				limit = {
					is_alive = yes
					NOT = { this = root }
				}
				save_scope_as = spn_cv_second
			}
		}
		var:spn_order_chapter ?= {
			ordered_in_list = {
				variable = spn_order_members
				limit = {
					is_alive = yes
					NOT = { this = root }
					spn_is_order_member_trigger = yes
				}
				order_by = spn_order_succession_score_value
				save_scope_as = spn_cv_second_alt
			}
		}
		if = {
			limit = { NOT = { exists = scope:spn_cv_second } }
			scope:spn_cv_second_alt ?= { save_scope_as = spn_cv_second }
		}
	}
	option = {
		name = spn_conclave.0001.a
		open_view_data = {
			view = activity_list_detail_host_window
			data = activity_type:spn_conclave
		}
	}
	# 2.40 (WORK-ORDER-2.40 A5): "Let another temple call it" passes the call on and ends the alert (the old "Not yet" kept
	# the flag, so vanilla's host-an-activity alert stayed for two years).
	option = {
		name = spn_conclave.0001.b
		trigger = { NOT = { var:spn_conclave_reason ?= flag:will } }
		custom_tooltip = spn_conclave.0001.b.tt
		hidden_effect = { spn_conclave_pass_on_effect = yes }
	}
	# A Grand Magus who called it at will: not now (the decision calls it again).
	option = {
		name = spn_conclave.0001.b_will
		trigger = { var:spn_conclave_reason ?= flag:will }
		custom_tooltip = spn_conclave.0001.b_will.tt
		if = {
			limit = { has_character_flag = spn_conclave_may_call }
			remove_character_flag = spn_conclave_may_call
		}
		remove_variable = spn_conclave_reason
	}
	after = {
		if = {
			limit = { has_variable = spn_conclave_passed }
			remove_variable = spn_conclave_passed
		}
	}
}

# .0410 -- Fill the Seats. A player Grand Magus names one member to an empty seat; again while seats and names remain
# (at most three times: spn_cv_seat_rounds).
spn_conclave.0410 = {
	type = character_event
	title = spn_conclave.0410.t
	desc = spn_conclave.0410.desc
	theme = learning
	override_background = { reference = magic_temple }
	left_portrait = { character = root animation = personality_rational }
	trigger = {
		is_alive = yes
		global_var:current_archmage_global ?= root
		spn_conclave_seat_open_trigger = yes
	}
	immediate = {
		# 2.39 (Boot 2, 16 Oct): each round reads its nominees afresh
		if = { limit = { exists = scope:spn_cv_n1 } clear_saved_scope = spn_cv_n1 }
		if = { limit = { exists = scope:spn_cv_n2 } clear_saved_scope = spn_cv_n2 }
		if = { limit = { exists = scope:spn_cv_n3 } clear_saved_scope = spn_cv_n3 }
