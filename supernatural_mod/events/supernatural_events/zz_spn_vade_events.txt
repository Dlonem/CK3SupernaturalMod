	type = character_event
	title = spn_vade.0100.t
	desc = {
		desc = spn_vade.0100.desc
		triggered_desc = {
			trigger = { has_variable = spn_vade_heard_many }
			desc = spn_vade.0100.desc_many
		}
	}
	theme = hunting
	override_background = { reference = wilderness_forest }
	left_portrait = { character = root animation = interested }
	right_portrait = { character = scope:spn_vade_heard_holder animation = idle }
	trigger = {
		is_alive = yes
		spn_is_champion_trigger = yes

		NOT = { exists = global_var:spn_vade_destroyed }

		name = spn_vade.0100.a

			base = 50

	option = {

# .0130 -- Someone Is Pulling at the Hide. The champion; scope:spn_th_caster, scope:spn_th_hide (Tear the Hide Away).
spn_vade.0130 = {
	type = character_event
	title = spn_vade.0130.t
	desc = spn_vade.0130.desc
	theme = hunting
	override_background = { reference = wilderness_forest }
	left_portrait = { character = root animation = rage }
	right_portrait = { character = scope:spn_th_caster animation = scheme }
	trigger = {
		is_alive = yes
		scope:spn_th_caster ?= { is_alive = yes }
		scope:spn_th_hide ?= {
			artifact_owner ?= root
			exists = var:spn_holds_vade_page
		}
	}
	option = {
		name = spn_vade.0130.a
		custom_tooltip = spn_vade.0130.a_tt
		hidden_effect = {
			random = {
				chance = spn_vade_hide_holds_value
				save_scope_value_as = { name = spn_th_held value = yes }
			}
			if = {
				limit = { exists = scope:spn_th_held }
				scope:spn_th_caster = {
					increase_wounds_no_death_effect = { REASON = fight }
					if = {
						limit = { is_ai = no }
						send_interface_toast = {
							type = event_toast_effect_bad
							title = spn_vade_toast_hide_held
							left_icon = this
							right_icon = root
						}
					}
				}
				if = {
					limit = { is_ai = no }
					send_interface_toast = {
						type = event_toast_effect_good
						title = spn_vade_toast_hide_held
						left_icon = root
						right_icon = scope:spn_th_caster
					}
				}
			}
			else = {
				spn_vade_draw_page_effect = { HIDE_ART = scope:spn_th_hide TO = scope:spn_th_caster }
				scope:spn_th_caster = {
					if = {
						limit = { is_ai = no }
						send_interface_toast = {
							type = event_toast_effect_good
							title = spn_vade_toast_hide_torn
							left_icon = this
							right_icon = root
						}
					}
				}
				if = {
					limit = {
						OR = {
							has_trait = maimed
							has_trait_rank = { trait = wounded rank = 2 }
							has_trait_rank = { trait = wounded rank = 3 }
						}
					}
					death = { death_reason = spn_death_hide_torn_away killer = scope:spn_th_caster }
				}
				else = {
					increase_wounds_no_death_effect = { REASON = fight }
					if = {
						limit = { is_ai = no }
						send_interface_toast = {
							type = event_toast_effect_bad
							title = spn_vade_toast_hide_torn
							left_icon = root
							right_icon = scope:spn_th_caster
						}
					}
				}
			}
		}
	}
}

# .0210 -- (hidden) the golem's search. The golem's master; scope:spn_golem. 60% a name, else he comes back wrong.
spn_vade.0210 = {
	type = character_event
	hidden = yes
	trigger = {
		is_alive = yes
		scope:spn_golem ?= { is_alive = yes }
	}
	on_trigger_fail = {
		scope:spn_golem ?= {
			if = {
				limit = { has_character_flag = spn_golem_searching }
				remove_character_flag = spn_golem_searching
			}
		}
	}
	immediate = {
		scope:spn_golem = {
			if = {
				limit = { has_character_flag = spn_golem_searching }
				remove_character_flag = spn_golem_searching
			}
			if = {
				limit = { has_character_modifier = spn_golem_away_modifier }
				remove_character_modifier = spn_golem_away_modifier
			}
		}
		random_list = {
			60 = { trigger_event = spn_vade.0211 }
			40 = {
				scope:spn_golem = { increase_wounds_no_death_effect = { REASON = fight } }
				trigger_event = spn_vade.0212
			}
		}
	}
}

	type = character_event
	hidden = yes
	trigger = {
		is_alive = yes

	immediate = {
