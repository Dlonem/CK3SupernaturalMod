# .0100 -- Under the Glove. The prisoner; scope:spn_gl_asker (§9.3).
spn_relic.0100 = {
	type = character_event
	title = spn_relic.0100.t
	desc = spn_relic.0100.desc
	theme = prison
	override_background = { reference = dungeon }
	left_portrait = { character = root animation = fear }
	right_portrait = { character = scope:spn_gl_asker animation = scheme }
	trigger = {
		is_alive = yes
		scope:spn_gl_asker ?= { is_alive = yes }
		is_imprisoned_by = scope:spn_gl_asker
	}
	immediate = {
		# Word gets out (25%, every use).
		random = {
			chance = 25
			liege ?= {
				if = {
					limit = { NOT = { this = scope:spn_gl_asker } }
					add_opinion = { target = scope:spn_gl_asker modifier = spn_burned_under_the_glove_opinion }
					if = {
						limit = { is_ai = no }
						send_interface_toast = {
							type = event_toast_effect_bad
							title = spn_relic_toast_glove_word
							left_icon = scope:spn_gl_asker
							right_icon = root
						}
					}
				}
			}
			scope:spn_gl_asker = {
				if = {
					limit = { exists = location.county }
					location.county = { save_scope_as = spn_gl_county }
					spn_mark_menace_effect = { COUNTY = scope:spn_gl_county CULPRIT = scope:spn_gl_asker KIND = witch SEVERITY = prowl }
				}
			}
		}
	}
	option = {
		name = spn_relic.0100.a
		if = {
			limit = {
				any_secret = { NOT = { is_known_by = scope:spn_gl_asker } }
			}
			random_secret = {
				limit = { NOT = { is_known_by = scope:spn_gl_asker } }
				reveal_to = scope:spn_gl_asker
			}
		}
		else_if = {
			limit = {
				any_known_secret = {
					NOT = { is_known_by = scope:spn_gl_asker }
					NOT = { secret_owner = scope:spn_gl_asker }
				}
			}
			random_known_secret = {
				limit = {
					NOT = { is_known_by = scope:spn_gl_asker }
					NOT = { secret_owner = scope:spn_gl_asker }
				}
				reveal_to = scope:spn_gl_asker
			}
		}
		else = {
			custom_tooltip = spn_relic.0100.a_nothing
		}
		ai_chance = {
			base = 50
			modifier = { add = 30 has_trait = craven }
			modifier = { add = 20 has_trait = honest }
		}
	}
	option = {
		name = spn_relic.0100.b
		increase_wounds_no_death_effect = { REASON = fight }
		stress_impact = { base = major_stress_impact_gain }
		scope:spn_gl_asker = {
			add_hook = {
				type = threat_hook
				target = root
			}
		}
		add_opinion = { target = scope:spn_gl_asker modifier = spn_burned_under_the_glove_opinion }
		ai_chance = {
			base = 50
			modifier = { add = 30 has_trait = brave }
			modifier = { add = 20 has_trait = stubborn }
			modifier = { add = 20 has_trait = deceitful }
		}
	}
	option = {
		name = spn_relic.0100.c
		trigger = { spn_is_wolf_trigger = yes }
		scope:spn_gl_asker = { add_prestige = -20 }
		ai_chance = { base = 200 }
	}
}

# .0200 -- Call the Dead. The caller; scope:spn_nc_phone (the Necrophone). Three calls and it cracks (§9.3).
spn_relic.0200 = {
	type = character_event
	title = spn_relic.0200.t
	desc = {
		desc = spn_relic.0200.desc
		triggered_desc = {
			trigger = { scope:spn_nc_phone ?= { var:spn_necrophone_calls ?= 2 } }
			desc = spn_relic.0200.desc_cracking
		}
	}
	theme = witchcraft
	override_background = { reference = study }
	left_portrait = { character = root animation = pondering }
	artifact = {
		target = scope:spn_nc_phone
		position = lower_right_portrait
		trigger = { exists = scope:spn_nc_phone }
	}
	trigger = {
		is_alive = yes
		scope:spn_nc_phone ?= { artifact_owner ?= root }
	}
	# How long until a lost hide wakes: read now, so the option's line can show it.
	immediate = {
		if = {
			limit = { has_variable = spn_nc_wake }
			remove_variable = spn_nc_wake
		}
		if = {
			limit = { exists = global_var:spn_hide_alpha_lost }
			set_variable = { name = spn_nc_wake value = global_var:spn_hide_alpha_lost }
		}
		else_if = {
			limit = { exists = global_var:spn_hide_graybeard_lost }
			set_variable = { name = spn_nc_wake value = global_var:spn_hide_graybeard_lost }
		}
		else_if = {
			limit = { exists = global_var:spn_hide_midnight_lost }
			set_variable = { name = spn_nc_wake value = global_var:spn_hide_midnight_lost }
		}
		else_if = {
			limit = { exists = global_var:spn_hide_silverback_lost }
			set_variable = { name = spn_nc_wake value = global_var:spn_hide_silverback_lost }
		}
		else_if = {
			limit = { exists = global_var:spn_hide_timber_lost }
			set_variable = { name = spn_nc_wake value = global_var:spn_hide_timber_lost }
		}
		else_if = {
			limit = { exists = global_var:spn_hide_tundra_lost }
			set_variable = { name = spn_nc_wake value = global_var:spn_hide_tundra_lost }
		}
		if = {
			limit = { exists = var:spn_nc_wake }
			change_variable = { name = spn_nc_wake subtract = current_year }
			if = {
				limit = { var:spn_nc_wake < 1 }
				set_variable = { name = spn_nc_wake value = 1 }
			}
		}
	}
	# Where is a page of the book? One held, unguarded, not by you.
	option = {
		name = spn_relic.0200.a
		trigger = {
			spn_vade_on_trigger = yes
			OR = {
				spn_relic_page_nameable_trigger = { N = 1 }
				spn_relic_page_nameable_trigger = { N = 2 }
				spn_relic_page_nameable_trigger = { N = 3 }
				spn_relic_page_nameable_trigger = { N = 4 }
			}
		}
		if = {
			limit = { spn_relic_page_nameable_trigger = { N = 1 } }
			global_var:spn_vade_page_1_art.artifact_owner = { save_scope_as = spn_nc_named }
		}
		else_if = {
			limit = { spn_relic_page_nameable_trigger = { N = 2 } }
			global_var:spn_vade_page_2_art.artifact_owner = { save_scope_as = spn_nc_named }
		}
		else_if = {
			limit = { spn_relic_page_nameable_trigger = { N = 3 } }
			global_var:spn_vade_page_3_art.artifact_owner = { save_scope_as = spn_nc_named }
		}
		else = {
			global_var:spn_vade_page_4_art.artifact_owner = { save_scope_as = spn_nc_named }
		}
		spn_vade_learn_holder_effect = { HOLDER = scope:spn_nc_named } # 2.40 (WORK-ORDER-2.40 B3): and a claim
		custom_tooltip = spn_vade_claim_given_tt
		custom_tooltip = spn_relic.0200.a_tt
		spn_relic_necrophone_used_effect = yes
	}
	# Where is the lost hide?
	option = {
		name = spn_relic.0200.b
		trigger = {
			OR = {
				exists = global_var:spn_hide_alpha_lost
				exists = global_var:spn_hide_graybeard_lost
				exists = global_var:spn_hide_midnight_lost
				exists = global_var:spn_hide_silverback_lost
				exists = global_var:spn_hide_timber_lost
				exists = global_var:spn_hide_tundra_lost
			}
		}
		custom_tooltip = spn_relic.0200.b_tt
		spn_relic_necrophone_used_effect = yes
	}
	# Where is the Seal of Solomon?
	option = {
		name = spn_relic.0200.c
		trigger = { exists = global_var:spn_seal }
		spn_seal_learn_effect = yes
		custom_tooltip = spn_relic.0200.c_tt
		spn_relic_necrophone_used_effect = yes
	}
	option = {
		name = spn_relic.0200.d
	}
}
