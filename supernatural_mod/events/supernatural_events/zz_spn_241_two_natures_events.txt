# Supernatural 2.41 -- THE WOLF WITH DEMON BLOOD (WORK-ORDER-2.41 §6 Option A, §7.1).
# Loc: localization/english/event_localization/zz_spn_241_flavour_l_english.yml.
#
# A demon-blooded wolf (demon0 + supernatural_werewolf) carries a hidden lean, var:spn_two_natures_lean: up is the wolf,
# down is the demon. Three growing-up beats move it (ages 8, 12, 16, birthday on_action below). At maturity Witchcraft's
# magic_ev3.75 hands him to .0010, The Two Natures, instead of making him a demon.
#
# Who sees a beat: the child, if he is a player; otherwise a living player parent, who decides how to raise him.
# AI-to-AI: the lean moves on its own, by the child's and his guardians' traits, at .0010.
#
#   .0001  The Bitten Hand    age 8
#   .0002  Two Shadows        age 12
#   .0003  The First Change   age 16
#   .0010  The Two Natures    maturity (from magic_ev3.75)
#   .0011  After              a year on, whichever won


##################################################
# The Bitten Hand (age 8)
##################################################
spn_two_natures.0001 = {
	type = character_event
	title = spn_two_natures.0001.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { root = scope:spn_tn_child }
				desc = spn_two_natures.0001.desc_child
			}
			desc = spn_two_natures.0001.desc_parent
		}
	}
	theme = family
	override_background = { reference = courtyard }
	left_portrait = {
		character = scope:spn_tn_child
		animation = anger
	}
	right_portrait = {
		character = scope:spn_tn_viewer
		animation = worry
	}
	trigger = {
		exists = scope:spn_tn_child
		scope:spn_tn_child = { is_alive = yes }
	}

	# Child: I said I was sorry. / Parent: I teach mercy.
	option = {
		name = {
			trigger = { root = scope:spn_tn_child }
			text = spn_two_natures.0001.a_child
		}
		name = {
			trigger = { NOT = { root = scope:spn_tn_child } }
			text = spn_two_natures.0001.a_parent
		}
		spn_two_natures_lean_effect = { ADD = 1 }
		custom_tooltip = spn_two_natures.lean_wolf_tt
	}
	# Child: He deserved it. / Parent: I teach strength.
	option = {
		name = {
			trigger = { root = scope:spn_tn_child }
			text = spn_two_natures.0001.b_child
		}
		name = {
			trigger = { NOT = { root = scope:spn_tn_child } }
			text = spn_two_natures.0001.b_parent
		}
		spn_two_natures_lean_effect = { ADD = -1 }
		custom_tooltip = spn_two_natures.lean_demon_tt
	}
}

##################################################
# The Two Natures (maturity)
# Root = the demon-blooded wolf. Fired by magic_ev3.75 (events/ev3.txt).
##################################################
spn_two_natures.0010 = {
	type = character_event
	title = spn_two_natures.0010.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { spn_two_natures_score_value > 0 }
				desc = spn_two_natures.0010.desc_wolf
			}
			triggered_desc = {
				trigger = { spn_two_natures_score_value < 0 }
				desc = spn_two_natures.0010.desc_demon
			}
			desc = spn_two_natures.0010.desc_even
		}
	}
	theme = dread
	override_background = { reference = wilderness_forest }
	left_portrait = {
		character = root
		animation = pain
	}

	trigger = {
		is_alive = yes
		has_trait = demon0
		has_trait = supernatural_werewolf
		NOT = { has_character_flag = spn_two_natures_done }
	}

	immediate = {
		add_character_flag = spn_two_natures_done
	}

	# The wolf wins.
	option = {
		name = spn_two_natures.0010.a
		custom_tooltip = spn_two_natures.0010.a.tt
		add_character_flag = spn_two_natures_wolf
		if = {
			limit = { spn_two_natures_score_value < 0 }
			add_stress = medium_stress_impact_gain
		}
		trigger_event = { id = spn_two_natures.0011 years = 1 }
		ai_chance = {
			base = 50
			modifier = { add = 40 spn_two_natures_score_value > 0 }
			modifier = { add = -40 spn_two_natures_score_value < 0 }
		}
	}

	# The demon wins.
	option = {
		name = spn_two_natures.0010.b
		# 2.41 audit -- only for one who may become a demon. Under "Mortals Becoming Demons: Players Only" an AI lost
		# the wolf below and then got nothing.
		trigger = { spn_may_become_demon_trigger = yes }
		custom_tooltip = spn_two_natures.0010.b.tt
		add_character_flag = spn_two_natures_demon
		if = {
			limit = { spn_two_natures_score_value > 0 }
			add_stress = medium_stress_impact_gain
		}
		# The wolf goes first: demon1 opposes supernatural_werewolf, and an opposite refuses add_trait. This is
		# spn_initiate.1002's cure, inline, so it runs before make_demon_effect rather than a tick after it.
		# 2.41 audit -- out of the pack first, as the cure decision does (10_spn_decisions.txt); an alpha's pack then
		# passes on. The effect tests for a pack itself.
		spn_pack_remove_member_effect = { WHO = root REASON = flag:left }
		remove_trait = supernatural_werewolf
		if = {
			limit = { has_trait = spn_known_werewolf }
			remove_trait = spn_known_werewolf
		}
		spn_strip_werewolf_perks_effect = yes
		every_secret = {
			limit = { secret_type = spn_secret_werewolf }
			remove_secret = yes
		}
		remove_variable = spn_age_aquired_immortality
		remove_variable = spn_yearly_pulses_since_trait_upgrade
		make_demon_effect = yes
		trigger_event = { id = spn_two_natures.0011 years = 1 }
		ai_chance = {
			base = 50
			modifier = { add = 40 spn_two_natures_score_value < 0 }
			modifier = { add = -40 spn_two_natures_score_value > 0 }
		}
	}
}
