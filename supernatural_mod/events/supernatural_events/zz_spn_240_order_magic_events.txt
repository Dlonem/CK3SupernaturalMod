# .2100 -- The Other Road. A player Mage tutor may teach his witch acolyte through Witchcraft's own Teach Sorcery.
spn_order.2100 = {
	type = character_event
	title = spn_order.2100.t
	desc = spn_order.2100.desc
	theme = learning
	override_background = { reference = magic_academy_lib }
	left_portrait = { character = root animation = thinking }
	right_portrait = {
		character = scope:spn_tp_tutee
		animation = interested
	}
	trigger = {
		is_alive = yes
		this = scope:spn_tp_tutor
		scope:spn_tp_tutee ?= {
			var:spn_order_tutor ?= root
			spn_order_can_teach_sorcery_trigger = { TUTOR = root }
		}
	}
	immediate = {
		spn_flavor_mark_effect = yes
	}
	option = {
		name = spn_order.2100.a
		custom_tooltip = spn_order.2100.a.tt
		open_interaction_window = {
			interaction = teach_sorcery
			actor = root
			recipient = scope:spn_tp_tutee
		}
	}
	option = {
		name = spn_order.2100.b
		custom_tooltip = spn_order.2100.b.tt
	}
}

# .2103 -- Where Were You Last Night? The acolyte's family asks. A secret kept is the Order's Silence.
spn_order.2103 = {
	type = character_event
	title = spn_order.2103.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { this = scope:spn_tp_tutee }
				desc = spn_order.2103.desc_tutee
			}
			desc = spn_order.2103.desc_tutor
		}
	}
	theme = secret
	override_background = { reference = bedchamber }
	left_portrait = { character = root animation = worry }
	right_portrait = {
		character = scope:spn_tp_kin
		animation = paranoia
	}
	trigger = {
		is_alive = yes
		scope:spn_tp_tutor ?= { is_alive = yes }
		scope:spn_tp_tutee ?= { is_alive = yes }
		scope:spn_tp_tutee ?= { var:spn_order_tutor ?= scope:spn_tp_tutor }
		scope:spn_tp_tutee ?= {
			OR = {
				primary_spouse ?= {
					is_alive = yes
					NOT = { exists = var:spn_order_chapter }
				}
				any_close_family_member = {
					is_alive = yes
					is_adult = yes
					NOT = { exists = var:spn_order_chapter }
				}
			}
		}
		spn_flavor_ready_trigger = yes
	}
	immediate = {
		spn_flavor_mark_effect = yes
		scope:spn_tp_tutee = { add_character_flag = { flag = spn_tp_seen_night years = 10 } }
		scope:spn_tp_tutee = {
			if = {
				limit = {
					primary_spouse ?= {
						is_alive = yes
						NOT = { exists = var:spn_order_chapter }
					}
				}
				primary_spouse = { save_scope_as = spn_tp_kin }
			}
			else = {
				random_close_family_member = {
					limit = {
						is_alive = yes
						is_adult = yes
						NOT = { exists = var:spn_order_chapter }
					}

			target = scope:spn_tp_kin

				desc = spn_order.2103.a.success
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
				}
				send_interface_toast = {
					type = event_toast_text_good
					title = spn_order.2103.a.success
					desc = spn_order.2103.a.success_desc
					left_icon = scope:spn_tp_kin
					spn_order_add_standing_effect = { VALUE = 3 }
					spn_order_deed_effect = { PREROGATIVE = flag:silence AMOUNT = 3 TOAST = no }
				}
			}
			50 = {
				desc = spn_order.2103.a.failure
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
				}
				send_interface_toast = {
					type = event_toast_text_neutral
					title = spn_order.2103.a.failure
					desc = spn_order.2103.a.failure_desc
					left_icon = scope:spn_tp_kin
					scope:spn_tp_kin = { add_opinion = { target = root modifier = suspicion_opinion opinion = -15 } }

			}
		}
	}
	option = {

		scope:spn_tp_kin = { add_opinion = { target = root modifier = disappointed_opinion opinion = -10 } }

# .2110 -- (hidden) The Rose Withered. 400 days after a rose was taken, nothing has come of it: no note on its way, no
# temple, no refusal. He is told plainly and may leave another rose at once.
spn_order.2110 = {
	type = character_event
	hidden = yes
	trigger = {
		is_alive = yes
		has_character_flag = spn_order_sought
		NOT = { exists = var:spn_order_chapter }
		NOT = { exists = var:spn_order_picker }
		NOT = { has_character_flag = spn_order_refused }
		NOT = { has_character_flag = spn_order_spurned }
	}
	immediate = {
		remove_character_flag = spn_order_sought
		remove_decision_cooldown = spn_order_seek_decision
		send_interface_toast = {
			type = event_toast_text_neutral
			title = spn_order_toast_sought_withered
			desc = spn_order_toast_sought_withered_desc
			left_icon = root
		}
	}
}

# .2121 -- (hidden) The Order Is Given. An AI agent arrives and answers it, by the ladder: a Powder, his temple, his own
# power. He refuses only if he holds his commander in contempt or has already broken the Order's rules.
spn_order.2121 = {
	type = character_event
	hidden = yes
	trigger = {
		is_alive = yes
		is_imprisoned = no
		spn_is_order_member_trigger = yes
		exists = scope:spn_rs_county
		scope:spn_rs_commander ?= {
			is_alive = yes
			spn_order_respondeo_exposure_open_trigger = { COUNTY = scope:spn_rs_county }
		}
	}
	immediate = {
		if = {
			limit = {
				OR = {
					has_character_flag = spn_order_broke_rules
					opinion = { target = scope:spn_rs_commander value <= -25 }
				}
			}
			spn_order_respondeo_refused_effect = yes
		}
		else_if = {
			limit = { spn_relic_can_spend_powder_trigger = yes }
			spn_order_respondeo_use_powder_effect = yes
			spn_order_respondeo_effect = { COUNTY = scope:spn_rs_county }
			spn_order_respondeo_answered_effect = yes
		}
		else_if = {
			limit = { spn_order_respondeo_chapter_can_pay_trigger = yes }
			spn_order_respondeo_chapter_pays_effect = yes
			spn_order_respondeo_effect = { COUNTY = scope:spn_rs_county }
			spn_order_respondeo_answered_effect = yes
		}
		else_if = {
			limit = { can_spend_standard_ps_trigger = yes }
			spend_standard_ps_effect = yes
			spn_order_respondeo_effect = { COUNTY = scope:spn_rs_county }
			spn_order_respondeo_answered_effect = yes
		}
		else = {
			spn_order_respondeo_unpaid_effect = yes
		}
	}
}
