# Supernatural 2.35 -- "The Hunt". WORK-ORDER-2.35-THE-HUNT §4, §5, §6.9.
#
# The chain: a creature leaves a trail on a county (zz_spn_menace_effects.txt) -> the county's lord notices
# (Something Stalks ...) -> he investigates (The Ledger of the Dead / The Spymaster's Report) or posts a bounty ->
# a hunter runs a Supernatural Hunt there (supernatural_hunt.txt) -> the creature gets a real, two-sided chance
# to answer (Whispers of a Hunter, Someone Is Asking Questions, The Hunter at Your Door) -> the county recovers
# (What the Hunt Was Worth). Nothing here spawns a monster; every target is a character who did something.
#
# TWO-SIDED, always (§5.1, vanilla's murder_outcome_reworked.0001 -> after -> .0002): decide once, save scopes,
# fire the other side's event from `after` or from the option. Order of choices: hunter (planning + Cornered) ->
# creature (Someone Is Asking Questions) -> if the creature wins, hunter (Overmatched: retreat or press) -> the kill
# last, after banking. Nobody dies without having chosen the option that risked it.
#
# Scopes: see zz_spn_hunt_effects.txt. Events refer to each other by title in comments.

# ==========================================================================================================
# §4 THE LORD FINDS OUT. Root = the county holder, scope:spn_menace_county = the county (set by the third mark).

# Nobody Here Remembers -- 2.35. The county has taken three compulsions inside five years and the holes
# have started to line up (spn_compel_leaves_a_trace_effect / spn_compel_left_a_hole_effect). There is no
# body and nothing to dig up: what the lord has is a town that is unusually certain nothing happened.
#
# Thomas, 8 Sep 2026: "if an entire town doesn't know anything about what a hunter clearly saw was true,
# then there would definitely be some funny business there." This is the funny business, and the loop it
# closes is the point of it: a creature that covers its tracks BADLY eventually creates the exact trail it
# was hiding, because Have It Looked Into writes a menace mark with the culprit's own name on it.
#
# Root = the county's holder. scope:spn_compel_hole_county = the county.
spn_hunt.0130 = {
	type = character_event
	title = spn_hunt.0130.t
	desc = spn_hunt.0130.desc
	theme = dread
	override_background = { reference = market }
	left_portrait = { character = root animation = worry }

	trigger = {
		is_alive = yes
		is_imprisoned = no
		exists = scope:spn_compel_hole_county
		scope:spn_compel_hole_county = {
			has_county_modifier = spn_compelled_modifier
			exists = var:spn_compel_culprit
			var:spn_compel_culprit = { is_alive = yes }
		}
	}

	immediate = {
		save_scope_as = spn_hunt_lord
	}

	# Have it looked into. Your learning against how well it was done.
	option = {
		name = spn_hunt.0130.a
		custom_tooltip = spn_hunt.0130.a.tt

			skill = learning
			value = spn_compel_hide_value
			localization = spn_compel_investigate_duel

				desc = spn_hunt.0130.a.success_tt
				compare_modifier = { value = scope:duel_value multiplier = 2 min = -40 }
				min = 5
				spn_compel_investigation_lands_effect = yes

			50 = {
				desc = spn_hunt.0130.a.failure_tt
				compare_modifier = { value = scope:duel_value multiplier = -2 min = -40 }
				min = 5
				add_stress = minor_stress_impact_gain
			}
		}
		ai_chance = {
			base = 40
			modifier = { add = 30 has_trait = paranoid }
			modifier = { add = 20 has_trait = diligent }
			modifier = { add = 20 learning >= 12 }
			modifier = { add = -25 has_trait = lazy }
		}
	}
	# Put the Trailsman on it. He reads ground for a living, and a county that has forgotten a week is
	# ground. Rewards the court position rather than duplicating it.
	option = {
		name = spn_hunt.0130.b
		trigger = {
			OR = {
				any_court_position_holder = { type = spn_trailsman_court_position }
				any_court_position_holder = { type = spn_trailsman_camp_officer }
			}
		}
		custom_tooltip = spn_hunt.0130.b.tt
		duel = {
			skill = learning
			value = spn_compel_hide_value
			localization = spn_compel_investigate_duel
			65 = {
				desc = spn_hunt.0130.a.success_tt
				compare_modifier = { value = scope:duel_value multiplier = 2 min = -40 }
				modifier = { add = 20 always = yes }
				min = 10
				spn_compel_investigation_lands_effect = yes
			}
			35 = {
				desc = spn_hunt.0130.a.failure_tt
				compare_modifier = { value = scope:duel_value multiplier = -2 min = -40 }
				min = 5
			}
		}
		ai_chance = {
			base = 70
			modifier = { add = 20 has_trait = paranoid }
		}
	}
	# Bad air, bad harvest, bad year.
	option = {
		name = spn_hunt.0130.c
		custom_tooltip = spn_hunt.0130.c.tt
		stress_impact = {
			paranoid = minor_stress_impact_gain

			modifier = { add = 30 has_trait = lazy }
			modifier = { add = 20 has_trait = content }

# ==========================================================================================================
# §6.9 THE HUNT. Root = the host, activity events. scope:activity is implicit; the shared scopes are bound.
# ==========================================================================================================

# The Trail -- the opener. The kind shows only to a hunter who can read it.
# 2.35 PART TWO (§8.5b) -- The Trail gained a way to turn round. The tracking-phase exit is deliberately the
# CHEAPEST of the four: no XP, a little prestige, no county refusal counted against him. A young hunter who
# rides out, looks at what is actually happening in this county and goes home has done nothing wrong, and the
# system should not punish him for the one decision that keeps him alive to make better ones.
spn_hunt.0100 = {
	type = activity_event
	title = spn_hunt.0100.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					spn_hunt_reads_the_kind_trigger = yes
					scope:activity = { var:spn_hunt_kind ?= flag:vampire }
				}
				desc = spn_hunt.0100.desc_vampire
			}
			triggered_desc = {
				trigger = {
					spn_hunt_reads_the_kind_trigger = yes
					scope:activity = { var:spn_hunt_kind ?= flag:werewolf }
				}
				desc = spn_hunt.0100.desc_werewolf
			}
			# 2.35 PART TWO -- the two new kinds, each behind its own read. A demon costs 65 (§4.2) because
			# nothing about a demon's kill looks like an animal's; a witch is read off the OTHER track
			# entirely, which is the whole point of the trait having two.
			triggered_desc = {
				trigger = {
					spn_hunt_reads_demon_trigger = yes
					scope:activity = { var:spn_hunt_kind ?= flag:demon }
				}
				desc = spn_hunt.0100.desc_demon
			}
			triggered_desc = {
				trigger = {
					spn_hunt_reads_witch_trigger = yes
					scope:activity = { var:spn_hunt_kind ?= flag:witch }
				}
				desc = spn_hunt.0100.desc_witch
			}
			desc = spn_hunt.0100.desc
		}
	}
	theme = hunting
	override_background = { reference = bp1_crossroads_inn } # 2.35a art pass -- was tavern. The Trail: you have just ridden into the county. Not the same tavern as every other scene.
	left_portrait = { character = root animation = marshal }

	immediate = {
		spn_hunt_bind_scopes_effect = yes
	}

	option = {
		name = spn_hunt.0100.a
		custom_tooltip = spn_hunt.0100.a.tt
	}

	# 2.35 PART TWO (§8.5b) -- "This is not my fight." The first and cheapest of the four off-ramps.
	#
	# It does not end the ACTIVITY -- script cannot walk a host out of his own activity -- it ends the HUNT:
	# the tracking roll is skipped, the confrontation never happens, and the conclusion pays a withdrawal.
	# No XP, a little prestige for having ridden out at all, and no refusal counted against the county,
	# because a man who looked and left before he ever knocked on a door has not been turned back by anything.
	option = {
		name = spn_hunt.0100.b
		custom_tooltip = spn_hunt.0100.b.tt
		scope:activity = { set_variable = { name = spn_hunt_walked_away value = yes } }
		spn_hunt_set_outcome_effect = { OUTCOME = withdrew }
		add_prestige = 10
		ai_chance = {
			base = 5
			modifier = { add = 40 has_trait = craven }
			modifier = { add = 25 has_trait = content }
			modifier = { add = 20 has_trait = lazy }
			# The projected odds, run for real. A hunter who can do the arithmetic does it here.
			modifier = { add = 45 spn_hunt_projected_odds_value < 30 }
			modifier = { add = 25 spn_hunt_projected_odds_value < 45 }
			modifier = { add = -40 has_trait = brave }
			modifier = { add = -35 has_trait = zealous }
			modifier = { add = -30 has_trait = vengeful }
			modifier = { add = -30 spn_hunter_has_nothing_else_trigger = yes }
		}
	}
}

	# that door learns exactly what you are, and unlike the form, that does not wear off.
	#
	# The form is timed rather than permanent because this is a door, not a decision -- the shapeshift
	# interaction (10_spn_character_interactions_shapeshift.txt) is where a wolf CHOOSES to live as one, and
	# nothing here should quietly take that choice over. Both halves of it, modifier and POD_bald_look_flag,
	# expire together so the portrait never gets stuck without its hair.
	option = {
		name = spn_hunt.0200.f
		trigger = {
			OR = {
				has_trait = supernatural_werewolf
				has_trait = supernatural_hybrid
			}
			is_shapeshifted_trigger = no
		}
		custom_tooltip = spn_hunt.0200.f.tt
		add_character_modifier = { modifier = POD_werewolf_form_lupus_modifier days = 30 }
		add_character_flag = { flag = POD_bald_look_flag days = 30 }
		# Seen. The hunter, and everyone who rode with him. A creature already known as what it is has no
		# secret left to lose here, and random_secret simply matches nothing.
		hidden_effect = {
			save_temporary_scope_as = spn_hunt_the_wolf
			scope:spn_hunter = { add_to_list = spn_hunt_saw_the_wolf }
			scope:spn_hunt_activity ?= {
				every_attending_character = {
					limit = { has_trait = supernatural_hunter  NOT = { this = scope:spn_hunter } }
					add_to_list = spn_hunt_saw_the_wolf
				}
			}
			every_in_list = {
				list = spn_hunt_saw_the_wolf
				save_temporary_scope_as = spn_hunt_witness
				scope:spn_hunt_the_wolf = {
					if = {
						limit = { has_trait = supernatural_hybrid }
						spn_give_knowledge_of_hybrid_secret = { CHARACTER = scope:spn_hunt_witness }
					}
					else = {
						spn_give_knowledge_of_lycan_secret = { CHARACTER = scope:spn_hunt_witness }
					}
				}
			}
		}
		spn_hunt_fight_effect = { DEFENDER = root }
		ai_chance = {
			base = 45
			modifier = { add = 30 has_trait = wrathful }
			modifier = { add = 25 has_trait = brave }
			modifier = { add = 20 has_trait = arrogant }
			modifier = { add = 20 prowess >= 15 }
			# Nothing left to hide.
			modifier = { add = 15 spn_hunt_is_known_creature_trigger = yes }
			# He came with the one metal it is written for. Turning into the thing silver was made to kill is
			# how wolves end up on a wall.
			modifier = {
				add = -40
				scope:spn_hunt_activity = {
					has_activity_option = { category = spn_hunt_preparations option = spn_hunt_prep_silver }
				}
			}
			modifier = { add = -35 has_trait = craven }
			modifier = { add = -25 has_trait = calm }
			modifier = { add = -25 has_trait = shy }
			modifier = { add = -20 has_trait = patient }
		}
	}
	# Send [proxy] in your place (D6).
	option = {
		name = spn_hunt.0200.d
		trigger = { exists = scope:spn_proxy }
		custom_tooltip = spn_hunt.0200.d.tt
		# 2.35h -- AND THE HUNTER IS TOLD, which he was not. Thomas, watching this happen to him: "I
		# cornered one guy, and then I said I'm gonna strike now, and then when I click it, it changes the
		# scene, and it's somebody else... there might be a missing part of that little conversation."
		# There was. The substitution happened correctly and silently: the fight resolved against a face
		# the hunter had never been shown, in the same click. spn_hunt.0221 is that missing beat, and it
		# gives him the decision the moment deserves -- kill the one who was sent, or refuse the trade.
		if = {
			limit = { scope:spn_proxy = { is_ai = yes } }
			scope:spn_proxy = { save_scope_as = spn_defender }
			scope:spn_hunter = {
				spn_hunt_compute_fight_bonus_effect = yes
				trigger_event = spn_hunt.0221
			}
		}
		else = {
			scope:spn_proxy = { trigger_event = spn_hunt.0220 }
		}
		ai_chance = {
			base = 20
			modifier = { add = 30 has_trait = callous }
			modifier = { add = 20 has_trait = craven }
			modifier = { add = 15 has_trait = arbitrary }
			modifier = { add = -30 has_trait = compassionate }
			modifier = { add = -20 has_trait = brave }
		}
	}
	# Make them an offer: fifty gold now and silence bought (The Price of Silence). Elders only.
	option = {
		name = spn_hunt.0200.e
		trigger = {
			spn_hunt_creature_track_value >= 75
			gold >= 100
		}
		custom_tooltip = spn_hunt.0200.e.tt

			triggered_desc = { trigger = { scope:spn_hunt_activity = { var:spn_hunt_outcome = flag:hunter_beaten } } desc = spn_hunt.0300.desc_hunter_beaten }

	}

		name = spn_hunt.0300.a

				exists = scope:spn_culprit

				scope:spn_hunt_activity = {
					OR = {
						var:spn_hunt_outcome = flag:fled
						var:spn_hunt_outcome = flag:bought
						var:spn_hunt_outcome = flag:hunter_beaten

		}
	}

# ==========================================================================================================
# 2.35a -- IT IS DONE. The hunter's side of a won fight, which did not exist until the first boot found it.
#
# Every other way a confrontation could end already had a window on the hunter's side -- Overmatched when he
# lost, Gone when it fled, Turned Back when he was compelled. Winning had nothing, because the kill happened
# inside spn_hunt_hunter_won_effect, which runs in the DEFENDER's scope from an event on the AI culprit. The
# player rode out, picked how to go in, and then read a summary three weeks later.
#
# The death is performed HERE, in `after`, so the moment is on screen before it happens rather than after --
# and the standing ordering rule is better served, not worse: nothing in the mod reads the creature later
# than this line does.
# ==========================================================================================================
spn_hunt.0205 = {
	type = activity_event
	title = spn_hunt.0205.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:spn_defender = { is_a_demon_trigger = yes } }
				desc = spn_hunt.0205.desc_demon
			}
			triggered_desc = {
				trigger = { scope:spn_defender = { has_trait = supernatural_werewolf } }
				desc = spn_hunt.0205.desc_werewolf
			}
			triggered_desc = {
				trigger = { scope:spn_defender = { has_trait = supernatural_hybrid } }
				desc = spn_hunt.0205.desc_hybrid
			}
			triggered_desc = {
				trigger = { scope:spn_defender = { has_trait = supernatural_vampire } }
				desc = spn_hunt.0205.desc_vampire
			}
			desc = spn_hunt.0205.desc
		}
	}
	theme = hunting
	override_background = { reference = alley_night }
	left_portrait = { character = root animation = marshal }
	right_portrait = { character = scope:spn_defender animation = fear }

	trigger = {
		exists = scope:spn_defender
		scope:spn_defender = { is_alive = yes }
	}

	immediate = {
		spn_hunt_bind_scopes_effect = yes
	}

	# Finish it.
	option = {
		name = spn_hunt.0205.a
		custom_tooltip = spn_hunt.0205.a.tt
	}

	# 2.35a -- and the one it was not, if there is anything left of one. A creature that has already lost
	# does not have to die for the county's books to balance: a hunter can take it in instead, which is the
	# only mercy in this system that costs him something real.
	option = {
		name = spn_hunt.0205.b
		trigger = {
			is_landed_or_landless_administrative = yes
			scope:spn_defender = { is_imprisoned = no }
			NOT = { scope:spn_hunt_activity ?= { has_variable = spn_hunt_coven_running } }
		}
		custom_tooltip = spn_hunt.0205.b.tt
		imprison_character_effect = {
			TARGET = scope:spn_defender
			IMPRISONER = root
		}
		scope:spn_hunt_activity ?= { remove_variable = spn_hunt_kill_pending }
		add_trait_xp = { trait = supernatural_hunter track = monster_hunting value = { 5 12 } }
		ai_chance = {
			base = 10
			modifier = { add = 30 has_trait = just }
			modifier = { add = 25 has_trait = compassionate }
			modifier = { add = 20 has_trait = calm }
			modifier = { add = -40 has_trait = zealous }
			modifier = { add = -30 has_trait = wrathful }
			modifier = { add = -25 has_trait = sadistic }
		}
	}

	after = {
		# LAST, and it is the last thing in the whole chain.
		if = {
			limit = {
				exists = scope:spn_defender
				scope:spn_defender = { is_alive = yes  is_imprisoned = no }
			}
			scope:spn_hunt_activity ?= { remove_variable = spn_hunt_kill_pending }
			scope:spn_defender = { spn_hunt_perform_the_kill_effect = yes }
		}
	}
}
