#Market Day
spn_embrace.0001 = {
	type = character_event
	title = spn_embrace.0001.t
	desc = spn_embrace.0001.desc
	theme = intrigue

	trigger = {
		AND = {
			spn_has_supernatural_trait = no
			spn_has_hunter_trait = no
			#is_witch_trigger = no
		}
	}

	override_background = {
		reference = market
	}

	right_portrait = {
		character = root
		animation = idle
	}

	left_portrait = {
		character = scope:spn_new_created_witch
		animation = idle
	}

	immediate = {
		spn_create_peddler_effect = { WHO = root }
		scope:spn_created_witch = { save_scope_as = spn_new_created_witch }
	}

	option = {
		name = spn_embrace.0001.a
		custom_tooltip = spn_embrace.0001.a.tt
		if = {
			limit = { has_character_flag = spn_embrace_lycan_dec }
			random_list = {
				75 = {
					custom_tooltip = spn_embrace.0001.a.live.tt
					modifier = {
						add = {
							value = health
							subtract = {
								value = 4
							}
							multiply = {
								value = 2
							}
						}
					}
					trigger_event = spn_embrace.0002
				}
				25 = {
					custom_tooltip = spn_embrace.0001.a.die.tt
					modifier = {
						add = {
							value = health
							subtract = {
								value = 4
							}
							multiply = {
								value = -2
							}
						}
					}
					death = {
						death_reason = death_poison
					}
				} 
			}
		}
		else_if = {
			limit = { has_character_flag = spn_embrace_vampire_dec }
			random_list = {
				75 = {
					custom_tooltip = spn_embrace.0001.a.live.tt
					modifier = {
						add = {
							value = health
							subtract = {
								value = 4
							}
							multiply = {
								value = 2
							}
						}
					}
					trigger_event = spn_embrace.0003
				}
				25 = {
					custom_tooltip = spn_embrace.0001.a.die.tt
					modifier = {
						add = {
							value = health
							subtract = {
								value = 4
							}
							multiply = {
								value = -2
							}
						}
					}
					death = {
						death_reason = death_poison
					}
				} 
			}
		}
		else = {
			random_list = {
				#Three visible outcomes, none of which names a creature. WHICH one you become is
				#rolled in the hidden_effect below, so the tooltip no longer reads "50% Pale Moon,
				#50% Crimson Night". Death is down from an effective 20% to 15%; that and more has
				#gone into simply being swindled, so the 25 gold is a real gamble instead of a
				#guaranteed transformation with a death tax.
				55 = {
					custom_tooltip = spn_embrace.0001.a.live.tt
					modifier = {
						add = {
							value = health
							subtract = {
								value = 4
							}
							multiply = {
								value = 2
							}
						}
					}
					hidden_effect = {
						random_list = {
							50 = { trigger_event = spn_embrace.0002 } #Pale Moon
							50 = { trigger_event = spn_embrace.0003 } #Crimson Night
						}
					}
				}
				30 = {
					custom_tooltip = spn_embrace.0001.a.nothing.tt
					trigger_event = spn_embrace.0005 #Nothing At All
				}
				15 = {
					custom_tooltip = spn_embrace.0001.a.die.tt
					modifier = {
						add = {
							value = health
							subtract = {
								value = 4
							}
							multiply = {
								value = -2
							}
						}
					}
					death = {
						death_reason = death_poison
					}
				}
			}
		}
		if = {
			limit = {
				is_alive = yes
			}
			remove_short_term_gold = 25
			add_character_flag = {
				flag = spn_dissappear_spn_new_created_witch
			}
		}
	}

	option = {
		name = spn_embrace.0001.b
		custom_tooltip = spn_embrace.0001.b.tt
		add_character_flag = {
			flag = spn_dissappear_spn_new_created_witch
		}
	}

	after = { 
		trigger_event = spn_embrace.0004 #Vanish Peddler
	}
}

spn_embrace.1001 = {
	type = character_event

		AND = {
			spn_has_supernatural_trait = no
			spn_has_hunter_trait = no
			#is_witch_trigger = no
		}
	}

	right_portrait = {
		character = root

	}

	immediate = {
		spn_create_local_vampire_effect = { FROMCHAR = root }
		scope:spn_created_vampire = { 
			add_character_flag = {
				flag = supernatural_embrace_vampire
			}
			trigger_event = spn_initiate.0001 #Initialize Vampire
			set_variable = {
				name = spn_age_aquired_immortality
				value = 20
			}
			hidden_effect = {
				add_trait_xp = {
					trait = supernatural_vampire
					track = vampirism
					value = { 15 65 }
				}
			}
			save_scope_as = spn_new_created_vampire 
		}
	}

	option = {
		name = spn_embrace.1001.a
		custom_tooltip = spn_embrace.1001.a.tt
		trigger_event = spn_embrace.1002
	}

	option = {
		name = spn_embrace.1001.b
		custom_tooltip = spn_embrace.1001.b.tt
	}

	#after = { }
}

		}
		triggered_animation = {

		}
		triggered_animation = {
			trigger = {
				NOR = {
					has_trait = craven

				}

			animation = shock
		}

		name = spn_embrace.1002.a

			craven = major_stress_impact_gain
			brave = minor_stress_impact_loss
		}

			75 = { #Survive the battle
			    custom_tooltip = spn_embrace.1002.surv.tt
				trigger_event = spn_embrace.1003 #Survived Battle

					add = {
						value = prowess
					}
				}

				modifier = { add = 15 has_trait = giant }
				modifier = { add = 5 has_trait = strong }
				modifier = { add = 10 has_trait = athletic }
				modifier = { add = 10 has_trait = wrathful }

				modifier = { add = -15 has_trait = craven }
			}

				modifier = { add = -15 has_trait = giant }

				modifier = { add = -10 has_trait = athletic }
				modifier = { add = -10 has_trait = wrathful }

				modifier = { add = 15 has_trait = craven }
			}
		}
	}

		name = spn_embrace.1002.b
		custom_tooltip = spn_embrace.1002.b.tt
		stress_impact = {
			wrathful = major_stress_impact_gain
			calm = minor_stress_impact_loss
		}

			75 = { #Convince
				custom_tooltip = spn_embrace.1002.surv.tt
				trigger_event = spn_embrace.1004 #Convinced
				modifier = {
					add = {
						value = diplomacy
					}
				}
				modifier = { add = 15 has_trait = calm }
				modifier = { add = 5 has_trait = patient }
				modifier = { add = 10 has_trait = shrewd }
				modifier = { add = 15 has_trait = diplomat }

				modifier = { add = -15 has_trait = wrathful }
				modifier = { add = -10 has_trait = impatient }
			}
			25 = { #Die
				death = { death_reason = spn_death_throat_ripped_out }
				modifier = { add = -15 has_trait = calm }
				modifier = { add = -5 has_trait = patient }
				modifier = { add = -10 has_trait = shrewd }
				modifier = { add = -15 has_trait = diplomat }

				modifier = { add = 15 has_trait = wrathful }
				modifier = { add = 10 has_trait = impatient }
			}
		}
	}

	option = { #Run away
		name = spn_embrace.1002.c
		custom_tooltip = spn_embrace.1002.c.tt

			brave = major_stress_impact_gain

		}

			75 = { #Run away!
			    custom_tooltip = spn_embrace.1002.surv.tt
				trigger_event = spn_embrace.1005 #Got away
				modifier = { add = 15 has_trait = craven }

				modifier = { add = -15 has_trait = brave }

			}
			25 = { #Die
				death = { death_reason = spn_death_throat_ripped_out }
				modifier = { add = -15 has_trait = craven }
				modifier = { add = -15 has_trait = athletic }

				modifier = { add = 15 has_trait = brave }
			}
		}
	}

	option = { #Shocked
		name = spn_embrace.1002.d
		custom_tooltip = spn_embrace.1002.d.tt

		random_list = {
			75 = { #Barely survive
				custom_tooltip = spn_embrace.1002.surv.tt

#He comes back. Fired from the two non-turn outcomes of the incubation climax.
spn_embrace.3009 = {
	type = character_event
	title = spn_embrace.3009.t
	desc = spn_embrace.3009.desc
	theme = intrigue

	trigger = {
		exists = var:spn_deferred_hunter_trainer
		spn_has_supernatural_trait = no
		spn_has_hunter_trait = no
		NOT = { has_trait = ww_infected_wound }
	}

	immediate = {
		var:spn_deferred_hunter_trainer = { save_scope_as = spn_new_created_hunter }
	}

	left_portrait = {
		character = scope:spn_new_created_hunter
		animation = idle
	}

	right_portrait = {
		character = root
		animation = happiness
	}

	option = { #Finish what we started
		name = spn_embrace.3009.a
		custom_tooltip = spn_embrace.3009.a.tt

		remove_character_flag = spn_hunter_training_deferred
		remove_variable = spn_deferred_hunter_trainer

		trigger_event = spn_initiate.3001 #Initiate Hunter
		add_trait_xp = {
			trait = supernatural_hunter
			track = monster_hunting
			value = { 5 20 }
		}
		spn_assign_hunter_trainer = { CHARACTER = scope:spn_new_created_hunter }
		spn_give_knowledge_of_hunter_secret = {
			CHARACTER = scope:spn_new_created_hunter
		}
		scope:spn_new_created_hunter = {
			spn_give_knowledge_of_hunter_secret = {
				CHARACTER = root
			}
		}

		set_relation_mentor = scope:spn_new_created_hunter

		# 2.33 -- AI weighting. 3009 is genuinely AI-reachable: spn_offer_gift / spn_force_gift /
		# spn_ask_for_gift can all land an infected wound on an AI character, and when the fever
		# breaks back to human this offer collects the hunter trainer who was waiting on it.
		# Before this it was an unweighted coin flip. Taking up the spear is a matter of nerve,
		# grudge, and whether you have the time -- so boldness, martial aptitude and the debt owed
		# to the trainer push up; caution, contentment and the burden of a crown push down.
		ai_chance = {
			base = 40

			# --- nerve ---
			modifier = { has_trait = brave      add = 30 }
			modifier = { has_trait = craven     add = -35 }
			modifier = { has_trait = wrathful   add = 12 }
			modifier = { has_trait = calm       add = -8 }

			# --- appetite for the work ---
			modifier = { has_trait = ambitious  add = 18 }
			modifier = { has_trait = content    add = -30 }
			modifier = { has_trait = diligent   add = 15 }
			modifier = { has_trait = lazy       add = -20 }
			modifier = { has_trait = vengeful   add = 18 }
			modifier = { has_trait = forgiving  add = -10 }
			modifier = { has_trait = zealous    add = 22 }
			modifier = { has_trait = cynical    add = -12 }
			modifier = { has_trait = just       add = 10 }
			modifier = { has_trait = paranoid   add = -10 }
			modifier = { has_trait = shy        add = -8 }
			ai_value_modifier = {
				ai_boldness      = high_chance_impact_positive_ai_value
				ai_vengefulness  = low_chance_impact_positive_ai_value
				ai_zeal          = low_chance_impact_positive_ai_value
			}

			# --- can I actually do this? ---
			modifier = { has_trait = education_martial add = 20 }
			modifier = { add = { value = prowess  subtract = 8  multiply = 2  min = -20  max = 25 } }
			modifier = { age > 55  add = -20 }
			modifier = { has_trait = wounded  add = -10 }

			# --- the debt to the one who stood over you ---
			modifier = {
				OR = {
					has_relation_friend = scope:spn_new_created_hunter
					has_relation_best_friend = scope:spn_new_created_hunter
				}
				add = 20
			}
			modifier = { is_close_family_of = scope:spn_new_created_hunter  add = 15 }
			modifier = {
				OR = {
					has_relation_rival = scope:spn_new_created_hunter
					has_relation_nemesis = scope:spn_new_created_hunter
				}
				add = -40
			}
			opinion_modifier = { who = root  opinion_target = scope:spn_new_created_hunter  multiplier = 0.3 }

			# --- a crown leaves no time for the hunt ---
			modifier = { highest_held_title_tier = tier_duchy    add = -15 }
			modifier = { highest_held_title_tier = tier_kingdom  add = -30 }
			modifier = { highest_held_title_tier >= tier_empire  add = -45 }
		}
	}

	option = { #I have seen enough of monsters
		name = spn_embrace.3009.b
		custom_tooltip = spn_embrace.3009.b.tt

		remove_character_flag = spn_hunter_training_deferred
		remove_variable = spn_deferred_hunter_trainer

		stress_impact = {
			brave = minor_stress_impact_gain
			craven = minor_stress_impact_loss
		}

		# The mirror of .a: flat floor so the offer is never a certainty, lifted by the same
		# temperaments that make a person walk away from a spear.
		ai_chance = {
			base = 40

			modifier = { has_trait = craven     add = 30 }
			modifier = { has_trait = content    add = 25 }
			modifier = { has_trait = lazy       add = 20 }
			modifier = { has_trait = calm       add = 10 }
			modifier = { has_trait = shy        add = 8 }
			modifier = { has_trait = forgiving  add = 10 }
			modifier = { has_trait = cynical    add = 12 }
			modifier = { has_trait = brave      add = -25 }
			modifier = { has_trait = ambitious  add = -15 }
			modifier = { has_trait = zealous    add = -20 }
			modifier = { has_trait = vengeful   add = -15 }
			modifier = { has_trait = education_martial add = -15 }
			modifier = { age > 55  add = 20 }
			modifier = { highest_held_title_tier >= tier_kingdom  add = 30 }
			ai_value_modifier = { ai_boldness = -1 }
		}
	}
