##################################################################
# WEREWOLF WOUND INCUBATION - pulse engine + event pool
#
# Identifier = the ww_infected_wound trait. Climax (9000) removes it -> pulse (0001) self-terminates.
# Infection track XP is the spine (0-100). Options use RANDOM xp swings + stress trade-offs.
# Fighting the infection (lowering XP) costs stress; giving in (raising XP) relieves it.
# Names: use var-chain for the sire (scope: refs don't resolve in this build's event loc).
##################################################################


# ============================================================
# 0001 - PULSE DISPATCHER (hidden). No back-to-back repeats; soft caps on the generics.
# ============================================================
spn_wound_incubation.0001 = {
	hidden = yes
	trigger = { has_trait = ww_infected_wound }
	immediate = {
		random_list = {
			10 = { # The wound festers (generic, cap 2)
				trigger = { NOT = { var:spn_last_wound_event = 10 }  NOT = { var:spn_wound_c10 >= 2 } }
				set_variable = { name = spn_last_wound_event value = 10 }
				change_variable = { name = spn_wound_c10 add = 1 }
				trigger_event = spn_wound_incubation.0010
			}
			10 = { # Fevered nights (generic, cap 2)
				trigger = { NOT = { var:spn_last_wound_event = 11 }  NOT = { var:spn_wound_c11 >= 2 } }
				set_variable = { name = spn_last_wound_event value = 11 }
				change_variable = { name = spn_wound_c11 add = 1 }
				trigger_event = spn_wound_incubation.0011
			}
			9 = { # The wound worsens - risk of maiming (generic)
				trigger = { NOT = { var:spn_last_wound_event = 30 } }
				set_variable = { name = spn_last_wound_event value = 30 }
				trigger_event = spn_wound_incubation.0030
			}
			8 = { # A healer attends you (ONCE, needs a physician courtier)
				trigger = {
					NOT = { has_character_flag = spn_had_healer_event }
					any_courtier = { has_trait = lifestyle_physician }
				}
				trigger_event = spn_wound_incubation.0012
			}
			7 = { # A hunter grows suspicious (ONCE, needs a capable monster-hunter courtier)
				trigger = {
					NOT = { has_variable = spn_hunter_confronted }
					any_courtier = {
						has_trait = supernatural_hunter
						has_trait_xp = { trait = supernatural_hunter track = monster_hunting value >= 20 }
					}
				}
				trigger_event = spn_wound_incubation.0040
			}
			6 = { # The wolf stirs (yield vs resist) - one-shot
				trigger = { NOR = { has_character_flag = spn_yielding_to_the_change  has_character_flag = spn_resisting_the_change } }
				set_variable = { name = spn_last_wound_event value = 20 }
				trigger_event = spn_wound_incubation.0020
			}
			6 = { # The one who clawed you (any turning) - one-shot
				trigger = { has_variable = spn_conferred_sire  NOT = { has_variable = spn_sire_choice_made } }
				set_variable = { name = spn_last_wound_event value = 21 }
				trigger_event = spn_wound_incubation.0021
			}
			7 = { # A crisis of faith (one-shot; the event sets its own spn_faith_crisis_done flag)
				trigger = { NOT = { has_character_flag = spn_faith_crisis_done } }
				set_variable = { name = spn_last_wound_event value = 50 }
				trigger_event = spn_wound_incubation.0050
			}
		}
		trigger_event = { id = spn_wound_incubation.0001 days = { 90 120 } }
	}
}

					set_variable = { name = spn_marked_werewolf  value = root.var:spn_conferred_sire }

	option = { # [Brave/Dread] Cow them into silence
