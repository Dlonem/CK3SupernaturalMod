# Supernatural 2.40 -- the hide duel and The Hide Speaks (WORK-ORDER-2.40 F1, C10). Package Q6, events spn_hide.2100-.2199.
# AGOT's dragon duel (sighting .0001, ambush .0002, pursuit .0003, permission .1010) gave the shape; nothing is copied.
# The window is vanilla's duel_event (generic portraits, van/gui/event_windows/duel_event.gui:262-330), both fighters in
# vanilla's duel_wield_weapon stance. The roll is hidden; options say which way the fight leans in words, never in numbers.
# Every event re-checks that both are alive, free and still champions, and every option asks again before it acts.
# The result (spn_hide.0202, "Two Wolves on One Field") stays in zz_spn_hide_events.txt.

namespace = spn_hide

# ==================================================================================================
# .2100 -- Across the Field. To the champion who sees the other first (scope:spn_hduel_a).
# ==================================================================================================
spn_hide.2100 = {
	type = character_event
	window = duel_event
	title = spn_hide.2100.t
	desc = {
		desc = spn_hide.2100.desc
		first_valid = {
			triggered_desc = {
				trigger = { scope:spn_hduel_other = { has_trait = spn_champion_silverback } }
				desc = spn_hide_duel_sees_silverback
			}
			triggered_desc = {
				trigger = { scope:spn_hduel_other = { has_trait = spn_champion_timber } }
				desc = spn_hide_duel_sees_timber
			}
			triggered_desc = {
				trigger = { scope:spn_hduel_other = { has_trait = spn_champion_tundra } }
				desc = spn_hide_duel_sees_tundra
			}
			triggered_desc = {
				trigger = { scope:spn_hduel_other = { has_trait = spn_champion_graybeard } }
				desc = spn_hide_duel_sees_graybeard
			}
			triggered_desc = {
				trigger = { scope:spn_hduel_other = { has_trait = spn_champion_midnight } }
				desc = spn_hide_duel_sees_midnight
			}
			triggered_desc = {
				trigger = { scope:spn_hduel_other = { has_trait = spn_champion_alpha } }
				desc = spn_hide_duel_sees_alpha
			}
		}
		first_valid = {
			triggered_desc = {
				trigger = { scope:spn_hduel_my_chance >= 65 }
				desc = spn_hide_duel_lean_strong
			}
			triggered_desc = {
				trigger = { scope:spn_hduel_my_chance <= 35 }
				desc = spn_hide_duel_lean_weak
			}
			desc = spn_hide_duel_lean_even
		}
		triggered_desc = {
			trigger = { spn_hide_duel_wanted_trigger = { WHO = root FOE = scope:spn_hduel_other } }
			desc = spn_hide_duel_wanted
		}
	}
	theme = battle
	override_background = { reference = battlefield }
	left_portrait = {
		character = root
		# A wolf in his own shape shows the claws; the werewolf skin, when it exists, swaps this one line.
		triggered_animation = {
			trigger = { is_shapeshifted_trigger = yes }
			animation = aggressive_unarmed
		}
		scripted_animation = duel_wield_weapon
	}
	right_portrait = {
		character = scope:spn_hduel_other
		# A wolf in his own shape shows the claws; the werewolf skin, when it exists, swaps this one line.
		triggered_animation = {
			trigger = { is_shapeshifted_trigger = yes }
			animation = aggressive_unarmed
		}
		scripted_animation = duel_wield_weapon
	}
	trigger = {
		exists = scope:spn_hduel_a
		exists = scope:spn_hduel_b
		OR = {
			scope:spn_hduel_a = root
			scope:spn_hduel_b = root
		}
		scope:spn_hduel_a = { spn_hide_duel_live_trigger = { OTHER = scope:spn_hduel_b } }
	}
	immediate = {
		spn_hide_duel_other_effect = yes
		scope:spn_hduel_a = { spn_hide_duel_guard_effect = yes }
		scope:spn_hduel_b = { spn_hide_duel_guard_effect = yes }
	}
	option = {
		name = spn_hide.2100.a
		spn_clash_lean_tooltip_effect = { A = root B = scope:spn_hduel_other FORM = no }
		custom_tooltip = spn_hide_duel_throat_tt
		if = {
			limit = { spn_hide_duel_live_trigger = { OTHER = scope:spn_hduel_other } }
			spn_hide_duel_style_effect = { STYLE = throat }
			hidden_effect = {
				spn_hide_duel_guard_effect = yes
				scope:spn_hduel_other = {
					spn_hide_duel_guard_effect = yes
					trigger_event = spn_hide.2101
				}
			}
		}
		else = {
			spn_hide_duel_lapse_effect = yes
		}
		ai_chance = {
			base = 40
			modifier = {
				add = 30
				scope:spn_hduel_my_chance >= 65
			}
			modifier = {
				add = 20
				has_trait = brave
			}
			modifier = {
				add = 20
				has_trait = wrathful
			}
			modifier = {
				add = 25
				has_trait = spn_champion_midnight
			}
		}
	}
	option = {
		name = spn_hide.2100.b
		spn_clash_lean_tooltip_effect = { A = root B = scope:spn_hduel_other FORM = no }
		custom_tooltip = spn_hide_duel_wear_tt
		if = {
			limit = { spn_hide_duel_live_trigger = { OTHER = scope:spn_hduel_other } }
			spn_hide_duel_style_effect = { STYLE = wear }
			hidden_effect = {
				spn_hide_duel_guard_effect = yes
				scope:spn_hduel_other = {
					spn_hide_duel_guard_effect = yes
					trigger_event = spn_hide.2101
				}
			}
		}
		else = {
			spn_hide_duel_lapse_effect = yes
		}
		ai_chance = {
			base = 40
			modifier = {
				add = 15
				has_trait = spn_champion_silverback
			}
		}
	}
	option = {
		name = spn_hide.2100.c
		custom_tooltip = spn_hide_duel_slip_tt
		if = {
			limit = {
				scope:spn_hduel_other = {
					OR = {
						has_trait = spn_champion_midnight
						has_character_modifier = spn_let_it_take_me_modifier
					}
				}
			}
			custom_tooltip = spn_hide_duel_midnight_tt
		}
		if = {
			limit = { has_trait = spn_champion_timber }
			custom_tooltip = spn_hide_duel_timber_tt
		}
		if = {
			limit = { spn_hide_duel_live_trigger = { OTHER = scope:spn_hduel_other } }
			hidden_effect = {
				random_list = {
					# It saw you go: it may run you down.
					50 = {
						modifier = {
							add = 25
							scope:spn_hduel_other = {
								OR = {
									has_trait = spn_champion_midnight
									has_character_modifier = spn_let_it_take_me_modifier
								}
							}
						}
						modifier = {
							add = 15
							scope:spn_hduel_other.prowess > root.prowess
						}
						spn_hide_duel_guard_effect = yes
						scope:spn_hduel_other = {
							spn_hide_duel_guard_effect = yes
							trigger_event = spn_hide.2102
						}
					}
					# Lost in the fighting. The other never knew where you went.
					50 = {
						modifier = {
							add = 25
							has_trait = spn_champion_timber
						}
						modifier = {
							add = 15
							prowess > scope:spn_hduel_other.prowess
						}
						send_interface_toast = {
							type = event_toast_effect_neutral
							title = spn_hide_duel_slipped_t
							left_icon = root
							custom_tooltip = spn_hide_duel_slipped_tt
						}
						spn_hide_duel_missed_effect = { A = scope:spn_hduel_a B = scope:spn_hduel_b }
						scope:spn_hduel_a = { spn_hide_duel_clean_effect = yes }
						scope:spn_hduel_b = { spn_hide_duel_clean_effect = yes }
					}
				}
			}
		}
		else = {
			spn_hide_duel_lapse_effect = yes
		}
		ai_chance = {
			base = 20
			modifier = {
				add = 40
				scope:spn_hduel_my_chance <= 35
			}
			modifier = {
				add = 30
				has_trait = craven
			}
			# Graybeard does not step back.
			modifier = {
				factor = 0.5
				has_trait = spn_champion_graybeard
			}
		}
	}
}

			scope:spn_hduel_a ?= { spn_hide_duel_clean_effect = yes }
			scope:spn_hduel_b ?= { spn_hide_duel_clean_effect = yes }
			# The one who ran hears he got away; the one who let him go chose it and needs no word.
			scope:spn_hduel_other = {
				if = {
					limit = {
						is_alive = yes
						is_ai = no
					}
					trigger_event = spn_hide.0202
				}
			}
		}
		ai_chance = {
			base = 40
		}
	}
