spn_initiate.0001 { #Initialize Vampire
	hidden = yes

	trigger = {
		# 2.39 (Thomas's call, 17 Oct): a child is never made a vampire, except a designed ruler at game start (the embrace flag)
		trigger_if = {
			limit = { NOT = { has_character_flag = supernatural_embrace_vampire } }
			is_adult = yes
		}
		AND = {
			spn_has_supernatural_trait = no
			spn_has_hunter_trait = no
			#is_witch_trigger = no
		}
		# SPECIES GATE -- see common/scripted_triggers/zz_spn_species_triggers.txt.
		# This is the hard backstop. These four events are the ONLY doors into the creature traits,
		# so a clause here covers every route in -- interactions, AI actions, activities, and
		# anything added later -- not just the buttons that were known about when it was written.
		spn_is_valid_supernatural_target = yes
		spn_can_become_creature_trigger = yes # 2.40 (WORK-ORDER-2.40 F3): no golem, demon-blooded or Baba Yaga (an opposite trait refuses add_trait and the rest of the door ran anyway)
	}

	immediate = {
		#Apply Genetic Trait from Decision
		if = {
			limit = { has_character_flag = supernatural_embrace_vampire }
			add_trait = supernatural_vampire
			add_character_flag = immune_to_disease
			remove_character_flag = supernatural_embrace_vampire
		}
		else = {
			add_trait = supernatural_vampire
		}
		# 2.40 (WORK-ORDER-2.40 F3): the level the caller chose (a picker, a randomiser, an embrace), now that the trait exists
		spn_initiate_apply_pending_xp_effect = { TRAIT = supernatural_vampire TRACK = vampirism VAR = spn_initiate_xp }
		spn_initiate_clear_pending_xp_effect = yes
		# 2.40 (WORK-ORDER-2.40 F3): a value set before this door (the made vampires of the embrace events, 20) is kept; a cure removes it
		if = {
			limit = { NOT = { has_variable = spn_age_aquired_immortality } }
			set_variable = {
				name = spn_age_aquired_immortality
				value = age
			}
		}
		set_variable = {
			name = spn_yearly_pulses_since_trait_upgrade
			value = 0
		}
		if = {
			limit = { has_treatable_disease_trigger = yes }
			spn_remove_treatable_disease = yes
		}
		if = {
			limit = { has_variable = spn_vampire_blood_donor }
			spn_assign_conferred_vampire_blood = yes
		}
		add_character_flag = {
			flag = immune_to_disease
		}

		spn_remove_negative_age_traits = yes
		spn_embrace_health_cleanup = yes # 2.34
		add_character_flag = {
			flag = spn_recently_became_vampire
			days = 1
		}
		#Unconditional. This used to sit only inside the has_character_flag = supernatural_embrace_*
		#branch, so anyone turned by a path that does not set that flag received NO disease immunity.
		#2001 also strips the flag further up, so a werewolf becoming a hybrid came out LESS
		#protected than before. Setting a flag twice is harmless, so this is safe on every path.
		add_character_flag = immune_to_disease

		spn_give_vampire_secret_or_trait_effect = yes
		
		trigger_event = { #Removes Vampire Blood + Donor variable
			id = spn_miscellaneous_events.0002
		}

		spn_track_current_visual_age = { VISUALAGE = age }

		# 2.34: the turning memory (and the maker's). Last, after the sire bond is settled. Idempotent.
		spn_write_vampire_turning_memory_effect = yes
		
	}
}

spn_initiate.0002 { #Remove Vampire
	hidden = yes

	trigger = {
		has_trait = supernatural_vampire
		has_variable = spn_age_aquired_immortality
	}

	immediate = {
		remove_trait = supernatural_vampire
		remove_trait = spn_known_vampire
		remove_variable = spn_age_aquired_immortality
		remove_variable = spn_yearly_pulses_since_trait_upgrade
		remove_character_flag = immune_to_disease
		remove_character_flag = spn_has_conferred_sire
		remove_character_flag = spn_has_mother_sire
		remove_character_flag = spn_has_father_sire


		if = {
			limit = { any_secret = { type = spn_secret_vampire } }
			random_secret = { #Remove the secret
				limit = {
					secret_type = spn_secret_vampire
				}
				remove_secret = yes
			}
		}
	}
}

spn_initiate.1001 { #Initiate Werewolf
	hidden = yes

	trigger = {
		AND = {
			spn_has_supernatural_trait = no
			spn_has_hunter_trait = no
			#is_witch_trigger = no
		}
		# SPECIES GATE -- see common/scripted_triggers/zz_spn_species_triggers.txt.
		# This is the hard backstop. These four events are the ONLY doors into the creature traits,
		# so a clause here covers every route in -- interactions, AI actions, activities, and
		# anything added later -- not just the buttons that were known about when it was written.
		spn_is_valid_supernatural_target = yes
		spn_can_become_creature_trigger = yes # 2.40 (WORK-ORDER-2.40 F3): no golem, demon-blooded or Baba Yaga (an opposite trait refuses add_trait and the rest of the door ran anyway)
	}

	immediate = { 
		#Apply Genetic Trait from Decision
		if = {
			limit = { has_character_flag = supernatural_embrace_werewolf }
			add_trait = supernatural_werewolf
			remove_character_flag = supernatural_embrace_werewolf
		}
		else = {
			add_trait = supernatural_werewolf
		}
		# 2.40 (WORK-ORDER-2.40 F3): the level the caller chose, now that the trait exists
		spn_initiate_apply_pending_xp_effect = { TRAIT = supernatural_werewolf TRACK = lycanthropy VAR = spn_initiate_xp }
		spn_initiate_clear_pending_xp_effect = yes

		# Born-vs-turned tag (for the gathering activity & future content).
		# The wound/incubation climax sets spn_turned_werewolf BEFORE firing this event;
		# everyone else (born, ruler designer, console, etc.) defaults to born.
		if = {
			limit = {
				NOT = { has_character_flag = spn_turned_werewolf }
				# 2.37 -- a wolf a hide chose (It Knows You) is not born (§13.3).
				NOT = { has_character_flag = spn_hide_chosen_werewolf }
			}
			add_character_flag = spn_born_werewolf
			# 2.37 -- born wolves start at Pack Standing 25 (§13.3), once: the flag stops Found a Pack and the
			# Gathering granting it again.
			if = {
				limit = { NOT = { has_character_flag = spn_born_standing_given } }
				add_character_flag = spn_born_standing_given
				add_trait_xp = { trait = supernatural_werewolf track = wolf_pack value = 25 }
			}
		}

		set_variable = {
			name = spn_age_aquired_immortality
			value = age
		}

		if = {
			limit = { has_treatable_disease_trigger = yes }
			spn_remove_treatable_disease = yes
		}
		add_character_flag = {
			flag = immune_to_disease
		}

		add_character_flag = {
			flag = spn_recently_became_lycan
			days = 1
		}
		#Unconditional. This used to sit only inside the has_character_flag = supernatural_embrace_*
		#branch, so anyone turned by a path that does not set that flag received NO disease immunity.
		#2001 also strips the flag further up, so a werewolf becoming a hybrid came out LESS
		#protected than before. Setting a flag twice is harmless, so this is safe on every path.
		add_character_flag = immune_to_disease

		spn_give_lycan_secret_or_trait_effect = yes

		#spn_track_current_visual_age = { VISUALAGE = age }

		# 2.34: the turning memory. The wound path already wrote the clawed version in spn_complete_wound_turn;
		# this only writes the anonymous one when nothing exists yet. Idempotent.
		spn_write_werewolf_woke_memory_effect = yes
	}
}

spn_initiate.2001 { #Initiate Hybrid
	hidden = yes

	trigger = {
		# 2.39 (Thomas's call, 17 Oct): a child is never made a hybrid, except a designed ruler at game start (the embrace flag)
		trigger_if = {
			limit = { NOT = { has_character_flag = supernatural_embrace_hybrid } }
			is_adult = yes
		}
		AND = {
			OR = {
				has_trait = supernatural_werewolf
				spn_has_supernatural_trait = no
			}
			spn_has_hunter_trait = no
			#is_witch_trigger = no
		}	
		# SPECIES GATE -- see common/scripted_triggers/zz_spn_species_triggers.txt.
		# This is the hard backstop. These four events are the ONLY doors into the creature traits,
		# so a clause here covers every route in -- interactions, AI actions, activities, and
		# anything added later -- not just the buttons that were known about when it was written.
		spn_is_valid_supernatural_target = yes
		spn_can_become_creature_trigger = yes # 2.40 (WORK-ORDER-2.40 F3): no golem, demon-blooded or Baba Yaga (an opposite trait refuses add_trait and the rest of the door ran anyway)
	}

	immediate = {
		# 2.37 -- Pack Standing lives on the werewolf's wolf_pack track; read it before the trait goes (§13.3b).
		if = {
			limit = { has_trait = supernatural_werewolf }
			save_scope_value_as = { name = spn_carry_standing value = "has_trait_xp(supernatural_werewolf|wolf_pack)" }
		}
		#Removes Lycanthropy from New Hybrid
		remove_trait = supernatural_werewolf
		remove_trait = spn_known_werewolf
		spn_former_werewolf_effect = yes
		remove_character_flag = immune_to_disease
		# 2.36 -- the old werewolf sire goes, but not one a confer interaction assigned for this very turning
		# (the four hybrid confer sites mark that for a day). Stripping it unconditionally meant a hybrid made by
		# interaction never had its maker named (spn_write_hybrid_turning_memory_effect needs the flag).
		if = {
			limit = { NOT = { has_character_flag = spn_sire_just_conferred } }
			remove_character_flag = spn_has_conferred_sire
		}
		remove_character_flag = spn_sire_just_conferred
		remove_character_flag = spn_has_mother_sire
		remove_character_flag = spn_has_father_sire

		if = {
			limit = { any_secret = { type = spn_secret_werewolf } }
			random_secret = { #Remove the secret
				limit = {
					secret_type = spn_secret_werewolf
				}
				remove_secret = yes
			}
		}
		
		if = {
			limit = { has_character_flag = supernatural_embrace_hybrid }
			add_trait = supernatural_hybrid
			add_character_flag = immune_to_disease
			remove_character_flag = supernatural_embrace_hybrid
		}
		else = {
			add_trait = supernatural_hybrid
		}
		# 2.40 (WORK-ORDER-2.40 F3): the level the caller chose, on both tracks, now that the trait exists
		spn_initiate_apply_pending_xp_effect = { TRAIT = supernatural_hybrid TRACK = hybrid_vampirism VAR = spn_initiate_xp }
		spn_initiate_apply_pending_xp_effect = { TRAIT = supernatural_hybrid TRACK = hybrid_lycanthropy VAR = spn_initiate_xp }
		spn_initiate_clear_pending_xp_effect = yes
		# 2.37 -- ...and onto the hybrid's own wolf_pack track: a Second at 80 is still a Second (§13.3b).
		if = {
			limit = { exists = scope:spn_carry_standing }
			add_trait_xp = { trait = supernatural_hybrid track = wolf_pack value = scope:spn_carry_standing }
		}
		var:spn_pack ?= { spn_pack_refresh_effect = yes }

		set_variable = {
			name = spn_age_aquired_immortality
			value = age
		}

		if = {
			limit = { has_character_flag = vampire_blood_in_system }
			remove_character_flag = vampire_blood_in_system
			remove_character_modifier = ingested_vampire_blood_modifier
		}

		if = {
			limit = { has_variable = spn_vampire_blood_donor }
			spn_assign_conferred_vampire_blood = yes
		}

		if = {
			limit = { has_treatable_disease_trigger = yes }
			spn_remove_treatable_disease = yes
		}
		add_character_flag = {
			flag = immune_to_disease
		}
		
		spn_remove_negative_age_traits = yes
		spn_embrace_health_cleanup = yes # 2.34
		add_character_flag = {
			flag = spn_recently_became_hybrid
			days = 1
		}
		#Unconditional. This used to sit only inside the has_character_flag = supernatural_embrace_*
		#branch, so anyone turned by a path that does not set that flag received NO disease immunity.
		#2001 also strips the flag further up, so a werewolf becoming a hybrid came out LESS
		#protected than before. Setting a flag twice is harmless, so this is safe on every path.
		add_character_flag = immune_to_disease

		spn_give_hybrid_secret_or_trait_effect = yes

		trigger_event = { #Removes Vampire Blood + Donor variable
			id = spn_miscellaneous_events.0002
		}

		spn_track_current_visual_age = { VISUALAGE = age }

		# 2.34: the turning memory. Idempotent.
		spn_write_hybrid_turning_memory_effect = yes
	}
}
