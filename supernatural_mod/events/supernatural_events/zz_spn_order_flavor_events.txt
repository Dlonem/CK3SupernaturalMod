# Supernatural 2.39 -- the Order's new moments (WORK-ORDER-2.39 F14 §2-3). Loc:
# localization/english/event_localization/zz_spn_239_flavor_l_english.yml.
#
# Every moment here re-checks its whole situation in its own trigger when it arrives (the Long Night lesson). The chapter
# moments (.0350, .0351, .0360-.0363, .0370-.0372) are sent by the chapter's year (spn_order_chapter_year_effect, step
# 10): each trigger is the flavor gate plus the branch's own spn_order_cm_*_ok_trigger, and each immediate marks the gate
# and sets its ten-year "seen" flag, so a moment the gate turned away is not used up. The two follow-ups (.0352, .0353)
# answer the player's own choice: they do not wait for the gate, but they mark it.
#
# No option shows a chance. Hidden rolls are hidden; the duels are the mod's own pattern (The Errand, .0321).

# .0350 -- After the Others Have Gone. A player Temple Magus, Grand Magus or seat holder; an AI junior of his temple who
# wants him.
spn_order.0350 = {
	type = character_event
	title = spn_order.0350.t
	desc = spn_order.0350.desc
	theme = seduction
	override_background = { reference = corridor_night }
	left_portrait = { character = root animation = thinking }
	right_portrait = { character = scope:spn_adv_junior animation = flirtation }
	trigger = {
		is_alive = yes
		is_imprisoned = no
		spn_flavor_ready_trigger = yes
		spn_order_cm_advance_senior_ok_trigger = yes
	}
	immediate = {
		spn_flavor_mark_effect = yes
		add_character_flag = { flag = spn_cm_seen_0350 years = 10 }
		var:spn_order_chapter = {
			random_in_list = {
				variable = spn_order_members
				limit = { spn_order_advance_junior_trigger = { SENIOR = root } }
				save_scope_as = spn_adv_junior
			}
		}
	}
	# Lovers; the hook (spn_order_lover_set) lets the junior ask for a word later, if root is his patron.
	option = {
		name = spn_order.0350.a
		scope:spn_adv_junior = {
			set_relation_lover = {
				target = root
				reason = spn_rel_order_advance
			}
		}
		had_sex_with_effect = {
			CHARACTER = scope:spn_adv_junior
			PREGNANCY_CHANCE = pregnancy_chance
		}
		stress_and_fulfillment_impact = {
			chaste = minor_stress_impact_gain
			zealous = minor_stress_impact_gain
		}
		if = {
			limit = { is_married = yes }
			stress_and_fulfillment_impact = { honest = minor_stress_impact_gain }
		}
	}
	option = {
		name = spn_order.0350.b
		scope:spn_adv_junior = {
			add_opinion = {
				target = root
				modifier = refusal_opinion
				opinion = -10
			}
		}
		stress_and_fulfillment_impact = { compassionate = minor_stress_impact_loss }
	}
	option = {
		name = spn_order.0350.c
		scope:spn_adv_junior = {
			add_opinion = {
				target = root
				modifier = humiliated_opinion
				opinion = -20
			}
			if = {
				limit = { spn_is_order_member_trigger = yes }
				custom_tooltip = spn_order.0350.c.tt
				spn_order_flavor_standing_effect = { VALUE = -5 }
			}
		}
		stress_and_fulfillment_impact = {
			compassionate = minor_stress_impact_gain
			callous = minor_stress_impact_loss
			arrogant = minor_stress_impact_loss
		}
	}
}

# .0351 -- A Door Left Open. A player Neophyte, Acolyte or Medicum, 18 or older; the first AI senior who could hear it:
# his own Temple Magus, the Grand Magus, then a seat holder (never his tutor: Late Lessons is that pair).
spn_order.0351 = {
	type = character_event
	title = spn_order.0351.t
	desc = spn_order.0351.desc
	theme = seduction
	override_background = { reference = study }
	left_portrait = { character = root animation = worry }
	right_portrait = { character = scope:spn_adv_senior animation = interested }
	trigger = {
		is_alive = yes
		is_imprisoned = no
		spn_flavor_ready_trigger = yes
		spn_order_cm_advance_junior_ok_trigger = yes
	}
	immediate = {
		spn_flavor_mark_effect = yes
		add_character_flag = { flag = spn_cm_seen_0351 years = 10 }
		save_scope_as = spn_adv_junior
		if = {
			limit = {
				var:spn_order_chapter ?= {
					story_owner = { spn_order_advance_senior_trigger = { JUNIOR = root } }
				}
			}
			var:spn_order_chapter.story_owner = { save_scope_as = spn_adv_senior }
		}
		else_if = {
			limit = {
				global_var:current_archmage_global ?= {
					in_diplomatic_range = root
					spn_order_advance_senior_trigger = { JUNIOR = root }
				}
			}
			global_var:current_archmage_global = { save_scope_as = spn_adv_senior }
		}
		else_if = {
			limit = {
				global_var:current_loremaster_global ?= {
					in_diplomatic_range = root
					spn_order_advance_senior_trigger = { JUNIOR = root }
				}
			}
			global_var:current_loremaster_global = { save_scope_as = spn_adv_senior }
		}
		else_if = {
			limit = {
				global_var:current_black_mage_global ?= {
					in_diplomatic_range = root
					spn_order_advance_senior_trigger = { JUNIOR = root }
				}
			}
			global_var:current_black_mage_global = { save_scope_as = spn_adv_senior }
		}
		else = {
			global_var:current_admin_global ?= { save_scope_as = spn_adv_senior }
		}
	}
	# Say it. The senior decides by his nature and his opinion of you (spn_order_advance_accept_value), hidden.
	option = {
		name = spn_order.0351.a
		custom_tooltip = spn_order.0351.a.tt
		hidden_effect = {
			scope:spn_adv_senior = {
				random = {
					chance = spn_order_advance_accept_value
					save_scope_value_as = { name = spn_adv_yes value = yes }
				}
			}
			if = {
				limit = { exists = scope:spn_adv_yes }
				scope:spn_adv_senior = {
					set_relation_lover = {
						target = root
						reason = spn_rel_order_advance
					}
				}
				had_sex_with_effect = {
					CHARACTER = scope:spn_adv_senior
					PREGNANCY_CHANCE = pregnancy_chance
				}
			}
			else = {
				add_stress = minor_stress_gain
				scope:spn_adv_senior = {
					add_opinion = {
						target = root
						modifier = annoyed_opinion
						opinion = -10
					}
				}
				random = {
					chance = 25
					save_scope_value_as = { name = spn_adv_talk value = yes }
				}
				if = {
					limit = { exists = scope:spn_adv_talk }
					spn_order_flavor_standing_effect = { VALUE = -5 }
					send_interface_toast = {
						type = event_toast_text_bad
						title = spn_order.0351.talk
						desc = spn_order.0351.talk_desc
						left_icon = scope:spn_adv_senior
					}
				}
				else = {
					send_interface_toast = {
						type = event_toast_text_neutral
						title = spn_order.0351.no
						desc = spn_order.0351.no_desc
						left_icon = scope:spn_adv_senior
					}
				}
			}
		}
	}
	# Say it, and ask for his word as well. Only to a patron (his Temple Magus or the Grand Magus: seat holders do not
	# decide promotions), and only from the ambitious, the greedy and the arbitrary. Refused by a just or honest senior,
	# it is reported: the lesser charge (spn_order_favouritism), never spn_order_broke_rules. Trying to buy rank costs no
	# more than being given it (WORK-ORDER-2.39 §5.4).
	option = {
		name = spn_order.0351.b
		trigger = {
			OR = {
				has_trait = ambitious
				has_trait = greedy
				has_trait = arbitrary
			}
			scope:spn_adv_senior = { spn_order_patron_of_trigger = { JUNIOR = root } }
		}
		custom_tooltip = spn_order.0351.b.tt
		hidden_effect = {
			save_scope_value_as = { name = spn_adv_bold value = yes }
			scope:spn_adv_senior = {
				random = {
					chance = spn_order_advance_accept_value
					save_scope_value_as = { name = spn_adv_yes value = yes }
				}
			}
			if = {
				limit = { exists = scope:spn_adv_yes }
				# The favour is settled here and now, so the hook finds nothing left to ask: the flag comes first.
				add_character_flag = { flag = spn_order_favour_asked years = 10 }
				scope:spn_adv_senior = {
					set_relation_lover = {
						target = root
						reason = spn_rel_order_advance
					}
				}
				had_sex_with_effect = {
					CHARACTER = scope:spn_adv_senior
					PREGNANCY_CHANCE = pregnancy_chance
				}
				spn_order_favour_effect = { JUNIOR = root PATRON = scope:spn_adv_senior }
				send_interface_toast = {
					type = event_toast_text_good
					title = spn_order.0351.deal
					desc = spn_order.0351.deal_desc
					left_icon = scope:spn_adv_senior
				}
			}
			else = {
				scope:spn_adv_senior = {
					add_opinion = {
						target = root
						modifier = insulted_opinion
					}
				}
				if = {
					limit = {
						scope:spn_adv_senior = {
							OR = {
								has_trait = just
								has_trait = honest
							}
						}
					}
					add_character_flag = { flag = spn_order_favouritism years = 10 }
					send_interface_toast = {
						type = event_toast_text_bad
						title = spn_order.0351.reported
						desc = spn_order.0351.reported_desc
						left_icon = scope:spn_adv_senior
					}
				}
				else = {
					send_interface_toast = {
						type = event_toast_text_bad
						title = spn_order.0351.cold
						desc = spn_order.0351.cold_desc
						left_icon = scope:spn_adv_senior
					}
				}
			}
		}
	}
	option = {
		name = spn_order.0351.c
		add_learning_lifestyle_xp = minor_lifestyle_xp
		stress_and_fulfillment_impact = {
			lustful = minor_stress_impact_gain
			chaste = minor_stress_impact_loss
		}
	}
}
