# .0002 -- (hidden, no root) the AI seats. Waits twice for a player still deciding, then seats regardless.
spn_order.0002 = {
	scope = none
	hidden = yes
	trigger = {
		spn_order_on_trigger = yes
		NOT = { exists = global_var:spn_order_seeded }
	}
	immediate = {
		if = {
			limit = {
				any_player = { has_character_flag = spn_order_offer_pending }
				OR = {
					NOT = { exists = global_var:spn_order_seat_waits }
					global_var:spn_order_seat_waits < 2
				}
			}
			if = {
				limit = { NOT = { exists = global_var:spn_order_seat_waits } }
				set_global_variable = { name = spn_order_seat_waits value = 0 }
			}
			change_global_variable = { name = spn_order_seat_waits add = 1 }
		}
		else = {
			spn_order_seed_ai_effect = yes
			if = {
				limit = { has_global_variable = spn_order_seat_waits }
				remove_global_variable = spn_order_seat_waits
			}
		}
	}
}
