spn_gathering.0203 = {
	type = activity_event
	title = spn_gathering.0203.t
	desc = spn_gathering.0203.desc
	theme = hunting
	override_background = { reference = wilderness_forest }
	left_portrait = { character = root animation = personality_bold }
	right_portrait = { character = scope:spn_young_wolf animation = rage }
	trigger = {
		this = scope:host
		scope:activity = { has_current_phase = spn_gathering_run_phase }
		scope:activity = {
			any_attending_character = {
				NOT = { this = scope:host }
				spn_is_wolf_trigger = yes
				age < 25
			}
		}
		NOT = { scope:activity = { has_variable = spn_run_seen_0203 } }
	}
	immediate = {
		scope:activity = { set_variable = { name = spn_run_seen_0203 value = yes } }
		scope:activity = {
			random_attending_character = {
				limit = {
					NOT = { this = scope:host }
					spn_is_wolf_trigger = yes
					age < 25
				}
				save_scope_as = spn_young_wolf
			}
		}
	}
	option = {
		name = spn_gathering.0203.a
		spn_pack_gain_standing_effect = { VALUE = 3 }
		scope:spn_young_wolf = { add_opinion = { target = root modifier = respect_opinion opinion = 10 } }
		ai_chance = {
			base = 60
			modifier = { add = 20 has_trait = arrogant }
		}
	}
	option = {
		name = spn_gathering.0203.b
		scope:spn_young_wolf = { spn_pack_gain_standing_effect = { VALUE = 3 } }
		ai_chance = {

spn_gathering.0300 = {
	type = activity_event
	title = spn_gathering.0300.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:spn_ch_chosen = root }
				desc = spn_gathering.0300.desc_me
			}
			desc = spn_gathering.0300.desc_other
		}
	}
	theme = dread
	override_background = { reference = wilderness_forest }
	left_portrait = { character = root animation = worry }
	right_portrait = { character = scope:spn_ch_chosen animation = stunned }
	artifact = {
		target = scope:spn_ch_hide
		position = lower_right_portrait
		trigger = { exists = scope:spn_ch_hide }
	}
	trigger = {
		exists = scope:spn_ch_hide
		exists = scope:spn_ch_chosen
		scope:spn_ch_hide = { artifact_owner ?= root }
		scope:spn_ch_chosen = { is_alive = yes }
	}
	# It chose him.
	option = {
		name = spn_gathering.0300.a
		trigger = { scope:spn_ch_chosen = root }
		custom_tooltip = spn_gathering.0300.a.tt
		spn_hide_choosing_give_effect = yes
	}
	# It chose another. Let it go.
	option = {
		name = spn_gathering.0300.b
		trigger = { NOT = { scope:spn_ch_chosen = root } }
		custom_tooltip = spn_gathering.0300.b.tt
		scope:spn_ch_chosen = { spn_hide_choosing_give_effect = yes }
		ai_chance = {
			base = 70
		}
	}
	# Or fight for it.
	option = {
		name = spn_gathering.0300.c
		trigger = {
			NOT = { scope:spn_ch_chosen = root }
			spn_is_wolf_trigger = yes
			# 2.37o: an AI holder does not fight a player the hide chose; that player would be hurt and lose it without
			# a word (the review of 4 Oct, S3). A player holder may fight anyone.
			OR = {
				is_ai = no
				scope:spn_ch_chosen = { is_ai = yes }
			}
		}
		custom_tooltip = spn_gathering.0300.c.tt
		spn_clash_lean_tooltip_effect = { A = root B = scope:spn_ch_chosen FORM = yes }
		spn_creature_clash_effect = {
			A = root
			B = scope:spn_ch_chosen
			THEN = spn_hide_choosing_clash_after_effect
			FORM = yes
		}
		ai_chance = {
			base = 30
			modifier = { add = 30 has_trait = ambitious }
			modifier = { add = 20 has_trait = wrathful }
			modifier = { add = -30 has_trait = craven }
		}
	}
}
