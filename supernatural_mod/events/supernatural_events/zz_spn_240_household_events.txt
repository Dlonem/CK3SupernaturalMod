namespace = spn_household

# Supernatural 2.40 (WORK-ORDER-2.40 F2) -- household administration. The one event: Answer the Old Blood's choice.
# The thirst, the wolf and both go through the embrace level pickers (spn_initiate.0003 / .1003 / .2003), which since F3
# really land 0, 50 or 100, so the doors do the rest (secret, immunity, memory). The pit is Witchcraft's own door,
# make_demon_effect (gated by spn_may_become_demon_trigger, which a player always passes).

# The Old Blood Answers
spn_household.0001 = {
	type = character_event
	title = spn_household.0001.t
	desc = spn_household.0001.desc
	theme = default
	right_portrait = {
		character = root
		animation = thinking
	}

	trigger = {
		is_alive = yes
		spn_household_has_door_trigger = yes
		spn_has_supernatural_trait = no
		spn_has_hunter_trait = no
		spn_can_become_creature_trigger = yes
	}

	# The thirst.
	option = {
		name = spn_household.0001.a
		custom_tooltip = spn_household.0001.a.tt
		hidden_effect = { trigger_event = spn_initiate.0003 }
	}
	# The wolf.
	option = {
		name = spn_household.0001.b
		custom_tooltip = spn_household.0001.b.tt
		hidden_effect = { trigger_event = spn_initiate.1003 }
	}
	# Both.
	option = {
		name = spn_household.0001.c
		custom_tooltip = spn_household.0001.c.tt
		hidden_effect = { trigger_event = spn_initiate.2003 }
	}
	# The pit.
	option = {
		name = spn_household.0001.d
		trigger = { spn_may_become_demon_trigger = yes }
		custom_tooltip = spn_household.0001.d.tt
		hidden_effect = {
			if = {
				limit = { has_character_flag = spn_embrace_choosing }
				remove_character_flag = spn_embrace_choosing
			}
			make_demon_effect = yes
		}
	}
	# Not today.
	option = {
		name = spn_household.0001.e
		if = {
			limit = { has_character_flag = spn_embrace_choosing }
			remove_character_flag = spn_embrace_choosing
		}
	}
}
