			}
		}
	}

		if = {

		else = {
			scope:spn_fe_a = { save_scope_as = spn_feud_other }
		}
	}

		name = spn_pack.0200.a
	}
	# 2.40 (WORK-ORDER-2.40 C6): the alpha whose wolf lost it may call the holder out. The Challenge settles who is
	# stronger; it does not move the hide, and the text says so. Greyed when the Challenge's own test says no.
	option = {
		name = spn_pack.0200.b
		show_as_unavailable = {
			scope:spn_fe_a ?= root
			scope:spn_feud_taker ?= { is_alive = yes }
		}
		trigger = {
			scope:spn_fe_a ?= root
			scope:spn_feud_taker ?= {
				is_alive = yes
				NOT = { this = root }
			}
			spn_pack_challenge_shown_trigger = { ACTOR = root TARGET = scope:spn_feud_taker }
		}
		custom_tooltip = spn_pack.0200.b.tt
		open_interaction_window = {
			interaction = spn_challenge_interaction
			actor = root
			recipient = scope:spn_feud_taker
		}
	}
	# 2.40 (WORK-ORDER-2.40 C6): or let it go, both ways.
	option = {
		name = spn_pack.0200.c
		custom_tooltip = spn_pack.0200.c.tt
		if = {
			limit = { has_opinion_modifier = { target = scope:spn_feud_other modifier = spn_hide_feud_opinion } }
			remove_opinion = { target = scope:spn_feud_other modifier = spn_hide_feud_opinion }
		}
		scope:spn_feud_other = {
			if = {
				limit = { has_opinion_modifier = { target = root modifier = spn_hide_feud_opinion } }
				remove_opinion = { target = root modifier = spn_hide_feud_opinion }
			}
		}
		add_prestige = minor_prestige_loss
	}
}

	type = character_event
	title = spn_pack.0300.t

	theme = dread
	override_background = { reference = wilderness_forest }
	left_portrait = { character = root animation = personality_bold }

		is_alive = yes
		NOT = { exists = global_var:spn_ta_century }
		exists = scope:spn_ta_pack
		var:spn_pack ?= scope:spn_ta_pack
		spn_true_alpha_candidate_trigger = { PACK = scope:spn_ta_pack }
	}
	immediate = {
		set_global_variable = { name = spn_ta_century value = yes years = 100 }
		set_global_variable = { name = spn_ta_living value = root }
		add_trait = spn_true_alpha
		scope:spn_ta_pack.story_owner = { save_scope_as = spn_ta_old_alpha }
		spn_true_alpha_rises_effect = yes
	}
	option = {
		name = spn_pack.0300.a
		# 2.37o: re-checked when clicked: he still runs in that pack and does not lead it yet (the review of 4 Oct, F13).
		if = {
			limit = {
				var:spn_pack ?= scope:spn_ta_pack
				spn_is_alpha_trigger = no
			}
			custom_tooltip = spn_pack.0300.a.tt
			# 2.40 (WORK-ORDER-2.40 C5): the alpha he would replace answers: a player is asked (Red Eyes in Your Pack), an
			# AI alpha yields, fights or lets him go by his temper (spn_pack_true_alpha_ai_answer_effect).
			save_scope_as = spn_ta
			scope:spn_ta_pack.story_owner = { save_scope_as = spn_ta_old_alpha }

				limit = { scope:spn_ta_old_alpha = { is_ai = no } }

			}
			else = {
				spn_pack_true_alpha_ai_answer_effect = yes
			}
		}
		else = {
			custom_tooltip = spn_moment_passed_tt
		}
		ai_chance = {
			base = 60
			modifier = { add = 20 has_trait = just }
		}
	}
	option = {
		name = spn_pack.0300.b
		if = {
			limit = {
				var:spn_pack ?= scope:spn_ta_pack
				spn_is_alpha_trigger = no
			}
			custom_tooltip = spn_pack.0300.b.tt
			spn_true_alpha_found_own_effect = yes
		}
		else = {
			custom_tooltip = spn_moment_passed_tt
		}
		ai_chance = {
			base = 40
			modifier = { add = 30 has_trait = ambitious }
		}
