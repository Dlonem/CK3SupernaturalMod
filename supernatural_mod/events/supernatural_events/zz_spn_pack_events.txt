# .0011 -- An Invitation. The new wolf, three days after he turned; his maker's pack asks.

	type = character_event
	title = spn_pack.0011.t
	desc = spn_pack.0011.desc
	theme = hunting
	override_background = { reference = wilderness_forest }
	left_portrait = { character = root animation = interested }
	right_portrait = { character = scope:spn_inv_alpha animation = personality_bold }
	trigger = {
		is_alive = yes
		spn_is_wolf_trigger = yes
		NOT = { exists = var:spn_pack }
		scope:spn_inv_pack ?= { spn_pack_has_room_trigger = yes }
		scope:spn_inv_alpha = {
			is_alive = yes
			var:spn_pack ?= scope:spn_inv_pack
		}
	}
	option = {
		name = spn_pack.0011.a
		spn_pack_add_member_effect = { WHO = root PACK = scope:spn_inv_pack }
		ai_chance = {
			base = 50
			modifier = { add = 50 has_relation_ww_clawed_by = scope:spn_inv_alpha }
			modifier = { add = 50 is_close_family_of = scope:spn_inv_alpha }
			modifier = { add = 20 has_trait = gregarious }
			modifier = { add = -30 has_trait = shy }
			modifier = { add = -30 has_trait = paranoid }
		}
	}
	option = {
		name = spn_pack.0011.b
		ai_chance = {
			base = 20
			modifier = { add = 30 has_trait = shy }
			modifier = { add = 20 has_trait = paranoid }
		}
	}
}

# .0014 -- (hidden) carried through the fever by a pack at a Gathering (2.37b): he wakes in it. Fired a day after the

	hidden = yes
	trigger = {

		}
	}
}

	type = character_event

	theme = hunting
	override_background = { reference = wilderness_forest }
	left_portrait = { character = root animation = interested }
	right_portrait = { character = scope:spn_mo_asker animation = personality_rational }
	trigger = {
		is_alive = yes
		scope:spn_mo_asker = { is_alive = yes }
	}
	option = {
		name = spn_pack.0101.a
		# 2.37o: re-checked when clicked: you still lead the pack, it has room, and the wolf still runs with the other
		# (the review of 4 Oct, F13).

			limit = {
				exists = scope:spn_mo_pack
				exists = scope:spn_mo_other_pack
				var:spn_pack ?= scope:spn_mo_pack
				spn_is_alpha_trigger = yes

				scope:spn_mo_asker = {
					is_alive = yes
					spn_is_alpha_trigger = no

				}
			}
			custom_tooltip = spn_pack.0101.a.tt
			spn_pack_take_asker_effect = yes
		}
		else = {
			custom_tooltip = spn_moment_passed_tt
		}
		ai_chance = { base = 50 }
	}
	option = {
		name = spn_pack.0101.b
		ai_chance = { base = 50 }
	}

# .0300 -- The True Alpha. The globals and the trait first, then the choice.
spn_pack.0300 = {
	type = character_event
	title = spn_pack.0300.t
	desc = spn_pack.0300.desc
	theme = dread
	override_background = { reference = wilderness_forest }
	left_portrait = { character = root animation = personality_bold }
	right_portrait = { character = scope:spn_ta_old_alpha animation = worry }
	trigger = {
		is_alive = yes
		NOT = { exists = global_var:spn_ta_century }
		exists = scope:spn_ta_pack
		var:spn_pack ?= scope:spn_ta_pack
		spn_true_alpha_candidate_trigger = { PACK = scope:spn_ta_pack }
	}
	immediate = {
		set_global_variable = { name = spn_ta_century value = yes years = 100 }
		set_global_variable = { name = spn_ta_living value = root }
		add_trait = spn_true_alpha
		scope:spn_ta_pack.story_owner = { save_scope_as = spn_ta_old_alpha }
		spn_true_alpha_rises_effect = yes
	}
	option = {
		name = spn_pack.0300.a
		# 2.37o: re-checked when clicked: he still runs in that pack and does not lead it yet (the review of 4 Oct, F13).
		if = {
			limit = {
				var:spn_pack ?= scope:spn_ta_pack
				spn_is_alpha_trigger = no
			}
			custom_tooltip = spn_pack.0300.a.tt
			scope:spn_ta_pack = { spn_pack_set_alpha_effect = { NEW = root } }
			scope:spn_ta_old_alpha = {
				add_opinion = { target = root modifier = spn_true_alpha_rose_opinion }
			}
		}
		else = {
			custom_tooltip = spn_moment_passed_tt
		}
		ai_chance = {
			base = 60
			modifier = { add = 20 has_trait = just }
		}
	}
	option = {
		name = spn_pack.0300.b
		if = {
			limit = {
				var:spn_pack ?= scope:spn_ta_pack
				spn_is_alpha_trigger = no
			}
			custom_tooltip = spn_pack.0300.b.tt
			spn_true_alpha_found_own_effect = yes
		}
		else = {
			custom_tooltip = spn_moment_passed_tt
		}
		ai_chance = {
			base = 40
			modifier = { add = 30 has_trait = ambitious }
		}
	}
}
