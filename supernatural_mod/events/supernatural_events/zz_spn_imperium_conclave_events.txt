# .0210 -- What the Grand Magus Did. A player member who came and knows the Grand Magus's staged-rift secret.
spn_conclave.0210 = {
	type = activity_event
	title = spn_conclave.0210.t
	desc = spn_conclave.0210.desc
	theme = intrigue
	override_background = { reference = mc_shadows }
	left_portrait = { character = root animation = thinking }
	right_portrait = { character = scope:spn_cv_gm animation = personality_rational }
	trigger = {
		is_alive = yes
		scope:spn_cv_gm ?= { is_alive = yes }
	}
	option = {
		name = spn_conclave.0210.a
		trigger = {
			NOT = { scope:activity ?= { has_variable = spn_cv_unmasked } }
			scope:spn_cv_gm = {
				any_secret = {
					secret_type = spn_secret_staged_rift
					is_known_by = root
				}
			}
		}
		custom_tooltip = spn_conclave.0210.a.tt
		hidden_effect = {
			scope:activity ?= { set_variable = { name = spn_cv_unmasked value = yes days = 60 } }
			scope:spn_cv_gm = {
				random_secret = {
					limit = {
						secret_type = spn_secret_staged_rift
						is_known_by = root
					}
					expose_secret = root
				}
			}
		}
	}
	option = {
		name = spn_conclave.0210.b
		trigger = { NOT = { scope:activity ?= { has_variable = spn_cv_unmasked } } }
		custom_tooltip = spn_conclave.0210.b.tt
	}
	option = {
		name = spn_conclave.0210.c
		trigger = { scope:activity ?= { has_variable = spn_cv_unmasked } }
	}
}
