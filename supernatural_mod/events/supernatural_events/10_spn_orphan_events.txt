# Supernatural 2.35 PART TWO -- "The Bridge". THE TWO ORPHANS.
# WORK-ORDER-2.35-PART-TWO-THE-BRIDGE §10.
#
# Both are rare, both are set twenty years in the future, and both point the same fuse in opposite directions:
# the werewolf orphan is a child who will decide what to be, and the demon child is a child who will become
# something on a timer whatever anybody decides.
#
# THE WEREWOLF ORPHAN, 8% (E13: "yes, rare enough"), off a culprit whose track was 50 or better.
# THE DEMON CHILD, 6%. Thomas: "Could you make a demon orphan, like a baby demon? There's a part in the mod
# called Abaddon -- that was an incredible addition from CK2... perhaps there could be a demon baby as an
# Easter egg as well, and you can either kill it or exorcise it."
#
# The mechanics for the second one already exist upstream and nobody has used them this way.
# ms_child_demon_effect (00_summoning_magic_effects.txt:603) is Abaddon: an age-0 demon0 child placed in the
# summoner's own house. And demon0 is LATENT, not a demon -- is_a_demon_trigger is demon1 | demon2 only -- so
# the merciful option is genuinely available, and upstream already wrote the branch for it: magic_ev2.38
# option b does exactly `remove_trait = demon0`. Exorcising a demon child is that branch, reached a different
# way, and it is why "you can either kill it or exorcise it" turned out to be free.
#
# EVERY OPTION IN BOTH EVENTS CARRIES TRAIT-WEIGHTED ai_chance -- Thomas's standing note, honoured throughout.
# And in both, anything that ends in death is last.

# ==========================================================================================================
# What Was Left in the House. Root = the hunter. scope:spn_hunt_orphan_child set.
# ==========================================================================================================
spn_hunt.0600 = {
	type = character_event
	title = spn_hunt.0600.t
	desc = spn_hunt.0600.desc
	theme = family
	override_background = { reference = ruined_holding } # 2.35a art pass -- was bedchamber. What Was Left in the House: the house after the fight went through it.
	left_portrait = { character = root animation = disbelief }
	right_portrait = { character = scope:spn_hunt_orphan_child animation = fear }

	trigger = {
		is_alive = yes
		exists = scope:spn_hunt_orphan_child
		scope:spn_hunt_orphan_child = { is_alive = yes }
	}

	# Take the child.
	option = {
		name = spn_hunt.0600.a
		trigger = { is_landed_or_landless_administrative = yes }
		custom_tooltip = spn_hunt.0600.a.tt
		scope:spn_hunt_orphan_child = {
			set_employer = root
			add_character_flag = spn_raised_by_the_hunter
			set_variable = { name = spn_orphan_raiser value = root }
			# At sixteen the child chooses, and the choice reads the hunter's own record rather than a die.
			trigger_event = { id = spn_hunt.0601 years = 16 }
		}
		add_stress = minor_stress_impact_loss
		ai_chance = {
			base = 45
			modifier = { add = 35 has_trait = compassionate }
			modifier = { add = 25 has_trait = just }
			modifier = { add = 25 has_trait = forgiving }
			modifier = { add = 20 has_trait = generous }
			modifier = { add = -25 has_trait = callous }
			modifier = { add = -20 has_trait = paranoid }
		}
	}
	# Leave the child.
	option = {
		name = spn_hunt.0600.b
		custom_tooltip = spn_hunt.0600.b.tt
		scope:spn_hunt_orphan_child = {
			set_variable = { name = spn_orphan_raiser value = root }
			add_character_flag = spn_left_by_the_hunter
			trigger_event = { id = spn_hunt.0601 years = 16 }
		}
		stress_impact = {
			base = minor_stress_impact_gain
			compassionate = medium_stress_impact_gain
		}
		ai_chance = {
			base = 40
			modifier = { add = 30 has_trait = cynical }
			modifier = { add = 25 has_trait = content }
			modifier = { add = 20 has_trait = lazy }
			modifier = { add = -25 has_trait = compassionate }
		}
	}
	# Finish it. Anything that ends in death is last, and nothing here reads the child after this line.
	option = {
		name = spn_hunt.0600.c
		custom_tooltip = spn_hunt.0600.c.tt
		add_piety = { value = minor_piety_value multiply = -2 }
		stress_impact = {
			base = major_stress_impact_gain
			sadistic = minor_stress_impact_loss
			callous = minor_stress_impact_gain
			compassionate = massive_stress_impact_gain
			just = major_stress_impact_gain
		}
		if = {
			limit = { exists = scope:spn_hunt_county }
			# The county does not thank a man for this and it does not forget it either.
			scope:spn_hunt_county = {
				add_county_modifier = { modifier = spn_hunt_mob_justice_modifier years = 2 }
			}
		}
		scope:spn_hunt_orphan_child = { death = { death_reason = death_murder killer = root } }
		ai_chance = {
			base = 0
			modifier = { add = 25 has_trait = sadistic }
			modifier = { add = 20 has_trait = callous }
			modifier = { add = 15 has_trait = paranoid }
			modifier = { add = -100 has_trait = compassionate }
			modifier = { add = -100 has_trait = forgiving }
			modifier = { add = -60 has_trait = just }
		}
	}
}

# Sixteen years later. Root = the child. The reciprocating event from the other side of the fight, twenty
# years late -- and which side it lands on is decided by what the hunter did and what he became.
spn_hunt.0601 = {
	type = character_event
	title = spn_hunt.0601.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_character_flag = spn_raised_by_the_hunter }
				desc = spn_hunt.0601.desc_raised
			}
			desc = spn_hunt.0601.desc_left
		}
	}
	theme = family
	override_background = { reference = study }
	left_portrait = { character = root animation = thinking }
	# 2.35a -- the man this whole event is about was not on screen.
	right_portrait = { character = scope:spn_orphan_raiser animation = personality_content }

	trigger = {
		is_alive = yes
		is_adult = yes
		exists = var:spn_orphan_raiser
	}

	immediate = {
		var:spn_orphan_raiser = { save_scope_as = spn_orphan_raiser }
	}

	# Raised by him, and it took: a hunter. Reads the man's actual record -- a name the world knows and a
	# household that liked him -- rather than a coin flip.
	option = {
		name = spn_hunt.0601.a
		trigger = {
			has_character_flag = spn_raised_by_the_hunter
			scope:spn_orphan_raiser = { is_alive = yes }
			OR = {
				scope:spn_orphan_raiser = { spn_hunter_reputation_value >= 2 }
				opinion = { target = scope:spn_orphan_raiser value >= 20 }
			}
		}
		custom_tooltip = spn_hunt.0601.a.tt
		trigger_event = spn_initiate.3001
		ai_chance = { base = 100 }
	}
	# Raised by him, and it did not: a creature, and it leaves.
	option = {
		name = spn_hunt.0601.b
		trigger = {
			has_character_flag = spn_raised_by_the_hunter
			NOR = {
				AND = {
					scope:spn_orphan_raiser = { is_alive = yes }
					scope:spn_orphan_raiser = { spn_hunter_reputation_value >= 2 }
				}
				opinion = { target = scope:spn_orphan_raiser value >= 20 }
			}
		}
		custom_tooltip = spn_hunt.0601.b.tt
		add_character_flag = spn_orphan_went_its_own_way
		if = {
			limit = { scope:spn_orphan_raiser = { is_alive = yes } }
			set_relation_rival = scope:spn_orphan_raiser
		}
		ai_chance = { base = 100 }
	}
	# Left in the house, and it remembers exactly who walked out of it.
	option = {
		name = spn_hunt.0601.c
		trigger = { has_character_flag = spn_left_by_the_hunter }
		custom_tooltip = spn_hunt.0601.c.tt
		if = {
			limit = { scope:spn_orphan_raiser = { is_alive = yes } }
			set_relation_nemesis = scope:spn_orphan_raiser
			scope:spn_orphan_raiser = { trigger_event = { id = spn_hunt.0602 days = { 30 400 } } }
		}
		ai_chance = { base = 100 }
	}
}

# ==========================================================================================================
# The Thing in the Cradle. Root = the hunter. scope:spn_hunt_cradle_child set.
#
# The three-way of mercy, ambition and violence, with a twenty-year fuse on the middle one -- the same fuse
# as the werewolf orphan, pointed the other way.
# ==========================================================================================================
spn_hunt.0610 = {
	type = character_event
	title = spn_hunt.0610.t
	desc = spn_hunt.0610.desc
	theme = dread
	override_background = { reference = bp2_nursery } # 2.35a art pass -- was bedchamber. The Thing in the Cradle: it is, literally, a nursery.
	left_portrait = { character = root animation = disbelief }
	right_portrait = { character = scope:spn_hunt_cradle_child animation = sadness }

	trigger = {
		is_alive = yes
		exists = scope:spn_hunt_cradle_child
		scope:spn_hunt_cradle_child = {
			is_alive = yes
			has_trait = demon0
		}
	}

	# Exorcise it -- before it was ever a demon. Upstream's own branch (magic_ev2.38 option b), reached a
	# different way. The child is ordinary. magic_ev3.75 will fire at twenty and find nothing to work with.
	option = {
		name = spn_hunt.0610.a
		trigger = {
			custom_tooltip = {
				text = spn_hunt_exorcise_needs_tt
				spn_hunt_can_exorcise_trigger = yes
			}
		}
		custom_tooltip = spn_hunt.0610.a.tt
		scope:spn_hunt_cradle_child = {
			remove_trait = demon0
			add_character_flag = spn_was_exorcised
		}
		add_trait_xp = { trait = supernatural_hunter track = witch_hunting value = { 8 16 } }
		add_piety = { value = minor_piety_value multiply = 3 }
		add_stress = minor_stress_impact_loss
		ai_chance = {
			base = 55
			modifier = { add = 35 has_trait = compassionate }
			modifier = { add = 30 has_trait = just }
			modifier = { add = 25 has_trait = forgiving }
			# And for the zealous, who reads it as duty rather than mercy and arrives at the same door.
			modifier = { add = 30 has_trait = zealous }
		}
	}
	# Take it in and watch it. NOTHING IS REMOVED. At twenty or twenty-one magic_ev3.75 fires as it always
	# has, and you have a Greater Demon in your court that you raised.
	option = {
		name = spn_hunt.0610.b
		trigger = { is_landed_or_landless_administrative = yes }
		custom_tooltip = spn_hunt.0610.b.tt
		custom_tooltip = spn_hunt.0610.b.tt2
		scope:spn_hunt_cradle_child = {
			set_employer = root
			add_character_flag = spn_raised_by_the_hunter
		}
		ai_chance = {
			base = 20
			modifier = { add = 35 has_trait = ambitious }
			modifier = { add = 30 has_trait = arbitrary }
			modifier = { add = 25 has_trait = cynical }
			# Somebody who sees an opportunity rather than a problem.
			modifier = { add = 40 is_witch_trigger = yes }
			modifier = { add = -35 has_trait = zealous }
			modifier = { add = -30 has_trait = paranoid }
		}
	}
	# Behead it. LAST.
	option = {
		name = spn_hunt.0610.c
		custom_tooltip = spn_hunt.0610.c.tt
		# Character scope, exactly as upstream's own wh_attraction_value calls it -- the trigger takes the
		# faith as an argument rather than being evaluated inside a faith scope.
		if = {
			limit = { trait_is_criminal_in_faith_trigger = { TRAIT = trait:witch FAITH = root.faith GENDER_CHARACTER = root } }
			add_piety = { value = major_piety_value multiply = 1 }
		}
		else = {
			add_piety = { value = major_piety_value multiply = -1 }
		}
		stress_impact = {
			base = major_stress_impact_gain
			zealous = minor_stress_impact_loss
			sadistic = minor_stress_impact_loss
			callous = minor_stress_impact_gain
			compassionate = massive_stress_impact_gain
			forgiving = major_stress_impact_gain
		}
		scope:spn_hunt_cradle_child = { death = { death_reason = death_murder killer = root } }
		ai_chance = {
			base = 25
			modifier = { add = 40 has_trait = zealous }
			modifier = { add = 30 has_trait = sadistic }
			modifier = { add = 25 has_trait = callous }
			modifier = { add = 25 has_trait = paranoid }
			modifier = { add = -100 has_trait = compassionate }
			modifier = { add = -80 has_trait = forgiving }
		}
	}
	# Leave it where it is. Somebody else's problem. It grows up.
	option = {
		name = spn_hunt.0610.d
		custom_tooltip = spn_hunt.0610.d.tt
		ai_chance = {
			base = 30
			modifier = { add = 35 has_trait = craven }
			modifier = { add = 30 has_trait = content }
			modifier = { add = 20 has_trait = lazy }
		}
	}
}
