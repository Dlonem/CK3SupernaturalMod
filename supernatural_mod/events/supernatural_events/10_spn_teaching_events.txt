# Supernatural 2.35 PART TWO -- "The Bridge". THE FOUR TEACHING EVENTS.
# WORK-ORDER-2.35-PART-TWO-THE-BRIDGE §7.1.
#
# The trail system is now four kinds deep, has a cap, a global throttle, five ways to cover a body and three
# ways to clear a mark -- and a player who has just been turned knows none of it. These four exist so that
# nobody has to read a changelog to find out how the world works.
#
# EACH TEACHES ONE RULE BY SHOWING ITS CONSEQUENCE, NEVER BY EXPLAINING IT:
#   The First Body            -- your first mark.                        A county remembers.
#   Nobody Saw Anything       -- your first successful cover-up.         So it CAN be hidden.
#   The Man Asking Questions  -- your first mark reaching three.         So it can be followed.
#   What the Old Woman Told You -- your first prowl with no maker.       Nobody ever taught you any of this.
#
# ONE PER CHARACTER, EVER. The flags carry no duration, so they fire once in a playthrough and are fresh on
# the next one. All four are gated to players in the effects that fire them (zz_spn_menace_effects.txt,
# zz_spn_consequence_effects.txt): an AI creature does not need to be told what it just did, and a thousand
# queued events to tell it would be a real cost for no gain.
#
# NONE OF THEM HAS A MECHANICAL EFFECT worth the name. That is deliberate. A tutorial that pays you is a
# tutorial you farm.

# The First Body.
spn_hunt.0500 = {
	type = character_event
	title = spn_hunt.0500.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { is_a_demon_trigger = yes }
				desc = spn_hunt.0500.desc_demon
			}
			triggered_desc = {
				trigger = { has_trait = supernatural_werewolf }
				desc = spn_hunt.0500.desc_wolf
			}
			desc = spn_hunt.0500.desc
		}
	}
	theme = dread
	override_background = { reference = alley_night }
	left_portrait = { character = root animation = paranoia }

	trigger = { is_alive = yes }

	option = {
		name = spn_hunt.0500.a
		custom_tooltip = spn_hunt.0500.a.tt
	}
}

# The Man Asking Questions.
spn_hunt.0502 = {
	type = character_event
	title = spn_hunt.0502.t
	desc = spn_hunt.0502.desc
	theme = dread
	override_background = { reference = temple_church }
	left_portrait = { character = root animation = worry }

	trigger = { is_alive = yes }

	# Let him ask.
	option = {
		name = spn_hunt.0502.a
		custom_tooltip = spn_hunt.0502.a.tt
	}
	# Be somewhere else for a while. The third of the three ways to clear a trail, offered at the exact
	# moment a player first needs to know it exists.
	option = {
		name = spn_hunt.0502.b
		trigger = {
			spn_hunt_is_creature_trigger = yes
			NOT = { has_character_modifier = spn_gone_to_ground_modifier }
		}
		custom_tooltip = spn_hunt.0502.b.tt
		add_character_modifier = { modifier = spn_gone_to_ground_modifier years = 1 }
		spn_menace_gone_to_ground_effect = yes
	}
}

# What the Old Woman Told You. The mod's answer to Thomas's own complaint: "whomever turned me into a
# werewolf didn't teach me much about it."
spn_hunt.0503 = {
	type = character_event
	title = spn_hunt.0503.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_trait = supernatural_werewolf }
				desc = spn_hunt.0503.desc_wolf
			}
			desc = spn_hunt.0503.desc
		}
	}
	theme = secret
	override_background = { reference = village } # 2.35a art pass -- was tavern. What the Old Woman Told You: an almshouse doorway, not a tavern.
	left_portrait = { character = root animation = thinking }

	trigger = { is_alive = yes }

	# Listen.
	option = {
		name = spn_hunt.0503.a
		custom_tooltip = spn_hunt.0503.a.tt
		add_stress = minor_stress_impact_loss
	}

	option = {
		name = spn_hunt.0503.b
		custom_tooltip = spn_hunt.0503.b.tt
	}
}
