supernatural_ev1.01 = {
	title = supernatural_ev1.01
	# 2.32 -- the letter now says something different depending on who sent it, because a letter
	# from the vampire who MADE you does not read like a letter from an acquaintance.
	desc = {
		triggered_desc = {
			trigger = { has_relation_mmaker = scope:host }
			desc = supernatural_ev1.01.desc_sire
		}
		triggered_desc = {
			trigger = { has_relation_pprogeny = scope:host }
			desc = supernatural_ev1.01.desc_progeny
		}
		triggered_desc = {
			trigger = {
				OR = {
					has_trait = spn_known_vampire
					has_trait = spn_known_hybrid
				}
			}
			desc = supernatural_ev1.01.desc_known
		}
		desc = supernatural_ev1.01.desc
	}
	theme = secret
	override_background = { reference = study }
	left_portrait = { character = root animation = reading }
	right_portrait = { character = scope:host animation = scheme }

	trigger = {
		is_alive = yes
		is_imprisoned = no
		# 2.32 -- the two `value > 99` clauses are GONE. They stopped inviting a maxed character
		# at exactly the point the capped-track conversion now makes the evening worth attending;
		# together they were the dead end a maxed vampire sat in.
		NOT = { has_character_modifier = vf_recently_fed_modifier }
	}

	cooldown = {
		years = 2
	}

	option = {
		name = supernatural_ev1.01.a
		custom_tooltip = supernatural_ev1.01.a.tt

		# 2.32 -- this option had NO effect block at all. It was a pure notification: accepting it
		# did nothing and joined you to nothing, which is most of why the invitation felt hollow.
		trigger = {
			exists = scope:host.involved_activity
			can_join_activity = scope:host.involved_activity
		}
		accept_activity_invite = scope:host.involved_activity
		add_opinion = { target = scope:host modifier = pleased_opinion opinion = 15 }

		ai_chance = {
			base = 60
			modifier = { add = 30 has_relation_mmaker = scope:host }
			modifier = { add = 25 has_relation_pprogeny = scope:host }
			modifier = { add = 20 has_trait = gluttonous }
			modifier = { add = 15 has_trait = lustful }
			modifier = { add = 15 has_trait = gregarious }
			modifier = { add = -25 has_trait = paranoid }
			modifier = { add = -20 has_trait = craven }
			modifier = { add = -15 has_trait = content }
			modifier = { add = -10 has_trait = shy }
		}
	}

	option = {
		name = supernatural_ev1.01.b
		custom_tooltip = supernatural_ev1.01.b.tt

		# Declining costs the host something small -- an invitation refused is still an invitation
		# they extended, and they took a risk writing it down at all.
		scope:host = { add_opinion = { target = root modifier = disappointed_opinion opinion = -10 } }
		stress_impact = { base = minor_stress_impact_gain }

		ai_chance = {
			base = 40
			modifier = { add = 30 has_trait = paranoid }
			modifier = { add = 25 has_trait = craven }
			modifier = { add = 20 has_trait = content }
			modifier = { add = 15 has_trait = temperate }
			modifier = { add = -30 has_relation_mmaker = scope:host }
			modifier = { add = -20 has_trait = gluttonous }
		}
	}
}

supernatural_ev1.08 = {
	type = activity_event
	title = supernatural_ev1.08.title
	desc = supernatural_ev1.08.desc
	theme = intrigue
	override_background = {
		reference = bp1_wine_cellar
	}
	
	left_portrait = {
		character = root
		animation = disbelief
	}
	center_portrait = {
		character = scope:local_courtier
		animation = pain
	}
	right_portrait = {
		character = scope:vampire0
		animation = shock
	}
	
	# 2.32 -- this was `NOT = { A B C }`, which is NOT(A AND B AND C). The two traits are
	# `opposites`, so the AND could never be true and the whole trigger was permanently TRUE.
	# The obvious repair is NOR -- "an attendee who is not a vampire or hybrid" -- but that makes
	# the event UNREACHABLE, because can_be_activity_guest requires every guest to be one or the
	# other. Premise and guest list contradicted each other.
	#
	# Resolved by separating GUESTS from LOCALS, which is what the desc always described: the
	# injured man walks in through the tavern door, he was never on the invite list. So the
	# trigger asks for what the text actually needs -- another vampire here to help him inside
	# ("one of the attendees instinctively helps him") -- and the victim comes from the location.
	trigger = {
		is_alive = yes
		exists = scope:activity
		scope:activity = {
			any_attending_character = {
				count >= 1
				NOT = { this = root }
			}
		}
	}
	
	immediate = {
		hidden_effect = {
			# 2.32 -- "the door creaks open to reveal an injured man stumbling inside." He is a
			# local, and the code was drawing him from your own court. The old NOR also carried the
			# same double negative as ev1.09: NOR negates each member, so
			# `scope_is_close_to_root_trigger = no` resolved to `= yes` and REQUIRED a court member.
			vf_select_victim_effect = { SCOPE_NAME = local_courtier GENDER_FEMALE_CHANCE = 0 }
			scope:local_courtier = { add_trait = wounded_2 }
			involved_activity = {
				random_attending_character = {
					limit = {
						NOT = {
							THIS=ROOT
						}
					}
					save_scope_as = vampire0
				}
			}
		}
	}

	option = { #Feed on the Wounded Courtier
		name = supernatural_ev1.08.a
		# 2.32 -- both grants fired unconditionally. The traits are `opposites`, so the wrong half
		# errored on every single firing. Option c of this same event already branched correctly.
		if = {
			limit = { has_trait = supernatural_vampire }
			add_trait_xp = { trait = supernatural_vampire track = vampirism value = 10 }
		}
		else_if = {
			limit = { has_trait = supernatural_hybrid }
			add_trait_xp = { trait = supernatural_hybrid track = hybrid_vampirism value = 10 }
		}
		remove_short_term_gold = { 5 15 }
		# Standing rule: the irreversible step goes last, after every value that reads the
		# pre-change state has been banked.
		scope:local_courtier = {
			death = { death_reason = spn_death_drained killer = root }
		}
		
		ai_chance = {
			base = 50
			modifier = { add = 15 has_trait = gluttonous }
			modifier = { add = -10 has_trait = compassionate }
		}
	}

	option = { #Heal the Wounded Courtier
		name = supernatural_ev1.08.b
		add_prestige = { 35 75 }
		add_piety = { 40 85 }
		scope:local_courtier = {
			remove_trait = wounded_2
			add_character_modifier = {
				modifier = ingested_vampire_blood_modifier
			}
			add_character_flag = vampire_blood_in_system
			trigger_event = {
				id = spn_miscellaneous_events.0002
				years = 3
			}
		}
		remove_short_term_gold = { 5 15 }
		
		ai_chance = {
			base = 60
			modifier = { add = 20 has_trait = compassionate }
			modifier = { add = -15 has_trait = sadistic }
		}
	}

	option = { #Feed On & Heal the Wounded Couriter
		name = supernatural_ev1.08.c
		trigger = {
			OR = {
				has_trait_xp = {
					trait = supernatural_vampire
					track = vampirism
					value > 20
				}
				has_trait_xp = {
					trait = supernatural_hybrid
					track = hybrid_vampirism
					value > 15
				}
			}
		}
		add_prestige = { 35 75 }
		add_piety = { 40 85 }
		if = {
			limit = {
				has_trait = supernatural_vampire
			}
			add_trait_xp = {
				trait = supernatural_vampire
				track = vampirism
				value = 10
			}			
		}
		else_if = {
			limit = {
				has_trait = supernatural_hybrid
			}
			add_trait_xp = {
				trait = supernatural_hybrid
				track = hybrid_vampirism
				value = 10
			}
		}

		scope:local_courtier = {
			remove_trait = wounded_2
			add_character_modifier = {
				modifier = ingested_vampire_blood_modifier
			}
			add_character_flag = vampire_blood_in_system
			trigger_event = {
				id = spn_miscellaneous_events.0002
				years = 3
			}
		}
		remove_short_term_gold = { 5 15 }
		
		ai_chance = {
			base = 80
			modifier = { add = 20 has_trait = ambitious }
		}
	}
}
