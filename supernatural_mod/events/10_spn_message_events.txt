namespace = spn_message
spn_message.0001 { #Send confer vampirism acceptance message
	type = character_event
	hidden = yes

	immediate = {
		scope:recipient = {
			send_interface_message = {
				type = event_generic_neutral
				title = spn_received_vampirism
				
				right_icon = scope:actor

				show_as_tooltip = {
					scope:actor = {
						custom_tooltip = spn_msg_received_vampirism_interaction_message
					}
				}
			}
		}
		scope:actor = {
			send_interface_message = {
				type = event_generic_neutral
				title = spn_confered_vampirism
				
				right_icon = scope:recipient

				show_as_tooltip = {
					scope:recipient = {
						custom_tooltip = spn_msg_confered_vampirism_interaction_message
					}
				}
			}
		}
	}
}

spn_message.4001 { # Compulsion: success (notify both sides)
    type = character_event
    hidden = yes

    immediate = {
        # Recipient: vague—no actor face, no identity revealed
        scope:recipient = {
            send_interface_message = {
                type  = event_generic_neutral
                title = spn_compel_success_recipient
                # (no right_icon on purpose to keep it anonymous)
                show_as_tooltip = { custom_tooltip = spn_msg_compel_success_recipient }
            }
        }

        # Actor: show target face, explicit success
        scope:actor = {
            send_interface_message = {
                type  = event_generic_neutral
                title = spn_compel_success_actor
                right_icon = scope:recipient
                show_as_tooltip = {
                    scope:recipient = { custom_tooltip = spn_msg_compel_success_actor }
                }
            }
        }
    }
}
