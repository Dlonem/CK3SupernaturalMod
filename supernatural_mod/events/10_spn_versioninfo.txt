namespace = spn_versioninfo
#Main Info, Change global var name to get it to prompt during yearly pulse
spn_versioninfo.1000 = {
	type = character_event
	title = spn_versioninfo.1000.t
	
	theme = spn_info
	desc = spn_versioninfo.1000.desc

	trigger = {
		OR = {
			NOT = { exists = global_var:version_2_40 }
			exists = global_var:spn_versioninfo_today # 2.40 boot 3 (Q2): every player sent it on the same day sees it
		}
		NOT = { has_character_flag = spn_versioninfo_seen } # boot 3: one popup per player, even if two hooks send it that day
	}

	immediate = {
		set_global_variable = {
			name = version_2_40
			value = yes
		}
		set_global_variable = { name = spn_versioninfo_today value = yes days = 1 } # 2.40 boot 3 (Q2), see the trigger
		add_character_flag = { flag = spn_versioninfo_seen days = 2 }
	}

	option = {
		name = spn_versioninfo.1000.a
		custom_tooltip = spn_versioninfo.1000.a.tt
	}
}
