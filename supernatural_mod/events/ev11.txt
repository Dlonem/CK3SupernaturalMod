magic_ev11.4 = {
	hidden = yes
	scope = character
	trigger = {

		}
	}

			}
		}
	}
	
	immediate = {
		random_list = {
			2 = {
				#constructions
				trigger_event = magic_ev11.5
			}
			2 = {
				#logistics
				trigger_event = magic_ev11.6
			}
			2 = {
				#agriculture
				trigger = {
					is_ruler = yes
					government_has_flag = government_is_nomadic
					exists = capital_county
				}
				trigger_event = magic_ev11.7
			}
			2 = {
				#warfare
				trigger = {
					NOT = {
						has_trait = organizer
					}
				}
				trigger_event = magic_ev11.8
			}
			2 = {
				#ends
				trigger = {
					NOT = {
						has_character_flag = innovator_loop_flag
					}
				}
				trigger_event = magic_ev11.9
			}
			2 = {
				#armor
				trigger = {
					any_equipped_character_artifact = {
						artifact_slot_type = armor
						NOT = {
							has_artifact_modifier = wc_innovator_armor_modifier
						}
					}
				}
				trigger_event = magic_ev11.10
			}
			2 = {
				#management
				trigger = {
					OR = {
						exists = capital_county
						has_trait = governor
					}
				}
				trigger_event = magic_ev11.11
			}
			2 = {
				#gunpowder
				trigger = {
					NOR = {
						has_character_flag = wc_innovator_gunpowder_flag
						has_character_modifier = wc_innovator_pyrotechnician_modifier
					}
				}
				trigger_event = magic_ev11.12
			}
		}
	}
}
