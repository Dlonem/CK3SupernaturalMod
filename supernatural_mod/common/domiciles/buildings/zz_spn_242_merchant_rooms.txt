# Supernatural 2.42 -- Merchants of the Night. Two rooms for the merchant mansion, on the Old Vault's shape
# (zz_spn_241_vault_buildings.txt): internal, previous_building = merchant_company_main_01 (an internal room must have
# one), and vanilla's own "requires building X" line.

# THE SECOND LEDGER. A demon who deals with mortals keeps a second set of books beside the company's Ledger Room.
# Nothing in its name or text says what is entered in it.
spn_merchant_second_ledger_01 = {
	allowed_domicile_types = { merchant_company }
	can_construct_potential = {
		is_a_demon_trigger = yes
		has_perk = deals_with_mortals_perk
	}
	can_construct = {
		custom_tooltip = {
			text = spn_second_ledger_needs_ledger_tt
			domicile ?= { has_domicile_building_or_higher = merchant_company_ledger_room_01 }
		}
	}
	slot_type = internal
	previous_building = merchant_company_main_01
	construction_time = 90
	cost = { gold = merchant_company_upgrade_cost_tier_1_value }
	character_modifier = {
		diplomacy = 2
		intrigue = 2
		owned_scheme_secrecy_add = 10
	}
	ai_value = { value = merchant_company_ai_normal_new_internal_value }
	asset = {
		icon = "gfx/interface/icons/domicile_building/domicile_building_scribes.dds"
		texture = "gfx/interface/window_domiciles/merchant_company_grand_estate_western.dds"
		intersectionmask_texture = "gfx/interface/window_domiciles/merchant_company_grand_estate_western_mask.png"
	}
}

# THE ALCHEMIST'S STILL. A Mage, True Witch or Warlock who heads a company that already deals in medicines.
# Vanilla's trade good (Medicaments) and vanilla's rooms; no potion is ever a trade good.
spn_merchant_still_01 = {
	allowed_domicile_types = { merchant_company }
	can_construct_potential = {
		OR = {
			is_true_mage = yes
			has_trait = true_witch
			has_trait = warlock
		}
	}
	can_construct = {
		custom_tooltip = {
			text = spn_still_needs_medicines_tt
			domicile ?= {
				OR = {
					has_domicile_building_or_higher = merchant_company_medicaments_traders_01
					has_domicile_building_or_higher = merchant_company_apothecaries_hall_01
				}
			}
		}
	}
	slot_type = internal
	previous_building = merchant_company_main_01
	construction_time = 90
	cost = { gold = merchant_company_upgrade_cost_tier_1_value }
	character_modifier = {
		trade_category_medicaments_sell_price_mult = 0.0025
		trade_category_medicaments_xp_gain_mult = 0.05
	}
	ai_value = { value = merchant_company_ai_normal_new_internal_value }
	asset = {
		icon = "gfx/interface/icons/building_types/resource_buildings/medicaments_buildings.dds"
		texture = "gfx/interface/window_domiciles/merchant_company_grand_estate_western.dds"
		intersectionmask_texture = "gfx/interface/window_domiciles/merchant_company_grand_estate_western_mask.png"
	}
}
