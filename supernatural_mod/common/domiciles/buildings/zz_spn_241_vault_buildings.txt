# Supernatural 2.41 -- THE VAULT. Thomas, 17 Nov 2026: "some kind of treasure room, a mansion vault makes you less likely to
# get attacked. You're more safer to hostile schemes. It adds gold income... the amount of money you need is less because
# of how much you've already accrued."
#
# An internal room of the merchant mansion (vanilla 1.21 domicile type merchant_company), built on vanilla's own room
# shape (merchant_company_common_chest_01..: slot_type internal, previous_building chain, the mansion's own picture).
# Only a vampire or hybrid can raise one, and nothing in its name or text says why: a stranger sees an old family vault.
#   Safer:        enemy_hostile_scheme_success_chance_add (vanilla's own counter-scheme modifier).
#   Needs less:   domicile_build_gold_cost (building the mansion costs less) and a modest domicile_monthly_gold_add.
#   Accrued:      the second and third rooms need centuries -- an age a mortal merchant never reaches.
# Modest on purpose: an AI vampire must not snowball (the reason it is not a flat income).
spn_merchant_vault_01 = {
	allowed_domicile_types = { merchant_company }
	can_construct_potential = {
		OR = {
			has_trait = supernatural_vampire
			has_trait = supernatural_hybrid
		}
	}
	slot_type = internal
	previous_building = merchant_company_main_01
	construction_time = 90 # vanilla @merchant_company_upgrade_construction_time_01 (a file-local constant)
	cost = { gold = merchant_company_upgrade_cost_tier_1_value }
	character_modifier = {
		enemy_hostile_scheme_success_chance_add = -5
		domicile_build_gold_cost = -0.05
		domicile_monthly_gold_add = 0.5
	}
	ai_value = { value = merchant_company_ai_normal_new_internal_value }
	asset = {
		icon = "gfx/interface/icons/domicile_building/domicile_building_church_treasury.dds"
		texture = "gfx/interface/window_domiciles/merchant_company_grand_estate_western.dds"
		intersectionmask_texture = "gfx/interface/window_domiciles/merchant_company_grand_estate_western_mask.png"
	}
}
