# Supernatural 2.43 stage 7 (23 Nov 2026) -- CHARACTER scope (root = a hunter weighing the Hunt). A warm trail he could ride
# for this year: inside the hunt's own leash (is_location_valid's spn_max_hunt_distance: squared_distance_large, x25 for a
# camp) and not resting after a hunt (spn_hunt_recently_hunted). Read by the Hunt's ai_will_do and by the funnel.
spn_hunt_ai_trail_in_reach_trigger = {
	exists = location
	any_in_global_list = {
		variable = spn_menace_counties
		spn_trailsman_trail_is_warm_trigger = yes
		NOT = { has_variable = spn_hunt_recently_hunted }
		title_province ?= {
			trigger_if = {
				limit = { root = { is_landless_adventurer = yes } }
				squared_distance = { target = root.location value <= spn_hunt_ai_camp_leash_value }
			}
			trigger_else = {
				squared_distance = { target = root.location value <= squared_distance_large }
			}
		}
	}
}
