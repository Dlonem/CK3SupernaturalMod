# 2.39 (WORK-ORDER-2.39 F9): the game rule "Supernatural: Mortals Becoming Demons" (spn_mortal_demons_rule,
# common/game_rules/10_spn_rules.txt).
#
# Character scope: may this living character be made a Demon? Default: always, which is exactly 2.38. Players Only: a
# player character only. It is read in the four places that turn a living character into a demon, all in Witchcraft
# files and all through apply_upstream_edits.py (package P9 entries):
#   make_demon_effect (the demon-blooded heir coming of age, "The truth about me", the magic eruption, the Ascendancy
#     project event, the Bael letter's courtier), mr_ascendancy_effect (Ascendancy research), create_mage_effect (an AI
#     ruler made a world Mage) and magical_research_for_ai_effect (an AI mage's own research).
# Demons created by summoning (create_character from a demon template) never come through any of these, so they still
# appear. An AI character who is demon-blooded (demon0) simply stays so.
spn_may_become_demon_trigger = {
	OR = {
		NOT = { has_game_rule = spn_mortal_demons_players_only }
		is_ai = no
	}
}
