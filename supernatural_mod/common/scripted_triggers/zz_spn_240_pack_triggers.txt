# ==================================================================================================
# C8 Let It Take Me: at war for this purpose. The same rule as the war prompt (spn_hide_war_prompt_valid_trigger): his
# own war, or, holding no land, his lord's. Character scope = the champion.
# ==================================================================================================
spn_hide_lets_it_take_at_war_trigger = {
	OR = {
		is_at_war = yes
		AND = {
			is_landed = no
			liege_or_court_owner ?= { is_at_war = yes }
		}
	}
}

				NOT = { exists = var:spn_knows_shrine_tundra }

		AND = {

# ==================================================================================================
# C6 Offer Pack Terms. Root = anyone; A and B = the two alphas (A takes B's pack in). Neither pack was joined by the
# Unification (its co-alpha and its Evolved belong to the married rite), and together they fit under the cap.
# ==================================================================================================
spn_pack_terms_merge_ok_trigger = {
	# Boot 1 (22 Oct): guarded. The engine checks send options for characters who are not alphas, and an unguarded
	# var:spn_pack read logged a "Failed context switch" each time.
	$A$ = { exists = var:spn_pack }
	$B$ = { exists = var:spn_pack }
	custom_tooltip = {
		text = spn_pack_terms_unified_tt
		$A$.var:spn_pack ?= { NOT = { exists = var:spn_pack_unified } }
		$B$.var:spn_pack ?= { NOT = { exists = var:spn_pack_unified } }
	}
	trigger_if = {
		limit = {
			$A$ = { exists = var:spn_pack }
			$B$ = { exists = var:spn_pack }
		}
		custom_tooltip = {
			text = spn_unification_too_many_tt
			$B$.var:spn_pack.var:spn_pack_size <= {
				value = spn_pack_cap_value
				subtract = $A$.var:spn_pack.var:spn_pack_size
			}
		}
	}
}

# ==================================================================================================
# C4 Hold the Unification: a spouse who leads another pack, or who already runs in HOST's pack. Character scope = the
# spouse.
# ==================================================================================================
spn_unification_spouse_ok_trigger = {
	OR = {
		spn_is_alpha_trigger = yes
		trigger_if = {
			limit = { $HOST$ = { exists = var:spn_pack } }
			var:spn_pack ?= $HOST$.var:spn_pack
		}
		trigger_else = { always = no }
	}
}

# C4 .0305's re-check for a spouse who already runs in his pack. Root = the host. Nested, so nothing reads a pack that
# is gone.
spn_unification_same_pack_ok_trigger = {
	exists = var:spn_pack
	trigger_if = {
		limit = { exists = var:spn_pack }
		NOT = { var:spn_pack = { exists = var:spn_pack_unified } }
		$SPOUSE$ = {
			is_alive = yes
			is_spouse_of = root
			var:spn_pack ?= root.var:spn_pack
		}
	}
}

# ==================================================================================================
# Boot 2 (j) Offer Pack Terms: may $HOST$ know this alpha runs a pack? Root = the other alpha. The mod's own rule for
# knowing a wolf (spn_host_knows_wolf_trigger), a Blood Pact between them, or a moment between their packs that named
# him to $HOST$ (spn_pack_alpha_scented_effect: Another Pack's Scent, A Wolf Asks to Run With You, Young Wolves Fight).
# Every one is something $HOST$ was told or saw, so the menu entry never appears on a secret (B6).
# ==================================================================================================
spn_pack_alpha_known_trigger = {
	OR = {
		spn_host_knows_wolf_trigger = { HOST = $HOST$ }
		spn_pack_pact_bound_trigger = { WHO = $HOST$ }
		$HOST$ = { is_target_in_variable_list = { name = spn_alphas_scented target = prev } }
	}
}

# ==================================================================================================
# Boot 2 (k) Young Wolves Fight. Are the two packs at odds? A and B = the two alphas. A rivalry either way, a feud over
# a hide either way, or one thinks badly of the other; never between packs bound by a Blood Pact.
# ==================================================================================================
spn_pack_young_fight_at_odds_trigger = {
	NOT = { $A$ = { spn_pack_pact_bound_trigger = { WHO = $B$ } } }
	OR = {
		$A$ = {
			OR = {
				has_relation_rival = $B$
				has_relation_potential_rival = $B$
				has_opinion_modifier = { target = $B$ modifier = spn_hide_feud_opinion }
				opinion = { target = $B$ value <= -25 }
			}
		}
		$B$ = {
			OR = {
				has_relation_rival = $A$
				has_relation_potential_rival = $A$
				has_opinion_modifier = { target = $A$ modifier = spn_hide_feud_opinion }
				opinion = { target = $A$ value <= -25 }
			}
		}
	}
}

# Boot 3: the two alphas belong to two different houses (the same test as spn_pack_house_step_effect) and vanilla
# will track a relation between those houses (its own is_valid_to_start test, house_relation_is_valid_to_start_trigger,
# called as in vanilla's 10_tgp_test_interactions.txt:19). Gates the Back option's house line, so it never promises a
# step the engine would not take. Root = anyone.
spn_pack_young_fight_houses_differ_trigger = {
	scope:spn_mo_alpha = { exists = house }
	scope:spn_mo_other = { exists = house }
	NOT = { scope:spn_mo_alpha.house = scope:spn_mo_other.house }
	house_relation_is_valid_to_start_trigger = {
		HOUSE = scope:spn_mo_alpha.house
		OTHER_HOUSE = scope:spn_mo_other.house
	}
}
