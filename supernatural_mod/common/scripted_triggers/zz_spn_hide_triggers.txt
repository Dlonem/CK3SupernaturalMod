# Werewolf or hybrid. Never a vampire, hunter, witch or demon.
spn_is_wolf_trigger = {
	OR = {
		has_trait = supernatural_werewolf
		has_trait = supernatural_hybrid
	}
}

# Who a hide can take: an adult wolf, and the species gate (zz_spn_species_triggers.txt). A born pup waits.
spn_can_bear_hide_trigger = {
	spn_is_wolf_trigger = yes
	is_adult = yes
	spn_is_valid_supernatural_target = yes
}

# Will this hide take this wolf (§5.8, 2.37b)? Root = the wolf; scope:spn_fit_hide = the hide's flag. Two routes:
#  1. one trait the hide looks for, and more of those than of the traits it turns from (Thomas, 3 Oct: "even for at
#     least 1 trait"). Alpha still takes only a leader.
#  2. the score: half the wolf's track plus the traits, 40 (spn_hide_fit_value). A wolf deep in the blood can still be
#     chosen with no matching trait, and the Circle's +20 / +40 still counts here.
# Every place that asks "would it take him" reads this; the score alone is for ranking who it goes looking for.
spn_hide_accepts_wolf_trigger = {
	OR = {
		spn_hide_fit_value >= spn_hide_fit_bar_wolf_value
		AND = {
			spn_hide_liked_count_value >= 1
			spn_hide_liked_count_value > spn_hide_hated_count_value
			spn_hide_leader_ok_trigger = yes
		}
	}
}

# Alpha only chooses a leader (design §3.4): a pack's alpha, a duke or more, or with the Packs rule Off a True Werewolf
# or a full wolf. Every other hide: always. Root = the candidate; scope:spn_fit_hide = the hide's flag.
spn_hide_leader_ok_trigger = {
	trigger_if = {
		limit = { scope:spn_fit_hide = flag:alpha }
		OR = {
			spn_is_alpha_trigger = yes
			highest_held_title_tier >= tier_duchy
			AND = {
				has_game_rule = spn_packs_off
				OR = {
					has_trait = spn_true_werewolf
					spn_hide_lyc_full_trigger = yes
				}
			}
		}
	}
	trigger_else = { always = yes }
}

# "Worn", for claims (§5.4): equipped AND its owner carries this hide's champion trait. A wolf who has just put a
# hide on but has not been chosen yet does not strip claims.

	OR = {
		AND = {

		}
		AND = {
			var:spn_hide ?= flag:timber
			artifact_owner ?= { has_trait = spn_champion_timber }
		}
		AND = {
			var:spn_hide ?= flag:tundra
			artifact_owner ?= { has_trait = spn_champion_tundra }
		}
		AND = {
			var:spn_hide ?= flag:graybeard
			artifact_owner ?= { has_trait = spn_champion_graybeard }
		}
		AND = {
			var:spn_hide ?= flag:midnight
			artifact_owner ?= { has_trait = spn_champion_midnight }
		}

			var:spn_hide ?= flag:alpha

		}
	}
}

# Lycanthropy (hybrids: hybrid lycanthropy) at 100: Alpha's fit with the Packs rule Off (§5.6, §9.5). Read only on the
# trait he carries: has_trait_xp on a trait he lacks is never asked.
spn_hide_lyc_full_trigger = {
	trigger_if = {
		limit = { has_trait = supernatural_werewolf }
		has_trait_xp = { trait = supernatural_werewolf track = lycanthropy value >= 100 }
	}
	trigger_else_if = {
		limit = { has_trait = supernatural_hybrid }
		has_trait_xp = { trait = supernatural_hybrid track = hybrid_lycanthropy value >= 100 }
	}
	trigger_else = { always = no }
}

# Can $HOST$ plausibly know what this wolf is? The wolf version of spn_host_knows_supernatural_trigger
# (10_spn_triggers.txt:137). Evaluated in the CANDIDATE's scope.
spn_host_knows_wolf_trigger = {
	OR = {
		has_trait = spn_known_werewolf
		has_trait = spn_known_hybrid
		any_secret = {
			type = spn_secret_werewolf
			is_known_by = $HOST$
		}
		any_secret = {
			type = spn_secret_hybrid
			is_known_by = $HOST$
		}
		is_spouse_of = $HOST$
		is_close_family_of = $HOST$
		has_relation_ww_clawed = $HOST$
		has_relation_ww_clawed_by = $HOST$
		has_relation_spn_pack_mate = $HOST$
		# Same pack. trigger_if, not AND: CK3 does not short-circuit AND siblings, so an unguarded cross-scope var
		# read would error on every wolf outside a pack (the Long Night's rule 4).
		trigger_if = {
			limit = {
				exists = var:spn_pack
				$HOST$ = { exists = var:spn_pack }
			}
			var:spn_pack = $HOST$.var:spn_pack
		}
		trigger_else = { always = no }
	}
}

# ---- War hooks (§12.2, §12.5) ----
# Champion scope; scope:war set by on_war_started / on_join_war_as_secondary.
spn_hide_war_prompt_valid_trigger = {
	is_alive = yes
	NOT = { has_character_flag = spn_hide_war_dont_ask }
	NOT = { has_character_flag = spn_hide_war_prompted }
	# 2.37 (Thomas, 3 Oct: the prompt came back again and again): once per war. Allies join a war over weeks
	# (on_join_war_as_secondary), and the 3-day flag only covered the first few days of it.
	NOT = { is_target_in_variable_list = { name = spn_hide_war_asked target = scope:war } }
	OR = {
		is_participant_in_war = scope:war
		# His lord's war is his only when he holds no land: a knight or a courtier goes where his lord goes. A landed
		# vassal is asked about the wars he is in; vanilla never calls vassals into a liege's war
		# (00_war_and_peace_triggers.txt:167), so his lord's wars stay his own choice through Let the Wolf Run.
		AND = {
			is_landed = no
			liege_or_court_owner ?= { is_participant_in_war = scope:war }
		}
	}
}

			location ?= root.combat.location
		}
	}
}

# A pack's True Alpha candidate (§16). PACK = the story.
spn_true_alpha_candidate_trigger = {
	is_alive = yes
	is_adult = yes
	NOT = { this = $PACK$.story_owner }
	NOT = { has_trait = spn_true_alpha }
	NOT = { has_character_flag = spn_killed_a_wolf }
	spn_pack_standing_value >= 100
	calc_true_if = {
		amount >= 2
		has_trait = just
		has_trait = honest
		has_trait = compassionate
		has_trait = brave
		has_trait = diligent
	}
}

# 2.37o: a wolf of PACK who would ask to run with NEW_ALPHA's pack, for a reason (the review of 4 Oct, F8: it was any
# wolf, his own alpha's loyal son included). Not a loyal wolf, not his own alpha's close kin; and fickle, or he thinks
# little of his own alpha, or much of the other one.
spn_pack_moment_asker_trigger = {
	spn_pack_moment_member_trigger = { PACK = $PACK$ }
	NOT = { has_trait = loyal }
	NOT = { is_close_family_of = $PACK$.story_owner }
	OR = {
		has_trait = fickle
		opinion = {
			target = $PACK$.story_owner
			value <= -10
		}
		opinion = {
			target = $NEW_ALPHA$
			value >= 30
		}
	}
}

# ---- 2.37o: a hide lying idle (the yearly net, step 1; spn_hide_owner_changed) ----
# Artifact scope = a hide. Not in a shrine, held by an AI who is not its champion -- a mortal, a wolf it refused, a
# champion holding it as a spare -- and not a hunter's trophy. A player's hide is never idle: he decides.
spn_hide_lies_idle_trigger = {
	NOT = { exists = var:spn_hide_shrine_county }
	# Worn means a wolf is trying it on and the hide is deciding (the weekly check); a champion wears his own.
	is_equipped = no
	artifact_owner ?= {
		is_ai = yes
		NOT = { has_trait = supernatural_hunter }

		NOT = {
			AND = {
				spn_is_wolf_trigger = yes
				is_adult = no
			}
		}
	}
}

# ---- The Hunt (§11.2) ----
# Root = a PROVINCE (is_location_valid, province_score); scope:host = the hunter. An AI hunter does not ride on a known
# champion whose card is ready unless he means it. A player is never stopped.
spn_hide_ai_hunter_deterred_trigger = {

# Root = the culprit; HUNTER = the hunter. Monster hunting 75 AND a reason: a zealous, vengeful or brave man; a rival
# or nemesis; a man whose family or trainer the culprit killed; or a name the world tells stories about. Used only
# with a province as root (the culprit is read back from it for the killer test).
spn_hunter_means_it_trigger = {
	$HUNTER$ = {
		trigger_if = {
			limit = { has_trait = supernatural_hunter }
			has_trait_xp = { trait = supernatural_hunter track = monster_hunting value >= 75 }
		}
		trigger_else = { always = no }
		OR = {
			has_trait = zealous
			has_trait = vengeful
			has_trait = brave
			has_relation_rival = prev
			has_relation_nemesis = prev
			spn_hunter_reputation_value >= 3
			var:spn_hunter_trainer ?= { killer ?= root.county.var:spn_menace_culprit }
			any_close_family_member = {
				even_if_dead = yes
				is_alive = no
				killer ?= root.county.var:spn_menace_culprit
			}
		}
	}
}

# ==================================================================================================
# 2.37o: WHY A WOLF CHALLENGES (Thomas, 4 Oct). Root = the would-be challenger. Read by the Challenge interaction's
# ai_will_do, the Challenge the Alpha intent's ai_will_do, the fire's challenger test (AI challengers only), and the
# lines that tell a player why he was challenged. Nothing here reads prev, root or a saved scope.
# ==================================================================================================

# He would win: his share of the clash at least EDGE to one, by the clash's own weights. Exactly
#   (65 + d + X_me) M_me >= EDGE x (65 - d + X_foe) M_foe,   d = clamp(my prowess - foe prowess, +-25),
# rearranged so the foe-dependent clamp sits on the right. The chance is A^2 / (A^2 + B^2): EDGE 0.8 is about 39%,
# 1 is 50%, 1.15 is 57%, 1.3 is 63%.
spn_ai_clash_edge_trigger = {
	spn_ai_clash_own_weight_value >= {
		value = 65
		add = $FOE$.spn_ai_clash_x_value
		subtract = {
			value = prowess
			subtract = $FOE$.prowess
			min = -25
			max = 25
		}
		multiply = $FOE$.spn_ai_clash_mult_value
		multiply = $EDGE$
		subtract = {
			value = {
				value = prowess
				subtract = $FOE$.prowess
				min = -25
				max = 25
			}
			multiply = spn_ai_clash_mult_value
		}
	}
}

# TOO WEAK FOR THE PACK. His own alpha, and one of: he would beat him (57%); the alpha's body is failing (a champion's
# never does); the pack holds the alpha lower than him by 15 (refusing and losing fights costs standing); a craven
# alpha; or an alpha whose name the hunters know, over a pack that still hides.
spn_ai_alpha_weak_trigger = {
	spn_ai_same_pack_trigger = { WHO = $ALPHA$ }
	OR = {
		spn_ai_clash_edge_trigger = { FOE = $ALPHA$ EDGE = 1.15 }
		$ALPHA$ = {
			OR = {
				has_trait = infirm
				has_trait = incapable
				has_trait = blind
			}
		}
		spn_pack_standing_value >= {
			value = $ALPHA$.spn_pack_standing_value
			add = 15
		}
		AND = {
			$ALPHA$ = { has_trait = craven }
			NOT = { has_trait = craven }
		}
		AND = {
			$ALPHA$ = {
				OR = {
					has_trait = spn_known_werewolf
					has_trait = spn_known_hybrid
				}
			}
			NOR = {
				has_trait = spn_known_werewolf
				has_trait = spn_known_hybrid
			}
		}
	}
}

# CRUEL, AND AT ODDS WITH THE BETTER OF THE PACK. His own alpha has one of the vices, he has none, and: the alpha's vice is the one
# his own nature stands against (compassionate against sadistic or callous, just against arbitrary, honest against
# deceitful); or he has come to dislike the alpha (-20); or two of the pack have.
spn_ai_alpha_cruel_trigger = {
	spn_ai_same_pack_trigger = { WHO = $ALPHA$ }
	spn_ai_cruel_wolf_trigger = no
	$ALPHA$ = { spn_ai_cruel_wolf_trigger = yes }
	OR = {
		AND = {
			has_trait = compassionate
			$ALPHA$ = {
				OR = {
					has_trait = sadistic
					has_trait = callous
				}
			}
		}
		AND = {
			has_trait = just
			$ALPHA$ = { has_trait = arbitrary }
		}
		AND = {
			has_trait = honest
			$ALPHA$ = { has_trait = deceitful }
		}
		opinion = {
			target = $ALPHA$
			value <= -20
		}
		spn_ai_pack_turned_trigger = { ALPHA = $ALPHA$ }
	}
}

# THE WHOLE TEST FOR AN ALPHA. His own: too weak or cruel. Another pack's, or a lone wolf's: only a feud (beating
# another pack's alpha takes nothing from that pack).
spn_ai_alpha_challenge_cause_trigger = {
	spn_ai_challenge_not_close_trigger = { FOE = $ALPHA$ }
	trigger_if = {
		limit = { spn_ai_same_pack_trigger = { WHO = $ALPHA$ } }
		OR = {
			spn_ai_alpha_weak_trigger = { ALPHA = $ALPHA$ }
			spn_ai_alpha_cruel_trigger = { ALPHA = $ALPHA$ }
		}
	}
	trigger_else = {
		spn_ai_feud_trigger = { FOE = $ALPHA$ }
	}
}

}
