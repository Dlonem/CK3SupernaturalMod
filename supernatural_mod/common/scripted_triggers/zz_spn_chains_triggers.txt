# 2.39 (WORK-ORDER-2.39 F6): held in the Warded Chains. Root = the prisoner. A wolf (werewolf or hybrid, so never a pure
# vampire: to a vampire they are only chains) whose gaoler OWNS the chains, in any slot, equipped or not ("merely has it").
# Read by: Break the Chains (greyed), Release Me (greyed, a hybrid), Compel Mortal's Leave Prison (greyed), both wolf forms
# of the Shapeshift interaction (greyed), the hunt's "Let it out" (hidden), and spn_hide.0400's re-check.
spn_held_in_chains_trigger = {
	is_imprisoned = yes
	spn_is_wolf_trigger = yes                                       # zz_spn_hide_triggers.txt
	imprisoner ?= {
		any_character_artifact = { var:spn_relic ?= flag:chains }  # the relic idiom (zz_spn_order_decisions.txt, the Powder)
	}
}
