# Generalized "actor can physically reach the recipient" check (Force Gift now; reusable for future
# melee/touch-based options). Mirrors the contact conditions of has_physical_contanct2 but is our own,
# WITHOUT the magic-specific unlimited_spells_range game rule. Call inside scope:actor with scope:recipient set.
spn_has_physical_contact = {
	OR = {
		AND = {
			scope:actor = {
				OR = {
					is_landed_or_landless_administrative = yes
					AND = { is_in_army = yes  scope:recipient = { is_in_army = yes } }
					AND = { is_landless_ruler = yes  scope:recipient = { is_landless_ruler = yes } }
					AND = {
						exists = involved_activity
						exists = scope:recipient.involved_activity
						involved_activity = scope:recipient.involved_activity
					}
				}
			}
			trigger_if = {
				limit = { exists = scope:actor.location  exists = scope:recipient.location }
				scope:actor.location = scope:recipient.location
			}
		}
		# GENUINE proximity / in-your-power only (no relationship shortcuts - a hostile clawing
		# needs you to actually reach them, not just be their distant friend).
		scope:recipient = { is_vassal_of = scope:actor  NOT = { is_at_war_with = scope:actor } }
		scope:recipient = { is_imprisoned_by = scope:actor }
		scope:recipient = { is_courtier_of = scope:actor }
		scope:recipient = { is_foreign_court_or_pool_guest_of = scope:actor }
		# A neighbouring ruler is "right next to you" - within reach to claw (your realm + the realms bordering it).
		scope:actor = { character_is_realm_neighbor = scope:recipient }
	}
}

# Can $HOST$ plausibly know what this character is? Gates feeding-activity invitations.
# Evaluated in the CANDIDATE's scope. $HOST$ is the inviter.
#   ONLY IF: they carry the known trait openly, OR the host actually knows their secret,
#            OR they are the host's spouse or close family - those people already know.
#   NEVER IF: none of the above. An inviter cannot know what nobody has told them.
spn_host_knows_supernatural_trigger = {
	OR = {
		has_trait = spn_known_vampire
		has_trait = spn_known_hybrid
		any_secret = {
			type = spn_secret_vampire
			is_known_by = $HOST$
		}
		any_secret = {
			type = spn_secret_hybrid
			is_known_by = $HOST$
		}
		is_spouse_of = $HOST$
		is_close_family_of = $HOST$
		# 2.32 -- THE SIRE BOND, and it is deliberately NOT a secret check.
		# A maker knows what their progeny is because they made them. Routing that through
		# "the host knows their secret" would be both wrong and harmful: holding someone's
		# secret is exactly what enables exposure and blackmail, and a sire needs neither to
		# know. Both directions, because either of them can be the one throwing the evening.
		has_relation_mmaker = $HOST$
		has_relation_pprogeny = $HOST$
	}
}

spn_meets_dist_criteria = {
	# 2.31c -- the game-start sweep and the yearly pulse both select through this, and the creature
	# traits they hand out are gated four events downstream. Selecting a target that will be refused
	# is not free: spn_initiate.0006/.1006/.3006 call add_trait_xp unconditionally beside the
	# trigger_event, so a refused pick still writes XP for a trait nobody has. Filter the pick.
	spn_is_valid_supernatural_target = yes
	is_ai = yes
	is_adult = yes
	spn_has_supernatural_trait = no
	NOT = { has_trait = supernatural_hunter }
	highest_held_title_tier <= $TIER$
	health >= 3.5
	spn_total_skill >= 40
}

spn_possible_pregnancy = {
	NOT = { #Inheritance is off and one of the characters has a spn trait
		has_game_rule = spn_no_trait_inheritance
	}
	OR = {
		AND = { #Inheritance is on or Supernaturals Create Humans
			OR = {
				has_game_rule = spn_yes_trait_inheritance
				has_game_rule = spn_human_no_immortal_trait_inheritance
			}
			OR = {
				spn_has_supernatural_trait = yes
				$FATHER$ = { spn_has_supernatural_trait = yes }
			}
		}
		AND = { #Inheritance is Supernaturals Only
			has_game_rule = spn_immortal_only_trait_inheritance
			AND = {
				spn_has_supernatural_trait = yes
				$FATHER$ = { spn_has_supernatural_trait = yes }
			}
		}
		AND = { #Inheritance is Only Via Humans
			has_game_rule = spn_human_only_trait_inheritance

				AND = {
					spn_has_supernatural_trait = yes

					spn_has_supernatural_trait = no

#Determines whether a Demon pregnancy is allowed, based on the spn_enable_demon_inheritance game rule.
#ROOT is the mother, $FATHER$ is the biological father.
#Demon = is_a_demon_trigger (demon1 / demon2). Human = spn_has_supernatural_trait = no.
#This is intentionally independent of the werewolf spn_enable_trait_inheritance rule.
spn_demon_possible_pregnancy = {
	NOT = { has_game_rule = spn_demon_no_inheritance }
	OR = {
		AND = { #Demon + Demon (allowed under On and Demons Only)
			OR = {
				has_game_rule = spn_demon_yes_inheritance
				has_game_rule = spn_demon_immortal_only_inheritance
			}
			is_a_demon_trigger = yes
			$FATHER$ = { is_a_demon_trigger = yes }
		}
		AND = { #Demon + Human, either direction (allowed under On and Only Via Humans)
			OR = {
				has_game_rule = spn_demon_yes_inheritance
				has_game_rule = spn_demon_human_only_inheritance
			}
			OR = {
				AND = {
					is_a_demon_trigger = yes
					$FATHER$ = { spn_has_supernatural_trait = no }
				}
				AND = {
					spn_has_supernatural_trait = no
					$FATHER$ = { is_a_demon_trigger = yes }
				}
			}
		}
	}
}

# 2.32 -- ONE definition of "can I drink this person?", shared by every feeding site.
# Evaluated in the CANDIDATE's scope; root is the feeder.
#
# The per-site filters had drifted apart. Most excluded vampires and werewolves but not
# supernatural_hybrid and not supernatural_hunter, so your own court could hand you a
# hybrid or a monster hunter as prey. And only supernatural_ev1.26 routed through
# spn_is_valid_supernatural_target -- the AGOT patch's species hook -- so everywhere else,
# with the patch loaded, you could drink from one of the walking dead.
vf_valid_victim_trigger = {
	is_adult = yes
	is_alive = yes
	is_imprisoned = no
	NOT = { this = root }
	NOR = {
		has_trait = supernatural_vampire
		has_trait = supernatural_werewolf
		has_trait = supernatural_hybrid
		has_trait = supernatural_hunter
	}
	# `always = yes` in standalone play, overridden by the AGOT patch. The dead have no blood.
	spn_is_valid_supernatural_target = yes
}
