# Generalized "actor can physically reach the recipient" check (Force Gift now; reusable for future
# melee/touch-based options). Mirrors the contact conditions of has_physical_contanct2 but is our own,
# WITHOUT the magic-specific unlimited_spells_range game rule. Call inside scope:actor with scope:recipient set.
spn_has_physical_contact = {
	OR = {
		AND = {
			scope:actor = {
				OR = {
					is_landed_or_landless_administrative = yes
					AND = { is_in_army = yes  scope:recipient = { is_in_army = yes } }
					AND = { is_landless_ruler = yes  scope:recipient = { is_landless_ruler = yes } }
					AND = {
						exists = involved_activity
						exists = scope:recipient.involved_activity
						involved_activity = scope:recipient.involved_activity
					}
				}
			}
			trigger_if = {
				limit = { exists = scope:actor.location  exists = scope:recipient.location }
				scope:actor.location = scope:recipient.location
			}
		}
		# GENUINE proximity / in-your-power only (no relationship shortcuts - a hostile clawing
		# needs you to actually reach them, not just be their distant friend).
		scope:recipient = { is_vassal_of = scope:actor  NOT = { is_at_war_with = scope:actor } }
		scope:recipient = { is_imprisoned_by = scope:actor }
		scope:recipient = { is_courtier_of = scope:actor }
		scope:recipient = { is_foreign_court_or_pool_guest_of = scope:actor }
		# A neighbouring ruler is "right next to you" - within reach to claw (your realm + the realms bordering it).
		scope:actor = { character_is_realm_neighbor = scope:recipient }
	}
}

# Can $HOST$ plausibly know what this character is? Gates feeding-activity invitations.
# Evaluated in the CANDIDATE's scope. $HOST$ is the inviter.
#   ONLY IF: they carry the known trait openly, OR the host actually knows their secret,
#            OR they are the host's spouse or close family - those people already know.
#   NEVER IF: none of the above. An inviter cannot know what nobody has told them.
spn_host_knows_supernatural_trigger = {
	OR = {
		has_trait = spn_known_vampire
		has_trait = spn_known_hybrid
		any_secret = {
			type = spn_secret_vampire
			is_known_by = $HOST$
		}
		any_secret = {
			type = spn_secret_hybrid
			is_known_by = $HOST$
		}
		is_spouse_of = $HOST$
		is_close_family_of = $HOST$
		# 2.32 -- THE SIRE BOND, and it is deliberately NOT a secret check.
		# A maker knows what their progeny is because they made them. Routing that through
		# "the host knows their secret" would be both wrong and harmful: holding someone's
		# secret is exactly what enables exposure and blackmail, and a sire needs neither to
		# know. Both directions, because either of them can be the one throwing the evening.
		has_relation_mmaker = $HOST$
		has_relation_pprogeny = $HOST$
	}
}

				has_game_rule = spn_demon_yes_inheritance

			}
			OR = {
				AND = {
					is_a_demon_trigger = yes

				}
				AND = {
					spn_has_supernatural_trait = no

# Is this character TRAINER's apprentice? Either the training "not yet" state (spn_train_hunter, D2) or the
# bite-chain deferral (10_spn_bite_incubation.txt:824-833). Never a hunter already.
spn_is_apprentice_of_trigger = {
	NOT = { has_trait = supernatural_hunter }
	# 2.35a -- THE SAME FLOOR, on the third and last way into the trait. Gating the two interactions is not
	# enough on its own: the bite-chain deferral (10_spn_bite_incubation.txt) can leave a character carrying
	# spn_hunter_training_deferred without either interaction having been used, and this trigger is what
	# graduates an apprentice at the end of a hunt. A child who got here another way stops here.
	age >= 13
	OR = {
		AND = {
			has_character_flag = spn_hunter_apprentice
			var:spn_hunter_trainer ?= $TRAINER$
		}
		AND = {
			has_character_flag = spn_hunter_training_deferred
			var:spn_deferred_hunter_trainer ?= $TRAINER$
		}
	}
}

# A vampire, werewolf, hybrid -- or, since 2.35 PART TWO, a demon. The four things a Supernatural Hunt can be
# about. Still deliberately NOT spn_has_supernatural_trait: that also matches a witch, and a witch is hunted
# through the Witch kind on evidence rather than tracked through the field on a body (E1).
#
# The two guards are borrowed verbatim from banish_demon's own exclusions (00_magic_interactions.txt:1084) --
# an endgame-crisis demon is upstream's set piece and nobody's quarry, and a demon summoned in the last seven
# days is still standing in the circle somebody drew for it.
#
# is_a_demon_trigger (magic_triggers.txt:1209) is demon1 | demon2 and is upstream's, verified, unmodified.
# demon0 is deliberately NOT a demon here and never has been: a latent child is not a menace, and The Thing in
# the Cradle exists precisely because it is not one yet.
spn_hunt_is_creature_trigger = {
	OR = {
		has_trait = supernatural_vampire
		has_trait = supernatural_werewolf
		has_trait = supernatural_hybrid
		AND = {
			is_a_demon_trigger = yes
			NOT = { has_character_flag = summoned_by_abdul_flag }
			NOT = { has_character_flag = just_summoned_demon_flag }
		}
	}
}

# 2.35 PART TWO (§4.1) -- AND A DEMON HUNTS ALONE. Thomas: "It's not going to join a pack."
#
# Widening the creature trigger above would otherwise have put demons into the pack count for free, which is
# exactly the kind of unasked-for ripple the widening was supposed to be checked for. This is the trigger the

spn_hunt_joins_a_pack_trigger = {
	spn_hunt_is_creature_trigger = yes
	NOT = { is_a_demon_trigger = yes }
}

# 2.35 PART TWO (§5.4) -- MADE, OR SUMMONED. There is a person under a made demon and there is nothing under a
# summoned one, and this is the only question the exorcism's outcome turns on.
#
# PRIMARY: spn_was_mortal, one added line in upstream's make_demon_effect (§12), which every made-demon path
# runs -- the demon0 maturation (magic_ev3.75, magic_ev2.38), the magic eruption, the Ascendancy project,
# Abraxas's courtier. A summoned demon is created from a character template and never touches that effect.
#
# FALLBACK, written in beside it because a flag added this release cannot exist on a demon that was made in a
# save started before it: not of the Hell dynasty, and has a father. Every summoned template is house_hell or
# fatherless; a man who was turned has both a house of his own and parents.
spn_hunt_demon_was_mortal_trigger = {
	OR = {
		has_character_flag = spn_was_mortal
		AND = {
			NOT = { dynasty ?= dynasty:hell_dynasty }
			exists = father
		}
	}
}

# ACTIVITY scope. 2.35 PART TWO -- IS THIS HUNT FOLLOWING A TRAIL?
#
# The tracking pool's events used to ask `has_activity_option = spn_hunt_kind_monster`, which was the right
# question when the Monster kind was the only one that ever had a culprit. A Witch Hunt ridden to a witch
# trail now has one too, and it runs the same spine -- so a pool event about survivors, graves, mob justice,
# a crowded county, a compelled county or a fellow hunter should fire on it as well.
#
# The kind-SPECIFIC events (Two Different Animals, By the Moon, Not a Drop, Nothing Was Taken) keep their own
# var:spn_hunt_kind gate and are excluded by that, correctly.
spn_hunt_runs_the_trail_trigger = {
	OR = {
		has_activity_option = { category = special_type option = spn_hunt_kind_monster }
		has_variable = spn_hunt_witch_trail
	}
}

spn_hunt_culprit_in_reach_trigger = {
	# 2.35f -- GUARDED. As siblings these threw both halves of the same mistake (error.log 9 Sep 14:16):
	#   "squared_distance [ Expected a province or landed title scope, got 'Thomas Schwab' ]"
	#   "squared_distance [ province (target province/barony/county) was null ]"
	# The `exists = location` line does not gate the line under it -- CK3 evaluates every sibling of an AND
	# -- so a culprit with no location still had its location dereferenced, and the trigger was also being
	# reached with a CHARACTER as root instead of the activity province it documents. trigger_if gates both.
	trigger_if = {
		limit = {
			exists = scope:spn_culprit
			scope:spn_culprit = { exists = location }
		}
		# 2.35h -- AND THE STRING FORM WAS THE REST OF IT. Inside a raw "..." expression, `root` is the
		# SCRIPT root -- the hunter -- not the scope this trigger was called in, so the comparison was
		# handed a character even though the caller scopes correctly into the activity's province
		# (zz_spn_hunt_effects.txt:254). "Expected a province or landed title scope, got 'Janyra
		# Waynwood'", 14 times in one observer run. The block form asks the question from the scope it is
		# standing in and needs no root at all -- vanilla's own idiom, laamp_nm_contracts.txt:29-34.
		squared_distance = {
			target = scope:spn_culprit.location
			value <= spn_max_hunt_distance
		}
	}
	trigger_else = { always = no }
}
