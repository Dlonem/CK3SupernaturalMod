# Supernatural 2.34 -- memory helpers.

# Would vanilla's RelationToMeShort give a real word for $OTHER$ (my son, my lover, my vassal, my courtier),
# or would it fall through to "acquaintance"? The maker's memory (spn_made_a_vampire_memory) uses the
# relationship word when there is one and the plain name when there is not -- Thomas's brief:
# "my friend, my lover, my however", falling back to the name.
# Scope: a character. OTHER = the other character (an event target).
spn_relation_word_exists_trigger = {
	OR = {
		is_close_or_extended_family_of = $OTHER$
		is_spouse_of = $OTHER$
		is_concubine_of = $OTHER$
		$OTHER$ = { is_concubine_of = prev }
		has_relation_lover = $OTHER$
		has_relation_soulmate = $OTHER$
		has_relation_friend = $OTHER$
		has_relation_best_friend = $OTHER$
		has_relation_rival = $OTHER$
		has_relation_nemesis = $OTHER$
		has_relation_ward = $OTHER$
		has_relation_guardian = $OTHER$
		has_relation_mentor = $OTHER$
		has_relation_student = $OTHER$
		has_relation_bully = $OTHER$
		has_relation_victim = $OTHER$
		is_vassal_or_below_of = $OTHER$
		is_liege_or_above_of = $OTHER$
		is_courtier_of = $OTHER$
		$OTHER$ = { is_courtier_of = prev }
		is_imprisoned_by = $OTHER$
		$OTHER$ = { is_imprisoned_by = prev }
	}
}

# Did this character rise on a battlefield? Four memory types share the beat (see zz_spn_memories.txt).
spn_rose_in_battle_any_trigger = {
	OR = {
		any_memory = { has_memory_type = spn_rose_in_battle_memory }
		any_memory = { has_memory_type = spn_rose_in_battle_maker_memory }
		any_memory = { has_memory_type = spn_rose_in_battle_feller_memory }
		any_memory = { has_memory_type = spn_rose_in_battle_both_memory }
	}
}
