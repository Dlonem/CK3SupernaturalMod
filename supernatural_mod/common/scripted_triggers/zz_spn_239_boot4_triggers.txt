# Supernatural 2.39 -- Boot 4 (16 Oct; tool/pkgs/PB/HANDOFF.md §11). Where a temple's hall can be: read by
# spn_order_hall_home_effect and the chapter's year, step 2c (zz_spn_collegium_effects.txt).

# PROVINCE scope. A Collegium that no temple meets beneath (the adoption step 2c makes).
spn_order_hall_free_trigger = {
	has_building = spn_order_collegium_01
	NOT = { exists = var:spn_order_hall_of }
	NOT = { has_variable = spn_order_hall_let_go } # 2.39 (final pass, 17 Oct): a hall a temple let go stays bricked up (20 years)
}

# PROVINCE scope. A holding that could take a Collegium now: the seed's own tests (spn_order_hall_seed_site_trigger: a free
# slot, in a castle, city or temple, the only holding types whose buildings list carries it), none there yet, and nothing
# being built.
spn_order_hall_room_trigger = {
	free_building_slots > 0
	has_ongoing_construction = no
	OR = {
		has_holding_type = castle_holding
		has_holding_type = city_holding
		has_holding_type = church_holding
	}
	NOT = { has_building = spn_order_collegium_01 }
	NOT = { exists = var:spn_order_hall_of }
}

# CHARACTER scope (the lord above the temple). The Gnostic Council pays for a hall only under an AI lord who could hold one
# (the seed's character tests).
spn_order_hall_council_lord_trigger = {
	is_alive = yes
	is_ai = yes
	NOT = { government_has_flag = government_is_tribal }
	NOT = { government_has_flag = government_is_nomadic }
	NOT = { government_has_flag = government_is_wanua }
}
