mr_enabled_option1_trigger = {
	trigger_if = {
		limit = {
			 NOT = { exists = var:drawn_option1 }
		}
		always = no
	}
	trigger_else_if = {
		limit = {
			var:drawn_option1 = flag:ascendancy
		}
		has_trait = source
	}
	trigger_else_if = {
		limit = {
			var:drawn_option1 = flag:lich
		}
		custom_description = {
			text = not_necromancer_nor_corrupted
			OR = {
				has_trait = necromancer
				has_trait = corrupted
			}
		}
	}
	trigger_else_if = {
		limit = {
			var:drawn_option1 = flag:soul_transfer
		}
		custom_description = {
			text = legendary_mage_for_soul_transfer
			is_legendary_mage_trigger = yes
			NOT = {
				has_character_modifier = no_soul_modifier
			}
		}
	}
	trigger_else = {
		always = yes
	}
}

may_visit_witch_house_trigger = {
	NOT = {
		has_character_modifier = mr_visited_yaga_hut_modifier
	}
	has_knowledge_witch_house_trigger = yes
}
has_knowledge_witch_house_trigger = {
	OR = {
		trigger_if = {
			limit = {
				exists = dynasty
			}
			dynasty = {
				has_dynasty_perk = witchcraft_legacy_3
			}
		}
		has_character_flag = knowledge_witch_house_flag
		has_witch_house_trigger = yes
		AND = {
			is_independent_ruler = yes
			any_realm_province = {
				has_building_or_higher = witch_house
			}
		}
	}
}
