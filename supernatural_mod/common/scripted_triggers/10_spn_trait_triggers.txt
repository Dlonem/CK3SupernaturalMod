# TRUE if the character is suffering an illness, grievous wound, or wasting condition.
# Used by the gift interactions: someone staring down their own death has a real,
# personal reason to gamble on the curse, no matter how grim the odds.
spn_desperate_for_a_cure_trigger = {
	OR = {
		health < 3
		has_trait = ill
		has_trait = pneumonic
		has_trait = great_pox
		has_trait = early_great_pox
		has_trait = lovers_pox
		has_trait = leper
		has_trait = wounded_2
		has_trait = wounded_3
		has_trait = infirm
		has_trait = incapable
		has_trait = gout_ridden
		has_trait = consumption
		has_trait = cancer
		has_trait = typhus
		has_trait = bubonic_plague
		has_trait = smallpox
		has_trait = sickly
	}
}
