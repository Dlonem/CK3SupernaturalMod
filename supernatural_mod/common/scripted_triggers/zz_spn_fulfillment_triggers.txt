# Supernatural 2.36 -- who can be charged a Spiritual Fulfillment loss by common/scripted_effects/zz_spn_fulfillment_effects.txt.
# Character scope.
#
# Playable, because change_spiritual_fulfillment errors on anyone else (vanilla, spiritual_counsel_interaction.txt), and not
# cynical: a cynic has stopped caring what his faith condemns. The -75% / -25% multipliers would cancel for him under a
# Rite that outlaws his kind, but other modifiers can push that total either side of -100%, so it is decided here instead.
spn_weighs_loss_trigger = {
	is_playable_character = yes
	NOT = { has_trait = cynical }
}
