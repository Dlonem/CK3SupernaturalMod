# Does this character know $WHO$ is of the Order? (T17: "members you know".) In his own scope.
spn_order_knows_member_trigger = {
	OR = {
		this = $WHO$
		trigger_if = {
			limit = {
				exists = var:spn_order_chapter
				$WHO$ = { exists = var:spn_order_chapter }
			}
			var:spn_order_chapter = $WHO$.var:spn_order_chapter
		}
		trigger_else = { always = no }
		any_known_secret = {
			secret_type = spn_secret_blue_rose
			secret_owner = $WHO$
		}
	}
}

# A candidate for the Selection (§5.4). The cheap tests first: the walk's own limit is the same OR.
spn_order_selection_reach_trigger = {
	OR = {
		has_trait = source
		has_trait = source2
		has_trait = archmage
		AND = {
			age >= 12
			age <= 30
			learning >= 8 # 2.38e: was 12 (Thomas, 9 Oct); the 73-year run found no member's child in reach
			OR = {
				has_character_flag = spn_order_legacy # 2.38c: the child of a member who has died
				any_parent = { spn_is_order_member_trigger = yes }
			}
		}
		# 2.38b: witches and scholars, very picky (DESIGN-2.38b §6). The cheap tests first in each.
		AND = {
			learning >= 12
			is_adult = yes
			OR = {
				has_trait = witch
				has_trait = true_witch
				any_secret = { type = secret_witch }
			}
			NOT = { has_trait = callous }
			NOT = { has_trait = deceitful }
			spn_order_fit_value >= 10
		}
		AND = {
			learning >= 16
			age >= 16
			age <= 40
			OR = {
				has_trait = education_learning_2
				has_trait = education_learning_3
				has_trait = education_learning_4
				has_trait = education_learning_5
			}
			spn_order_fit_value >= 30
		}
		# 2.38e: the Order grows through courtiers and wanderers, not kings (Thomas, 9 Oct: "I would only want the order to be
		# bigger if most of them were like courtiers and not like landed"). A learned scholar who holds no land is welcome on
		# easier terms than a lord; the landed branch above stays as strict as it was.
		AND = {
			learning >= 12
			is_landed = no
			age >= 16
			age <= 50
			OR = {
				has_trait = education_learning_3
				has_trait = education_learning_4
				has_trait = education_learning_5
			}
			spn_order_fit_value >= 20
		}
		# A player who heard the rumour and kept watch, or who was named by a member (DESIGN-2.38b §8).
		has_character_flag = spn_order_eyes_open
		has_character_flag = spn_order_sponsored
	}
}
spn_order_candidate_trigger = {
	spn_order_selection_reach_trigger = yes
	spn_is_order_member_trigger = no
	NOT = { exists = var:spn_order_chapter }
	NOT = { has_character_flag = spn_order_refused }
	NOT = { has_character_flag = spn_order_passed_over } # 2.38b: a player Temple Magus said not this one
	age >= 12
	is_imprisoned = no
	spn_order_can_join_trigger = yes
}

			}
