# Supernatural -- religion triggers (CK3 1.20).
#
# 1.20 moved every "is this trait a crime here?" trigger from the FAITH to the RITE: vanilla deleted
# trait_is_criminal_in_faith_trigger and its family and replaced them with *_in_rite_trigger versions that
# take RITE = <rite> and read rite_has_parameter. Characters and counties hold a Rite now, and a Faith can hold
# several Rites with different crime doctrines, so the Rite is the only place the answer is right.
#
# These are vanilla 1.20's own blocks (00_religious_triggers.txt), verbatim, with Supernatural's branches added
# for the four KNOWN creature traits -- the same extension the 1.19 versions of this file made to the faith
# triggers. Overriding vanilla's names (rather than shipping private copies) is deliberate: vanilla's own
# crime, secret and AI code calls these, so a known vampire is a criminal wherever vanilla asks, not only
# where Supernatural asks.
#
# The 1.19 *_in_faith_trigger copies are gone; every Supernatural caller now uses the rite versions.
# RE-DIFF AFTER EVERY CK3 UPDATE: these are vanilla shadows.

trait_is_criminal_in_rite_trigger = {
	OR = {
		#KINSLAYER
		AND = {
			OR = {
				$TRAIT$ = trait:kinslayer_3
				$TRAIT$ = trait:kinslayer_2
				$TRAIT$ = trait:kinslayer_1
			}
			$RITE$ = { rite_has_parameter = kinslaying_any_dynasty_member_crime }
		}
		AND = {
			OR = {
				$TRAIT$ = trait:kinslayer_3
				$TRAIT$ = trait:kinslayer_2
			}
			$RITE$ = { rite_has_parameter = kinslaying_extended_family_crime }
		}
		AND = {
			$TRAIT$ = trait:kinslayer_3
			$RITE$ = { rite_has_parameter = kinslaying_close_kin_crime }
		}
		#WITCH
		AND = {
			$TRAIT$ = trait:witch
			$RITE$ = { rite_has_parameter = witchcraft_illegal }
		}
		#CANNIBAL
		AND = {
			$TRAIT$ = trait:cannibal
			NOT = {
				$RITE$ = { rite_has_parameter = cannibalism_legal }
			}
		}
		#SODOMITE
		AND = {
			$TRAIT$ = trait:sodomite
			$RITE$ = { rite_has_parameter = homosexuality_illegal }
		}
		#DEVIANT
		AND = {
			$TRAIT$ = trait:deviant
			$RITE$ = { rite_has_parameter = deviancy_illegal }
		}
		#INCEST
		AND = {
			$TRAIT$ = trait:incestuous
			NOT = {
				$RITE$ = { rite_has_parameter = allows_unrestricted_marriage }
			}
		}
		#ADULTERER/FORNICATOR
		AND = {
			OR = {
				$TRAIT$ = trait:adulterer
				$TRAIT$ = trait:fornicator
			}
			OR = {
				AND = {
					$RITE$ = { rite_has_parameter = adultery_female_crime }
					$GENDER_CHARACTER$ = { is_female = yes }
				}
				AND = {
					$RITE$ = { rite_has_parameter = adultery_male_crime }
					$GENDER_CHARACTER$ = { is_male = yes }
				}
			}
		}
		# Devoted is criminal when monasticism is forbidden
		AND = {
			$TRAIT$ = trait:devoted
			$RITE$ = { rite_has_parameter = devoted_trait_criminal }
		}
		#VAMPIRE (Supernatural)
		AND = {
			$TRAIT$ = trait:spn_known_vampire
			$RITE$ = { rite_has_parameter = spn_vampirism_illegal }
		}
		#WEREWOLF (Supernatural)
		AND = {
			$TRAIT$ = trait:spn_known_werewolf
			$RITE$ = { rite_has_parameter = spn_lycanthropy_illegal }
		}
		#HYBRID (Supernatural)
		AND = {
			$TRAIT$ = trait:spn_known_hybrid
			$RITE$ = { rite_has_parameter = spn_hybrid_illegal }
		}
		#HUNTER (Supernatural)
		AND = {
			$TRAIT$ = trait:spn_known_hunter
			$RITE$ = { rite_has_parameter = spn_hunter_illegal }
		}
	}
}

has_negative_attitude_towards_trait_trigger = {
	# Direct trait comparison (was flag:$TRAIT$; that produced unsuppressable flag:trait:X log noise).
	NOR = {
		has_trait = $TRAIT$
		#INCESTUOUS
		AND = {
			$TRAIT$ = trait:incestuous
			any_secret = { secret_type = secret_incest }
		}
		#WITCH
		AND = {
			$TRAIT$ = trait:witch
			any_secret = { secret_type = secret_witch }
		}
		#VAMPIRE
		AND = {
			$TRAIT$ = trait:spn_known_vampire	
		    any_secret = { secret_type = spn_secret_vampire }
		}
		#WEREWOLF
		AND = {
			$TRAIT$ = trait:spn_known_werewolf	
			any_secret = { secret_type = spn_secret_werewolf }
		}
		#HYBRID
		AND = {
			$TRAIT$ = trait:spn_known_hybrid	
		    any_secret = { secret_type = spn_secret_hybrid }
		}
		#Hunter
		AND = {
			$TRAIT$ = trait:spn_known_hunter
		    any_secret = { secret_type = spn_secret_hunter }
		}
		#CANNIBAL
		AND = {
			$TRAIT$ = trait:cannibal
			any_secret = { secret_type = secret_cannibal }
		}
		#SODOMITE
		AND = {
			$TRAIT$ = trait:sodomite
			OR = {
				has_sexuality = homosexual
				has_sexuality = bisexual
			}
			any_secret = {
				secret_type = secret_homosexual
			}
		}
		#ADULTERER
		AND = {
			$TRAIT$ = trait:adulterer
			OR = {
				has_trait = lustful
				AND = {
					is_married = yes
					any_relation = { type = lover NOT = { is_spouse_of = prev } }
				}
				any_relation = {
					type = lover
					is_married = yes
					NOT = { is_spouse_of = prev }
				}
			}
		}
		#FORNICATOR
		AND = {
			$TRAIT$ = trait:fornicator
			OR = {
				has_trait = lustful
				any_relation = { type = lover always = yes }
			}
		}
	}

	rite = { save_temporary_scope_as = my_rite }
	OR = {
		trait_is_shunned_in_rite_trigger = { TRAIT = $TRAIT$ RITE = scope:my_rite GENDER_CHARACTER = $GENDER_CHARACTER$ }
		trait_is_criminal_in_rite_trigger = { TRAIT = $TRAIT$ RITE = scope:my_rite GENDER_CHARACTER = $GENDER_CHARACTER$ }
	}
}
