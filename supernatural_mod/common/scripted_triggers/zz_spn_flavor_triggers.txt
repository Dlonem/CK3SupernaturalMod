# Supernatural 2.39 -- flavor that is spread out, not bunched: Order romance, neophyte stress, the council's moments and the
# pack's night home (WORK-ORDER-2.39 F14). The gate itself (spn_flavor_ready_trigger) and the romance test
# (spn_romance_ok_trigger) are in 10_spn_triggers.txt.
#
# The chapter-moment selectors run from the chapter's year, where ROOT is the story, so no trigger here reads root.
# Each spn_order_cm_*_ok_trigger saves the candidate as scope:spn_cm_who first and reads that instead; the same trigger
# is the event's own re-check when it arrives, so selector and event can never disagree.

# ==================================================================================================
# SMALL TESTS
# ==================================================================================================
# 2.39 (F14): this character and $WHO$ hold the same rank in the Order. Ranks are flags, so each is compared by name.
spn_order_same_rank_trigger = {
	OR = {
		AND = {
			var:spn_order_rank ?= flag:neophyte
			$WHO$ = { var:spn_order_rank ?= flag:neophyte }
		}
		AND = {
			var:spn_order_rank ?= flag:acolyte
			$WHO$ = { var:spn_order_rank ?= flag:acolyte }
		}
		AND = {
			var:spn_order_rank ?= flag:medicum
			$WHO$ = { var:spn_order_rank ?= flag:medicum }
		}
		AND = {
			var:spn_order_rank ?= flag:magistratus
			$WHO$ = { var:spn_order_rank ?= flag:magistratus }
		}
	}
}

# 2.39 (F14): a junior of the Order: a Neophyte, an Acolyte or a Medicum, and not a Temple Magus.
spn_order_is_junior_trigger = {
	exists = var:spn_order_chapter
	OR = {
		var:spn_order_rank ?= flag:neophyte
		var:spn_order_rank ?= flag:acolyte
		var:spn_order_rank ?= flag:medicum
	}
	spn_is_temple_magus_trigger = no
}

# 2.39 (F14 §2): this character is $JUNIOR$'s patron: the junior's own Temple Magus, or the Grand Magus. Seat holders are
# partners in the moments, never patrons: they do not decide promotions.
spn_order_patron_of_trigger = {
	save_temporary_scope_as = spn_pt_patron
	spn_is_order_member_trigger = yes
	NOT = { this = $JUNIOR$ }
	$JUNIOR$ = {
		spn_order_is_junior_trigger = yes
		OR = {
			var:spn_order_chapter ?= { story_owner = scope:spn_pt_patron }
			global_var:current_archmage_global ?= scope:spn_pt_patron
		}
	}
}

# 2.39 (F14 §2): a member who could have seen $JUNIOR$'s favour: an AI member of the junior's rank in his temple, neither
# the junior nor the patron. Root = the candidate witness.
spn_order_favour_peer_trigger = {
	is_alive = yes
	is_ai = yes
	is_imprisoned = no
	NOT = { this = $JUNIOR$ }
	NOT = { this = $PATRON$ }
	spn_order_same_rank_trigger = { WHO = $JUNIOR$ }
}
# ...and one who resents him: his rival, or thinks -10 or less of him.
spn_order_favour_jealous_trigger = {
	spn_order_favour_peer_trigger = { JUNIOR = $JUNIOR$ PATRON = $PATRON$ }
	OR = {
		has_relation_rival = $JUNIOR$
		opinion = {
			target = $JUNIOR$
			value <= -10
		}
	}
}

# THE SUBJECTS OF THE NEW MOMENTS. Root = the candidate; the parameter is the player the moment is for.

# R1, After the Others Have Gone: an AI junior of the senior's temple who wants him. Never his tutee (Late Lessons is
# that pair), never already his lover or spouse.
spn_order_advance_junior_trigger = {
	is_alive = yes
	is_ai = yes
	is_imprisoned = no
	NOT = { this = $SENIOR$ }
	# Below him in rank (a seat holder can be of any rank).
	OR = {
		var:spn_order_rank ?= flag:neophyte
		AND = {
			var:spn_order_rank ?= flag:acolyte
			$SENIOR$ = { spn_order_medicum_plus_trigger = yes }
		}
		AND = {
			var:spn_order_rank ?= flag:medicum
			$SENIOR$ = { var:spn_order_rank ?= flag:magistratus }
		}
	}
	NOT = { var:spn_order_tutor ?= $SENIOR$ }
	NOT = { is_spouse_of = $SENIOR$ }
	NOT = { has_relation_lover = $SENIOR$ }
	OR = {
		has_trait = lustful
		has_trait = ambitious
		has_trait = gregarious
		has_trait = seducer
		has_trait = rakish
	}
	opinion = {
		target = $SENIOR$
		value >= 25
	}
	spn_romance_ok_trigger = { CHARACTER = $SENIOR$ }
	trigger_if = {
		limit = {
			OR = {
				is_married = yes
				any_relation = { type = lover }
			}
		}
		might_cheat_on_every_partner_trigger = yes
	}
}

# R2, A Door Left Open: an AI senior the junior might say it to. Never his tutor (Late Lessons is that pair).

	NOT = { is_spouse_of = $JUNIOR$ }
	NOT = { has_relation_lover = $JUNIOR$ }
	spn_romance_ok_trigger = { CHARACTER = $JUNIOR$ }
	trigger_if = {
		limit = {
			OR = {
				is_married = yes
				any_relation = { type = lover }
			}
		}
		might_cheat_on_every_partner_trigger = yes
	}
}

# T1, A Brother Asks for Help: an AI member of the Temple Magus's temple who broke the rules and has not been judged.
spn_order_brother_asker_trigger = {
	is_alive = yes
	is_ai = yes
	is_imprisoned = no
	NOT = { this = $HEAD$ }
	spn_is_order_member_trigger = yes
	has_character_flag = spn_order_broke_rules
	NOT = { has_character_flag = spn_conclave_judged }
	# Never the sitting Grand Magus: the chair's charges are the Conclave's (M3) and Invocare's to read, among them an
	# exposed staged rift (F12), which no Temple Magus can quietly bury.
	NOT = { global_var:current_archmage_global ?= this }
}

# T2, A Feud in the Chapter: an AI member of the Temple Magus's temple who could be one of the feuding pair.
spn_order_feud_member_trigger = {
	is_alive = yes
	is_ai = yes
	is_imprisoned = no
	NOT = { this = $HEAD$ }
	spn_is_order_member_trigger = yes
}
# ...and the pair: rivals, or each thinks -25 or less of the other.
spn_order_feuding_trigger = {
	NOT = { this = $WITH$ }
	OR = {
		has_relation_rival = $WITH$
		AND = {
			opinion = {
				target = $WITH$
				value <= -25
			}
			$WITH$ = {
				opinion = {
					target = prev
					value <= -25
				}
			}
		}
	}
}
# ...a member of the feuding pair who has his other half in the same temple. Root = the candidate.
spn_order_feud_has_partner_trigger = {
	spn_order_feud_member_trigger = { HEAD = $HEAD$ }
	save_temporary_scope_as = spn_cf_first
	var:spn_order_chapter ?= {
		any_in_list = {
			variable = spn_order_members
			spn_order_feud_member_trigger = { HEAD = $HEAD$ }
			spn_order_feuding_trigger = { WITH = scope:spn_cf_first }
		}
	}
}
# T2, A Feud in the Chapter: this Temple Magus has two AI members at odds. Root = the Temple Magus.
spn_order_chapter_feud_trigger = {
	save_temporary_scope_as = spn_cf_head
	spn_is_temple_magus_trigger = yes
	var:spn_order_chapter ?= {
		any_in_list = {
			variable = spn_order_members
			spn_order_feud_has_partner_trigger = { HEAD = scope:spn_cf_head }
		}
	}
}
# T2, A Feud on the Council: two AI seat holders, $A$ and $B$ (global variables, so each may be unset).
spn_order_council_feud_pair_trigger = {
	$A$ ?= {
		is_alive = yes
		is_ai = yes
		save_temporary_scope_as = spn_cf_a
	}
	$B$ ?= {
		is_alive = yes
		is_ai = yes
		spn_order_feuding_trigger = { WITH = scope:spn_cf_a }
	}
}
# The Grand Magus's three seats, any two of them. Root = the Grand Magus.
spn_order_council_feud_trigger = {
	global_var:current_archmage_global ?= this
	OR = {
		spn_order_council_feud_pair_trigger = { A = global_var:current_loremaster_global B = global_var:current_black_mage_global }
		spn_order_council_feud_pair_trigger = { A = global_var:current_loremaster_global B = global_var:current_admin_global }
		spn_order_council_feud_pair_trigger = { A = global_var:current_black_mage_global B = global_var:current_admin_global }
	}
}

# T3, Where Do You Go at Night?: the member's AI spouse, at his side, not of the Order.

# THE PACK'S NIGHT HOME (F14 §4)
# ==================================================================================================
# A player wolf due The Pack Comes Home: adult and free, not in the last 18 months (D19: never two years running), the
# flavor gate open, and a pack to come home to.
spn_pack_home_due_trigger = {
	is_alive = yes
	is_ai = no
	is_adult = yes
	is_imprisoned = no
	NOT = { has_character_flag = spn_pack_home_seen }
	spn_flavor_ready_trigger = yes
	spn_pack_home_company_trigger = yes # 2.39 (Boot 1, 16 Oct): a pack, not a lone alpha
}
# 2.39 (Boot 1, 16 Oct): the night is a pack's. He and at least two other wolves of his pack, adult, alive and free:
# every opening reads "the pack came home", and "Speak at the fire" speaks to them (Thomas's run: it came to a lone alpha,
# "a pack party of one"). Character scope; he is counted himself.
spn_pack_home_company_trigger = {
	var:spn_pack ?= {
		any_in_list = {
			variable = spn_pack_members
			is_alive = yes
			is_adult = yes
			is_imprisoned = no
			count >= 3
		}
	}
}
# An AI wolf of the pack who can be one of the night's pair.
spn_pack_home_pair_trigger = {
	is_alive = yes
	is_ai = yes
	is_adult = yes
	is_imprisoned = no
}
