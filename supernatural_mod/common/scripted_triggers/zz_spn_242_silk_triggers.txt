# Supernatural 2.42 -- Merchants of the Night (WORK-ORDER-2.42 v3). Triggers.

# CHARACTER scope: a debtor. Has he refused, at least once, to repay a loan from $LENDER$?
# Vanilla 1.21 story_cycle_loan: var:lender, var:reminders counts refusals. Shared by Collect in Blood and Settle in Souls.
spn_debtor_refused_me_trigger = {
	any_owned_story = {
		story_type = story_cycle_loan
		var:lender ?= $LENDER$
		exists = var:reminders
		var:reminders >= 1
	}
}

# CHARACTER scope: a lender who does not die. He is the one who sees a debt pass from father to son.
spn_undying_lender_trigger = {
	OR = {
		has_trait = supernatural_vampire
		has_trait = supernatural_hybrid
		is_a_demon_trigger = yes
	}
}
