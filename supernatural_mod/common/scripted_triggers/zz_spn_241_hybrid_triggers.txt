# Supernatural 2.41 (WORK-ORDER-2.41 §6, C3 Option A) -- the werewolf door (spn_initiate.1001). As
# spn_can_become_creature_trigger, but demon blood is allowed: the wolf with demon blood.
spn_can_become_werewolf_trigger = {
	NOR = {
		has_trait = golem
		has_trait = yaga
	}
	spn_is_valid_supernatural_target = yes
}

# Supernatural 2.41 (WORK-ORDER-2.41 §5.2) -- the six-child cap. Read in spn_on_genetic_birth (10_spn_birthday_on_actions.txt),
# where the newborn is root and scope:mother / scope:real_father are the parents.
spn_hybrid_parent_at_child_cap_trigger = {
	OR = {
		scope:mother ?= {
			has_trait = supernatural_hybrid
			is_alive = yes # 2.41 audit: no variables on the dead
			has_variable = spn_number_of_supernatural_children
			trigger_if = {
				limit = { has_variable = spn_number_of_supernatural_children }
				var:spn_number_of_supernatural_children >= spn_hybrid_child_cap_value
			}
		}
		scope:real_father ?= {
			has_trait = supernatural_hybrid
			is_alive = yes # 2.41 audit: no variables on the dead
			has_variable = spn_number_of_supernatural_children
			trigger_if = {
				limit = { has_variable = spn_number_of_supernatural_children }
				var:spn_number_of_supernatural_children >= spn_hybrid_child_cap_value
			}
		}
	}
}
