# Supernatural 2.35 -- who you can hang it on.
#
# Thomas, 9 Sep 2026: "if you compelled a mortal to end their life, there would be a body. And that would be
# confusing. If anything, you could do that to get the hunters off your trail. That could be an opportunity
# there. Maybe at some point, it could be cool to have them pretend to be a werewolf or vampire to try and
# definitely get the hunters off your trail."
#
# So a compelled death is not simply a mark against you. Done well it is a mark against SOMEBODY ELSE, and
# the mark carries their kind -- so a vampire who dresses a body as a wolf's work sends the county's next
# hunter to a wolf's door. Done well but with nobody around to blame, it is no mark at all. Done badly it is
# exactly what it looks like.
#
# Character scope: the candidate. scope:spn_body_actor = the creature doing the dressing.
#
# 2.35 PART TWO -- was scope:actor. The family left the compel namespace and is reachable from any kill site
# now, so the scope it reads has to be one that exists outside a character interaction.
#
# Note that spn_hunt_is_creature_trigger took demons this release, so a demon standing in the county can be
# framed like anything else -- and the mark it gets carries a demon's kind (spn_body_blame_effect).

spn_body_frameable_trigger = {
	is_alive = yes
	is_adult = yes
	is_imprisoned = no
	spn_hunt_is_creature_trigger = yes
	NOT = { this = scope:spn_body_actor }
	# Not your own blood, and not the blood that made you. Whatever else a creature is, it does not send
	# hunters to its own line -- and the maker/progeny relations are the mod's spine for that line.
	NOT = { has_relation_pprogeny = scope:spn_body_actor }
	NOT = { has_relation_mmaker = scope:spn_body_actor }
	NOT = { has_relation_ww_clawed = scope:spn_body_actor }
	NOT = { has_relation_ww_clawed_by = scope:spn_body_actor }
	# Somebody already hiding cannot be found to be blamed, and somebody the county already hunts needs no
	# help from you.
	NOT = { has_character_modifier = spn_gone_to_ground_modifier }
}
