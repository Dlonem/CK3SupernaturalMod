# Supernatural 2.41 -- Glitterhoof (WORK-ORDER-2.41 §3.4). Character scope: the ruler.
spn_glitterhoof_eligible_trigger = {
	is_adult = yes
	OR = {
		has_trait = lunatic
		has_trait = possessed
	}
	NOT = { has_character_flag = spn_had_glitterhoof }
	# 2.41 boot (Thomas, 3 Nov): a landless adventurer may name the horse his Second (second_camp_officer, the camp's
	# second-in-command). A landed ruler names him Chancellor.
	OR = {
		AND = {
			is_landed = yes
			is_landless_adventurer = no
			highest_held_title_tier >= tier_county
		}
		is_landless_adventurer = yes
	}
	is_at_war = no
	is_incapable = no
	is_imprisoned = no
	trigger_if = {
		limit = { is_ai = yes }
		highest_held_title_tier >= tier_kingdom
	}
}

# CHARACTER scope: a horse at the player's court that the unlock can still be used on.
spn_glitterhoof_can_unlock_trigger = {
	has_trait = spn_horse
	has_variable = can_not_marry
	is_adult = yes
}

# CHARACTER scope. The horse a ruler named, as against a foal of his line. 2.41 audit: flagged at creation
# (spn_glitterhoof.0001); one named before 2.41 has no flag and is known by the marriage gate, which only he was created
# with (foals are born without it).
spn_is_the_glitterhoof_trigger = {
	has_trait = spn_horse
	is_alive = yes
	OR = {
		has_character_flag = spn_the_glitterhoof
		has_variable = can_not_marry
	}
}
