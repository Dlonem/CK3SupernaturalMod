# Supernatural 2.39 -- "The Hermetic Order of the Blue Rose, Part 2": the witness misfire (WORK-ORDER-2.39 F3).
#
# The four kinds of spell, for "The Spell Goes Wrong" (spn_vade.0600). Caster scope, reading var:spn_mf_spell: the spell
# number Witchcraft's set_spell_id gave the cast that misfired (its names: LastSpellCast, 01_magic_custom_loc.txt). The
# event reads them only in its immediate, while the variable still exists. A number on none of the first three lists is a
# utility spell (the fourth trigger), so a spell Witchcraft adds later falls there by itself.

# Disabling: a misfire may (rarely) not stop, and kill the target. Cast a Curse 2, Domination 3, Powerful Curse 5, Power
# Strike 9, Inception 32 and its kinds 321-329, Antimagic Seal 33, Psychic Surgery 38.
spn_vade_misfire_disabling_trigger = {
	OR = {
		var:spn_mf_spell = 2
		var:spn_mf_spell = 3
		var:spn_mf_spell = 5
		var:spn_mf_spell = 9
		var:spn_mf_spell = 32
		var:spn_mf_spell = 33
		var:spn_mf_spell = 38
		AND = {
			var:spn_mf_spell >= 321
			var:spn_mf_spell <= 329
		}
	}
}

# Already deadly: it turns on the caster or his court (wounds only). Banish Demon 6, Use Ancient Singularity 7, Death Wish
# 8, Plague Spell 13, Fire Storm 15, Use Raw Magic 18, Soul Transfer 19, Tornado 30, the curse on an enemy's lands 31,
# Earthquake 35.
spn_vade_misfire_deadly_trigger = {
	OR = {
		var:spn_mf_spell = 6
		var:spn_mf_spell = 7
		var:spn_mf_spell = 8
		var:spn_mf_spell = 13
		var:spn_mf_spell = 15
		var:spn_mf_spell = 18
		var:spn_mf_spell = 19
		var:spn_mf_spell = 30
		var:spn_mf_spell = 31
		var:spn_mf_spell = 35
	}
}

# Helpful: it overshoots. Healing 4, Aura Enhancement 10, Bless the Land 11, Hypercognition 14, Ritual of Greater
# Protection 23, Ritual of Life 28, Thorn Walls 34, Stop the Cataclysm 36.
spn_vade_misfire_helpful_trigger = {
	OR = {
		var:spn_mf_spell = 4
		var:spn_mf_spell = 10
		var:spn_mf_spell = 11
		var:spn_mf_spell = 14
		var:spn_mf_spell = 23
		var:spn_mf_spell = 28
		var:spn_mf_spell = 34
		var:spn_mf_spell = 36
	}
}

# Utility: everything else (Aura Reading 1, Shadow Step 12, Mana Siphon 16, Mind Scan 17, Triangulation 20, Summon Demon
# 21, Transmutation 22, Scrying 24, Summon Demonic Army 25, the Ultimate Spell 26, Change the Future 29, Teleportation 37).
spn_vade_misfire_utility_trigger = {
	spn_vade_misfire_disabling_trigger = no
	spn_vade_misfire_deadly_trigger = no
	spn_vade_misfire_helpful_trigger = no
}
