# resolves, and expires by itself at 200 days -- comfortably past the 75-115 day wait -- so a chain that dies

# fails open instead of locking the character out forever.

	icon = prowess_positive
	prowess = 1
	monthly_martial_lifestyle_xp_gain_mult = 0.1
}
