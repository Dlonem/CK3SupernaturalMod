# 1.20.1.2). Witchcraft's artifact crafting (00_creating_artifact_effects.txt) and one Witchcraft event (ev3) still hand
# them out, and artifacts in running saves carry them, so the mod keeps both names alive here.

# domain_tax_mult is still a valid modifier in 1.21 (vanilla uses it 28 times), so that one is unchanged.

# with artifact_fertility_growth_N (county_fertility_growth_mult, same numbers), and this follows that exact swap.
artifact_domain_tax_mult_2_modifier = { domain_tax_mult = 0.02 }
artifact_development_growth_12_modifier = { county_fertility_growth_mult = 0.24 }
