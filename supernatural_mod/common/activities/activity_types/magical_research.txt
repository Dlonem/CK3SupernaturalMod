magical_research = {
	is_shown = {
		#has_bp2_dlc_trigger = yes
		highest_held_title_tier > tier_barony
		is_adult = yes
		is_true_mage = yes
	}

	activity_group_type = grand

	can_start_showing_failures_only = {
		is_available_adult = yes
	}

	is_valid = {
		scope:host = {
			#Getting imprisoned will invalidate
			is_imprisoned = no
			#Becoming incapable will invalidate
			NOT = { is_incapable = yes }
		}
	}

	on_invalidated = {
		#incapable
		if = {
			limit = {
				scope:host = { is_incapable = yes }
			}
			scope:activity = {
				activity_type = { save_scope_as = activity_type }
			}
			scope:host = {
				trigger_event = activity_system.0330
			}
			every_attending_character = {
				limit = { NOT = { this = scope:host } }
				trigger_event = activity_system.0331
			}
		}
		# Host is imprisoned
		else_if = {
			limit = {
				scope:host = { is_imprisoned = yes } 
			}
			scope:host = { trigger_event = bp2_adult_education.0005 }
		}
	}

	on_host_death = {
		#cancel the activity by not setting a new host
		#free the students
		scope:activity = {
			every_attending_character = {
				limit = {
					has_character_flag = generated_character_uni
				}
				remove_character_flag = generated_character_uni
			}
		}
		# Send your entourage home
		scope:host = {
			current_travel_plan = {
				every_entourage_character = { set_location_to_default = yes }
			}
		}
	}

	###################
	# PHASES
	###################

	phases = {
		magical_research_phase = {
			is_predefined = yes

			on_phase_active = {
				scope:host = {
					#remove flags or other setup
					if = {
						limit = {
							exists = var:mr_final_option
						}
						remove_variable = mr_final_option
					}
					if = {
						limit = {
							exists = var:mr_assistant
						}
						remove_variable = mr_assistant
					}
				}
				if = {
					limit = { this = scope:host }
					scope:activity = {
						#Phase progression
						#progress_activity_phase_after = { months = 6 }	#used in the 'mr_launch_effect'
						#initial progress boost
						activity_location = { save_scope_as = location }
						
					}
					#Arrival event
					if = {
						limit = {
							scope:activity = {
								has_variable = activity_magic_expedition
							}
							scope:location = {
								has_building_or_higher = witch_house
							}
						}
						trigger_event = magic_ev11.153
					}
					else_if = {
						limit = {
							scope:activity = {
								has_variable = activity_magic_expedition
							}
							scope:location = {
								has_building_or_higher = horned_god_house
							}
						}
						trigger_event = magic_ev11.92
					}
					else_if = {
						limit = {
							scope:activity = {
								has_variable = activity_magic_expedition
							}
							scope:location = {
								has_building_or_higher = mage_house
							}
						}
						trigger_event = magic_ev11.80
					}
					else = {
						trigger_event = { on_action = magical_research_arrival_events }
					}
				}
			}

			on_monthly_pulse = {
				if = {
					limit = {
						this = scope:host
						exists = var:mr_final_option
						scope:activity = {
							NOT = { has_variable = activity_magic_expedition }
						}
					}
					trigger_event = { on_action = magical_research_destination_events }				
				}
			}

			on_end = {
				if = {
					limit = {
						this = scope:host
						scope:activity = {
							NOT = { has_variable = activity_magic_expedition }
						}
					}
					mr_resolve_research_effect = yes		
				}
			}
		}
		magical_research_calibration_phase = {
			is_predefined = yes
			on_enter_phase = {
				if = {
					limit = { this = scope:host }
					if = {
						limit = {
							scope:activity = {
								has_variable = activity_magic_expedition
							}
						}
						#send_interface_toast = {
						#	title = test
						#}
						scope:activity = {
							progress_activity_phase_after = { days = 1 }
						}
					}
				}
			}
			on_phase_active = {
				if = {
					limit = { this = scope:host }
					if = {
						limit = {
							scope:activity = {
								has_variable = activity_magic_expedition
							}
						}
						#send_interface_toast = {
						#	title = test2
						#}
						scope:activity = {
							progress_activity_phase_after = { days = 1 }
						}
					}
					else = {
						mr_consider_additional_reward_from_bt_effect = yes
					}
					
					#scope:activity = {
					#	#Phase progression
					#	progress_activity_phase_after = { days = 1 }
					#	#initial progress boost
					#	activity_location = { save_scope_as = location }
					#}
				}
			}

			on_monthly_pulse = {
				
			}

			on_end = {
				
			}
		}
