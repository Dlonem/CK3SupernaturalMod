# Supernatural 2.37 -- the Werewolf Gathering (WORK-ORDER-2.37 §14). Claims the empty stub. The shape of the mod's own
# vampire_feeding.txt: three phases, our own pool on_action (spn_gathering_pulse), the feeding's travel backgrounds and
# window characters. The Gathering needs no holding: the AI stays in its own realm, a player may go as far as the hunt.
werewolf_gathering = {
	is_shown = {
		OR = {
			is_landed_or_landless_administrative = yes
			is_landless_adventurer = yes
		}
		is_playable_character = yes
		is_available_adult = yes
		trigger_if = {
			limit = { is_ai = yes }
			is_at_war = no
		}
		spn_is_wolf_trigger = yes
		NOT = { has_character_modifier = spn_gone_to_ground_modifier }
	}

	can_start_showing_failures_only = {
		# 2.37e: the engine does not hold the host to the cooldown below; every vanilla activity with one checks it here.
		NOT = { is_activity_type_on_cooldown = werewolf_gathering }
		is_available = yes
	}

	is_valid = {
		NOT = { has_variable = activity_invalidated }
		scope:host = {
			OR = {
				is_landed_or_landless_administrative = yes
				is_landless_adventurer = yes
			}
			is_alive = yes
			is_imprisoned = no
			NOT = { is_incapable = yes }
		}
	}

	on_start = {
		scope:host = { trigger_event = spn_gathering.0001 }
	}

	on_invalidated = {
		if = {
			limit = { scope:host = { is_incapable = yes } }
			scope:activity = {
				activity_type = { save_scope_as = activity_type }
			}
			scope:host = { trigger_event = activity_system.0330 }
			every_attending_character = {
				limit = { NOT = { this = scope:host } }
				trigger_event = activity_system.0331
			}
		}
	}

	on_host_death = {
		scope:activity = {
			set_variable = { name = activity_invalidated }
		}
	}

	###################
	# COST
	###################
	cost = {
		gold = {
			add = {
				value = standard_camp_party_activity_cost
				desc = hunt_base_cost
			}
		}
	}

	ui_predicted_cost = {
		gold = {
			value = standard_camp_party_activity_cost
			divide = 5
			ceiling = yes
			multiply = 5
		}
	}
	cooldown = { years = 5 }

	wait_time_before_start = { weeks = 1 }
	max_guest_arrival_delay_time = { months = 3 }
	is_single_location = yes

	###################
	# AI VALUES (§14.5). Root = the character weighing it.
	###################
	ai_will_do = {
		value = 20
		if = { limit = { has_trait = gregarious } add = 20 }
		if = { limit = { has_trait = ambitious } add = 10 }
		if = { limit = { spn_is_alpha_trigger = yes } add = 30 }
		add = {
			value = spn_pack_standing_value
			divide = 5
		}
		if = { limit = { has_trait = shy } add = -20 }
		if = { limit = { has_trait = paranoid } add = -15 }
		if = { limit = { has_trait = content } add = -10 }
		# Alpha's Call the Packs (§6.5).
		if = { limit = { has_character_flag = spn_called_the_packs } add = 100 }
	}

	ai_check_interval = 60

	ai_will_select_province = {
		value = { value = 20 add = scope:score }
	}

	###################
	# PHASES (§14.2). The host's on_phase_active moves the clock; the Run's pool comes from spn_gathering_pulse.
	###################
	phases = {
		spn_gathering_calling_phase = {
			is_predefined = yes
			order = 1
			on_phase_active = {
				if = {
					limit = { this = scope:host }
					scope:activity = { progress_activity_phase_after = { weeks = 2 } }
					# The Pack Bows: a champion has come.
					if = {
						limit = {
							scope:activity = {
								any_attending_character = { spn_is_champion_trigger = yes }
							}
						}
						scope:activity = {
							random_attending_character = {
								limit = { spn_is_champion_trigger = yes }
								save_scope_as = spn_bow_champ
							}
							every_attending_character = {
								limit = {
									is_ai = no
									NOT = { this = scope:spn_bow_champ }
								}
								trigger_event = spn_gathering.0110
							}
						}
					}
				}
			}
		}

		spn_gathering_run_phase = {
			is_predefined = yes
			order = 2
			# 2.37b -- on_enter_phase ADDED, the hunt's 2.35 lesson (supernatural_hunt.txt, spn_confrontation_phase).
			# A phase after the first is entered PASSIVE and waits until something calls progress_activity_phase_after;
			# only then is it made active and on_phase_active runs. Without this the Gathering sat on "waiting for the
			# Run" forever: Thomas's 3 Oct save, passive since 1082.9.2, seven months.
			on_enter_phase = {
				if = {
					limit = { this = scope:host }
					scope:activity = { progress_activity_phase_after = { days = 1 } }
				}
			}
			on_phase_active = {
				if = {
					limit = { this = scope:host }
					scope:activity = { progress_activity_phase_after = { weeks = 3 } }
				}
			}
			on_weekly_pulse = {
				if = {
					limit = { this = scope:host }
					trigger_event = { on_action = spn_gathering_pulse }
				}
			}
			on_end = {
				if = {
					limit = { this = scope:host }
					spn_gathering_run_end_effect = yes
				}
			}
		}

		spn_gathering_circle_phase = {
			is_predefined = yes
			order = 3
			# 2.37b -- the same: entered passive, made active here.
			on_enter_phase = {
				if = {
					limit = { this = scope:host }
					scope:activity = { progress_activity_phase_after = { days = 1 } }
				}
			}
			on_phase_active = {
				if = {
					limit = { this = scope:host }
					scope:activity = { progress_activity_phase_after = { weeks = 1 } }
					spn_gathering_circle_effect = yes
				}
			}
		}
	}

	###################
	# LOCATION (§14.1). The hunt's own leash for players; the AI stays home.
	###################
	# 2.37b -- THE MAP PINS, the hunt's 2.35a lesson again (supernatural_hunt.txt, province_score). max_province_icons
	# only LIMITS pins to the top-scored provinces; it does not make them. The score gave every province on the map a
	# flat 10 and every forest 30, so the six pins went to six forests anywhere in the world and none near Thomas.
	# Now: one place per county (its seat), the hunt's reach (near, or anywhere in your own liege's realm), and a score
	# that starts at zero and ranks woods, your own ground and nearness.
	is_location_valid = {
		exists = county
		custom_tooltip = {
			text = spn_wg_location_capital_tt
			OR = {
				is_county_capital = yes
				# A camp: domicile_realm gives a landless AI his camp's province and nothing else (_activity_type.info:91).
				scope:host = { is_landless_adventurer = yes }
			}
		}
		custom_tooltip = {
			text = spn_wg_location_too_far_tt
			OR = {
				vf_host_distance_to_province <= spn_max_hunt_distance
				AND = {
					exists = scope:host.top_liege
					county ?= { holder ?= { top_liege ?= { this = scope:host.top_liege } } }
				}
			}
		}
	}
	province_filter = all
	ai_province_filter = domicile_realm
	max_province_icons = 5
	allow_zero_guest_invites = yes
	open_invite = no

	# root = the province; scope:host = the host.
	province_score = {
		value = 0
		# The woods are the point: the biggest term.
		if = {
			limit = {
				OR = {
					terrain = forest
					terrain = taiga
				}
			}
			add = 60
		}
		# Your own ground first, then your liege's realm (the hunt's two terms).
		if = {
			limit = { county ?= { holder ?= { OR = { this = scope:host  is_vassal_or_below_of = scope:host } } } }
			add = 45
		}
		else_if = {
			limit = {
				exists = scope:host.top_liege
				county ?= { holder ?= { top_liege ?= { this = scope:host.top_liege } } }
			}
			add = 20
		}
		# Nearer is better.
		if = {
			limit = { NOT = { vf_host_distance_to_province <= squared_distance_medium } }
			add = -25
		}
		if = {
			limit = { NOT = { vf_host_distance_to_province <= squared_distance_large } }
			add = -35
		}
	}
	# The hunt's shape: a base line, then one first_valid per tag (supernatural_hunt.txt province_description).
	province_description = {
		desc = spn_wg_prov_base
		first_valid = {
			triggered_desc = {
				trigger = {
					OR = {
						terrain = forest
						terrain = taiga
					}
				}
				desc = spn_wg_tag_forest
			}
		}
		first_valid = {
			triggered_desc = {
				trigger = { county ?= { holder ?= { OR = { this = scope:host  is_vassal_or_below_of = scope:host } } } }
				desc = spn_wg_tag_own
			}
		}
		first_valid = {
			triggered_desc = {
				trigger = { NOT = { vf_host_distance_to_province <= squared_distance_large } }
				desc = spn_wg_tag_far
			}
		}
	}
