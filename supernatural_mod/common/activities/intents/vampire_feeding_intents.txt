embrace_the_night_intent = {
	icon = intent_crash_wedding

	# AI behavior settings
	ai_will_do = {
		value = 50

		# 2.33 -- PERSONALITY. None of the three intents referenced a single trait, so an AI host
		# picked its PURPOSE for the evening on stress and age alone. Root is the character picking
		# the intent (_activity_type.info:182).
		# The indulgent intent -- revel in it. Where appetite and vanity go.
		if = { limit = { has_trait = gluttonous }    add = 25 }
		if = { limit = { has_trait = lustful }       add = 20 }
		if = { limit = { has_trait = arrogant }      add = 15 }
		if = { limit = { has_trait = sadistic }      add = 15 }
		if = { limit = { has_trait = gregarious }    add = 12 }
		if = { limit = { has_trait = temperate }     add = -25 }
		if = { limit = { has_trait = chaste }        add = -20 }
		if = { limit = { has_trait = humble }        add = -15 }
		if = { limit = { has_trait = paranoid }      add = -15 }
		if = { limit = { has_trait = zealous }       add = -12 }

		# Increase chance for reveler
		if = {
			limit = {
				has_trait_xp = {
					trait = lifestyle_reveler
					value > 25
				}
			}
			add = 20
		}
		if = {
			limit = {
				has_trait_xp = {
					trait = lifestyle_reveler
					value > 50
				}
			}
			add = 30
		}
		if = {
			limit = {
				has_trait_xp = {
					trait = lifestyle_reveler
					value > 75
				}
			}
			add = 40
		}

		# Increase chance for younger vampires
		if = {
			limit = {
				age < 40
			}
			add = 15
		}

		# Decrease chance for older vampires
		if = {
			limit = {
				age > 70
			}
			add = -40
		}
	}

	auto_complete = yes
	on_invalidated = { trigger_event = activity_system.0090 }

	scripted_animation = {
		animation = toast_goblet
	}
}
