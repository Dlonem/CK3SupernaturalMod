# Supernatural 2.38 -- "The Blue Rose": the Order's intents (WORK-ORDER-2.38 §5.15, §5.21). Witchcraft's activities list them
# through apply_upstream_edits.py (E9, E10: the Mage Summit; E11, E12: the Grand Rite). Their effects run in those
# activities' on_complete (spn_order_summit_complete_effect, spn_coven_rite_complete_effect), which read the intent, or for
# AI members the Order membership (Witchcraft gives its AI guests flags, not intents: ms_load_intents_for_ai_effect).
# The names a player sees name no society (review D should-fix 2): "Seek Kindred Minds", "Speak for My Circle".
# Loc: localization/english/spn_order_l_english.yml.

# Meet in the Masks ("Seek Kindred Minds"): members you know to be members are glad of you.
spn_meet_in_the_masks_intent = {
	icon = intent_mystic
	is_shown = {
		spn_order_on_trigger = yes
		spn_is_order_member_trigger = yes
	}
	auto_complete = yes
	on_invalidated = { trigger_event = activity_system.0090 }
	scripted_animation = {
		animation = personality_rational
	}
	ai_will_do = {
		value = 40
		if = {
			limit = { has_trait = gregarious }
			add = 20
		}
	}
}

# Bring the Coven to the Order (the Grand Rite's host): every Mage of your coven who is not a member is offered the rose.
spn_bring_coven_to_order_intent = {
	icon = intent_knowledge
	is_shown = {
		spn_order_on_trigger = yes
		spn_order_reach_built_trigger = yes
		spn_order_magistratus_plus_trigger = yes
		exists = var:spn_coven
	}
	auto_complete = yes
	on_invalidated = { trigger_event = activity_system.0090 }
	scripted_animation = {
		animation = personality_rational
	}
	ai_will_do = {
		value = 0
		if = {
			limit = {
				has_trait = grandmaster_coven
				var:spn_order_rank ?= flag:magistratus
			}
			add = 30
		}
	}
}

spn_seek_the_chair_intent = {
	icon = intent_influence
	is_shown = {
		spn_order_on_trigger = yes
		spn_is_temple_magus_trigger = yes
	}
	is_valid = {
		custom_tooltip = {
			text = spn_conclave_chair_empty_tt
			OR = {
				NOT = { exists = global_var:current_archmage_global }
				# After the vote, the intent has done its work (the winner, and those who lost fairly).
				involved_activity ?= { has_variable = spn_cv_vote_closed }
			}
		}
	}
	auto_complete = yes
	on_invalidated = { trigger_event = activity_system.0090 }
	scripted_animation = {
		animation = personality_bold
	}
	ai_will_do = {
		value = 30
		if = { limit = { has_trait = ambitious } add = 60 }
		if = { limit = { has_character_flag = spn_order_aspires_chair } add = 60 }
		if = { limit = { has_trait = content } add = -40 }
	}
}
