# Supernatural 2.38b -- the Conclave's guests (DESIGN-2.38b §4): every member of the Order in the host's reach, from the
# temple lists (no world walk; the shape of activity_invite_rule_pack, activity_invite_rules_supernatural.txt).
activity_invite_rule_spn_order = {
	effect = {
		every_in_global_list = {
			variable = spn_order_chapters
			limit = {
				is_alive = yes
				exists = var:spn_order_chapter
			}
			var:spn_order_chapter = {
				every_in_list = {
					variable = spn_order_members
					limit = {
						is_alive = yes
						is_adult = yes
						is_imprisoned = no
						NOT = { this = root }
						in_diplomatic_range = root
					}
					add_to_list = characters
				}
			}
		}
	}
}
