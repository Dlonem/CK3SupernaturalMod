# Supernatural 2.38f -- what a creature lifestyle's experience reads as in a tooltip. Without these the engine looked for
# keys like add_vampire_lifestyle_xp_global_past and logged them missing (Thomas's 233-year run). Vanilla's own pattern:
# common/effect_localization/00_character_effects.txt, add_learning_lifestyle_xp.
add_vampire_lifestyle_xp = {
	global = GAINS_VAMPIRE_LIFESTYLE_XP
	global_past = GAINED_VAMPIRE_LIFESTYLE_XP
	first = I_GAIN_VAMPIRE_LIFESTYLE_XP
	first_past = I_GAINED_VAMPIRE_LIFESTYLE_XP
	third = TARGET_GAINS_VAMPIRE_LIFESTYLE_XP
	third_past = TARGET_GAINED_VAMPIRE_LIFESTYLE_XP
}
