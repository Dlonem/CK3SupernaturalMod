						type = conquest

							tier = tier_county

							holder = scope:attacker
							change = scope:change
						}
					}
					resolve_title_and_vassal_change = scope:change
				}
			}
		}
		every_ruler = {
			limit = {
				is_ruler = yes
				is_landed_or_landless_administrative = yes
				is_independent_ruler = yes
				is_at_war = no

					THIS = scope:attacker
