#For when a spn genetic birth occurs
spn_on_genetic_birth = {
	trigger = {
		# SPECIES GATE -- see common/scripted_triggers/zz_spn_species_triggers.txt. Nothing that is
		# not a person inherits any of this, whatever its parents are.
		spn_is_valid_supernatural_target = yes
		# 2.41 audit -- and never a foal. The gate above reads spn_horse on the newborn, who may not have the trait yet
		# (spn_glitterhoof_on_birth adds it on this same on_action).
		NOR = {
			scope:mother ?= { has_trait = spn_horse }
			scope:real_father ?= { has_trait = spn_horse }
		}
		# 2.41 (WORK-ORDER-2.41 §5.2) -- a hybrid parent with six supernatural children has the seventh born human.
		# Supernatural children only (C4); the cap never shows in a tooltip.
		NOT = { spn_hybrid_parent_at_child_cap_trigger = yes }
		OR = {
			AND = { #Werewolf / Vampire / Hybrid path (original behaviour)
				NOR = {
					has_game_rule = spn_no_trait_inheritance 
					has_game_rule = spn_human_no_immortal_trait_inheritance
					has_game_rule = spn_human_only_trait_inheritance
				}
				OR = {
					scope:mother = { spn_has_supernatural_trait = yes }
					scope:real_father = { spn_has_supernatural_trait = yes }
					# 2.41 boot 8 -- a carrier of wolf blood (zz_spn_241_wolf_blood_values.txt)
					scope:mother ?= { is_alive = yes has_variable = spn_wolf_blood }
					scope:real_father ?= { is_alive = yes has_variable = spn_wolf_blood }
				}
			}
			AND = { #Demon path, gated by its own rule so it works even if werewolf inheritance is off
				NOT = { has_game_rule = spn_demon_no_inheritance }
				OR = {
					scope:mother = { is_a_demon_trigger = yes }
					scope:real_father = { is_a_demon_trigger = yes }
				}
			}
		}
	}
	effect = {
		if = { #Demon parent(s) -> chance of a demon-blooded (demon0) child, per spn_enable_demon_inheritance. Checked first so Demons never fall into the werewolf branches below.
			limit = {
				NOT = { has_game_rule = spn_demon_no_inheritance }
				OR = {
					scope:mother = { is_a_demon_trigger = yes }
					scope:real_father = { is_a_demon_trigger = yes }
				}
			}

			random = {
				chance = spn_demon_birth_chance
				add_trait = demon0
				create_character_memory = { type = spn_demon_blood_memory } # 2.34: anchors the "Something in the Blood" event
				#demon0 on its own is personality flavour - no immortality, no powers. Without a
				#maturation event the child stays demon-blooded for life. The witchcraft half's
				#on_action_demon_inherited schedules this, but only when BOTH parents are demons,
				#so the Demon+Human default (which the game rule advertises) never matured anyone.
				#Reported by Omni, Jun 2026. Double-scheduling is harmless: magic_ev3.75 requires
				#has_trait = demon0 and the first firing removes it.
				trigger_event = { id = magic_ev3.75 years = { 20 21 } }
				spn_update_number_of_supernatural_children_for_parents = yes
			}
			# 2.41 (WORK-ORDER-2.41 §6, C3 Option A) -- THE WOLF WITH DEMON BLOOD. A wolf or hybrid parent's blood is now
			# offered as well, the way the wolf-parent branches below offer it: the flag, then spn_initiate.9000's own
			# inheritance roll. Born both is the demon roll times the wolf roll, so it is rare. magic_ev3.75 hands a
			# demon-blooded wolf to The Two Natures instead of making him a demon.
			if = {
				limit = {
					NOR = {
						has_game_rule = spn_no_trait_inheritance
						has_game_rule = spn_human_no_immortal_trait_inheritance
						has_game_rule = spn_human_only_trait_inheritance
					}
					OR = {
						scope:mother = { OR = { has_trait = supernatural_werewolf has_trait = supernatural_hybrid } }
						scope:real_father = { OR = { has_trait = supernatural_werewolf has_trait = supernatural_hybrid } }
					}
				}
				if = {
					limit = { scope:mother = { OR = { has_trait = supernatural_werewolf has_trait = supernatural_hybrid } } }
					spn_assign_mother_sire = { MOTHER = scope:mother }
				}
				else = {
					spn_assign_father_sire = { FATHER = scope:real_father }
				}
				if = {
					limit = { NOT = { has_trait = demon0 } }
					spn_update_number_of_supernatural_children_for_parents = yes
				}
				add_character_flag = { flag = spn_baby_werewolf }
			}
		}
		else_if = { #Both Parents Lycan
			limit = {
				scope:real_father = { has_trait = supernatural_werewolf }
				scope:mother = { has_trait = supernatural_werewolf }
			}

			spn_give_knowledge_of_lycan_secret = { CHARACTER = scope:mother }
		
			if = {
				limit = { scope:father = scope:real_father }
				spn_give_knowledge_of_lycan_secret = { CHARACTER = scope:father }
			}
			else_if = {
				limit = {
					scope:real_father = { has_character_flag = pregnancy_real_father_knows }
				}
				spn_give_knowledge_of_lycan_secret = { CHARACTER = scope:real_father }
			}

			spn_update_number_of_supernatural_children_for_parents = yes

			add_character_flag = {
				flag = spn_baby_werewolf
			}
		}
		else_if = { #Both Parents Supernaturals, but either not the same or are hybrids
			limit = {
				OR = {
					has_game_rule = spn_yes_trait_inheritance
					has_game_rule = spn_immortal_only_trait_inheritance
				}
				OR = {
					AND = {
						scope:real_father = { has_trait = supernatural_hybrid }
						scope:mother = { has_trait = supernatural_werewolf }
					}
					AND = {
						scope:real_father = { has_trait = supernatural_werewolf }
						scope:mother = { has_trait = supernatural_hybrid }
					}
				}
			}

			spn_give_knowledge_of_hybrid_secret = { CHARACTER = scope:mother }
		
			if = {
				limit = { scope:father = scope:real_father }
				spn_give_knowledge_of_hybrid_secret = { CHARACTER = scope:father }
			}
			else_if = {
				limit = {
					scope:real_father = { has_character_flag = pregnancy_real_father_knows }
				}
				spn_give_knowledge_of_hybrid_secret = { CHARACTER = scope:real_father }
			}

			spn_update_number_of_supernatural_children_for_parents = yes

			add_character_flag = {
				flag = spn_baby_werewolf #Because Hybrids cannot be born v2.08
			}
		}
		else_if = { #Mother is Werewolf
			limit = {
				#NOR = {											#Commented out on 11/14/22, letting werewolves procreate in the Vampire setting
				#	has_game_rule = spn_vampire_trait_inheritance
				#}
				scope:mother = { has_trait = supernatural_werewolf }
			}

			spn_assign_mother_sire = {
				MOTHER = scope:mother
			}

			spn_update_number_of_supernatural_children_for_parents = yes

			add_character_flag = {
				flag = spn_baby_werewolf
			}
		}
		else_if = { #Mother is Hybrid
			limit = {
				# 2.41 boot 8 -- this was an OR of the game rule and the mother's trait, so under the default rule
				# (spn_werewolf_trait_inheritance) it took EVERY child that reached it: a hybrid father's child got the
				# human mother as its sire, and a vampire parent's child was born a werewolf. Now the mother must be one.
				scope:mother = { has_trait = supernatural_hybrid }
				OR = {
					has_game_rule = spn_yes_trait_inheritance
					has_game_rule = spn_werewolf_trait_inheritance
					has_game_rule = spn_vampire_trait_inheritance
					has_game_rule = spn_immortal_only_trait_inheritance
				}
			}

			spn_assign_mother_sire = {
				MOTHER = scope:mother
			}

			# 2.41 (§5.2) -- the only branch that did not count the child, so a hybrid mother's taper and cap never moved.
			spn_update_number_of_supernatural_children_for_parents = yes

			add_character_flag = {
				flag = spn_baby_werewolf
			}
		}
		else_if = { #Father is Lycan
		 	limit = {
		 		NOR = {
		 			has_game_rule = spn_vampire_trait_inheritance
		 			has_game_rule = spn_immortal_only_trait_inheritance
		 			has_game_rule = spn_yes_trait_inheritance
		 		}
		 		scope:real_father = { has_trait = supernatural_werewolf }
		 	}

		 	spn_assign_father_sire = {
		 		FATHER = scope:real_father
		 	}

		 	spn_update_number_of_supernatural_children_for_parents = yes

		 	add_character_flag = {
		 		flag = spn_baby_werewolf
		 	}
		}
		else_if = { #Father is hybrid
		 	limit = {
		 		NOR = {
		 			has_game_rule = spn_vampire_trait_inheritance
		 			has_game_rule = spn_immortal_only_trait_inheritance
		 			has_game_rule = spn_yes_trait_inheritance
		 			#has_game_rule = spn_no_hybrid_trait_inheritance
		 		}
		 		scope:real_father = { has_trait = supernatural_hybrid }
		 	}

		 	spn_assign_father_sire = {
		 		FATHER = scope:real_father
		 	}

		 	spn_update_number_of_supernatural_children_for_parents = yes

		 	add_character_flag = {
		 		flag = spn_baby_werewolf
		 	}
		}
		# 2.41 boot 8 -- THE THROWBACK. Neither parent is a creature, but one carries wolf blood. Not counted on the
		# parents (they are not creatures) and no sire: nobody here made him.
		else_if = {
			limit = {
				OR = {
					scope:mother ?= { is_alive = yes has_variable = spn_wolf_blood }
					scope:real_father ?= { is_alive = yes has_variable = spn_wolf_blood }
				}
				NOT = { scope:mother ?= { spn_has_supernatural_trait = yes } }
				NOT = { scope:real_father ?= { spn_has_supernatural_trait = yes } }
			}
			add_character_flag = spn_baby_werewolf
			add_character_flag = { flag = spn_wb_throwback days = 1 }
		}

		# 2.41 boot 8 -- the blood decides (zz_spn_241_wolf_blood_effects.txt).
		spn_wolf_birth_roll_effect = yes
		trigger_event = spn_initiate.9000 #Genetic Init
	}
}

#Used to Age up characters
spn_on_immortality_birthday = {
	trigger = {
		OR = {
			has_trait = supernatural_vampire
			has_trait = supernatural_hybrid
		}
		NOT = { has_character_flag = spn_overwrite_visual_age }
		NOT = { has_game_rule = spn_no_automatic_ageing }
	}

	effect = {
		if = {
			limit = {
				OR = {
					AND = {
						has_game_rule = spn_stops_at_portrait_maturity 
						age <= 25 
					}
					AND = {
						has_game_rule = spn_gradual_ageing
						age <= 16
					}
				}
				spn_has_supernatural_trait = yes
			}
			spn_track_current_visual_age = { VISUALAGE = age }
		}
		else_if = {
			limit = {
				spn_has_supernatural_trait = yes
				has_game_rule = spn_gradual_ageing
				age > 16
			}
			spn_track_current_visual_age = { VISUALAGE = spn_calc_gradual_visual_age }
		}
	}
}

#Check Pregnancy
spn_check_pregnancy = {
	effect = {
		if = { #Humans
			limit = {
				spn_has_supernatural_trait = no
				scope:real_father = { spn_has_supernatural_trait = no }
			}
			trigger_event = {
				on_action = spn_on_pregnancy_mother_events
			}
		}
		else_if = { #Demon pregnancy (gated by spn_enable_demon_inheritance, independent of the werewolf rule)
			limit = {
				spn_demon_possible_pregnancy = { FATHER = scope:real_father }
			}
			#debug_log = "Demon Pregnancy Possible"
			random = {
				chance = spn_pregnancy_chance

				save_scope_value_as = {
					name = spn_preg_success
					value = yes
				}
			}

			if = {
				limit = {
					exists = scope:spn_preg_success
				}
				#debug_log = "Demon Pregnancy Success"
				trigger_event = {
					on_action = spn_on_pregnancy_mother_events
				}
			}
			else = {
				#debug_log = "Demon Pregnancy Failure"
				end_pregnancy = yes
				pregnancy_cleanup_effect = yes
			}
		}
		else_if = { #One or more of the baby makers is Supernatural
			limit = {
				spn_possible_pregnancy = { FATHER = scope:real_father }
			}
			#debug_log = "Supernatural Pregnancy Possible"
			random = {
				chance = spn_pregnancy_chance
				
				save_scope_value_as = {
					name = spn_preg_success
					value = yes
				}
			}

			if = {
				limit = {
					exists = scope:spn_preg_success
				}
				#debug_log = "Supernatural Pregnancy Success"
				trigger_event = {
					on_action = spn_on_pregnancy_mother_events
				}
			}
			else = {
				#debug_log = "Supernatural Pregnancy Failure"
				end_pregnancy = yes
				pregnancy_cleanup_effect = yes
			}
		}
		else = { #Terminate Pregnancy
			end_pregnancy = yes
			pregnancy_cleanup_effect = yes
		}
	}
}

#Call for base game pregnancy events
spn_on_pregnancy_mother_events = {
	events = {
		pregnancy.0001 # Universal hidden maintenance event
		pregnancy.1001 # Mother gives birth to current husband's child.
		pregnancy.1101 # Mother gives birth to ex-husband's child.
		pregnancy.2001 # Unwed mother gives birth to bastard child.
		pregnancy.2002 # Married mother gives birth to bastard child.
		pregnancy.2050 # Married mother in same-sex relationship.
	}
}
