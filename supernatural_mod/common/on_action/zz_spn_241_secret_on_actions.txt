# Supernatural 2.41 (WORK-ORDER-2.41 §8, the Discord question "do vampires have secrets?").
# A creature turned under a rite that accepts his kind holds neither the secret nor the known trait
# (spn_give_*_secret_or_trait_effect, 20_spn_secret_effects.txt: the one-day spn_recently_became_* flag withholds the
# trait). If his rite later condemns him -- he converts, or his faith reforms -- nothing ever handed him the secret, so
# a spymaster had nothing to find. Once a year, inside the creature gate (zz_spn_240_perf_on_actions.txt), the same
# effects that run at the turning run again, but only for a creature who has neither AND whose rite now condemns him.
# Accepting rites change nothing here.
spn_241_secret_rite_check = {
	trigger = {
		OR = {
			has_trait = supernatural_vampire
			has_trait = supernatural_werewolf
			has_trait = supernatural_hybrid
		}
	}
	effect = {
		if = {
			limit = {
				has_trait = supernatural_hybrid
				NOR = {
					has_trait = spn_known_hybrid
					any_secret = { secret_type = spn_secret_hybrid }
				}
				OR = {
					spn_secret_hybrid_is_shunned_trigger = { OWNER = this }
					spn_secret_hybrid_is_criminal_trigger = { OWNER = this }
				}
			}
			spn_give_hybrid_secret_or_trait_effect = yes
		}
		else_if = {
			limit = {
				has_trait = supernatural_vampire
				NOR = {
					has_trait = spn_known_vampire
					any_secret = { secret_type = spn_secret_vampire }
				}
				OR = {
					spn_secret_vampire_is_shunned_trigger = { OWNER = this }
					spn_secret_vampire_is_criminal_trigger = { OWNER = this }
				}
			}
			spn_give_vampire_secret_or_trait_effect = yes
		}
		else_if = {
			limit = {
				has_trait = supernatural_werewolf
				NOR = {
					has_trait = spn_known_werewolf
					any_secret = { secret_type = spn_secret_werewolf }
				}
				OR = {
					spn_secret_werewolf_is_shunned_trigger = { OWNER = this }
					spn_secret_werewolf_is_criminal_trigger = { OWNER = this }
				}
			}
			spn_give_lycan_secret_or_trait_effect = yes
		}
	}
}
