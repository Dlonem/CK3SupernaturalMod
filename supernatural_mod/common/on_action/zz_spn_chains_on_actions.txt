# Supernatural 2.39 -- The Warded Chains on imprisonment (WORK-ORDER-2.39 F6). An APPEND through `on_actions = { }`, which
# merges across files (the idiom of zz_spn_hide_on_actions.txt), beside Witchcraft's own on_imprison append
# (yearly2_on_actions.txt); no vanilla on_action is copied or shadowed.
# No scripted trigger may live in this folder (2.35d: the parser stops at it).

# Code on-action (vanilla prison_on_actions.txt): root = the prisoner, scope:imprisoner = the one who holds him.
on_imprison = {
	on_actions = {
		spn_chains_on_imprison
	}
}

# A wolf already wearing its other shape when it is taken by a gaoler who owns the chains is made to put it down. The
# cheapest test first: almost nobody being imprisoned is a wolf.
spn_chains_on_imprison = {
	trigger = {
		spn_is_wolf_trigger = yes
		is_shapeshifted_trigger = yes        # 10_spn_triggers.txt
		scope:imprisoner ?= {
			any_character_artifact = { var:spn_relic ?= flag:chains }
		}
	}
	effect = {
		POD_shapeshift_remove_werewolf_form_effect = yes    # 20_spn_remove_werewolf_shapeshift_effects.txt
	}
}
