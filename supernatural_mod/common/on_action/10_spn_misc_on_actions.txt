#Checks to see if Lifetime events need to be reset
spn_check_lifetime_flags_for_spn = {
	trigger = {
		is_ai = no # cheapest discriminator first - this hook fires for every character every year
		spn_has_supernatural_trait = yes
		NOT = { has_game_rule = no_spn_lifetime_events_reset }
		spn_has_lifetime_flags = yes
	}
	effect = {
		spn_add_years_since_lifetime_reset = yes
		if = { 
			limit = { 
				has_game_rule = 1_spn_lifetime_events_reset
				var:spn_years_since_lifetime_reset >= 1
			} 
			spn_reset_years_since_lifetime_reset = yes
			spn_remove_lifetime_flags = yes
		}
		else_if = { 
			limit = { 
				has_game_rule = 10_spn_lifetime_events_reset
				var:spn_years_since_lifetime_reset >= 10
			} 
			spn_reset_years_since_lifetime_reset = yes
			spn_remove_lifetime_flags = yes
		}
		else_if = { 
			limit = { 
				has_game_rule = 25_spn_lifetime_events_reset
				var:spn_years_since_lifetime_reset >= 25
			} 
			spn_reset_years_since_lifetime_reset = yes
			spn_remove_lifetime_flags = yes
		}
		else_if = { 
			limit = { 
				has_game_rule = 50_spn_lifetime_events_reset
				var:spn_years_since_lifetime_reset >= 50
			} 
			spn_reset_years_since_lifetime_reset = yes
			spn_remove_lifetime_flags = yes
		}
		else_if = { 
			limit = { 
				has_game_rule = 75_spn_lifetime_events_reset
				var:spn_years_since_lifetime_reset >= 75
			} 
			spn_reset_years_since_lifetime_reset = yes
			spn_remove_lifetime_flags = yes
		}
		else_if = { 
			limit = { 
				has_game_rule = 100_spn_lifetime_events_reset
				var:spn_years_since_lifetime_reset >= 100
			} 
			spn_reset_years_since_lifetime_reset = yes
			spn_remove_lifetime_flags = yes
		}
	}
}

#Recover Wounds for Supernatural Regenerative Traits
spn_recover_immortal_wounds = {
	trigger = {
		spn_has_supernatural_trait = yes
		OR = {
			has_trait = wounded
			spn_char_has_physical_injury = yes
		}
	}
	effect = {
		spn_remove_wounds = yes
	}
}

#Recover from Disfigurement - Vampires and Hybrids only (intense regeneration; not werewolves)
spn_recover_disfigurement = {
	trigger = {
		OR = {
			has_trait = supernatural_vampire
			has_trait = supernatural_hybrid
		}
		has_trait = disfigured
	}
	effect = {
		remove_trait = disfigured
	}
}

#Recover from Incapable Due to Old Age, To remove incapable from vamps/hybrids v2.08
spn_age_trait_cleanup = {
	trigger = {
		AND = {
			OR = {
				has_trait = supernatural_vampire
				has_trait = supernatural_hybrid
			}
			OR = {
				has_trait = infirm
				has_trait = clouded_eyes
				has_trait = faltering_heart
				has_trait = withering_mind
				has_trait = fragile_bones
				has_trait = incapable
			}
		}
	}
	effect = {
		spn_remove_negative_age_traits = yes
	}
}

# DELIBERATE: is_ai = no is load-bearing here. Do not remove it.
#
# This hook hangs off random_yearly_everyone_pulse -- EVERY living character, every year, not just
# playables. The roll below is 3% per character per year. Over a 30-year adult lifetime that is
# 1 - 0.97^30 = ~60% chance a given character draws the chain at least once; over a century,
# 1 - 0.97^100 = ~95%. Against an AGOT-sized population (~191,000 characters) that does not read as
# 'the supernatural is out there somewhere' -- it saturates the world inside two generations and every
# court on the map ends up half-turned.
#
# The spread is supposed to come from somewhere else: world-seeding at game start
# (spn_trait_dist.0001), the confer interactions, which have real ai_potential and ai_will_do and so
# spread at a rate the AI actually weighs, and -- since 2.33 -- dying in battle with vampire blood in
# the system. Those are throttled. A blind yearly dice roll against the whole population is not.
#
# Verified 27 Aug 2026: every event downstream of spn_random_embrace.0001 is player-only as a result
# (35 events in 10_spn_embrace.txt). That is intended, not an oversight -- do not 'fix' it by adding
# ai_chance to those options, and do not open the gate to make them reachable.
spn_check_for_random_embrace = {
	trigger = {
		is_ai = no
		spn_has_supernatural_trait = no
		spn_has_hunter_trait = no
		is_witch_trigger = no # 2.34: the six downstream gates have this commented out, so a witch kept being turned into a werewolf (Workshop report, Jun 2026)
		OR = { 
			has_game_rule = spn_yes_embrace_events 
			has_game_rule = spn_only_embrace_events
		}
	}
	effect = {
		random_list = {
			3 = {
				trigger_event = {
					id = spn_random_embrace.0001 #Random Embrace
					days = { 1 80 }
				}
			}
			97 = { }
		}
	}
}

# DELIBERATE: is_ai = no is load-bearing here too -- same reasoning as spn_check_for_random_embrace
# above. This one is on yearly_playable_pulse (landed characters only) with a 6-year cooldown flag,
# so it is the gentler of the two, but a confrontation is a story beat aimed at a person watching it.
# Firing it for the AI spends the chain on characters nobody sees and burns the once-each flags
# (spn_had_confrontation_1/2/3) on rulers who will never read them.
#
# Verified 27 Aug 2026: every event in 10_spn_confrontation.txt is player-only as a result (12 events since 2.34; the
# earlier count of 11 included two .9000/.9001 tooltip keys).
# Intended. Do not add ai_chance to them, and do not open the gate.
spn_confrontations = {
	trigger = {
		is_ai = no
		spn_has_supernatural_trait = yes
		NOT = { has_character_flag = spn_had_confrontation }
		OR = { 
			has_game_rule = spn_yes_embrace_events 
			has_game_rule = spn_only_super_events
		}
	}
	effect = {
		random_list = {
			10 = {
				modifier = {
					add = {
						value = {
							value = intrigue
							max = 35 #  Changed from 25 -> 35 in v2.15
						}
						multiply = { value = -0.2 }
					}
				}
				random_list = {
					0 = {
						modifier = {
							add = {
								if = {
									limit = {
										NOT = {
											has_character_flag = spn_had_confrontation_1 
										} 
									}
									value = 50
								}
							}
						}
						trigger_event = {
							id = spn_confrontation.0001
							days = { 1 80 }
						}
						add_character_flag = spn_had_confrontation_1
					}
					0 = {
						modifier = {
							add = {
								if = {
									limit = { NOT = { has_character_flag = spn_had_confrontation_2 } }
									value = 50
								}
							}
						}
						trigger_event = {
							id = spn_confrontation.0011
							days = { 1 80 }
						}
						add_character_flag = spn_had_confrontation_2
					}
					0 = {
						modifier = {
							add = {
								if = {
									limit = { NOT = { has_character_flag = spn_had_confrontation_3 } }
									value = 50
								}
							}
						}
						trigger_event = {
							id = spn_confrontation.0021
							days = { 1 80 }
						}
						add_character_flag = spn_had_confrontation_3
					}
				}
			}
			90 = {
				modifier = {
					add = {
						value = {
							value = intrigue
							max = 35 # Changed from 25 -> 35 in v2.15
						}
						multiply = { value = 0.2 }
					}
				}
			}
		}
		add_character_flag = { 
			flag = spn_had_confrontation  
			years = 6
		}
	}
}

#Retroactive: before 2.31a, immune_to_disease was only added inside the
#has_character_flag = supernatural_embrace_* branch of the initiate events, so most existing
#supernaturals never received it - and hybrids had it stripped by 2001 and never restored.
#This stamps it once on anyone already turned. Self-terminating: after the first pass the
#NOT check fails forever, so the cost is one trigger evaluation per character per year.
# 2.33 -- BACKSTOP for the battle-transition chain. spn_in_transition no longer expires on a timer
# (see 20_spn_transition_effects.txt for why), so the only way it can outlive its chain is if the
# blood clears first: spn_miscellaneous_events.0002 is hidden and fires on schedule at 3 years,
# stripping vampire_blood_in_system and spn_vampire_blood_donor, while spn_transition.0001 may still
# be sitting unread in the player's event queue. Answering it then can no longer complete anything.
# This releases the character instead of leaving them set_knight_status = forbid forever.
#
# Also catches the character having been turned by some other route while the event sat unread.
# Self-terminating: after one pass the flag is gone and the trigger fails forever.
spn_transition_stuck_cleanup = {
	trigger = {
		has_character_flag = spn_in_transition
		OR = {
			spn_has_supernatural_trait = yes
			NOT = { has_character_flag = vampire_blood_in_system }
		}
	}
	effect = {
		remove_character_flag = spn_in_transition
		# 2.34: guarded like the sibling code in 20_spn_transition_effects.txt -- set_knight_status errors for anyone not in
		# a court's knight pool, which threw a log line every time this fired on a landless adventurer.
		if = {
			limit = { has_character_flag = spn_knight_status_forbidden }
			set_knight_status = default
			remove_character_flag = spn_knight_status_forbidden
		}
	}
}
