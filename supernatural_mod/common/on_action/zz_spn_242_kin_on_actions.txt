# Supernatural 2.42 -- Blood for Blood. Root = the dying, scope:killer = whoever the death names (vanilla on_death).
on_death = {
	on_actions = { spn_bfb_on_death }
}
spn_bfb_on_death = {
	trigger = {
		has_character_flag = spn_slain_by_hunter
		exists = scope:killer
		scope:killer = { is_alive = yes }
	}
	effect = {
		spn_bfb_record_effect = yes
	}
}
