# Supernatural 2.41 (WORK-ORDER-2.41 §7.1) -- the demon-blooded wolf's growing-up beats, on birthdays.
on_birthday = {
	on_actions = { spn_two_natures_birthday }
}
spn_two_natures_birthday = {
	trigger = {
		has_trait = demon0
		has_trait = supernatural_werewolf
		OR = {
			age = 8
			age = 12
			age = 16
		}
	}
	effect = {
		if = {
			limit = { age = 8 }
			spn_two_natures_beat_effect = { EVENT = spn_two_natures.0001 }
		}
		else_if = {
			limit = { age = 12 }
			spn_two_natures_beat_effect = { EVENT = spn_two_natures.0002 }
		}
		else = {
			spn_two_natures_beat_effect = { EVENT = spn_two_natures.0003 }
		}
	}
}
