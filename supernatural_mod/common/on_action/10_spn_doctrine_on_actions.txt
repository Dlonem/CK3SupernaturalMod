#Used to assign all criminal doctrines on game start
#
# The work lives in a SCRIPTED EFFECT, not inline here, and that is deliberate. A second mod that
# wants a different answer -- the AGOT compatibility patch does, because AGOT has no rf_pagan -- can
# then override one scripted effect instead of redefining this on_action. Two on_action definitions
# with an `effect` block each make the engine log
#     "There is more than one 'effect' defined using most recent: <file>"
# every single load, and the loser is silently discarded. One indirection removes both problems.
spn_assign_all_religions_immortal_doctrines = {
	effect = {
		spn_assign_immortal_doctrines_effect = yes
	}
}
