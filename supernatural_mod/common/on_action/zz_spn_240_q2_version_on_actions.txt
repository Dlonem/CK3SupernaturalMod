# Supernatural 2.40 boot 3 (Q2): the version popup reaches a running save without waiting for a 1 January with a player.
# spn_show_versioninfo (10_spn_version_on_actions.txt) runs on 1 January and on a new game, for every_player; a 1 January
# spent in observe mode reaches nobody. These two hooks send the same popup when a player's character has its quarterly
# pulse, and when a player changes character (succession, or taking a character). The event's own trigger
# (events/10_spn_versioninfo.txt) shows it once per version, and to every player sent it on that same day.
quarterly_playable_pulse = {
	on_actions = { spn_show_versioninfo_player }
}
on_player_character_change = {
	on_actions = { spn_show_versioninfo_player }
}
spn_show_versioninfo_player = {
	trigger = { is_ai = no }
	effect = {
		every_player = {
			trigger_event = spn_versioninfo.1000
		}
	}
}
