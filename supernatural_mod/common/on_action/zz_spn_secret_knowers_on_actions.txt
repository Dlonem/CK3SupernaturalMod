# Supernatural 2.34a -- one-time repair for saves that already carry unknown secrets.
#
# The fix in zz_spn_secret_knowers_effects.txt only runs when a secret is CREATED. A save from 2.34 or
# earlier is full of secrets that were created with no knower, and the spymaster and the scrying orb
# will keep "finding" them until somebody is told. This walks every living character who holds one of
# the four supernatural secrets, once per save, and tells the same people the creation-time fix would
# have told. Guarded by a global variable so it never runs twice; on_game_start_after_lobby fires on
# every load, so a save that predates 2.34a is repaired the first time it is opened under it.
# on_actions = { } sub-lists merge across files, so this adds to on_game_start_after_lobby without
# touching 10_spn_on_actions.txt.

on_game_start_after_lobby = {
	on_actions = {
		spn_secret_knowers_repair
	}
}

spn_secret_knowers_repair = {
	trigger = {
		NOT = { has_global_variable = spn_secret_knowers_repaired }
	}
	effect = {
		set_global_variable = {
			name = spn_secret_knowers_repaired
			value = yes
		}
		every_living_character = {
			limit = {
				any_secret = {
					OR = {
						secret_type = spn_secret_vampire
						secret_type = spn_secret_werewolf
						secret_type = spn_secret_hybrid
						secret_type = spn_secret_hunter
					}
				}
			}
			spn_reveal_secret_to_those_who_made_me_effect = { TYPE = spn_secret_vampire }
			spn_reveal_secret_to_those_who_made_me_effect = { TYPE = spn_secret_werewolf }
			spn_reveal_secret_to_those_who_made_me_effect = { TYPE = spn_secret_hybrid }
			spn_reveal_secret_to_those_who_made_me_effect = { TYPE = spn_secret_hunter }
		}
	}
}
