@chance_of_nothing = 75

# 2.35h -- THE RATES ON THIS PULSE WERE THE REASON NOBODY EVER SAW EITHER OF THESE EVENTS.
#
# Thomas, 9 Sep 2026: "I have yet to try to host a supernatural hunt, click on witch hunt, and then see
# somewhere in the world that there's a witch situation occurring." Here is why, in one multiplication.
#
# OLD: chance_of_no_event 25 (so 75% the on_action runs at all), then chance_to_happen 10 (10% the events
# are evaluated -- "a percentage chance determining whether the events involved will be evaluated at all",
# _on_actions.info:57), then weights 100 = 0 against 10 and 10, so 10/120 = 8.3% for either event.
#
#     0.75 x 0.10 x 0.083 = 0.62% per pulse, and the pulse is every one to two years.
#
# Expected wait for a hunter to be told about a witch: somewhere north of two hundred years. It was not
# rare, it was off. Nothing was broken -- three separate dampeners were multiplied together, each of which
# looks reasonable on its own line.
#
# NEW: 0.90 x 0.40 x 0.30 = 10.8% per pulse for each of the two, ~21% for one of them. Against a pulse of
# one to two years that is one of these arriving roughly every seven years for a given hunter, on top of
# everything the 2.35 trail layer gives him. That is an occasional piece of news, which is what it was
# always meant to be.

random_yearly_playable_pulse = {
    on_actions = {
        delay = { days = { 365 730 } }  # Pulse triggers once every 1 to 2 years
        spn_hunter_yearly_random_on_action
    }
}

spn_hunter_yearly_random_on_action = {
    random_on_action = {
        chance_of_no_event = { value = 10 }
        1 = spn_hunter_on_action
    }
}

spn_hunter_on_action = {
	trigger = {
		spn_has_supernatural_trait = no
	}
	random_events = {
		chance_to_happen = 40
		40 = 0  # nothing

		# Witch or Monster Hunter events (the events gate on the hunter trait themselves)
		30 = spn_hunters.0001  # Witch spotted
		30 = spn_hunters.0010  # Monster spotted	
	}
}
