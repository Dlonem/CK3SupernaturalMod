# Supernatural 2.40 (WORK-ORDER-2.40 G12): the trails map is started at game start and, on an older save, on its first
# 1 January under 2.40. Its own on_start syncs every trail already in the world, once (WORK-ORDER-2.40 §4 item 7).
# The situation is unique, so its existence is the "started" flag; if it is ever ended, the next 1 January starts it again.

on_game_start_after_lobby = {
	on_actions = {
		spn_trails_start
	}
}

yearly_global_pulse = {
	on_actions = {
		spn_trails_start
	}
}

spn_trails_start = {
	trigger = {
		NOT = { exists = situation:spn_trails }
	}
	effect = {
		start_situation = {
			type = spn_trails
		}
	}
}
