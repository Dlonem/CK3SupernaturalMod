# Supernatural 2.43 -- the hunter funnel. Every adult hunter, once a year inside the creature pulse, counts himself into
# the stages a hunt needs. spn_243_hunter_funnel_roll_effect (in the January census) moves the year's counts into
# spn_tc_f_* global variables, which the save keeps. Nothing here changes what anybody does.
spn_yearly_creature_pulse = {
	on_actions = { spn_243_hunter_funnel }
}
spn_243_hunter_funnel = {
	trigger = {
		has_trait = supernatural_hunter
		is_adult = yes
	}
	effect = {
		spn_tc_bump_effect = { NAME = spn_tc_fy_hunters }
		if = { limit = { spn_hunt_can_host_trigger = yes } spn_tc_bump_effect = { NAME = spn_tc_fy_hosts } }
		if = { limit = { has_court_position = spn_trailsman_court_position } spn_tc_bump_effect = { NAME = spn_tc_fy_trailsmen } }
		if = { limit = { has_trait_xp = { trait = supernatural_hunter track = monster_hunting value >= 20 } } spn_tc_bump_effect = { NAME = spn_tc_fy_xp20 } }
		if = {
			limit = {
				exists = location
				any_in_global_list = {
					variable = spn_menace_counties
					spn_trailsman_trail_is_warm_trigger = yes
					title_province ?= { squared_distance = { target = root.location value <= squared_distance_large } }
				}
			}
			spn_tc_bump_effect = { NAME = spn_tc_fy_near }
			if = { limit = { spn_hunt_can_host_trigger = yes } spn_tc_bump_effect = { NAME = spn_tc_fy_host_near } }
		}
		# 2.43 stage 7: the AI hosts the Hunt's +50 applies to (a warm trail inside his leash, not resting)
		if = {
			limit = {
				is_ai = yes
				spn_hunt_can_host_trigger = yes
				spn_hunt_ai_trail_in_reach_trigger = yes
			}
			spn_tc_bump_effect = { NAME = spn_tc_fy_host_reach }
		}
	}
}
