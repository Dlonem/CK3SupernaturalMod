# Supernatural 2.37 -- "The Six Hides": every hook (WORK-ORDER-2.37 §19.1). Each is an APPEND through `on_actions = { }`,
# which merges across files (the idiom of zz_spn_menace_on_actions.txt), so no vanilla on_action is copied or shadowed.
# Each starts with the cheapest test that is false almost everywhere.
#
# No scripted trigger may live in this folder (2.35d: the parser stops at it); every trigger used here is in
# common/scripted_triggers/zz_spn_hide_triggers.txt.

# ==================================================================================================
# DEATH (§8.0) -- the one on_death hook, all its jobs in order. Root is about to die; never read death_reason here.
# Gated per step by its own rule, so one shared hook cannot let one rule switch off the other's bookkeeping.
# ==================================================================================================
on_death = {
	on_actions = {
		spn_hide_on_death
	}
}

spn_hide_on_death = {
	trigger = {
		OR = {
			spn_is_wolf_trigger = yes
			spn_is_champion_trigger = yes
			has_trait = spn_true_alpha
			exists = var:spn_pack
			exists = var:spn_hide_wolf_hide
			has_trait = ww_infected_wound # 2.37e: step 0
		}
	}
	effect = {
		save_scope_as = spn_dying
		# 0. One who dies in the wound's fever comes off his sire's list (the Gathering's Fevered rule; 2.37e, the
		#    death outcome never runs spn_clear_wound_incubation_state).
		if = {
			limit = {
				has_trait = ww_infected_wound
				var:spn_conferred_sire ?= { is_alive = yes }
			}
			var:spn_conferred_sire = {
				if = {
					limit = { is_target_in_variable_list = { name = spn_my_fevered target = scope:spn_dying } }
					remove_list_variable = { name = spn_my_fevered target = scope:spn_dying }
				}
			}
		}
		# 1. Whoever kills a wolf is marked for life: never a True Alpha (§16). Either rule.
		if = {
			limit = {
				spn_is_wolf_trigger = yes
				scope:killer ?= { is_alive = yes } # 2.37e: ?= (no short-circuit beside an exists)
			}
			scope:killer = { add_character_flag = spn_killed_a_wolf }
		}
		# 2. A pack's alpha: the killer goes onto the story for on_owner_death to read (§13.6). Packs rule.
		if = {
			limit = {
				spn_packs_on_trigger = yes
				spn_is_alpha_trigger = yes
				exists = scope:killer
			}
			var:spn_pack = { set_variable = { name = spn_pack_killer value = scope:killer } }
		}
		# 3. A champion: the hide moves now, before vanilla inheritance (§8). Hides rule.
		if = {
			limit = {
				spn_hides_on_trigger = yes
				spn_is_champion_trigger = yes
			}
			spn_hide_exit_effect = { REASON = flag:death }
		}
		# 4. A member leaves the pack; an alpha's succession is the story's on_owner_death. Packs rule.
		if = {
			limit = {
				spn_packs_on_trigger = yes
				exists = var:spn_pack
			}
			spn_pack_remove_member_effect = { WHO = scope:spn_dying REASON = flag:death }
		}
		# 5. The True Alpha dies: the century still runs, but nobody "lives" now. Either rule.
		if = {
			limit = { global_var:spn_ta_living ?= scope:spn_dying }
			remove_global_variable = spn_ta_living
		}
		# 6. The wolf a lost hide was reserved for (§9.5) dies: the hide goes back to waiting. Hides rule.
		if = {
			limit = {
				spn_hides_on_trigger = yes
				exists = var:spn_hide_wolf_hide
			}
			if = {
				limit = { var:spn_hide_wolf_hide = flag:silverback }
				if = {
					limit = { global_var:spn_hide_silverback_reserved ?= scope:spn_dying }
					spn_hide_rewait_effect = { HIDE = silverback }
				}
			}
			if = {
				limit = { var:spn_hide_wolf_hide = flag:timber }
				if = {
					limit = { global_var:spn_hide_timber_reserved ?= scope:spn_dying }
					spn_hide_rewait_effect = { HIDE = timber }
				}
			}
			if = {
				limit = { var:spn_hide_wolf_hide = flag:tundra }
				if = {
					limit = { global_var:spn_hide_tundra_reserved ?= scope:spn_dying }
					spn_hide_rewait_effect = { HIDE = tundra }
				}
			}
			if = {
				limit = { var:spn_hide_wolf_hide = flag:graybeard }
				if = {
					limit = { global_var:spn_hide_graybeard_reserved ?= scope:spn_dying }
					spn_hide_rewait_effect = { HIDE = graybeard }
				}
			}
			if = {
				limit = { var:spn_hide_wolf_hide = flag:midnight }
				if = {
					limit = { global_var:spn_hide_midnight_reserved ?= scope:spn_dying }
					spn_hide_rewait_effect = { HIDE = midnight }
				}
			}
			if = {
				limit = { var:spn_hide_wolf_hide = flag:alpha }
				if = {
					limit = { global_var:spn_hide_alpha_reserved ?= scope:spn_dying }
					spn_hide_rewait_effect = { HIDE = alpha }
				}
			}
			remove_variable = spn_hide_wolf_hide
		}
	}
}

# ==================================================================================================
# OWNER CHANGE (§5.3, §8 `soul`, §9.1). Root = the artifact; scope:owner, scope:old_owner, scope:old_holy_site
# (inventory_on_actions.txt:2-8).
# ==================================================================================================
on_artifact_changed_owner = {
	on_actions = {
		spn_hide_owner_changed
	}
}

spn_hide_owner_changed = {
	trigger = {
		has_variable = spn_hide
	}
	effect = {
		save_scope_as = spn_moved_hide
		# Out of a shrine into someone's hands, by any route (vanilla's siege theft, our own takes, the Church's
		# reverse, the zealot sweep): the marker goes. Idempotent with the holy-site hook.
		if = {
			limit = {
				exists = var:spn_hide_shrine_county
				exists = scope:owner
			}
			spn_hide_shrine_left_effect = yes
		}
		spn_hide_save_champion_effect = yes
		# 2.37o: the idle count (the yearly net) runs only while the hide stays idle. In a player's hands, its
		# champion's or a shrine's, it starts again from nothing.
		if = {
			limit = {
				exists = var:spn_hide_idle_since
				spn_hide_lies_idle_trigger = no
			}
			remove_variable = spn_hide_idle_since
		}
		# Witchcraft's soul transfer (§8 `soul`): every artifact moves to the new body and the player goes with it.
		# The hide follows the soul, not the husk.
		if = {
			limit = {
				exists = scope:owner
				exists = scope:old_owner
				exists = scope:spn_hide_champ
				scope:owner = { has_character_flag = just_soul_transfered_flag }
				scope:old_owner = scope:spn_hide_champ
			}
			scope:old_owner = { spn_hide_exit_effect = { REASON = flag:soul } }
			# The exit ran on the husk after the hide had already moved, so it could not free the hide: free it here.
			# If the new body is chosen, the champion effect guards it again.
			scope:spn_moved_hide = {
				if = {
					limit = { has_variable = artifact_prevent_transfer }
					remove_variable = artifact_prevent_transfer
				}
			}
			scope:owner = {
				if = {
					limit = { spn_can_bear_hide_trigger = yes }
					spn_hide_make_champion_dispatch_effect = { ARTIFACT = scope:spn_moved_hide }
				}
				else = {
					spn_hide_start_check_effect = yes
				}
			}
		}
		# The hide cannot leave a living champion: anything that moved it is undone (§5.3).
		else_if = {
			limit = {
				exists = scope:spn_hide_champ
				scope:spn_hide_champ = { is_alive = yes }
				NOT = { scope:owner ?= scope:spn_hide_champ }
			}
			set_owner = {
				target = scope:spn_hide_champ
				history = {
					type = given
					recipient = scope:spn_hide_champ
				}
			}
			equip_artifact_to_owner_replace = yes
		}
		else = {
			# A hide passing from one pack's wolf to another pack's wolf is a feud (§17.5).
			if = {
				limit = {
					exists = scope:owner
					exists = scope:old_owner
					scope:owner = { exists = var:spn_pack }
					scope:old_owner = { exists = var:spn_pack }
				}
				if = {
					limit = { NOT = { scope:owner.var:spn_pack = scope:old_owner.var:spn_pack } }
					scope:owner.var:spn_pack.story_owner = { save_scope_as = spn_mv_a }
					scope:old_owner.var:spn_pack.story_owner = { save_scope_as = spn_mv_b }
					spn_hide_feud_effect = { A = scope:spn_mv_b B = scope:spn_mv_a }
				}
			}
			scope:owner ?= { spn_hide_start_check_effect = yes }
		}
	}
}

# ==================================================================================================
# SHRINES (§9.1). Root = the artifact; scope:holy_site (now), scope:old_holy_site (before). "Out" runs before
# "Into": a shrine-to-shrine move has both, and the other order would delete the new marker.
# ==================================================================================================
on_artifact_changed_holy_site_owner = {
	on_actions = {
		spn_hide_holy_site_changed
	}
}

spn_hide_holy_site_changed = {
	trigger = {
		has_variable = spn_hide
	}
	effect = {
		save_scope_as = spn_shrine_hide
		if = {
			limit = { exists = scope:old_holy_site }
			spn_hide_shrine_left_effect = yes
		}
		# Not before seeding is done: seeding places its own markers and sets spn_hides_seeded as its last step. 2.37o:
		# nor in seeding's own wake (spn_hides_seeding, one day): this hook runs after the effect that fired it.
		if = {
			limit = {
				exists = scope:holy_site
				exists = global_var:spn_hides_seeded
				NOT = { exists = global_var:spn_hides_seeding }
			}
			scope:holy_site.holy_site_county = { save_scope_as = spn_shrine_county }
			spn_hide_shrine_entered_effect = { COUNTY = scope:spn_shrine_county FAITH = scope:spn_shrine_county.holder.faith }
			spn_hide_word_travels_effect = yes
		}
	}
}

# Root = the side. Both sides' hooks fire; the first does everything and the second finds nothing left.
spn_hide_combat_end = {
	trigger = {
		OR = {
			global_var:spn_hide_champion_silverback ?= { spn_hide_in_this_battle_trigger = yes }
			global_var:spn_hide_champion_timber ?= { spn_hide_in_this_battle_trigger = yes }
			global_var:spn_hide_champion_tundra ?= { spn_hide_in_this_battle_trigger = yes }

			global_var:spn_hide_champion_alpha ?= { spn_hide_in_this_battle_trigger = yes }
		}
	}
	effect = {
		save_scope_as = spn_battle_side
		global_var:spn_hide_champion_silverback ?= {
			if = {
				limit = { spn_hide_in_this_battle_trigger = yes }
				add_to_list = spn_battle_champs
			}
		}
		global_var:spn_hide_champion_timber ?= {
			if = {
				limit = { spn_hide_in_this_battle_trigger = yes }
				add_to_list = spn_battle_champs
			}
		}
		global_var:spn_hide_champion_tundra ?= {
			if = {
				limit = { spn_hide_in_this_battle_trigger = yes }
				add_to_list = spn_battle_champs
			}
		}
		global_var:spn_hide_champion_graybeard ?= {
			if = {
				limit = { spn_hide_in_this_battle_trigger = yes }
				add_to_list = spn_battle_champs
			}
		}
		global_var:spn_hide_champion_midnight ?= {
			if = {
				limit = { spn_hide_in_this_battle_trigger = yes }
				add_to_list = spn_battle_champs
			}
		}
		global_var:spn_hide_champion_alpha ?= {
			if = {
				limit = { spn_hide_in_this_battle_trigger = yes }
				add_to_list = spn_battle_champs
			}
		}
		spn_hide_battle_aftermath_effect = yes
	}

# ==================================================================================================
# THE HUNT (§9.5). hunt_outcome_end_events fires for the host on day 35 of every hunt (hunt.txt:5029-5031);
# hunt_end_events is defined but nothing in 1.20 calls it. Neither trigger reads the white-animal story, and neither
# reads hunt_success: it is set by an outcome event fired from this same on_action, so it is not there yet.
# ==================================================================================================
hunt_outcome_end_events = {
	on_actions = {
		spn_hide_hunt_outcome
	}
}

spn_hide_hunt_outcome = {
	trigger = {
		exists = var:spn_hide_wolf_hide
		scope:activity ?= { has_activity_option = { category = special_type option = hunt_type_legendary } }
		# 2.37e: and it is a wolf being hunted. Vanilla can hand the same player a second white animal (bp3_roaming.0120,
		# mpo_nerge.1031); killing that one must not pay the hide. The hunt copies the county's animal_type at its start
		# (hunt.txt:5466-5492), and the wake's sighting set it to flag:wolf.
		scope:activity ?= { var:animal_type ?= flag:wolf }
	}
	effect = {
		scope:activity = {
			set_variable = { name = spn_hide_wolf_hide value = root.var:spn_hide_wolf_hide }
		}
		add_character_flag = { flag = spn_hide_outcome_pending years = 1 } # 2.37e: was 10 days (see spn_hide.0301)
	}
	events = {
		delay = { days = 3 }
		spn_hide.0301
	}
}
