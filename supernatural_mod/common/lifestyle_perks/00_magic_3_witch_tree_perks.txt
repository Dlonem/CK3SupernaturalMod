protective_runes_perk = {
	name = protective_amulets_perk
	lifestyle = magic_lifestyle
	tree = witch
	position = { 2 0 }
	icon = node_magic

	auto_selection_weight = {
		value = 0
		if = {
			limit = {
				OR = {
					has_trait = witch
					any_secret = { type = secret_witch }
				}
			}
			add = 2000
		}
		if = {
			limit = { can_start_new_lifestyle_tree_trigger = no } # 2.34 (Supernatural): the tree has a clause in the trigger now
			multiply = 0
		}
	}

	can_be_picked = {
		OR = {
			has_trait = witch
			any_secret = { type = secret_witch }
			has_focus=arcane_focus
			has_trait = true_witch		
		}
		
	}
	
	effect = {

		#custom_tooltip = protective_runes_perk_effect4
		#custom_tooltip = protective_runes_perk_effect2
		#custom_tooltip = protective_runes_perk_effect3
		
		if = {
			limit = {
				NOT = {
					has_character_flag = produce_mamulet
				}
			}
			if = {
				limit = {
					has_character_flag = yaga_flag
				}
				create_yaga_amulet_artifact_effect = yes
			}
			else = {
				create_witch_amulet_artifact_effect = yes
			}
			
			add_character_flag = produce_mamulet
		}
		hidden_effect = {
			merge_mamulets_4_ai_effect = yes
			if = {
				limit = {
					is_ai = yes
					NOT = {
						any_character_artifact = {
							has_variable = mamulet
							is_equipped = yes
						}
					}
				}
				random_character_artifact = {
					limit = {
						has_variable = mamulet
					}
					equip_artifact_to_owner_replace = yes
				}
			}
			if = {
				limit = {
					#is_ai = no
					NOR = {
						has_trait = witch_hunter
						has_trait = true_witch
					}
				}
				random = {
					chance = 95
					trigger_event = {
						id = magic_ev2.46
						months = { 3 125 }
					}
				}
			}
		}
	}
}
