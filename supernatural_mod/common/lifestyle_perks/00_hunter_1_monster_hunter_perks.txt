# Well-Rounded
wellrounded_perk = {
	lifestyle = hunter_lifestyle
	tree = monster_hunter
	position = { 1 0 }
	icon = node_hunter
	
	can_be_picked = {
		has_hunter_trigger = yes
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_hunter_interest_trigger = yes
			}
			add = 3000
		}
		if = {
			limit = {
				has_focus = hunter_monster_hunter_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = faith_defender_perk } # 2.34: vanilla's sibling-root guard -- already in the tree via the other root
			}
			multiply = 0
		}
	}
	
	character_modifier = {
		stewardship = 2
		learning = 2
		diplomacy = 2
	}
	
}

# Faithful Zealot
faith_defender_perk = {
	lifestyle = hunter_lifestyle
	tree = monster_hunter
	auto_selection_weight = {
		# 2.32 -- only the ROOT perk of each of the six trees carried a weight, so the AI had
		# nothing telling it to climb toward the capstone: it entered a tree and stopped.
		# Witchcraft's demon tree weights all sixteen of its perks and was the template.
		# Same shape as the root -- interested characters push hard, the matching focus
		# multiplies -- so progression stays personality-driven rather than flat.
		value = 11
		if = {
			limit = { has_hunter_interest_trigger = yes }
			add = 3000
		}
		if = {
			limit = { has_focus = hunter_monster_hunter_focus }
			multiply = 5
		}
		if = { # 2.34: this root had no guard at all
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = wellrounded_perk }
			}
			multiply = 0
		}
	}
	position = { 3 0 }
	icon = node_hunter
	
	can_be_picked = {
		has_hunter_trigger = yes
	}
	
	#parent = wellrounded_perk
	
	character_modifier = {
		different_faith_opinion = 15
		faith_creation_piety_cost_mult = -0.25
		clergy_opinion = 15
		realm_priest_opinion = 25
	}
}
