
### HUNTER - WITCH HUNTING PERKS


# Anti-Mage
antimage_perk = {
	lifestyle = hunter_lifestyle
	tree = witch_hunter
	position = { 2.15 0 }
	icon = node_hunter

	can_be_picked = {
		has_hunter_trigger = yes
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_hunter_interest_trigger = yes
			}
			add = 3000
		}
		if = {
			limit = {
				has_focus = hunter_witch_hunter_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}
	
	# 2.34: the root perk carried no payload at all (the monster-hunter root gives +2/+2/+2). The ability its tooltip
	# advertises is now real: antimage_perk is an OR branch on witch_hunting_decision.
	character_modifier = {
		learning = 2
		intrigue = 2
		prowess = 2
	}
	effect = {
		custom_description_no_bullet = {
			text = antimage_perk_custom_effect
		}
	}
}


# Mage Slayer
mageslayer_perk = {
	lifestyle = hunter_lifestyle
	tree = witch_hunter
	auto_selection_weight = {
		# 2.32 -- only the ROOT perk of each of the six trees carried a weight, so the AI had
		# nothing telling it to climb toward the capstone: it entered a tree and stopped.
		# Witchcraft's demon tree weights all sixteen of its perks and was the template.
		# Same shape as the root -- interested characters push hard, the matching focus
		# multiplies -- so progression stays personality-driven rather than flat.
		value = 11
		if = {
			limit = { has_hunter_interest_trigger = yes }
			add = 3000
		}
		if = {
			limit = { has_focus = hunter_witch_hunter_focus }
			multiply = 5
		}
	}
	position = { 1.15 1.5 }
	icon = node_hunter

	can_be_picked = {
		has_hunter_trigger = yes
	}
	
	parent = antimage_perk
	
	character_modifier = {
		health = 0.15
		martial = 3
		intrigue = 2
	}
}

# Protection Runes
protection_runes_perk = {
	lifestyle = hunter_lifestyle
	tree = witch_hunter
	auto_selection_weight = {
		value = 11
		if = {
			limit = { has_hunter_interest_trigger = yes }
			add = 3000
		}
		if = {
			limit = { has_focus = hunter_witch_hunter_focus }
			multiply = 5
		}
	}
	position = { 0 3.5 } 
	icon = node_hunter

	can_be_picked = {
		has_hunter_trigger = yes
	}
	
	parent = mageslayer_perk
	
	effect = {
		custom_tooltip = protective_runes_perk_effect
		custom_tooltip = protective_runes_perk_effect2
		# 2.34: the witch-tree perk is `protective_runes_perk` (all 31 protection gates test that id); this hunter perk is
		# `protection_runes_perk`, one letter off, and conferred nothing. Deliberate design -- a hunter cannot be a witch,
		# so granting the witch-tree perk is the clean way to give them its protection.
		hidden_effect = { add_perk = protective_runes_perk }
	}
}
