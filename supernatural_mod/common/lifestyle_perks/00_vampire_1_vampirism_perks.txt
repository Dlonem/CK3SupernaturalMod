
### VAMPIRE - UNDEAD IMMORTALITY PERKS
#
# Tree layout = a "drop of blood" teardrop, symmetric about column x = 1.0:
#       row0            Daywalker (1.5)                <- top point
#       row1        Kiss (0.5)      Celerity (2.5)
#       row2   MasterShadows(0)  Compulsion(1.5)  Domination(3)   <- widest bulge
#       row3      VampNest(0)    Bloodthirsty(1.5)  Retinue(3)
#       row4            TrueVampire (1.5)            <- bottom point
#
# IMPORTANT: keep x >= 0. The old layout put MasterShadows and VampNest at x = -0.5;
# both rendered and NEITHER could be hovered, tooltipped or clicked, which severed the
# tree (reported by CleverPolarBear, Lord Zymeth, Noa3 and TobbieR between Dec 2025 and
# Jun 2026, one of them reproduced with Supernatural as the only active mod).
# The whole teardrop was shifted +0.5 rather than reshaped, so the silhouette is identical.
# Across all 18 vanilla lifestyle_perks files the complete set of X values is {0,1,2,3,4}.


# Become a Daywalker
daywalker_perk = {
	lifestyle = vampire_lifestyle
	tree = vampirism
	position = { 1.5 0 }
	icon = node_vampire

	can_be_picked = {
		has_vampirism_trigger = yes
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_vampire_interest_trigger = yes
			}
			add = 3000
		}
		if = {
			limit = {
				has_focus = vampire_vampirism_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}
	
	character_modifier = {
        intrigue = 4
        diplomacy = 3
        stewardship = 3
        martial = 2
		learning = 3
	}
	
	# 2.42 (Supernatural): Merchants of the Night. Added on top of every line above; nothing is taken away (WORK-ORDER-2.42 v3, 4).
	government_character_modifier = { # a merchant company (merchants and republics)
		flag = government_has_merchant_company
		character_travel_safety_mult = 0.1
	}
}

# Vampire's Kiss
vampire_seduction_perk = {
	lifestyle = vampire_lifestyle
	tree = vampirism
	auto_selection_weight = {
		# 2.32 -- only the ROOT perk of each of the six trees carried a weight, so the AI had
		# nothing telling it to climb toward the capstone: it entered a tree and stopped.
		# Witchcraft's demon tree weights all sixteen of its perks and was the template.
		# Same shape as the root -- interested characters push hard, the matching focus
		# multiplies -- so progression stays personality-driven rather than flat.
		value = 11
		if = {
			limit = { has_vampire_interest_trigger = yes }
			add = 3000
		}
		if = {
			limit = { has_focus = vampire_vampirism_focus }
			multiply = 5
		}
	}
	position = { 0.5 1 }
	icon = node_vampire

	can_be_picked = {
		has_vampirism_trigger = yes
	}
	
	parent = daywalker_perk
	
	effect = {
		custom_description_no_bullet = {
			text = graceful_recovery_perk_effect
		}
		custom_description_no_bullet = {
			text = mortal_adoration_perk_effect
		}
		custom_description_no_bullet = {
			text = spn_sire_bond_perk_effect # 2.40 boot 2 (QS, item h): the Sire Bond, zz_spn_240_sire_puppet_types.txt
		}
	}
	
	character_modifier = {
		#seduce_scheme_power_mult = 0.10
		seduce_scheme_phase_duration_add = -3
		attraction_opinion = 10
		}
	# 2.42 (Supernatural): Merchants of the Night. Added on top of every line above; nothing is taken away (WORK-ORDER-2.42 v3, 4).
	government_character_modifier = { # a merchant company (merchants and republics)
		flag = government_has_merchant_company
		trade_competitiveness_add = 5
	}
}

	icon = node_vampire

	can_be_picked = {
		AND = {
			has_vampirism_trigger = yes
			has_perk = vampire_seduction_perk
		}
	}

	character_modifier = {
        hard_casualty_modifier = -0.1
        knight_limit = 1
	}
	# 2.41 (CK3 1.21): the perk swaps with the government, the way vanilla perks do (tax_man_perk):
	# a merchant cannot raise men-at-arms, so those lines become something he can use.
	government_character_modifier = {
		flag = government_is_merchant
		invert_check = yes
		supernatural_animal_max_size_add = 1
		supernatural_monster_max_size_add = 1
		men_at_arms_maintenance_per_dread_mult = -0.005
	}
	government_character_modifier = {
		flag = government_is_merchant
		monthly_prestige_gain_per_dread_add = 0.01
		enemy_hostile_scheme_success_chance_add = -5
	}
	# 2.42 (Supernatural): Merchants of the Night. Added on top of every line above; nothing is taken away (WORK-ORDER-2.42 v3, 4).
	government_character_modifier = { # a merchant company (merchants and republics)
		flag = government_has_merchant_company
		owned_scheme_secrecy_add = 15
		trade_competitiveness_per_knight_mult = 0.005
	}
}
