
### VAMPIRE - UNDEAD IMMORTALITY PERKS
#
# Tree layout = a "drop of blood" teardrop, symmetric about column x = 1.0:
#       row0            Daywalker (1)                <- top point
#       row1        Kiss (0)      Celerity (2)
#       row2   MasterShadows(-0.5) Compulsion(1) Domination(2.5)  <- widest bulge
#       row3      VampNest(0)   Bloodthirsty(1)   Retinue(2)
#       row4            TrueVampire (1)            <- bottom point
#
# IMPORTANT: keep x within roughly [-0.5, 2.5]. Nodes further left (x <= -1)
# still RENDER but fall outside the tree's clickable area and can't be hovered
# or selected. -0.5 and 2.5 are confirmed-clickable extremes; stay inside them.


# Become a Daywalker
daywalker_perk = {
	lifestyle = vampire_lifestyle
	tree = vampirism
	position = { 1 0 }
	icon = node_vampire

	can_be_picked = {
		has_vampirism_trigger = yes
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_vampire_interest_trigger = yes
			}
			add = 3000
		}
		if = {
			limit = {
				has_focus = vampire_vampirism_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}
	
	character_modifier = {
        intrigue = 4
        diplomacy = 3
        stewardship = 3
        martial = 2
		learning = 3
	}
	
}
