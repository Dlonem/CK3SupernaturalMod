magical_experiments_p1_perk = {
	lifestyle = magic_lifestyle
	tree = magic
	position = { 1.80 0 }
	icon = node_magic

	auto_selection_weight = {
		value = 0
		if = {
			limit = {
				OR = {
					has_trait = witch
					any_secret = { type = secret_witch }
				}
				OR = {
					any_character_artifact = { has_variable = magic1 }
					has_trait = source
					#has_trait = source2
					#has_trait = source3
					#has_trait = source4
					has_character_flag = arcane_unlocked_flag
					has_power_place_trigger = yes
				}
			}
			add = 4000
		}
	}

	can_be_picked = {
		OR = {
			has_focus = arcane_focus
			is_true_mage = yes
		}
	}
	
	effect = {
		if = {
			limit = {
				has_character_flag = special_magic_character
			}
			add_learning_skill = 1
		}
		else = {
			random_list = {
				80 = {
					send_interface_toast = {
						title = magical_experiments_p1_perk_toast
						add_learning_skill = 1
					}
					
				}
				5 = {
					trigger = {
						NOT = {
							has_trait = disfigured
						}
					}
					add_trait = disfigured
				}
				5 = {
					trigger = {
						NOR = {
							has_trait = wounded_1
							has_trait = wounded_2
							has_trait = wounded_3
						}
					}
					add_trait = wounded_1
				}
				5 = {
					trigger = {
						NOT = {
							has_trait = one_eyed
						}
					}
					add_trait = one_eyed
				}
				5 = {
					trigger = {
						NOT = {
							has_trait = one_legged
						}
					}
					add_trait = one_legged
				}
			}
		}
		
		#if = { limit = { NOT = { has_character_modifier = power_saturated1 } }
		#	add_character_modifier = power_saturated1
		#}
		gain_standard_ps_effect = yes
		custom_tooltip = magical_experiments_p1_perk_effect
		hidden_effect = {
			if = {
				limit = {
					is_ai = no
				}
				random = {
					chance = 80
					trigger_event = {
						id = magic_ev2.58	#create mage
						years = { 2 15 }
					}
				}
			}
			if = {
				limit = {
					NOT = {
						has_trait = archmage
					}
				}
				random = {
					chance = 99
					trigger_event = {
						id = magic_ev2.59	#witch cooperation
						months = { 6 95 }
					}
				}
			}
			if = {
				limit = {
					NOT = {
						has_character_flag = special_magic_character
					}
					has_trait = archmage
				}
				remove_trait = archmage
			}
		}
	}
}
