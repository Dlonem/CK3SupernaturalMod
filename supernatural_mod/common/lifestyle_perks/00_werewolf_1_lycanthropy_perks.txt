
### WEREWOLF - THE CURSE OF THE BEAST


# Curse of the Wolf
curse_wolf_perk = {
	lifestyle = werewolf_lifestyle
	tree = lycanthropy
	position = { 1 0 }
	icon = node_werewolf

	can_be_picked = {
		has_lycanthropy_trigger = yes
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_werewolf_interest_trigger = yes
			}
			add = 3000
		}
		if = {
			limit = {
				has_focus = werewolf_lycanthropy_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}
	
	character_modifier = {
        intrigue = 3
        diplomacy = 3
        stewardship = 3
        martial = 5
		prowess = 4
	}
	
	# 2.43 (Supernatural): Merchants of the Night, as the 2.42 blocks.
	government_character_modifier = { # a merchant company (merchants and republics)
		flag = government_has_merchant_company
		# 2.43 (Thomas, 22 Nov): the creature starts ahead in trade
		max_trade_goods_on_route = 1
		trade_capacity_add = 25
	}
}


# Wolf Politics
wolf_politics_perk = {
	lifestyle = werewolf_lifestyle
	tree = lycanthropy
	auto_selection_weight = {
		# 2.32 -- only the ROOT perk of each of the six trees carried a weight, so the AI had
		# nothing telling it to climb toward the capstone: it entered a tree and stopped.
		# Witchcraft's demon tree weights all sixteen of its perks and was the template.
		# Same shape as the root -- interested characters push hard, the matching focus
		# multiplies -- so progression stays personality-driven rather than flat.
		value = 11
		if = {
			limit = { has_werewolf_interest_trigger = yes }
			add = 3000
		}
		if = {
			limit = { has_focus = werewolf_lycanthropy_focus }
			multiply = 5
		}
	}
	position = { 0 1 }
	icon = node_werewolf

	can_be_picked = {
		has_lycanthropy_trigger = yes
	}
	
	parent = curse_wolf_perk
	
	effect = {
		#custom_description_no_bullet = {
		#	text = unshackled_lust_perk_effect
		#}
		custom_tooltip = spn_wolf_politics_pack_tt # 2.37
	}
	
	character_modifier = {
		child_opinion = 10
		close_relative_opinion = 20
		spouse_opinion = 15
		courtier_and_guest_opinion = 10
		prisoner_opinion = 5
	}
	# 2.41 (CK3 1.21): the perk swaps with the government, the way vanilla perks do (tax_man_perk):
	# a landless adventurer or merchant has no vassals or counties, so those lines become something he can use.
	government_character_modifier = {
		flag = government_uses_landless_lifestyle_perks
		invert_check = yes
		vassal_opinion = 10
		fellow_vassal_opinion = 15
	}
	government_character_modifier = {
		flag = government_uses_landless_lifestyle_perks
		monthly_prestige = 0.25
	}
	# 2.42 (Supernatural): Merchants of the Night. Added on top of every line above; nothing is taken away (WORK-ORDER-2.42 v3, 4).
	government_character_modifier = { # a merchant company (merchants and republics)
		flag = government_has_merchant_company
		trait_track_caravaneer_xp_gain_mult = 0.25
	}
}
