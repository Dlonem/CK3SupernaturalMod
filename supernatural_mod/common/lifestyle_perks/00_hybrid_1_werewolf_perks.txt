lycanthropic_nature_perk = {
	lifestyle = hybrid_lifestyle
	tree = werewolf
	position = { 0 0 }
	icon = node_werewolf

	can_be_picked = {
		has_hybrid_trigger = yes
	}
	
	auto_selection_weight = {
		# 2.32 -- only the ROOT perk of each of the six trees carried a weight, so the AI had
		# nothing telling it to climb toward the capstone: it entered a tree and stopped.
		# Witchcraft's demon tree weights all sixteen of its perks and was the template.
		# Same shape as the root -- interested characters push hard, the matching focus
		# multiplies -- so progression stays personality-driven rather than flat.
		value = 11
		if = {
			limit = {
				has_hybrid_interest_trigger = yes
			}
			add = 3000
		}
		if = {
			limit = {
				has_focus = hybrid_hybrid_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = bloodline_perk } # 2.34: vanilla's sibling-root guard (the commented line named a trait, not the sibling root)
			}
			multiply = 0
		}
	}
	
	character_modifier = {
        intrigue = 3
        diplomacy = 2
        stewardship = 3
        martial = 4
		prowess = 4
	}
	# 2.42 (Supernatural): Merchants of the Night. Added on top of every line above; nothing is taken away (WORK-ORDER-2.42 v3, 4).
	government_character_modifier = { # a merchant company (merchants and republics)
		flag = government_has_merchant_company
		character_travel_safety_mult = 0.1
	}
}
