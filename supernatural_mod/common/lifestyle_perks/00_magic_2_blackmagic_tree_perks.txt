	lifestyle = magic_lifestyle
	tree = blackmagic
	position = { 2 0 }
	icon = node_blackmagic

	auto_selection_weight = { # 2.34 (Supernatural): the three magic trees had no AI weighting past their roots
		value = 0
		if = {
			limit = {
				OR = {
					has_trait = witch
					any_secret = { type = secret_witch }
				}
			}
			add = 2000
		}
		if = {
			limit = { can_start_new_lifestyle_tree_trigger = no } # 2.34 (Supernatural): the tree has a clause in the trigger now
			multiply = 0
		}
	}

	can_be_picked = {
		has_focus = blackmagic_focus
	}

	effect = {
		custom_tooltip = evil_eye_perk_effect
		if = {
			limit = {
				has_trait = lifestyle_mystic
			}
			add_trait_xp = {
				trait = lifestyle_mystic
				value = 10
			}
		}
	}
}

warlock_perk = {
	lifestyle = magic_lifestyle
	tree = blackmagic
	position = { 2 2.4 }
	icon = true_warlock

	auto_selection_weight = { # 2.34 (Supernatural): the three magic trees had no AI weighting past their roots
		value = 0
		if = {
			limit = {
				OR = {
					has_trait = witch
					any_secret = { type = secret_witch }
				}
			}
			add = 2000
		}
	}

	parent = summoner_p8_perk

	can_be_picked = {
		AND = {
			has_focus = blackmagic_focus
			has_perk = summoner_p8_perk
		}
		# 2.34: warlock is an opposite of the hunter traits; a perk must never be purchasable when its trait cannot be added
		custom_description = {
			text = spn_capstone_blocked_by_hunter
			NOR = { has_trait = witch_hunter has_trait = supernatural_hunter }
		}
	}

	trait = warlock
	
	effect = {
		custom_tooltip = warlock_perk_tt
		add_trait_force_tooltip = warlock
		if = {
			limit = {
				has_trait = grandmaster_coven
			}
			add_trait_xp = {
				trait = grandmaster_coven
				value = 20
			}
		}
		set_focus = no		
		hidden_effect = {
			if = {
				limit = {
					has_trait = archmage
				}
				add_trait_xp = {
					trait = archmage
					track = secret_knowledge
					value = 10
				}
			}

			if = {
				limit = {
					has_character_modifier = power_saturated1
				}
				remove_character_modifier = power_saturated1
				gain_standard_ps_effect = yes
			}
			else_if = {
				limit = {
					has_character_modifier = power_saturated2
				}
				remove_character_modifier = power_saturated2
				gain_standard_ps_effect = yes
			}
		}
	}
}
