### CHOOSE LIFESTYLE
action_can_choose_lifestyle = {
	type = alert
	check_create_action = {
		
		if = {
			limit = {
				is_adult = yes
				has_any_focus = no
				can_select_lifestyle_focus = yes
			}
			try_create_important_action = {
				important_action_type = action_can_choose_lifestyle
				actor = root
			}
		}
		
	}

	effect = {
		scope:actor = {
			if = {
				limit = { is_adult = yes }
				open_view_data = {
					view = lifestyle
				}
			}
			else = {
				open_view_data = {
					view = character_focus
				}
			}
		}
	}
}

action_can_spend_perk_point = {
	type = alert
	check_create_action = {
		
		if = {
			limit = { # Future-proofing
				has_any_focus = yes
				NOR = {
					has_focus_intrigue = yes
					has_focus_diplomacy = yes
					has_focus_learning = yes
					has_focus_martial = yes
					has_focus_stewardship = yes
					# FIXED 21 Aug 2026. This NOR is the list of focuses that ALREADY HAVE their own
					# perk-point alert, so this catch-all skips them. It had it backwards both ways:
					#
					#   vampire / werewolf / hybrid / hunter were EXCLUDED, and nothing anywhere in this
					#   mod raises an alert for them -- so a character on a creature focus with an unspent
					#   perk point was told nothing at all. That is the bug worth fixing here.
					#
					#   witch / arcane (action_lifestyle_magic) and manipulation / destruction
					#   (action_lifestyle_demonic) DO have their own alerts and were NOT excluded, so a
					#   mage got told the same thing twice.
					#
					#   has_focus_wanderer had been dropped; vanilla excludes it because
					#   action_lifestyle_wanderer covers it (game/common/important_actions/00_lifestyle.txt:207).
					#
					# blackmagic_focus is deliberately absent: nothing covers it, so it falls through to this
					# alert, which is what it should do.
					has_focus_wanderer = yes
					has_focus = witch_focus
					has_focus = arcane_focus
					has_focus = manipulation_focus
					has_focus = destruction_focus
				}
				perk_points > 0
				is_adult = yes
			}
			try_create_important_action = {
				important_action_type = action_can_spend_perk_point
				actor = root
			}
		}
		
	}

	effect = {
		scope:actor = {
			open_view_data = {
				view = lifestyle
			}
		}
	}
}
