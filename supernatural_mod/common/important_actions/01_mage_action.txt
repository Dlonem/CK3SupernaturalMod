action_can_host_magic_research_activity = {
	type = alert
	check_create_action = {
		if = {
			limit = {
				is_ai = no
				is_true_mage = yes
				NOT = { is_activity_type_on_cooldown = magical_research }
				NOT = { is_game_view_open = activity_planner }
				is_available_adult = yes
			}
			try_create_important_action = {
				important_action_type = action_can_host_magic_research_activity
				#activity_type = magical_research
			}
		}
	}

	effect = {
		#scope:activity_type = { open_view_data = activity_list_detail_host_window }
		open_view_data = {
			view = activity_list_detail_host_window
			data = activity_type:magical_research
		}
	}
}

action_can_host_mages_summit_activity = {
	type = alert
	check_create_action = {
		if = {
			limit = {
				is_ai = no
				is_true_mage = yes
				is_landed_or_landless_administrative = yes
				NOT = { is_activity_type_on_cooldown = mages_summit }
				NOT = { is_game_view_open = activity_planner }
				is_available_adult = yes
				gold >= mage_summit_cost_final_value
				OR = {
					OR = {
						is_legendary_mage_trigger = yes
						AND = {
							OR = {
								has_trait = yaga
								has_trait = archmage
							}
							prestige_level >= very_high_prestige_level
						}
						AND = {
							OR = {
								has_trait = yaga
								has_trait = archmage
							}
							exists = house
							house = {
								has_house_modifier = magic_tradition_modifier
							}
						}
					}
				}
			}
			try_create_important_action = {
				important_action_type = action_can_host_mages_summit_activity
				#activity_type = magical_research
			}
		}
	}

	effect = {
		#scope:activity_type = { open_view_data = activity_list_detail_host_window }
		open_view_data = {
			view = activity_list_detail_host_window
			data = activity_type:mages_summit
		}
	}
}
