spn_leave_pack_decision = {
	decision_group_type = supernatural # 2.37: the Supernatural Decisions group (zz_spn_decision_group_types.txt)
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}
	desc = spn_leave_pack_decision_desc
	selection_tooltip = spn_leave_pack_decision_tooltip
	confirm_text = spn_leave_pack_decision_confirm
	ai_check_interval = 12
	is_shown = {
		exists = var:spn_pack
		spn_is_alpha_trigger = no
	}
	effect = {
		spn_pack_remove_member_effect = { WHO = root REASON = flag:left }
	}
	ai_potential = {
		exists = var:spn_pack
	}
	ai_will_do = {
		base = 0
		# 2.37o (the review of 4 Oct, F15): a wolf leaves for a reason -- he and his alpha have no use for each other.
		# Fickle makes it likelier, never the reason; with kin re-inviting him it made a leave-and-return loop.
		# His alpha has no use for him (the alpha's own opinion of him; it read reverse_opinion, which is his of the alpha,
		# the review of 4 Oct, P1).
		modifier = {
			add = 30
			var:spn_pack ?= {
				story_owner = {
					opinion = {
						target = root
						value < -25
					}
				}
			}
		}
		# He has no use for his alpha.
		modifier = {
			add = 30
			opinion = {
				target = var:spn_pack.story_owner
				value <= -25
			}
		}
		modifier = { add = 20 has_trait = fickle }
		modifier = { add = -40 has_trait = loyal }
		modifier = {
			factor = 0
			opinion = {
				target = var:spn_pack.story_owner
				value > -25
			}
			var:spn_pack ?= {
				story_owner = {
					opinion = {
						target = root
						value >= -25
					}
				}
			}
		}
	}
}
