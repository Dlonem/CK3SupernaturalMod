# 2.40 (WORK-ORDER-2.40 A11): Hand On Your Temple. A Temple Magus steps down: he stays as a Magistratus, or works outside
# the temple with the Adepti. Another Magistratus must be free to lead it. The 10-year wait is a flag set only when he
# actually steps down, so "Not yet" costs nothing.
spn_order_step_down_decision = {
	decision_group_type = spn_order # 2.40 (boot 2, item i): was supernatural
	picture = { reference = "gfx/interface/illustrations/decisions/decision_misc.dds" }
	desc = spn_order_step_down_decision_desc
	selection_tooltip = spn_order_step_down_decision_tooltip
	confirm_text = spn_order_step_down_decision_confirm
	ai_check_interval = 60
	is_shown = {
		spn_order_on_trigger = yes
		spn_is_temple_magus_trigger = yes
	}
	is_valid = {
		custom_tooltip = {
			text = spn_order_step_down_heir_tt
			spn_order_step_down_heir_trigger = yes
		}
		custom_tooltip = {
			text = spn_order_step_down_recent_tt
			NOT = { has_character_flag = spn_order_stepped_down }
		}
	}
	effect = {
		custom_tooltip = spn_order_step_down_decision_effect_tt
		trigger_event = spn_order.2050
	}
	ai_potential = {
		spn_order_on_trigger = yes
		spn_is_temple_magus_trigger = yes
		NOT = { global_var:current_archmage_global ?= this }
	}
	# Reason- and personality-gated, never flat: an old head, or a content one, may pass the temple on. Never the Grand
	# Magus (the chair is his, and a temple is his voice in it).
	ai_will_do = {
		base = 0
		modifier = {
			add = 20
			age >= 65
		}
		modifier = {
			add = 20
			has_trait = content
			age >= 50 # orchestrator (R1 review): a content head steps aside in later life, not at any age
		}
		modifier = {
			add = 10
			has_trait = humble
		}
		modifier = {
			factor = 0
			OR = {
				has_trait = ambitious
				has_trait = arrogant
			}
		}
		modifier = {
			factor = 0
			global_var:current_archmage_global ?= this
		}
	}
}
