# Bind the Vade Maecum (§12.6): four unsealed pages or the whole book, and a living firstborn son to pay with.
spn_vade_bind_decision = {
	decision_group_type = spn_craft # 2.40 (boot 2, item i): was supernatural
	picture = { reference = "gfx/interface/illustrations/decisions/decision_destiny_goal.dds" }
	desc = spn_vade_bind_decision_desc
	selection_tooltip = spn_vade_bind_decision_tooltip
	confirm_text = spn_vade_bind_decision_confirm
	ai_check_interval = 24
	is_shown = {
		is_adult = yes # 2.39 (Boot 3, 16 Oct): a child does not
		spn_vade_on_trigger = yes
		has_trait = archmage
		any_character_artifact = {
			OR = {
				has_variable = spn_vade_page
				has_variable = spn_vade_book
			}
		}
	}
	is_valid = {
		custom_tooltip = {
			text = spn_vade_bind_four_tt
			OR = {
				any_character_artifact = { has_variable = spn_vade_book }
				any_character_artifact = {
					count >= 4
					has_variable = spn_vade_page
					NOT = { has_variable = spn_page_sealed }
				}
			}
		}
		trigger_if = {
			limit = { var:spn_vade_firstborn ?= { is_alive = yes } }
			custom_tooltip = {
				text = spn_vade_bind_son_tt
				always = yes
			}
		}
		trigger_else = {
			custom_tooltip = {
				text = spn_vade_bind_no_son_tt
				always = no
			}
		}
		NOT = { has_trait = spn_vade_bound }
		custom_tooltip = {
			text = spn_vade_bind_running_tt
			NOT = { exists = global_var:spn_vade_contract_running }
		}
	}
	effect = {
		custom_tooltip = spn_vade_bind_decision_effect_tt
		spn_vade_bind_effect = yes
	}
	ai_potential = {
		has_trait = archmage
		OR = {
			has_trait = callous
			has_trait = sadistic
		}
	}
	ai_will_do = {
		base = 0
		modifier = { add = 40 has_trait = ambitious }
		modifier = { add = 30 has_trait = arrogant }
		modifier = { add = 40 has_character_flag = spn_vade_tempted }
	}
}
