# A Daughter Chapter: a full chapter's Magistratus at standing 75 takes up to five who follow him to a new seat. Fifty years
# between daughters of one chapter.
spn_order_daughter_decision = {
	decision_group_type = supernatural
	picture = { reference = "gfx/interface/illustrations/decisions/decision_dynasty_house.dds" }
	desc = spn_order_daughter_decision_desc
	selection_tooltip = spn_order_daughter_decision_tooltip
	confirm_text = spn_order_daughter_decision_confirm
	ai_check_interval = 60
	is_shown = {
		spn_order_on_trigger = yes
		spn_order_reach_built_trigger = yes
		var:spn_order_rank ?= flag:magistratus
		spn_is_temple_magus_trigger = no
		is_landed = yes
	}
	is_valid = {
		custom_tooltip = {
			text = spn_order_daughter_standing_tt
			exists = var:spn_order_standing
			var:spn_order_standing >= 75
		}
		custom_tooltip = {
			text = spn_order_daughter_full_tt
			var:spn_order_chapter ?= { spn_order_has_room_trigger = no }
		}
		custom_tooltip = {
			text = spn_order_daughter_cd_tt
			var:spn_order_chapter ?= { NOT = { has_variable = spn_order_daughter_cd } }
		}
		custom_tooltip = {
			text = spn_order_daughter_spaced_tt
			spn_order_seat_spaced_trigger = yes
		}
		custom_tooltip = {
			text = spn_order_charter_room_tt
			NOT = {
				any_in_global_list = {
					variable = spn_order_chapters
					count >= spn_order_max_chapters_value
				}
			}
		}
	}
	effect = {
		save_scope_as = spn_dc_founder
		var:spn_order_chapter = {
			save_scope_as = spn_dc_parent
			set_variable = { name = spn_order_daughter_cd value = yes years = 50 }
			# Up to five who follow him (AI members who like him; a player member is never moved without asking).
			ordered_in_list = {
				variable = spn_order_members
				limit = {
					is_ai = yes
					is_alive = yes
					NOT = { this = scope:spn_dc_founder }
					NOT = { this = scope:spn_dc_parent.story_owner }
					NOT = { var:spn_order_rank ?= flag:neophyte }
					opinion = { target = scope:spn_dc_founder value >= 20 }
				}
				order_by = var:spn_order_standing
				max = 5
				check_range_bounds = no
				add_to_temporary_list = spn_dc_followers
			}
		}
		capital_county = { save_scope_as = spn_dc_seat }
		spn_order_found_chapter_effect = { HEAD = scope:spn_dc_founder SEAT = scope:spn_dc_seat }
		if = {
			limit = { exists = scope:spn_new_chapter }
			every_in_list = {
				list = spn_dc_followers
				# 2.39 (Boot 4, 16 Oct): his rank goes over as a flag literal, read before he leaves the old temple
				if = {
					limit = { var:spn_order_rank ?= flag:magistratus }
					spn_order_remove_member_effect = { WHO = this REASON = flag:transfer }
					spn_order_add_member_effect = { WHO = this CHAPTER = scope:spn_new_chapter RANK = flag:magistratus }
				}
				else_if = {
					limit = { var:spn_order_rank ?= flag:medicum }
					spn_order_remove_member_effect = { WHO = this REASON = flag:transfer }
					spn_order_add_member_effect = { WHO = this CHAPTER = scope:spn_new_chapter RANK = flag:medicum }
				}
				else = {
					spn_order_remove_member_effect = { WHO = this REASON = flag:transfer }
					spn_order_add_member_effect = { WHO = this CHAPTER = scope:spn_new_chapter RANK = flag:acolyte }
				}
				spn_order_grant_membership_effect = yes
			}
		}
	}
	ai_potential = {
		var:spn_order_rank ?= flag:magistratus
		is_landed = yes
	}
	ai_will_do = {
		base = 10
		modifier = { add = 20 has_trait = ambitious }
	}
}

# 2.38b: Ask the Temple for Help with Your Research (DESIGN-2.38b §7b). A member in the middle of Witchcraft's Magical
# Research: a Magistratus of his temple comes, and the project needs one breakthrough fewer (E31).
spn_order_research_help_decision = {
	decision_group_type = supernatural
	picture = { reference = "gfx/interface/illustrations/decisions/decision_misc.dds" }
	desc = spn_order_research_help_decision_desc
	selection_tooltip = spn_order_research_help_decision_tooltip
	confirm_text = spn_order_research_help_decision_confirm
	ai_check_interval = 24
	cooldown = { years = 5 }
	is_shown = {
		spn_order_on_trigger = yes
		spn_is_order_member_trigger = yes
	}
	is_valid = {
		custom_tooltip = {
			text = spn_order_research_help_in_research_tt
			involved_activity ?= { has_activity_type = magical_research }
		}
		custom_tooltip = {
			text = spn_order_research_help_standing_tt
			exists = var:spn_order_standing
			var:spn_order_standing >= 10
		}
		custom_tooltip = {
			text = spn_order_research_help_magistratus_tt
			var:spn_order_chapter ?= {
				any_in_list = {
					variable = spn_order_members
					NOT = { this = root }
					spn_order_magistratus_plus_trigger = yes
				}
			}
		}
	}
	effect = {
		spn_order_add_standing_effect = { VALUE = -10 }
		custom_tooltip = spn_order_research_help_decision_effect_tt
		trigger_event = spn_order.0330
	}
	ai_potential = { spn_is_order_member_trigger = yes }
	ai_will_do = { base = 50 }
}

	decision_group_type = supernatural

	ai_potential = {
