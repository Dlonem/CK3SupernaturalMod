spn_vanish_decision = {
	decision_group_type = supernatural # 2.37: the Supernatural Decisions group (zz_spn_decision_group_types.txt)
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}
	desc = spn_vanish_decision_desc
	selection_tooltip = spn_vanish_decision_tooltip
	confirm_text = spn_vanish_decision_confirm
	ai_check_interval = 12
	cooldown = { years = 10 }
	is_shown = {
		has_trait = spn_champion_timber
		trigger_if = {
			limit = { exists = var:spn_hide_beat_timber }
			var:spn_hide_beat_timber >= 3
		}
		trigger_else = { always = no }
	}
	is_valid = {
		location ?= {
			county ?= {
				custom_tooltip = {
					text = spn_vanish_no_trail_tt
					var:spn_menace_culprit ?= root
				}
			}
		}
		custom_tooltip = {
			text = spn_vanish_hunt_running_tt
			NOT = {
				any_activity = {
					has_activity_type = supernatural_hunt
					activity_location.county ?= root.location.county
				}
			}
		}
	}
	effect = {
		location ?= {
			county ?= {
				save_scope_as = spn_vanish_county
				spn_menace_clear_county_effect = yes
			}
		}
		if = {
			limit = {
				exists = scope:spn_vanish_county
				is_target_in_variable_list = { name = spn_my_trails target = scope:spn_vanish_county }
			}
			remove_list_variable = { name = spn_my_trails target = scope:spn_vanish_county }
		}
	}
	ai_potential = {
		has_trait = spn_champion_timber
		location ?= { county ?= { var:spn_menace_culprit ?= root } }
	}
	ai_will_do = {
		base = 10
		modifier = {
			add = 15
			has_trait = shy
		}
		modifier = {
			add = 15
			has_trait = paranoid
		}
		modifier = {
			add = 15
			has_trait = vengeful
		}
		modifier = {
			add = 15
			has_trait = wrathful
		}
	}
}
