# BREAK THE CHAINS (F5). All six champions, while imprisoned; ten years between, shown in the panel by the decision's own
# cooldown (no variable). Vanilla's own Escape from Prison is a decision shown only in prison (80_major_decisions.txt),
# which is what proves the panel works from a cell. The hide's other decisions use the same ten-year cooldown.
# The roll is spn_hide.0400 (zz_spn_chains_events.txt). No death anywhere in the chain. Player-initiated, so it neither
# checks nor marks the flavor gate (F14).
# ==================================================================================================
spn_break_the_chains_decision = {
	decision_group_type = spn_wolf # 2.40 (boot 2, item i): was supernatural
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_prison.dds"   # vanilla escape_from_prison_decision's
	}
	desc = spn_break_the_chains_decision_desc
	selection_tooltip = spn_break_the_chains_decision_tooltip
	confirm_text = spn_break_the_chains_decision_confirm
	ai_check_interval = 12
	cooldown = { years = 10 }
	is_shown = {
		is_imprisoned = yes
		spn_is_champion_trigger = yes
		spn_is_valid_supernatural_target = yes    # the compatibility patch's species hook (§6)
	}
	is_valid_showing_failures_only = {
		is_adult = yes
		is_incapable = no
		# The counter, met in play: greyed out with its reason while the gaoler owns the Warded Chains (F6).
		custom_description = {
			text = spn_held_in_chains_custom
			spn_held_in_chains_trigger = no
		}
	}
	effect = {
		imprisoner = { save_scope_as = imprisoner }
		# Vanilla's idiom (escape_from_prison_decision): both outcomes named, no odds shown.
		show_as_tooltip = {
			random_list = {
				50 = {
					show_chance = no
					desc = spn_break_the_chains_success_tt
					custom_tooltip = spn_break_the_chains_success_outcome_tt
				}
				50 = {
					show_chance = no
					desc = spn_break_the_chains_failure_tt
					custom_tooltip = spn_break_the_chains_failure_outcome_tt
				}
			}
		}
		custom_tooltip = spn_break_the_chains_decision_effect_tt
		trigger_event = spn_hide.0400
	}
	ai_potential = {
		is_imprisoned = yes
		spn_held_in_chains_trigger = no
	}
	# Weighted by torture, a reason to fear the axe, and temperament (§6 AI).
	ai_will_do = {
		base = 40
		modifier = { add = { value = spn_hide_bond_value multiply = 0.5 } }
		modifier = { add = 200 has_character_modifier = recently_tortured }        # vanilla 00_event_modifiers.txt
		modifier = { add = 100 imprisoner ?= { has_execute_reason = root } }
		modifier = { add = 20 has_trait = brave }
		modifier = { add = 20 has_trait = wrathful }
		modifier = { add = 20 has_trait = impatient }
		modifier = { add = -30 has_trait = craven }
		# A wolf nobody knows, under a faith that makes it a crime, has the most to lose by being seen (Let the Wolf Run's line).
		modifier = {
			add = -40
			NOR = {
				has_trait = spn_known_werewolf
				has_trait = spn_known_hybrid
			}
			rite ?= { rite_has_doctrine = spn_doctrine_lycanthropy_crime }
		}
	}
}

# ==================================================================================================
# WARD THE CHAINS (F6, D13 (a)+(x)). Prepare the Powder's shape (zz_spn_order_decisions.txt): five years between, standard
# power, gold. Mages and True Witches by trait, and the Order's Magistrati, so a game with the Order rule Off still has
# makers. The working rings, as Ask to Unbind's does: a champion nearby hears a Mage working wards.

spn_ward_chains_decision = {
	decision_group_type = spn_craft # 2.40 (boot 2, item i): was supernatural
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}
	desc = spn_ward_chains_decision_desc
	selection_tooltip = spn_ward_chains_decision_tooltip
	confirm_text = spn_ward_chains_decision_confirm
	ai_check_interval = 60
	cooldown = { years = 5 }
	is_shown = {
		spn_can_ward_chains_trigger = yes
		spn_is_valid_supernatural_target = yes
	}
	is_valid = {
		can_spend_standard_ps_trigger = yes        # magic_triggers.txt
	}
	cost = {
		gold = medium_gold_value                   # vanilla 01_dynamic_values.txt
	}
	effect = {
		spend_standard_ps_effect = yes             # 00_variety_magic_effects.txt
		spn_relic_make_chains_effect = { OWNER = root }
		# The ringing: other champions and senses nearby hear it, run as the caster (Ask to Unbind's shape).
		custom_tooltip = spn_ward_chains_ring_tt
		save_scope_as = magic_user
		send_message_4_magic_sense_effect = yes
	}
	ai_potential = {
		spn_can_ward_chains_trigger = yes
		NOT = { any_character_artifact = { var:spn_relic ?= flag:chains } }
	}
	# A Mage wards chains while he holds a wolf he knows of, or after one broke out of his cells; otherwise never.
	ai_will_do = {
		base = 0
		modifier = {
			add = 50
			any_prisoner = {
				OR = {
					has_trait = spn_known_werewolf
					has_trait = spn_known_hybrid
					any_secret = {
						OR = {
							secret_type = spn_secret_werewolf
							secret_type = spn_secret_hybrid
						}
						is_known_by = root
					}
				}
			}
		}
		modifier = {
			add = 30
			has_character_flag = spn_wolf_broke_out
		}
	}
}
