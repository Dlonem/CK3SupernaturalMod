# Supernatural 2.40 (WORK-ORDER-2.40 F2) -- household administration, mirrored on the Long Night's
# (common/decisions/zz_ln_household_decisions.txt). See zz_spn_240_household_triggers.txt for the key.
#
# EVERY GATE IS IN is_shown: a decision that fails is_valid is drawn greyed with its requirements, which would show the
# door to every player; one that fails is_shown does not exist. is_valid only ever says why a shown one cannot be taken
# right now, and only Thomas's house ever sees these.
# Boot 1 (22 Oct, Thomas): their own small group, "The Household", instead of the Major group. No AI ever: ai_check_interval 0, ai_potential never, ai_will_do 0. No cooldowns:
# every effect is safe to repeat (the Long Night's 1.0.9 rule).
# The embrace rule (Embrace Decisions Off, the default) does not hide the first one: it is his own gift (D7/F2, option a).
# The hides rule does hide the second.

# ---- 1. Answer the Old Blood: vampire, werewolf, hybrid or demon. It returns if he is ever cured. ----------------------
spn_household_blood_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_personal_religious.dds" }
	desc = spn_household_blood_decision_desc
	selection_tooltip = spn_household_blood_decision_tt
	confirm_text = spn_household_blood_decision_confirm
	decision_group_type = spn_household
	ai_check_interval = 0

	is_shown = {
		spn_household_has_door_trigger = yes
		spn_has_supernatural_trait = no
		spn_has_hunter_trait = no
		spn_can_become_creature_trigger = yes
		NOT = { has_character_flag = spn_embrace_choosing }
	}

	is_valid_showing_failures_only = {
		is_available_adult_or_is_commanding = yes
	}

	effect = {
		custom_tooltip = spn_household_blood_decision_effect_tt
		hidden_effect = {
			add_character_flag = { flag = spn_embrace_choosing days = 30 } # cleared by the level picker or by .0001's after
			trigger_event = spn_household.0001
		}
	}

	ai_potential = { always = no }
	ai_will_do = { base = 0 }
}
