# Supernatural 2.41 -- Glitterhoof (WORK-ORDER-2.41 §3.4, C5).
# The yearly roll for an eligible ruler: 10% for a player (about one decade on average), a quarter of that for an
# AI king or emperor (the trigger already keeps AI below kingdom tier out).
spn_glitterhoof_chance_value = {
	value = 10
	if = {
		limit = { is_ai = yes }
		multiply = 0.25
	}
}

# The Horse Question (§3.8): horses at your court and among your vassals.
spn_horse_count_value = {
	value = 0
	every_courtier = {
		limit = { has_trait = spn_horse }
		add = 1
	}
	every_vassal_or_below = {
		limit = { has_trait = spn_horse }
		add = 1
	}
}
