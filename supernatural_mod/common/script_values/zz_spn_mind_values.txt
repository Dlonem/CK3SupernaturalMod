# Supernatural 2.39 -- the wolf remembers (WORK-ORDER-2.39 F17 (a)). Every knob in one file; none is ever shown.

# Root = the wiped wolf, after the ring was counted. 10 + 8 per counted ring, +10 at Hide Bond 50 or more, at most 60:
# 18% on the first ringing and 34% on the third, "especially when the ringing kept happening".
spn_pm_ring_return_chance_value = {
	value = 10
	if = {
		limit = { has_variable = spn_pm_rings }
		add = {
			value = var:spn_pm_rings
			multiply = 8
		}
	}
	if = {
		limit = { spn_hide_bond_value >= 50 }    # zz_spn_hide_values.txt (0 for anyone who is not a champion)
		add = 10
	}
	max = 60
}

# Root = the wiped wolf (spn_order.0196, the slow pull). 8% plus 0.12 per point of the wolf's track: 8-20%.
spn_pm_pull_chance_value = {
	value = 8
	add = {
		value = spn_hunt_creature_track_value    # 20_spn_utility_values.txt
		multiply = 0.12
	}
	max = 20
}
