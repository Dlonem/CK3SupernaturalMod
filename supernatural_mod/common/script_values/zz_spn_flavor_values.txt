# Supernatural 2.39 -- the values of F14 (WORK-ORDER-2.39 F14): the chapter moment's chance and the hidden rolls of Order
# romance. No option text ever shows one of these numbers.

# The chance each player of a chapter has of a chapter moment in the chapter's year (expert call D-F2: about one every
# 2.5 years). It was the literal 20, once per chapter, before 2.39.
spn_order_moment_chance_value = 40

# An AI junior lover asks his patron for a word (junior's scope).
spn_order_favour_ask_chance_value = {
	value = 20
	if = {
		limit = { has_trait = ambitious }
		add = 30
	}
	if = {
		limit = { has_trait = greedy }
		add = 20
	}
	if = {
		limit = { has_trait = arbitrary }
		add = 10
	}
	if = {
		limit = { has_trait = honest }
		add = -20
	}
	if = {
		limit = { has_trait = content }
		add = -20
	}
	min = 0
	max = 80
}

# An AI patron agrees to speak for his lover (patron's scope).
spn_order_favour_grant_chance_value = {

		limit = { has_trait = arbitrary }

	}
	if = {
		limit = { has_trait = lustful }

	}

		limit = { has_trait = just }
		add = -30
	}
	if = {
		limit = { has_trait = honest }
		add = -20
	}
	if = {
		limit = { has_trait = zealous }
		add = -10
	}
	min = 5

}

# The talk (junior's scope; reads scope:spn_fe_junior and scope:spn_fe_patron, saved by spn_order_favour_effect): +10 for
# each AI member of his rank in his temple who resents him, +10 when the patron is the Grand Magus, less his intrigue
# above 10.
spn_order_favour_scandal_chance_value = {
	value = 15
	if = {
		limit = { exists = var:spn_order_chapter }
		var:spn_order_chapter = {
			every_in_list = {
				variable = spn_order_members
				limit = { spn_order_favour_jealous_trigger = { JUNIOR = scope:spn_fe_junior PATRON = scope:spn_fe_patron } }
				add = 10
			}
		}
	}
	if = {
		limit = { scope:spn_fe_patron = { global_var:current_archmage_global ?= this } }
		add = 10
	}
	if = {
		limit = { intrigue > 10 }
		subtract = {
			value = intrigue
			subtract = 10
		}
	}
	min = 5
	max = 60
}

# A Door Left Open: does the senior say yes? (senior's scope; scope:spn_adv_junior is the one asking, and
# scope:spn_adv_bold is set when he also asks for something.)
spn_order_advance_accept_value = {
	value = 30
	add = {
		value = "opinion(scope:spn_adv_junior)"
		divide = 2
	}
	if = {
		limit = { has_trait = lustful }
		add = 20
	}
	if = {
		limit = { has_trait = gregarious }
		add = 10
	}
	if = {
		limit = { has_trait = chaste }
		add = -30
	}
	if = {
		limit = { has_trait = just }
		add = -15
	}
	if = {
		limit = { has_trait = zealous }
		add = -10
	}
	if = {
		limit = { exists = scope:spn_adv_bold }
		add = -15
	}
	min = 5
	max = 85
}
