# Supernatural 2.41 (WORK-ORDER-2.41 §7.1) -- nature against nurture. Character scope: the demon-blooded wolf.
# Up is the wolf, down is the demon. The growing-up beats (var:spn_two_natures_lean), his own temper, and the people who
# raised him: "you can be a good demon" (Thomas, 2 Nov).
spn_two_natures_score_value = {
	value = 0
	if = {
		limit = { exists = var:spn_two_natures_lean }
		add = var:spn_two_natures_lean
	}
	if = { limit = { has_trait = compassionate } add = 1 }
	if = { limit = { has_trait = forgiving } add = 1 }
	if = { limit = { has_trait = calm } add = 1 }
	if = { limit = { has_trait = wrathful } add = -1 }
	if = { limit = { has_trait = sadistic } add = -2 }
	if = { limit = { has_trait = callous } add = -1 }
	every_relation = {
		type = guardian
		if = { limit = { has_trait = compassionate } add = 1 }
		if = { limit = { has_trait = sadistic } add = -1 }
	}
	every_parent = {
		limit = { is_alive = yes }
		if = { limit = { has_trait = compassionate } add = 1 }
		if = { limit = { is_a_demon_trigger = yes } add = -1 }
	}
}
