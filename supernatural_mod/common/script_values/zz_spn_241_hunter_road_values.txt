# Supernatural 2.41 boot 10 (5 Nov 2026) -- HUNTERS TAKE TO THE ROAD. zz_spn_241_hunter_road_effects.txt has the why.

# A hunter needs this much of either track to go on the road (written as a literal 20 in spn_hunter_road_eligible_trigger;
# kept here as the documented number). Thomas: "greater than 20 XP in either trait track. That way
# we're not just making hunters willy-nilly."
spn_hunter_road_xp_value = 20

# Each year, the chance an eligible hunter goes.
spn_hunter_road_chance_value = 35 # 2.43: was 20 (3 hunters took the road in 71 years)

# How many hunter camps the world can carry: one for every two live trails (read from the census, which runs each
# January), never fewer than 2 or more than 8. Only camps made here are counted.
# CHARACTER scope (a hunter). The most track a quarry may have before an AI hunter leaves it alone
# (supernatural_hunt.txt, ai_will_select_province).
spn_hunt_ai_overmatch_value = {
	value = spn_hunt_hunter_track_value
	# 2.43 (balance read of the 22 Nov co-op save: 36 trails, 0 hunts). Immortals gain track every year and a hunter only
	# by hunting, so at +25 nearly every quarry in the world was "too strong" for nearly every hunter, and none rode.
	# Now +35 alone, and +10 for each other hunter he can bring (up to three): a band takes on what one man would not.
	add = 35
	if = { limit = { any_courtier_or_guest = { count >= 1 is_adult = yes has_trait = supernatural_hunter } } add = 10 }
	if = { limit = { any_courtier_or_guest = { count >= 2 is_adult = yes has_trait = supernatural_hunter } } add = 10 }
	if = { limit = { any_courtier_or_guest = { count >= 3 is_adult = yes has_trait = supernatural_hunter } } add = 10 }
}

spn_hunter_camp_cap_value = {
	value = 0
	if = {
		limit = { exists = global_var:spn_tc_live }
		add = global_var:spn_tc_live
		multiply = 0.5
	}
	round = yes
	min = 2
	max = 8
}
