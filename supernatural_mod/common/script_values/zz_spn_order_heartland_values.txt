# Supernatural 2.38f -- the Order's heartland (Thomas, 9 Oct: "an emphasis in Western Europe").
#
# Read by spn_order_seat_score_value (zz_spn_order_values.txt), which orders the Mages a temple may be seeded around: a
# Mage in Western Europe is taken first while one exists, and the spacing rule still spreads the other temples over the
# world. Character scope.
#
# KEPT IN ITS OWN FILE ON PURPOSE. These are vanilla-map regions. On a map that does not have them (AGOT and other total
# conversions) the region test fails the game's load check and reads false: the lean is simply gone, the seat score works
# as before, and error.log shows one line at startup. A compatibility patch can override this file, by name, with its own
# regions. Nothing else in the mod names a vanilla region for the Order.
spn_order_heartland_value = {
	value = 0
	if = {
		limit = {
			location ?= { geographical_region = world_europe_west }
		}
		add = 40
	}
}
