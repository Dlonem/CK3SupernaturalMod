# Supernatural 2.39 -- how rare the Collegium is (Thomas's call, 16 Oct; zz_spn_239_collegium_rarity_triggers.txt).

# The most ordinary Collegia the AI raises (Thomas: up to 7; each temple's hall is extra).
spn_collegia_ordinary_cap_value = 7

# Any scope. Collegia standing or being raised with no temple beneath them: the world list (spn_collegium_register_effect)
# less the halls, and less those being raised for a temple (spn_collegium_for_hall, set on_start). Under this rule the list
# holds a dozen or so provinces, so this costs little.
spn_collegia_ordinary_value = {
	value = 0
	every_in_global_list = {
		variable = spn_collegia
		limit = {
			NOT = { exists = var:spn_order_hall_of }
			NOT = { has_variable = spn_collegium_for_hall }
		}
		add = 1
	}
}

# What the Gnostic Council pays to raise a temple's own hall (Boot 3; zz_spn_collegium_effects.txt step 2c): the
# Collegium's own price (vanilla's expensive_building_tier_2_cost, 350).
spn_order_hall_council_cost_value = 350
