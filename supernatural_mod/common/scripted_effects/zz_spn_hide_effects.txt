# Supernatural 2.37 -- "The Six Hides": the hide's effects (WORK-ORDER-2.37 §4.12, §5, §6, §7.3, §8, §9).
#
# STANDING RULES THAT SHAPE THIS FILE
#  * Bank before anything irreversible: every value a later line reads is saved first; the death, the destroy and the
#    removal of a trait are the last thing each effect does with what it read.
#  * Never use `root` inside a scripted effect to mean "the character this effect is about": these are called from
#    on_death, from events, from interactions and from each other, and root is whatever started the chain. Every
#    effect saves its own subject first (save_scope_as) and reads that.
#  * Six hides, six names. A hide is a flag on its artifact (var:spn_hide = flag:<hide>) and a champion carries the
#    same flag (var:spn_champion_of), so dispatchers turn a flag into the HIDE parameter once and the per-hide effects
#    read traits and globals by name.
#  * Nothing iterates over the world except where it is said so, with why.

# ==================================================================================================
# A HIDE'S FLAG INTO A SAVED SCOPE VALUE (2.37b). save_scope_value_as takes a flag only as a literal (flag:x), the way
# vanilla writes it 3,482 times. Handed a variable that holds one, it evaluates it as a number and fails: error.log
# 3 Oct, "Value of wrong type in ... spn_hide.0009 ... Got value of type 'flag'". Everything after read a 'value', so
# every scope:spn_fit_hide = flag:<hide> test was false and no hide ever counted a single trait. One literal per hide.
# FROM = anything that holds a hide flag (scope:x.var:spn_hide, var:spn_champion_of); NAME = the scope to save it as.
# ==================================================================================================
spn_hide_save_flag_effect = {
	if = {
		limit = { $FROM$ = flag:silverback }
		save_scope_value_as = { name = $NAME$ value = flag:silverback }
	}
	else_if = {
		limit = { $FROM$ = flag:timber }
		save_scope_value_as = { name = $NAME$ value = flag:timber }
	}
	else_if = {
		limit = { $FROM$ = flag:tundra }
		save_scope_value_as = { name = $NAME$ value = flag:tundra }
	}
	else_if = {
		limit = { $FROM$ = flag:graybeard }
		save_scope_value_as = { name = $NAME$ value = flag:graybeard }
	}
	else_if = {
		limit = { $FROM$ = flag:midnight }
		save_scope_value_as = { name = $NAME$ value = flag:midnight }
	}
	else_if = {
		limit = { $FROM$ = flag:alpha }
		save_scope_value_as = { name = $NAME$ value = flag:alpha }
	}
}

# ==================================================================================================
# CREATION -- Witchcraft's own create_artifact shape (00_creating_artifact_effects.txt). OWNER = a character.
# A hide that comes back after being lost reads the same global history, so it is the same hide.
# ==================================================================================================
spn_hide_create_effect = {
	$OWNER$ = {
		create_artifact = {
			name = spn_hide_$HIDE$_name
			description = spn_hide_$HIDE$_desc
			template = spn_hide_template
			type = spn_hide
			visuals = pelt
			modifier = spn_hide_$HIDE$_modifier
			rarity = illustrious
			decaying = no
			history = { type = created_before_history }
			save_scope_as = spn_new_hide
		}
	}
	# 2.37 -- guarded: an event option that creates a hide (The Wolf That Was More Than a Wolf, the debug events) is walked
	# for its tooltip without create_artifact running, and the bare read errored (the Found a Pack class).
	if = {
		limit = { exists = scope:spn_new_hide }
		scope:spn_new_hide = {
			set_variable = { name = spn_hide value = flag:$HIDE$ }
		}
		# Only a hide that was really made takes the hide off the lost record.
		if = {
			limit = { exists = global_var:spn_hide_$HIDE$_lost }
			remove_global_variable = spn_hide_$HIDE$_lost
		}
		if = {
			limit = { exists = global_var:spn_hide_$HIDE$_heard_by }
			remove_global_variable = spn_hide_$HIDE$_heard_by
		}
	}
	$OWNER$ = { spn_hide_start_check_effect = yes }
}

# Lost: the hide waits 5-20 years (Alpha 15-40, design §3.4) and then goes looking (§9.5). The history is in globals.
spn_hide_record_lost_effect = {
	set_global_variable = {
		name = spn_hide_$HIDE$_lost
		value = {
			value = current_year
			add = { integer_range = { min = spn_hide_lost_wait_min_$HIDE$_value max = spn_hide_lost_wait_max_$HIDE$_value } }
		}
	}
	if = {
		limit = { exists = global_var:spn_hide_$HIDE$_reserved }
		remove_global_variable = spn_hide_$HIDE$_reserved
	}
}

# A chase lapsed, nobody passed the bar, or the reserved wolf died: back to waiting, 2-5 years.
spn_hide_rewait_effect = {
	set_global_variable = {
		name = spn_hide_$HIDE$_lost
		value = {
			value = current_year
			add = { integer_range = { min = 2 max = 5 } }
		}
	}
	if = {
		limit = { exists = global_var:spn_hide_$HIDE$_reserved }
		remove_global_variable = spn_hide_$HIDE$_reserved
	}
}

# ==================================================================================================
# THE WEEKLY CHECK's chain (§5.5). One per holder; the pending flag is the guard against two chains.
# ==================================================================================================
spn_hide_start_check_effect = {
	if = {
		limit = { NOT = { has_character_flag = spn_hide_check_pending } }
		add_character_flag = { flag = spn_hide_check_pending days = 8 }
		trigger_event = { id = spn_hide.0001 days = 1 }
	}
}

# ==================================================================================================
# CLAIMS (§5.4). Artifact scope. A hide worn by its own champion carries no claims; each stripped claimant whose pack
# lost the hide is recorded as a feud (§17.5) -- never the owner's own house, never his own pack.
# ==================================================================================================
spn_hide_strip_claims_effect = {
	save_scope_as = spn_strip_hide
	artifact_owner ?= { save_scope_as = spn_strip_owner }
	every_artifact_claimant = {
		if = {
			limit = {
				spn_pack_politics_built_trigger = yes
				exists = scope:spn_strip_owner
				exists = var:spn_pack
				scope:spn_strip_owner = { exists = var:spn_pack }
			}
			# Nested, not one limit: CK3 does not short-circuit AND siblings, and the cross-scope read below would
			# error for a claimant or owner outside a pack.
			if = {
				limit = { NOT = { var:spn_pack = scope:spn_strip_owner.var:spn_pack } }
				var:spn_pack.story_owner = { save_scope_as = spn_feud_a }
				scope:spn_strip_owner.var:spn_pack.story_owner = { save_scope_as = spn_feud_b }
				spn_hide_feud_effect = { A = scope:spn_feud_a B = scope:spn_feud_b }
			}
		}
		remove_personal_artifact_claim = scope:spn_strip_hide
	}
	every_artifact_house_claimant = {
		remove_house_artifact_claim = scope:spn_strip_hide
	}
}

# ==================================================================================================
# MAKING A CHAMPION (§5.9). Character scope; scope:spn_mc_hide = the artifact. Order matters: bank first.
# Called through spn_hide_make_champion_dispatch_effect, which turns the artifact's flag into HIDE.
# ==================================================================================================
spn_hide_make_champion_dispatch_effect = {
	$ARTIFACT$ = { save_scope_as = spn_mc_hide }
	# 2.37o: a hide that takes a champion is not idle; its count starts again (the review of 4 Oct, P6).
	scope:spn_mc_hide = {
		if = {
			limit = { exists = var:spn_hide_idle_since }
			remove_variable = spn_hide_idle_since
		}
	}
	if = {
		limit = { scope:spn_mc_hide.var:spn_hide = flag:silverback }
		spn_hide_make_champion_effect = { HIDE = silverback }
	}
	else_if = {
		limit = { scope:spn_mc_hide.var:spn_hide = flag:timber }
		spn_hide_make_champion_effect = { HIDE = timber }
	}
	else_if = {
		limit = { scope:spn_mc_hide.var:spn_hide = flag:tundra }
		spn_hide_make_champion_effect = { HIDE = tundra }
	}
	else_if = {
		limit = { scope:spn_mc_hide.var:spn_hide = flag:graybeard }
		spn_hide_make_champion_effect = { HIDE = graybeard }
	}
	else_if = {
		limit = { scope:spn_mc_hide.var:spn_hide = flag:midnight }
		spn_hide_make_champion_effect = { HIDE = midnight }
	}
	else_if = {
		limit = { scope:spn_mc_hide.var:spn_hide = flag:alpha }
		spn_hide_make_champion_effect = { HIDE = alpha }
	}
}

spn_hide_make_champion_effect = {
	save_scope_as = spn_mc_champ
	# 1. One body, one hide: a champion of another hide is Two Wolves, never here.
	if = {
		limit = { spn_is_champion_trigger = no }
		# 2. Read the history. The start is the hide's legacy (half the Bond of the last who left his mark), or three
		#    quarters of his own kept Bond if he wore it before -- the higher of the two (design §2.3, "it remembers him").
		save_scope_value_as = {
			name = spn_mc_legacy
			value = {
				value = 0
				if = {
					limit = { exists = global_var:spn_hide_$HIDE$_legacy }
					value = global_var:spn_hide_$HIDE$_legacy
				}
			}
		}
		save_scope_value_as = {
			name = spn_mc_kept
			value = {
				value = 0
				if = {
					limit = { exists = var:spn_hide_bond_kept_$HIDE$ }
					value = var:spn_hide_bond_kept_$HIDE$
					multiply = 0.75
				}
			}
		}
		# 3. The trait and its starting Bond.
		add_trait = spn_champion_$HIDE$
		set_variable = { name = spn_champion_of value = flag:$HIDE$ }
		if = {
			limit = { scope:spn_mc_kept > scope:spn_mc_legacy }
			add_trait_xp = { trait = spn_champion_$HIDE$ track = hide_bond value = scope:spn_mc_kept }
		}
		else_if = {
			limit = { scope:spn_mc_legacy > 0 }
			add_trait_xp = { trait = spn_champion_$HIDE$ track = hide_bond value = scope:spn_mc_legacy }
		}
		# 4. The tongues: every culture that left its mark on this hide.
		every_in_global_list = {
			variable = spn_hide_$HIDE$_marks
			save_temporary_scope_as = spn_mc_tongue
			scope:spn_mc_champ = {
				# 2.37i: only a tongue he lacks (Thomas's 4 Oct run: "Already knows language").
				if = {
					limit = { NOT = { knows_language_of_culture = scope:spn_mc_tongue } }
					learn_language_of_culture = scope:spn_mc_tongue
				}
			}
		}
		# 5. The world's record.
		set_global_variable = { name = spn_hide_champion_$HIDE$ value = scope:spn_mc_champ }
		if = {
			limit = { NOT = { is_target_in_global_variable_list = { name = spn_hide_$HIDE$_champions target = scope:spn_mc_champ } } }
			add_to_global_variable_list = { name = spn_hide_$HIDE$_champions target = scope:spn_mc_champ }
		}
		# 6. The guards: nothing vanilla moves an artifact carrying artifact_prevent_transfer, and it carries no claims.
		scope:spn_mc_hide = {
			set_variable = artifact_prevent_transfer
			spn_hide_strip_claims_effect = yes
		}
		# 7. It is on him (_replace takes an occupied slot; plain equip fails on a full trinket row).
		scope:spn_mc_hide = {
			if = {
				limit = { is_equipped = no }
				equip_artifact_to_owner_replace = yes
			}
		}
		spn_hide_start_check_effect = yes
		if = {
			limit = { has_character_flag = spn_hide_fever_bonus }
			remove_character_flag = spn_hide_fever_bonus
		}
		# 7c. The pack learns it has a champion; Timber's runs alone outside one.
		if = {
			limit = { exists = var:spn_pack }
			var:spn_pack = { spn_pack_refresh_effect = yes }
		}
		else_if = {
			limit = { has_trait = spn_champion_timber }
			remove_all_character_modifier_instances = spn_alone_in_the_woods_modifier
			add_character_modifier = { modifier = spn_alone_in_the_woods_modifier years = 2 }
		}
		spn_hide_ageless_effect = yes # 2.37: an immortal champion's gift
		spn_hide_keep_whole_effect = yes # 2.37i: and old age falls away from him (Thomas, 4 Oct)
		# 8. The memory: the first time only (the 2.34 rules).
		if = {
			limit = { NOT = { any_memory = { has_memory_type = spn_hide_chosen_memory } } }
			create_character_memory = { type = spn_hide_chosen_memory }
		}
		# 9. Everyone knows what a known wolf carries.
		spn_hide_refresh_dread_effect = yes
		# 10. Beat 1 goes through the beat check like every beat: banked before it is fired, so a high starting Bond
		#     cannot schedule beat 2 beside it. Skipped if he has already seen it on this hide.
		if = {
			limit = {
				trigger_if = {
					limit = { exists = var:spn_hide_beat_$HIDE$ }
					var:spn_hide_beat_$HIDE$ < 1
				}
			}
			set_variable = { name = spn_hide_beat_$HIDE$ value = 1 }
			add_character_flag = { flag = spn_hide_beat_cooldown years = 2 }
			save_scope_value_as = { name = spn_beat_hide value = flag:$HIDE$ }
			trigger_event = { id = spn_hide.0100 days = 14 }
		}
	}
}

# 2.37i (Thomas, 4 Oct: a 76-year-old Champion of Alpha was blind, incapable, infirm and losing his mind). A hide
# keeps its champion whole. The vampire's own cleanup (spn_remove_negative_age_traits: infirm, clouded eyes, a
# faltering heart, a withering mind, fragile bones, incapable) plus blindness, which clouded eyes become at their
# worst (vanilla blind_on_max_level). Run when he is chosen, and yearly by spn_age_trait_cleanup with the vampires.
spn_hide_keep_whole_effect = {
	spn_remove_negative_age_traits = yes
	if = {
		limit = { has_trait = blind }
		remove_trait = blind
	}
}

spn_hide_refresh_dread_effect = {
	if = {
		limit = { has_character_modifier = spn_hide_known_dread_modifier }
		remove_character_modifier = spn_hide_known_dread_modifier
	}
	if = {
		limit = { has_character_modifier = spn_hide_known_dread_timber_modifier }
		remove_character_modifier = spn_hide_known_dread_timber_modifier
	}
	if = {
		limit = {
			spn_is_champion_trigger = yes
			OR = {
				has_trait = spn_known_werewolf
				has_trait = spn_known_hybrid
			}
		}
		if = {
			limit = { has_trait = spn_champion_timber }
			add_character_modifier = { modifier = spn_hide_known_dread_timber_modifier years = 2 }
		}
		else = {
			add_character_modifier = { modifier = spn_hide_known_dread_modifier years = 2 }
		}
	}
}

# 2.37 (Thomas, 3 Oct: "they know Latin because the hide knows Latin"): at Hide Bond 25 the hide takes its champion's
# tongue on its own, no question asked. Whoever wears it after him learns it (spn_hide_make_champion_effect step 4). The
# list is the same one Leave My Mark, the seeding and the wake write, and a culture already in it is not added twice, so
# this is safe to run on every gain and every year. Champion scope.
spn_hide_tongue_effect = {
	if = {
		limit = {
			spn_is_champion_trigger = yes

			spn_hide_bond_value >= spn_hide_bond_tongue_value
		}
	if = {
		limit = { has_trait = spn_champion_silverback }
		spn_hide_tongue_one_effect = { HIDE = silverback }
	}
	else_if = {
		limit = { has_trait = spn_champion_timber }
		spn_hide_tongue_one_effect = { HIDE = timber }
	}
	else_if = {
		limit = { has_trait = spn_champion_tundra }
		spn_hide_tongue_one_effect = { HIDE = tundra }
	}
	else_if = {
		limit = { has_trait = spn_champion_graybeard }
		spn_hide_tongue_one_effect = { HIDE = graybeard }
	}
	else_if = {
		limit = { has_trait = spn_champion_midnight }
		spn_hide_tongue_one_effect = { HIDE = midnight }
	}
	else_if = {
		limit = { has_trait = spn_champion_alpha }
		spn_hide_tongue_one_effect = { HIDE = alpha }
	}
	}
}

spn_hide_tongue_one_effect = {
	if = {
		limit = { NOT = { is_target_in_global_variable_list = { name = spn_hide_$HIDE$_marks target = culture } } }
		add_to_global_variable_list = { name = spn_hide_$HIDE$_marks target = culture }
		if = {
			limit = { is_ai = no }
			send_interface_toast = {
				type = event_toast_effect_good

			}

# 2.37 (Thomas, 3 Oct): the Bond grows a little every year the hide is worn, more in a pack, so a champion who never
# goes to war still hears the hide. Each beat needs only +5 beyond the last beat's +15, and beats stay two years apart,
# so a worn hide reaches full Bond in about 10-15 years without war. Global scope, once a year: six records at most.
spn_hide_yearly_bond_effect = {
	spn_hide_yearly_bond_one_effect = { HIDE = silverback }
	spn_hide_yearly_bond_one_effect = { HIDE = timber }
	spn_hide_yearly_bond_one_effect = { HIDE = tundra }
	spn_hide_yearly_bond_one_effect = { HIDE = graybeard }
	spn_hide_yearly_bond_one_effect = { HIDE = midnight }
	spn_hide_yearly_bond_one_effect = { HIDE = alpha }
}

spn_hide_yearly_bond_one_effect = {
	global_var:spn_hide_champion_$HIDE$ ?= {
		if = {
			limit = {
				is_alive = yes
				has_trait = spn_champion_$HIDE$
			}
			if = {
				limit = {
					any_character_artifact = {
						var:spn_hide ?= flag:$HIDE$
						spn_hide_worn_by_champion_trigger = yes
					}
				}
				if = {
					limit = {
						spn_packs_on_trigger = yes
						spn_in_pack_trigger = yes
					}
					spn_hide_add_bond_effect = { VALUE = spn_hide_bond_year_pack_value }
				}
				else = {
					spn_hide_add_bond_effect = { VALUE = spn_hide_bond_year_value }
				}
			}
			spn_hide_tongue_effect = yes
		}
	}
}

# §6.3. If the next beat's threshold is met and two years have passed since the last, bank the beat FIRST, then fire
# it 7-30 days later. Falling below a threshold never undoes a beat; climbing back never repeats one.
spn_hide_check_beat_effect = {
	# 2.37: the tongue first (every Bond gain and the weekly check pass through here), so it is in the hide before any
	# later beat says it is, an older save included.
	spn_hide_tongue_effect = yes
	if = {
		limit = {
			spn_is_champion_trigger = yes
			NOT = { exists = var:spn_champion_of }
		}
		spn_hide_heal_champion_var_effect = yes
	}
	if = {
		limit = {
			spn_is_champion_trigger = yes
			NOT = { has_character_flag = spn_hide_beat_cooldown }
		}
		if = {
			limit = { has_trait = spn_champion_silverback }
			save_scope_value_as = {
				name = spn_beat_seen
				value = {
					value = 0
					if = {
						limit = { exists = var:spn_hide_beat_silverback }
						value = var:spn_hide_beat_silverback
					}
				}
			}
		}
		if = {
			limit = { has_trait = spn_champion_timber }
			save_scope_value_as = {
				name = spn_beat_seen
				value = {
					value = 0
					if = {
						limit = { exists = var:spn_hide_beat_timber }
						value = var:spn_hide_beat_timber
					}
				}
			}
		}
		if = {
			limit = { has_trait = spn_champion_tundra }
			save_scope_value_as = {
				name = spn_beat_seen
				value = {
					value = 0
					if = {
						limit = { exists = var:spn_hide_beat_tundra }
						value = var:spn_hide_beat_tundra
					}
				}
			}

# ==================================================================================================
# THE CLASH (§7.3). Creature against creature: Challenge, champion against champion, the ringing's chase.
#   A, B   = the two sides
#   THEN   = a scripted effect run after the roll; it reads scope:spn_clash_winner and scope:spn_clash_loser
#   FORM   = yes when both count as in wolf form (Challenge: "a brawl, no swords"), else no
# Both weights and the outcome are banked before THEN runs. The roll is hidden: no odds are printed (his rule against
# telegraphed outcomes); spn_clash_lean_tooltip_effect says which way it leans in words.
# ==================================================================================================
spn_creature_clash_effect = {
	# A second clash in the same event (a battle, the yearly moment) must not read the first one's outcome.
	if = {
		limit = { exists = scope:spn_clash_a_won }
		clear_saved_scope = spn_clash_a_won
	}
	if = {
		limit = { exists = scope:spn_clash_wolf_form }
		clear_saved_scope = spn_clash_wolf_form
	}
	$A$ = { save_scope_as = spn_clash_a }
	$B$ = { save_scope_as = spn_clash_b }
	if = {
		limit = { always = $FORM$ }
		save_scope_value_as = { name = spn_clash_wolf_form value = yes }
	}
	scope:spn_clash_b = { save_scope_as = spn_clash_foe }
	scope:spn_clash_a = {
		save_scope_value_as = { name = spn_clash_wa value = spn_creature_clash_weight_value }
	}
	scope:spn_clash_a = { save_scope_as = spn_clash_foe }
	scope:spn_clash_b = {
		save_scope_value_as = { name = spn_clash_wb value = spn_creature_clash_weight_value }
	}
	hidden_effect = {
		random = {
			chance = spn_clash_chance_a_value
			save_scope_value_as = { name = spn_clash_a_won value = yes }
		}
	}
	# Hidden, all of it: in a tooltip the else branch would print as if B had already won.
	hidden_effect = {
		if = {
			limit = { exists = scope:spn_clash_a_won }
			scope:spn_clash_a = { save_scope_as = spn_clash_winner }
			scope:spn_clash_b = { save_scope_as = spn_clash_loser }
		}
		else = {
			scope:spn_clash_b = { save_scope_as = spn_clash_winner }
			scope:spn_clash_a = { save_scope_as = spn_clash_loser }
		}
		$THEN$ = yes
	}
}

# Words, not odds: the lean from A's side. Same weights the clash will use.
spn_clash_lean_tooltip_effect = {
	$A$ = { save_scope_as = spn_clash_a }
	$B$ = { save_scope_as = spn_clash_b }
	# 2.37e: FORM as the real clash has it (the Ringing fights out of the wolf's shape, FORM = no); it was always yes.
	# The weight reads exists = scope:spn_clash_wolf_form, so it is cleared and set exactly as spn_creature_clash_effect does.
	if = {
		limit = { exists = scope:spn_clash_wolf_form }
		clear_saved_scope = spn_clash_wolf_form
	}
	if = {
		limit = { always = $FORM$ }
		save_scope_value_as = { name = spn_clash_wolf_form value = yes }
	}
	scope:spn_clash_b = { save_scope_as = spn_clash_foe }
	scope:spn_clash_a = { save_scope_value_as = { name = spn_clash_wa value = spn_creature_clash_weight_value } }
	scope:spn_clash_a = { save_scope_as = spn_clash_foe }
	scope:spn_clash_b = { save_scope_value_as = { name = spn_clash_wb value = spn_creature_clash_weight_value } }
	if = {
		limit = { spn_clash_chance_a_value >= 65 }
		custom_tooltip = spn_clash_lean_strong_tt
	}
	else_if = {
		limit = { spn_clash_chance_a_value <= 35 }
		custom_tooltip = spn_clash_lean_weak_tt
	}
	else = {
		custom_tooltip = spn_clash_lean_even_tt
	}
}

spn_hide_exit_hide_effect = {
	save_scope_as = spn_exit_champ
	# 1. Save the hide (it may already be gone: destroyed in the days it was off). A scope left from an earlier exit
	#    in this chain must not stand in for it.
	if = {
		limit = { exists = scope:spn_exit_hide }
		clear_saved_scope = spn_exit_hide
	}
	random_character_artifact = {
		limit = { var:spn_hide ?= flag:$HIDE$ }
		save_scope_as = spn_exit_hide
	}
	# 2. Bank: his Bond for "it remembers him", the legacy if he left his mark, and the last champion's name.
	set_variable = { name = spn_hide_bond_kept_$HIDE$ value = spn_hide_bond_value }
	if = {
		limit = { has_character_flag = spn_hide_left_mark_$HIDE$ }
		set_global_variable = {
			name = spn_hide_$HIDE$_legacy
			value = {
				value = spn_hide_bond_value
				divide = 2
			}
		}
	}
	set_global_variable = { name = spn_hide_$HIDE$_last value = scope:spn_exit_champ }
	# 3. The guard off the artifact.
	scope:spn_exit_hide ?= {
		if = {
			limit = { has_variable = artifact_prevent_transfer }
			remove_variable = artifact_prevent_transfer
		}
	}
	# 4. The trait off.
	remove_trait = spn_champion_$HIDE$
	remove_variable = spn_champion_of
	# 5. The world's record -- only if it still points at him, so a stale champion's exit never erases a newer one.
	if = {
		limit = { global_var:spn_hide_champion_$HIDE$ ?= scope:spn_exit_champ }
		remove_global_variable = spn_hide_champion_$HIDE$
	}
	# 6. Move the hide, by reason.
	if = {
		limit = { scope:spn_exit_reason = flag:death }
		spn_hide_move_on_death_effect = { HIDE = $HIDE$ }
	}
	else_if = {
		limit = { scope:spn_exit_reason = flag:lost }
		# The weekly check sent this only after it found the hide nowhere in the world.
		spn_hide_record_lost_effect = { HIDE = $HIDE$ }
	}
	# 2.37e: unbound, abandoned, lost blood: it comes off him and stays in his hands, unguarded. Unbound will also not
	# look at him again for ten years: worn and unrefused, the weekly check offered it straight back within days and
	# the mage's work bought nothing. A mortal who abandoned the blood no longer wears it either, so It Knows You (which
	# needs it off) can find him. two_wolves: the event that sent it handles both hides. soul: it is on the new body.
	else_if = {
		limit = {
			OR = {
				scope:spn_exit_reason = flag:unbound
				scope:spn_exit_reason = flag:abandoned
				scope:spn_exit_reason = flag:lost_blood
			}
		}
		scope:spn_exit_hide ?= {
			if = {
				limit = { is_equipped = yes }
				unequip_artifact_from_owner = yes
			}
		}
		if = {
			limit = { scope:spn_exit_reason = flag:unbound }
			add_character_flag = { flag = spn_hide_refused_$HIDE$ years = 10 }
		}
	}
	# 7. Claims, for a death only, and only after the move and only if the hide still exists: his house, and his pack's
	#    alpha if that is not himself. An unbound or abandoning champion keeps his own hide and must hold no claim on it.
	if = {
		limit = {
			scope:spn_exit_reason = flag:death
			exists = scope:spn_exit_hide
		}
		house ?= { add_house_artifact_claim = scope:spn_exit_hide }
		# 2.37e: nested -- the story_owner read errored on every packless champion's death (no short-circuit).
		if = {
			limit = { exists = var:spn_pack }
			if = {
				limit = { NOT = { var:spn_pack.story_owner ?= scope:spn_exit_champ } }
				var:spn_pack.story_owner ?= {
					if = {
						limit = { is_alive = yes }
						add_personal_artifact_claim = scope:spn_exit_hide
					}
				}
			}
		}
	}
	# 8. The memory, for the living. A dead man's memories are written for nobody.
	if = {
		limit = {
			NOT = { scope:spn_exit_reason = flag:death }
			NOT = { any_memory = { has_memory_type = spn_hide_left_me_memory } }
		}
		create_character_memory = { type = spn_hide_left_me_memory }
	}
	# 9. The pack learns it has one champion fewer; Timber's gift and the ageless gift go with the hide.
	if = {
		limit = { has_character_modifier = spn_alone_in_the_woods_modifier }
		remove_all_character_modifier_instances = spn_alone_in_the_woods_modifier
	}
	if = {
		limit = { has_character_modifier = spn_hide_ageless_modifier }
		remove_all_character_modifier_instances = spn_hide_ageless_modifier
	}
	if = {
		limit = { exists = var:spn_pack }
		var:spn_pack = { spn_pack_refresh_effect = yes }
	}
	# 10. What only a champion carries goes too. The card's timer and the gift flags stay: they are his, once a life.
	spn_hide_refresh_dread_effect = yes
	if = {
		limit = { has_character_modifier = spn_let_it_take_me_modifier }
		remove_character_modifier = spn_let_it_take_me_modifier
	}
	if = {
		limit = { has_character_flag = spn_hide_wolf_runs }
		remove_character_flag = spn_hide_wolf_runs
	}
	if = {
		limit = { has_character_flag = spn_hide_off_since }
		remove_character_flag = spn_hide_off_since
	}
}

# §8 death row. Scope: the dying champion. The killer if he is alive and a person (the species gate keeps a dragon or
# a wight from taking it); else the primary heir; else lost, which means destroyed -- bank, destroy, then record it
# lost, or vanilla inheritance would hand the artifact on while the lost record makes a second one.
spn_hide_move_on_death_effect = {
	if = {
		limit = { exists = scope:spn_exit_hide }
		if = {
			limit = {
				scope:killer ?= { # 2.37e: ?= (most champions die with no killer)
					is_alive = yes
					NOT = { this = scope:spn_exit_champ }
					spn_is_valid_supernatural_target = yes
				}
			}
			scope:spn_exit_hide = {
				set_owner = {
					target = scope:killer
					history = {
						type = taken_in_battle
						actor = scope:spn_exit_champ
						recipient = scope:killer
					}
				}
			}
			scope:killer = { spn_hide_start_check_effect = yes }
			# What follows the hide: the hunter's trophy event, or Two Wolves for a champion who killed a champion.
			if = {
				limit = { scope:killer = { has_trait = supernatural_hunter } }
				scope:killer = {
					save_scope_as = spn_trophy_hunter
					trigger_event = { id = spn_hide.0320 days = 2 }
				}
			}
			# Deferred, never inline: this runs inside the dead man's exit, and a second exit nested here would
			# overwrite the scopes the first is still reading (its claims, its pack).
			else_if = {
				limit = { scope:killer = { spn_is_champion_trigger = yes } }
				scope:spn_exit_hide = { save_scope_as = spn_offered_hide }
				scope:killer = {
					if = {
						limit = { is_ai = no }
						trigger_event = { id = spn_hide.0012 days = 3 }
					}
					else = {
						trigger_event = { id = spn_hide.0016 days = 3 }
					}
				}
			}
		}
		else_if = {
			limit = { scope:spn_exit_champ.primary_heir ?= { is_alive = yes } }
			scope:spn_exit_champ.primary_heir = { save_scope_as = spn_exit_heir }
			scope:spn_exit_hide = {
				set_owner = {
					target = scope:spn_exit_heir
					history = {
						type = inherited
						actor = scope:spn_exit_champ
						recipient = scope:spn_exit_heir
					}
				}
			}
			scope:spn_exit_heir = { spn_hide_start_check_effect = yes }
		}
		else = {
			destroy_artifact = scope:spn_exit_hide
			clear_saved_scope = spn_exit_hide
			spn_hide_record_lost_effect = { HIDE = $HIDE$ }
		}
	}
	else = {
		spn_hide_record_lost_effect = { HIDE = $HIDE$ }
	}
}

# An AI champion who has come to hold a second hide (a clash on the field, a death; spn_hide.0016): he keeps the one
# that fits him better and hands the other to the best-fitting wolf in his pack, else keeps it in his chest (§5.10).
spn_hide_ai_two_wolves_effect = {
	save_scope_as = spn_two_ai
	$ARTIFACT$ = { save_scope_as = spn_two_new }
	random_character_artifact = {
		limit = { var:spn_hide ?= scope:spn_two_ai.var:spn_champion_of }
		save_scope_as = spn_two_old
	}
	spn_hide_save_flag_effect = { NAME = spn_fit_hide FROM = var:spn_champion_of }
	save_scope_value_as = { name = spn_two_fit_old value = spn_hide_fit_value }
	spn_hide_save_flag_effect = { NAME = spn_fit_hide FROM = scope:spn_two_new.var:spn_hide }
	save_scope_value_as = { name = spn_two_fit_new value = spn_hide_fit_value }
	# Keep the better fit. Swapping: the exit banks the old one's history first, then the new one is made his.
	if = {
		limit = {
			exists = scope:spn_two_old
			scope:spn_two_fit_new > scope:spn_two_fit_old
		}
		spn_hide_exit_effect = { REASON = flag:two_wolves }
		spn_hide_make_champion_dispatch_effect = { ARTIFACT = scope:spn_two_new }
		scope:spn_two_old = { save_scope_as = spn_two_spare }
	}
	else = {
		scope:spn_two_new = { save_scope_as = spn_two_spare }
	}
	# The one he put aside will not ask him again for five years; the one he kept grows (§5.10, as the player's choice).
	spn_hide_mark_refused_effect = { FROM = scope:spn_two_spare.var:spn_hide YEARS = 5 }
	spn_hide_add_bond_effect = { VALUE = spn_hide_bond_two_wolves_value }
	# The spare goes to the best-fitting wolf in his pack (by standing, the pack's own ranking), else stays in his chest.
	if = {
		limit = {
			exists = var:spn_pack
			var:spn_pack = {
				any_in_list = {
					variable = spn_pack_members
					NOT = { this = scope:spn_two_ai }
					spn_can_bear_hide_trigger = yes
					spn_is_champion_trigger = no
					is_ai = yes # 2.37o: a player is not handed a hide with no word (the review of 4 Oct, F12)
				}
			}
		}
		var:spn_pack = {
			ordered_in_list = {
				variable = spn_pack_members
				limit = {
					NOT = { this = scope:spn_two_ai }
					spn_can_bear_hide_trigger = yes
					spn_is_champion_trigger = no
					is_ai = yes # 2.37o: a player is not handed a hide with no word (the review of 4 Oct, F12)
				}
				order_by = spn_pack_standing_value
				save_scope_as = spn_two_heir
			}
		}
		scope:spn_two_spare = {
			if = {
				limit = { is_equipped = yes }
				unequip_artifact_from_owner = yes
			}
			set_owner = {
				target = scope:spn_two_heir
				history = {
					type = given
					actor = scope:spn_two_ai
					recipient = scope:spn_two_heir
				}

# ==================================================================================================
# AFTER THE BATTLE (§12.5). Root = the side; list spn_battle_champs = the champions who fought here.
# ==================================================================================================
spn_hide_battle_aftermath_effect = {
	# 1. Read first: was anything seen?
	if = {
		limit = {
			any_in_list = {
				list = spn_battle_champs
				has_character_flag = spn_feat_seen
			}
		}
		random_in_list = {
			list = spn_battle_champs
			limit = { has_character_flag = spn_feat_seen }
			save_scope_as = spn_men_saw_champ
		}
		# What they saw, read before the cleanup: how many of the other side's knights he killed, and whether there were
		# two beasts (a clash, or two champions seen).
		save_scope_value_as = {
			name = spn_men_saw_kills
			value = {
				value = 0
				if = {
					limit = { scope:spn_men_saw_champ = { exists = var:spn_feat_kills } }
					add = scope:spn_men_saw_champ.var:spn_feat_kills
				}
			}
		}
		if = {
			limit = {
				OR = {
					any_in_list = {
						list = spn_battle_champs
						has_variable_list = spn_clash_done_with
					}
					any_in_list = {
						list = spn_battle_champs
						count >= 2
						has_character_flag = spn_feat_seen
					}
				}
			}
			save_scope_value_as = { name = spn_men_saw_two value = yes }
		}
		# 2. What the Men Saw: players on either side who are not champions. Those on the side that lost the knights are
		# told it was their own.
		if = {
			limit = {
				scope:spn_battle_side = {
					OR = {
						side_commander ?= scope:spn_men_saw_champ
						side_primary_participant ?= scope:spn_men_saw_champ
						any_side_knight = { this = scope:spn_men_saw_champ }
					}
				}
			}
			scope:spn_battle_side.enemy_side.side_primary_participant ?= { save_scope_as = spn_men_saw_victims }
			scope:spn_battle_side = { spn_hide_men_saw_side_effect = { OURS = no } }
			scope:spn_battle_side.enemy_side = { spn_hide_men_saw_side_effect = { OURS = yes } }
		}
		else = {
			scope:spn_battle_side.side_primary_participant ?= { save_scope_as = spn_men_saw_victims }
			scope:spn_battle_side = { spn_hide_men_saw_side_effect = { OURS = yes } }
			scope:spn_battle_side.enemy_side = { spn_hide_men_saw_side_effect = { OURS = no } }
		}
	}
	# Two Wolves on One Field, for every champion who clashed (the event names one of them).
	every_in_list = {
		list = spn_battle_champs
		limit = {
			is_alive = yes
			has_variable_list = spn_clash_done_with
		}
		random_in_list = {
			variable = spn_clash_done_with
			save_scope_as = spn_field_foe
		}
		if = {
			limit = { is_ai = no }
			trigger_event = spn_hide.0202
		}
	}
	# 3. Clean up, so a second battle in the same province starts fresh.
	every_in_list = {
		list = spn_battle_champs
		spn_hide_battle_cleanup_effect = yes
	}
}

# One hide. CHAMPION = yes: an AI wolf if one fits. SHRINE = yes: else a shrine (or a treasury). Anything not placed is
# recorded lost.
spn_hide_seed_one_effect = {
	if = {
		limit = { always = $CHAMPION$ }
		save_scope_value_as = { name = spn_fit_hide value = flag:$HIDE$ }
		random_living_character = {
			limit = {
				is_ai = yes
				has_trait = supernatural_werewolf
				spn_can_bear_hide_trigger = yes
				spn_is_champion_trigger = no
				trigger_if = {
					limit = { has_trait = supernatural_werewolf }
					has_trait_xp = { trait = supernatural_werewolf track = lycanthropy value >= 50 }
				}
				trigger_else = { always = no }
				spn_hide_accepts_wolf_trigger = yes
			}
			weight = {
				base = 1
				modifier = { add = spn_hide_fit_value }
			}
			save_scope_as = spn_seed_wolf
		}
		if = {
			limit = { exists = scope:spn_seed_wolf }
			spn_hide_create_effect = { HIDE = $HIDE$ OWNER = scope:spn_seed_wolf }
			# Its first mark: the culture of the county it was placed in (§5.12).
			scope:spn_seed_wolf.location.county ?= {
				add_to_global_variable_list = { name = spn_hide_$HIDE$_marks target = culture }
			}
			scope:spn_seed_wolf = {
				spn_hide_make_champion_dispatch_effect = { ARTIFACT = scope:spn_new_hide }
			}
			save_scope_value_as = { name = spn_seed_done value = yes }
			clear_saved_scope = spn_seed_wolf
		}
	}
	if = {
		limit = {
			NOT = { exists = scope:spn_seed_done }
			always = $SHRINE$
		}
		spn_hide_seed_shrine_effect = { HIDE = $HIDE$ }
	}
	if = {
		limit = { NOT = { exists = scope:spn_seed_done } }
		spn_hide_record_lost_effect = { HIDE = $HIDE$ }
	}
	if = {
		limit = { exists = scope:spn_seed_done }
		clear_saved_scope = spn_seed_done
	}
}

# Into a shrine by vanilla's own method (02_starting_holy_site_relics_effects.txt:1592-1599): created in the site
# holder's inventory, then set_holy_site_owner + equip. Only holy sites with a free relic slot whose holder shares a
# faith that holds the site; else the treasury of the highest-tier mortal ruler; else lost.
spn_hide_seed_shrine_effect = {
	every_religion_global = {
		every_faith = {
			save_temporary_scope_as = spn_seed_faith
			every_holy_site = {
				limit = {
					holy_site_artifact_count < 3
					holy_site_county = {
						exists = holder
						holder.faith = scope:spn_seed_faith
						holder = { spn_is_hide_mortal_trigger = yes }
					}
				}
				add_to_temporary_list = spn_seed_sites_$HIDE$
			}
		}
	}
	if = {
		limit = {
			any_in_list = {
				list = spn_seed_sites_$HIDE$
				holy_site_artifact_count < 3
			}
		}
		# A list per hide, and the free slot tested again: an earlier hide may have filled this site.
		random_in_list = {
			list = spn_seed_sites_$HIDE$
			limit = { holy_site_artifact_count < 3 }
			save_scope_as = spn_seed_site
		}
		scope:spn_seed_site.holy_site_county = { save_scope_as = spn_seed_county }
		scope:spn_seed_county.holder = { save_scope_as = spn_seed_holder }
		spn_hide_create_effect = { HIDE = $HIDE$ OWNER = scope:spn_seed_holder }
		scope:spn_new_hide = {
			set_holy_site_owner = {
				target = scope:spn_seed_site
				generate_history = no
			}
			equip_artifact_to_owner = yes
			spn_hide_shrine_entered_effect = { COUNTY = scope:spn_seed_county FAITH = scope:spn_seed_holder.faith }
		}
		add_to_global_variable_list = { name = spn_hide_$HIDE$_marks target = scope:spn_seed_county.culture }
		save_scope_value_as = { name = spn_seed_done value = yes }
	}
	else = {
		ordered_ruler = {
			limit = { spn_is_hide_mortal_trigger = yes }
			order_by = highest_held_title_tier
			save_scope_as = spn_seed_holder
		}
		if = {
			limit = { exists = scope:spn_seed_holder }
			spn_hide_create_effect = { HIDE = $HIDE$ OWNER = scope:spn_seed_holder }
			scope:spn_seed_holder.capital_county ?= {
				add_to_global_variable_list = { name = spn_hide_$HIDE$_marks target = culture }
			}
			save_scope_value_as = { name = spn_seed_done value = yes }
		}
	}
}

# ==================================================================================================
# A LOST HIDE WAKES (§9.5). No root. The Tale's listener first, if he passes; else a weighted pick among living wolves
# who pass its bar (a world walk, once per wake, never per year). Nobody: a new wake date.
# ==================================================================================================
spn_hide_wake_effect = {
	save_scope_value_as = { name = spn_fit_hide value = flag:$HIDE$ }
	if = {
		limit = {
			global_var:spn_hide_$HIDE$_heard_by ?= {
				is_alive = yes
				spn_can_bear_hide_trigger = yes
				spn_is_champion_trigger = no
				NOT = { has_character_flag = spn_hide_refused_$HIDE$ }
				spn_hide_accepts_wolf_trigger = yes
			}
		}
		global_var:spn_hide_$HIDE$_heard_by = { save_scope_as = spn_wake_wolf }
	}
	else = {
		random_living_character = {
			limit = {
				spn_can_bear_hide_trigger = yes
				spn_is_champion_trigger = no
				NOT = { has_character_flag = spn_hide_refused_$HIDE$ }
				spn_hide_accepts_wolf_trigger = yes
			}
			weight = {
				base = 1
				modifier = { add = spn_hide_fit_value }
			}
			save_scope_as = spn_wake_wolf
		}
	}
	if = {
		limit = { exists = global_var:spn_hide_$HIDE$_heard_by }
		remove_global_variable = spn_hide_$HIDE$_heard_by
	}
	if = {
		limit = { NOT = { exists = scope:spn_wake_wolf } }
		spn_hide_rewait_effect = { HIDE = $HIDE$ }
	}
	# A player who can hunt (the hunt's own is_shown, hunt.txt:2-3), not already chasing a white animal: the legendary
	# wolf, vanilla's chase (story_cycle_hunt_mystical_animal) and vanilla's legendary hunt.
	else_if = {
		limit = {
			scope:spn_wake_wolf = {
				is_ai = no
				highest_held_title_tier > tier_barony
				exists = capital_county
				NOT = { any_owned_story = { story_type = story_cycle_hunt_mystical_animal } }
			}
		}
		scope:spn_wake_wolf = {
			set_variable = { name = animal_type value = flag:wolf }
			create_story = story_cycle_hunt_mystical_animal
			random_owned_story = {
				limit = { story_type = story_cycle_hunt_mystical_animal }
				# The owner may be standing outside his realm; the legendary option needs a sub-realm county.
				set_variable = { name = legendary_county value = scope:spn_wake_wolf.capital_county }
				set_variable = { name = spn_hide_wolf value = flag:$HIDE$ }
			}
			# Stored on him too: the story can end on day 35, before anything after the hunt reads it.
			set_variable = { name = spn_hide_wolf_hide value = flag:$HIDE$ }
			capital_county = {
				hunt_create_sighting_effect = { TYPE = legendary ANIMAL = flag:wolf OWNER = scope:spn_wake_wolf }
			}
			trigger_event = spn_hide.0300
		}
		set_global_variable = { name = spn_hide_$HIDE$_reserved value = scope:spn_wake_wolf years = 2 }
	}
	# An AI, a landless wolf, a player who cannot hunt or is already chasing one: It Came Out of the Trees.
	else = {
		spn_hide_create_effect = { HIDE = $HIDE$ OWNER = scope:spn_wake_wolf }
		scope:spn_new_hide = { save_scope_as = spn_offered_hide }
		scope:spn_wake_wolf = { trigger_event = spn_hide.0303 }
	}
	if = {
		limit = { exists = scope:spn_wake_wolf }
		clear_saved_scope = spn_wake_wolf
	}
}

					}

	}

	else_if = {
		limit = {
			exists = global_var:spn_hide_$HIDE$_lost
			global_var:spn_hide_$HIDE$_lost <= current_year
		}
		spn_hide_wake_effect = { HIDE = $HIDE$ }
	}
	# Neither in the world nor lost nor worn: something destroyed it without a hook. It comes back the long way.
	else_if = {

			NOT = { exists = global_var:spn_hide_$HIDE$_lost }
			NOT = { any_in_list = { list = spn_net_present var:spn_hide = flag:$HIDE$ } }
		}

		spn_hide_record_lost_effect = { HIDE = $HIDE$ }
	}
	# 2.37i, made general in 2.37o: a shrine says it keeps a hide only while the hide is there. Every year, not only the

					}
				}
			}
			remove_county_modifier = spn_hide_relic_kept_$HIDE$_modifier
		}
	}
}

# spn_hide.0001 still carried Silverback weeks after It Will Not Have You, and Alpha, worn the whole time, was never
# offered. A temporary scope is never handed to a triggered event. spn_offered_hide / spn_off_hide are still saved for
# the event that reads them, right before it is fired.
# ==================================================================================================
spn_hide_weekly_case_effect = {
	# Is he this hide's champion? Read once, in nested ifs (CK3 does not short-circuit AND siblings).
	save_temporary_scope_value_as = { name = spn_wk_mine value = flag:no }
	if = {
		limit = { exists = var:spn_champion_of }
		if = {
			limit = { scope:spn_wk_hide.var:spn_hide = var:spn_champion_of }
			save_temporary_scope_value_as = { name = spn_wk_mine value = flag:yes }
		}
	}
	if = {
		limit = { scope:spn_wk_mine = flag:yes }
		# 1. His, worn: no claims, no transfer, the beat check.
		if = {
			limit = { scope:spn_wk_hide = { is_equipped = yes } }
			scope:spn_wk_hide = {
				if = {
					limit = { NOT = { has_variable = artifact_prevent_transfer } }
					set_variable = artifact_prevent_transfer
				}
				spn_hide_strip_claims_effect = yes
			}
			spn_hide_check_beat_effect = yes
			if = {
				limit = { has_character_flag = spn_hide_off_since }
				remove_character_flag = spn_hide_off_since
			}
			# 2.37o: worn again, ageless again. The yearly net gives the gift only to a hide worn on 1 January, so one
			# taken off over the new year left him without it for a year (the review of 4 Oct, N6).
			if = {
				limit = {
					is_immortal = yes
					NOT = { has_character_modifier = spn_hide_ageless_modifier }
				}
				spn_hide_ageless_effect = yes
			}
		}
		# 2. His, off. A player is asked in a week; an AI champion never takes it off, so whatever took it off, it goes
		# back on (§21).
		else = {
			if = {
				limit = { is_ai = yes }
				scope:spn_wk_hide = { equip_artifact_to_owner_replace = yes }
			}
			else_if = {
				limit = {
					NOT = { has_character_flag = spn_hide_off_since }
					NOT = { exists = scope:spn_wk_asked }
				}
				add_character_flag = { flag = spn_hide_off_since days = 30 }
				save_temporary_scope_value_as = { name = spn_wk_asked value = yes }
				scope:spn_wk_hide = { save_scope_as = spn_off_hide }
				trigger_event = { id = spn_hide.0014 days = 7 }
			}
		}
	}
	# 3. Not his, worn: a wolf has just put it on. The hide decides in a few days; a champion of another hide gets
	# Two Wolves instead.
	else_if = {
		limit = {
			scope:spn_wk_hide = {
				is_equipped = yes
				NOT = { spn_hide_refused_trigger = { WHO = scope:spn_wk } }
			}
			NOT = { has_character_flag = spn_hide_takes_hold_pending }
			NOT = { exists = scope:spn_wk_offered }
			spn_can_bear_hide_trigger = yes
		}
		add_character_flag = { flag = spn_hide_takes_hold_pending days = 10 }
		save_temporary_scope_value_as = { name = spn_wk_offered value = yes }
		scope:spn_wk_hide = { save_scope_as = spn_offered_hide }
		if = {
			limit = { spn_is_champion_trigger = yes }
			if = {
				limit = { is_ai = yes }
				trigger_event = { id = spn_hide.0016 days = 3 }
			}
			else = {
				trigger_event = { id = spn_hide.0012 days = 3 }
			}
		}
		else = {
			trigger_event = { id = spn_hide.0009 days = { 3 7 } }
		}
	}
	# 4. Unworn, a mortal: once a year, a rare ask.
	else_if = {
		limit = {
			scope:spn_wk_hide = {
				is_equipped = no
				NOT = { spn_hide_refused_trigger = { WHO = scope:spn_wk } }
			}
			spn_is_hide_mortal_trigger = yes
			is_adult = yes
			NOT = { has_character_flag = spn_hide_mortal_roll }
			NOT = { exists = scope:spn_wk_offered }
		}
		add_character_flag = { flag = spn_hide_mortal_roll years = 1 }
		spn_hide_save_flag_effect = { NAME = spn_fit_hide FROM = scope:spn_wk_hide.var:spn_hide }
		if = {
			limit = {
				spn_hide_fit_value >= spn_hide_fit_bar_mortal_value
			}
			random = {
				chance = spn_hide_mortal_chance_value
				save_temporary_scope_value_as = { name = spn_wk_offered value = yes }
				scope:spn_wk_hide = { save_scope_as = spn_offered_hide }
				trigger_event = { id = spn_hide.0013 days = { 1 7 } }
			}
		}
	}
	# 5. Unworn, a wolf who is not its champion. A player chooses for himself (the inventory). An AI wolf who is not
	# already a champion has it put on for him -- vanilla's own AI never will (can_benefit outranks ai_score) -- and then
	# the hide decides, as in case 3.
	else_if = {
		limit = {
			is_ai = yes
			scope:spn_wk_hide = {
				is_equipped = no
				NOT = { spn_hide_refused_trigger = { WHO = scope:spn_wk } }
			}
			spn_can_bear_hide_trigger = yes
			spn_is_champion_trigger = no
			NOT = { has_character_flag = spn_hide_takes_hold_pending }
			NOT = { exists = scope:spn_wk_offered }
		}
		scope:spn_wk_hide = { equip_artifact_to_owner_replace = yes }
		add_character_flag = { flag = spn_hide_takes_hold_pending days = 10 }
		save_temporary_scope_value_as = { name = spn_wk_offered value = yes }
		scope:spn_wk_hide = { save_scope_as = spn_offered_hide }
		trigger_event = { id = spn_hide.0009 days = { 3 7 } }
	}
}

# 2.37 -- Send for a Hide (spn_hide.0380). Character scope. When word comes back depends only on how far the shrine is,
# measured with vanilla's own distance values (00_distance_values.txt), so it works on any map. COUNTY is banked first:
# the caller passes a var: path, and a parameter is pasted text.
spn_hide_errand_send_effect = {
	$COUNTY$ = { save_scope_as = spn_errand_to }
	if = {
		limit = {
			location ?= {
				squared_distance = {
					target = scope:spn_errand_to.title_province
					value <= squared_distance_large
				}
			}
		}
		trigger_event = { id = spn_hide.0381 days = { 45 90 } }
	}
	else_if = {
		limit = {
			location ?= {
				squared_distance = {
					target = scope:spn_errand_to.title_province
					value <= squared_distance_huge
				}
			}
		}
		trigger_event = { id = spn_hide.0381 days = { 90 180 } }
	}
	else = {
		trigger_event = { id = spn_hide.0381 days = { 180 365 } }
	}
}

# Debug (§24): give me this hide. Character scope. 2.37 (Thomas, 3 Oct: a second Silverback sat in a shrine while he wore
# the one the console made): if the hide is already in the world it is handed over -- out of its shrine or off its holder,
# set_owner as vanilla's siege theft does (steal_artifact_in_siege_effect), and the owner hook clears the shrine marker --
# so the world never holds two of one hide. A hide worn by its living champion is left alone (unbind him first). A lost
# hide is made.
spn_debug_give_hide_effect = {
	save_scope_as = spn_dbg_taker
	every_artifact = {
		limit = { var:spn_hide ?= flag:$HIDE$ }
		save_scope_as = spn_dbg_old_hide
	}
	if = {
		limit = { NOT = { exists = scope:spn_dbg_old_hide } }
		spn_hide_create_effect = { HIDE = $HIDE$ OWNER = scope:spn_dbg_taker }
	}
	else_if = {
		limit = {
			scope:spn_dbg_old_hide ?= { # ?=: a tooltip walk reads this limit without the loop having run
				NOT = { artifact_owner ?= scope:spn_dbg_taker }
				NOT = {
					artifact_owner ?= {
						is_alive = yes
						var:spn_champion_of ?= flag:$HIDE$
					}
				}
			}
		}
		scope:spn_dbg_old_hide ?= {
			set_owner = {
				target = scope:spn_dbg_taker
				history = {
					type = given
					recipient = scope:spn_dbg_taker
				}
			}
		}
		scope:spn_dbg_taker = { spn_hide_start_check_effect = yes }
	}
	else = {
		debug_log = "spn_debug_give_hide_effect: that hide is already yours or worn by its living champion"
	}
}

# Character scope = the one who listens. He learns ONE thing he lacked, in order: where a hide is (held or enshrined),
# that a hide is lost, the True Alpha. SEND = yes: a player hears it as Word at the Fire. A wolf at standing 50
# (lycanthropy 50 with the Packs rule Off) who learns who holds an unworn hide gets a claim on it (§5.4).
spn_hide_word_at_fire_effect = {
	save_scope_as = spn_waf_who
	if = {
		limit = { exists = scope:spn_waf_hide }
		clear_saved_scope = spn_waf_hide
	}
	if = {
		limit = { exists = scope:spn_waf_holder }
		clear_saved_scope = spn_waf_holder
	}
	save_scope_value_as = { name = spn_waf_kind value = flag:nothing }
	if = {
		limit = {
			any_in_list = {
				list = spn_all_hides
				spn_hide_fact_lacked_trigger = { WHO = scope:spn_waf_who }
			}
		}
		random_in_list = {
			list = spn_all_hides
			limit = { spn_hide_fact_lacked_trigger = { WHO = scope:spn_waf_who } }
			save_scope_as = spn_waf_hide
		}
		if = {
			limit = { scope:spn_waf_hide = { exists = var:spn_hide_shrine_county } }
			save_scope_value_as = { name = spn_waf_kind value = flag:shrine }
			scope:spn_waf_hide.var:spn_hide_shrine_county = { save_scope_as = spn_waf_county }
			spn_hide_learn_where_effect = { HIDE_SCOPE = scope:spn_waf_hide WHERE = scope:spn_waf_county WANTS = yes }
		}
		else_if = {
			limit = { scope:spn_waf_hide = { exists = artifact_owner } }
			save_scope_value_as = { name = spn_waf_kind value = flag:held }
			scope:spn_waf_hide.artifact_owner = { save_scope_as = spn_waf_holder }
			spn_hide_learn_where_effect = { HIDE_SCOPE = scope:spn_waf_hide WHERE = scope:spn_waf_holder WANTS = no }
			if = {
				limit = {
					NOT = { scope:spn_waf_holder = scope:spn_waf_who }
					scope:spn_waf_hide = { is_equipped = no }
					spn_is_wolf_trigger = yes
					spn_hide_claim_ready_trigger = yes
				}
				add_personal_artifact_claim = scope:spn_waf_hide
			}
		}
	}
	else_if = {
		limit = { spn_hide_lost_fact_lacked_trigger = yes }
		random_list = {
			1 = {
				trigger = {
					exists = global_var:spn_hide_silverback_lost
					NOT = { exists = var:spn_knows_lost_silverback }
				}
				set_variable = { name = spn_knows_lost_silverback value = yes years = 10 }
				save_scope_value_as = { name = spn_waf_lost value = flag:silverback }
			}
			1 = {
				trigger = {
					exists = global_var:spn_hide_timber_lost
					NOT = { exists = var:spn_knows_lost_timber }
				}
				set_variable = { name = spn_knows_lost_timber value = yes years = 10 }
				save_scope_value_as = { name = spn_waf_lost value = flag:timber }
			}
			1 = {
				trigger = {
					exists = global_var:spn_hide_tundra_lost
					NOT = { exists = var:spn_knows_lost_tundra }
				}
				set_variable = { name = spn_knows_lost_tundra value = yes years = 10 }
				save_scope_value_as = { name = spn_waf_lost value = flag:tundra }
			}
			1 = {
				trigger = {
					exists = global_var:spn_hide_graybeard_lost
					NOT = { exists = var:spn_knows_lost_graybeard }
				}
				set_variable = { name = spn_knows_lost_graybeard value = yes years = 10 }
				save_scope_value_as = { name = spn_waf_lost value = flag:graybeard }
			}
			1 = {
				trigger = {
					exists = global_var:spn_hide_midnight_lost
					NOT = { exists = var:spn_knows_lost_midnight }
				}
				set_variable = { name = spn_knows_lost_midnight value = yes years = 10 }
				save_scope_value_as = { name = spn_waf_lost value = flag:midnight }
			}
			1 = {
				trigger = {
					exists = global_var:spn_hide_alpha_lost
					NOT = { exists = var:spn_knows_lost_alpha }
				}
				set_variable = { name = spn_knows_lost_alpha value = yes years = 10 }
				save_scope_value_as = { name = spn_waf_lost value = flag:alpha }
			}
		}
		save_scope_value_as = { name = spn_waf_kind value = flag:lost }
	}
	else_if = {
		limit = {
			exists = global_var:spn_ta_living
			NOT = { global_var:spn_ta_living = scope:spn_waf_who }
			trigger_if = {
				limit = { exists = var:spn_knows_true_alpha }
				NOT = { var:spn_knows_true_alpha = global_var:spn_ta_living }
			}
			trigger_else = { always = yes }
		}
		set_variable = { name = spn_knows_true_alpha value = global_var:spn_ta_living years = 100 }
		global_var:spn_ta_living = { save_scope_as = spn_waf_ta }
		save_scope_value_as = { name = spn_waf_kind value = flag:true_alpha }
	}
	if = {
		limit = {
			always = $SEND$
			is_ai = no
		}
		trigger_event = spn_gathering.0301
	}
}

						has_relation_lover = scope:spn_th_midnight
					}
				}
				set_relation_rival = {
					target = scope:spn_th_midnight
					reason = spn_rival_two_hides
				}
			}
			trigger_event = spn_gathering.0304
		}
		scope:spn_th_midnight = { trigger_event = spn_gathering.0304 }
	}
	# --- The Unification (§15): the host asked for it, and his spouse leads another pack and is here.
	if = {
		limit = {
			spn_packs_on_trigger = yes
			scope:host = {
				has_activity_intent = spn_hold_the_unification_intent
				spn_is_alpha_trigger = yes
			}
			scope:activity = {
				any_attending_character = {
					spn_unification_partner_trigger = { HOST = scope:host }
					# 2.37o: the rite answers the host alone, so an AI host never merges a player's pack into his own;
					# the player holds the Unification himself (the review of 4 Oct, S1; pass j's rule).
					OR = {
						is_ai = yes
						scope:host = { is_ai = no }
					}
				}
			}
		}
		scope:activity = {
			random_attending_character = {
				limit = {
					spn_unification_partner_trigger = { HOST = scope:host }
					OR = {
						is_ai = yes
						scope:host = { is_ai = no }
					}
				}
				save_scope_as = spn_un_spouse
			}
		}
		if = {
			limit = {
				scope:host = { NOT = { any_secret = { NOT = { is_known_by = scope:spn_un_spouse } } } }
				scope:spn_un_spouse = { NOT = { any_secret = { NOT = { is_known_by = scope:host } } } }
			}
			scope:host = { trigger_event = spn_gathering.0305 }
		}
		else = {
			scope:host = { trigger_event = spn_gathering.0306 }
			scope:spn_un_spouse = {
				if = {
					limit = { is_ai = no }
					trigger_event = spn_gathering.0306
				}
			}
		}
	}
}

# The end (§14.2 on_complete). Pack Standing by intent (Packs On), lycanthropy, Hide Bond, the born grant, Call the
# Packs' favours, and kindred blood between packs and faiths.
spn_gathering_complete_effect = {
	# 2.37b -- the Fevered: the pack sat with him through the Gathering and will see him through the fever. The turn
	# is likelier and death less likely at its end (spn_resolve_wound_climax), the fever's last choice gains Let the
	# Pack Carry Me, and if he turns he wakes in the host's pack (spn_pack.0014). Not more infection: a higher infection
	# track raises the climax's death weight (20_spn_sire_effects.txt), the opposite of what the pack is for.
	scope:host = { save_scope_as = spn_gc_host }
	scope:activity = {
		every_attending_character = {
			limit = {
				has_trait = ww_infected_wound
				NOT = { this = scope:host }
			}
			add_character_flag = spn_pack_carried
			set_variable = { name = spn_pack_carried_by value = scope:host }
			if = {
				limit = { is_ai = no }
				trigger_event = spn_gathering.0401
			}
		}
	}
	scope:activity = {
		every_attending_character = {
			limit = { spn_is_wolf_trigger = yes }
			save_temporary_scope_as = spn_gc_wolf
			if = {
				limit = { spn_packs_on_trigger = yes }
				if = {
					limit = { has_activity_intent = spn_earn_my_place_intent }
					spn_pack_gain_standing_effect = { VALUE = 20 }
				}
				else = {
					spn_pack_gain_standing_effect = { VALUE = 10 }
				}
				spn_pack_born_standing_effect = yes
			}
			if = {
				limit = { has_trait = supernatural_werewolf }
				add_trait_xp = { trait = supernatural_werewolf track = lycanthropy value = 5 }
			}
			else_if = {
				limit = { has_trait = supernatural_hybrid }
				add_trait_xp = { trait = supernatural_hybrid track = hybrid_lycanthropy value = 5 }
			}
			if = {
				limit = { spn_is_champion_trigger = yes }
				spn_hide_add_bond_effect = { VALUE = spn_hide_bond_gathering_value }
			}
			if = {
				limit = { spn_pack_politics_built_trigger = yes }
				scope:activity = {
					every_attending_character = {
						limit = {
							NOT = { this = scope:spn_gc_wolf }
							spn_is_wolf_trigger = yes
							spn_gathering_kindred_blood_trigger = { OTHER = scope:spn_gc_wolf }
						}
						add_opinion = { target = scope:spn_gc_wolf modifier = spn_kindred_blood_opinion }
					}
				}
			}
		}
	}
	# Call the Packs (§6.5): a favour from every alpha who came. 2.37o: and the call is spent (an AI's lasts five years,
	# so it would otherwise reach a second Gathering).
	if = {
		limit = { scope:host = { has_character_flag = spn_called_the_packs } }
		scope:host = { remove_character_flag = spn_called_the_packs }
		scope:activity = {
			every_attending_character = {
				limit = {
					NOT = { this = scope:host }
					spn_is_alpha_trigger = yes
				}
				save_temporary_scope_as = spn_gc_alpha
				scope:host = {
					if = {
						limit = { NOT = { has_hook = scope:spn_gc_alpha } }
						add_hook = { type = favor_hook target = scope:spn_gc_alpha }
					}
				}
			}
		}
	}
}

# THE FIELD (§12.3, §12.4). Root = the champion; scope:combat_side = his side.

spn_hide_feat_effect = {
	save_scope_as = spn_feat_champ
	# The budget, on the battle's first pick.
	if = {
		limit = { NOT = { has_variable = spn_feat_budget } }
		set_variable = { name = spn_feat_budget value = spn_hide_feat_budget_value days = 60 }
		set_variable = { name = spn_feat_province value = scope:combat_side.combat.location days = 60 }
	}
	if = {
		limit = { exists = scope:spn_feat_bitten }
		clear_saved_scope = spn_feat_bitten
	}
	# Counterplay: a hunter in their ranks. A quarter of his picks, the hunter finds him instead.
	if = {
		limit = {
			scope:combat_side.enemy_side = {
				any_side_knight = {
					is_alive = yes
					spn_has_hunter_trait = yes
				}
			}
		}
		random = {
			chance = 25
			save_scope_value_as = { name = spn_feat_bitten value = yes }
		}
	}
	if = {
		limit = { exists = scope:spn_feat_bitten }
		scope:combat_side.enemy_side = {
			random_side_knight = {
				limit = {
					is_alive = yes
					spn_has_hunter_trait = yes
				}
				save_temporary_scope_as = spn_feat_hunter
			}
		}
		# Posted on the hunter's side: "our" hunter, their beast.
		scope:combat_side.enemy_side = {
			battle_event = {
				key = spn_hide_feat_hunter
				left_portrait = scope:spn_feat_hunter
				right_portrait = root
				type = wound
				target_right = yes
			}
		}
		increase_wounds_no_death_effect = { REASON = fight }
		change_variable = { name = spn_feat_budget add = -1 }
	}
	else = {
		spn_hide_feat_kill_one_effect = yes
		# A second kill at 50%, while the budget lasts.
		if = {
			limit = {
				is_alive = yes
				var:spn_feat_budget > 0
				scope:combat_side.enemy_side = {
					any_side_knight = { spn_hide_feat_victim_trigger = { CHAMP = scope:spn_feat_champ } }
				}
			}
			random = {
				chance = 50
				spn_hide_feat_kill_one_effect = yes
			}
		}
	}
	# The secret: once a battle, a chance the enemy commander understands what he saw.
	if = {
		limit = {
			is_alive = yes
			NOT = { has_character_flag = spn_feat_revealed }
			has_character_flag = spn_feat_seen
		}
		add_character_flag = { flag = spn_feat_revealed days = 60 }
		random = {
			chance = spn_hide_war_reveal_chance_value
			scope:combat_side.enemy_side.side_commander ?= { save_scope_as = spn_feat_witness }
			if = {
				limit = { exists = scope:spn_feat_witness }
				if = {
					limit = { has_trait = supernatural_hybrid }
					spn_give_knowledge_of_hybrid_secret = { CHARACTER = scope:spn_feat_witness }
				}
				else = {
					spn_give_knowledge_of_lycan_secret = { CHARACTER = scope:spn_feat_witness }
				}
			}
		}
	}
}

# One kill. Root = the champion.
spn_hide_feat_kill_one_effect = {
	scope:combat_side.enemy_side = {
		random_side_knight = {
			limit = { spn_hide_feat_victim_trigger = { CHAMP = scope:spn_feat_champ } }
			save_temporary_scope_as = enemy_knight
		}
	}
	if = {
		limit = {
			is_alive = yes
			exists = scope:enemy_knight
			scope:enemy_knight = { is_alive = yes }
		}
		change_variable = { name = spn_feat_budget add = -1 }
		if = {
			limit = { exists = var:spn_feat_kills }
			change_variable = { name = spn_feat_kills add = 1 }
		}
		else = {
			set_variable = { name = spn_feat_kills value = 1 days = 60 }
		}
		random_list = {
			1 = {
				scope:combat_side = {
					battle_event = {
						key = spn_hide_feat_1
						left_portrait = root
						right_portrait = scope:enemy_knight
						type = death
						target_right = yes
					}
				}
			}
			1 = {
				scope:combat_side = {
					battle_event = {
						key = spn_hide_feat_2
						left_portrait = root
						right_portrait = scope:enemy_knight
						type = death
						target_right = yes
					}
				}
			}
			1 = {
				scope:combat_side = {
					battle_event = {
						key = spn_hide_feat_3
						left_portrait = root
						right_portrait = scope:enemy_knight
						type = death
						target_right = yes
					}
				}
			}
		}
		# The first kill of the battle is weighed by his own rite, once (the 2.36 rule).
		if = {
			limit = { NOT = { has_character_flag = spn_feat_weighed } }
			add_character_flag = { flag = spn_feat_weighed days = 60 }
			spn_creature_act_fulfillment_effect = { ACT = kill }
		}
		if = {
			limit = { NOT = { has_character_flag = spn_feat_seen } }
			add_character_flag = { flag = spn_feat_seen days = 60 }
		}
		spn_hide_feat_bond_effect = yes
		# A name, at ten. Never over one he already has.
		if = {
			limit = { exists = var:spn_hide_feats }
			change_variable = { name = spn_hide_feats add = 1 }
		}
		else = {
			set_variable = { name = spn_hide_feats value = 1 }
		}
		if = {
			limit = {
				var:spn_hide_feats >= 10
				has_any_nickname = no
			}
			give_nickname = nick_the_wolf
		}
		add_prestige = knight_prestige_gain_on_kill
		# The death last.
		scope:enemy_knight = {
			death = {
				death_reason = spn_death_torn_in_battle
				killer = scope:spn_feat_champ
			}
		}
	}
}

# Champion against champion (§12.4). The foe is picked INSIDE the effect: combat-phase effects run after every pick
# of the pulse, so two opposing champions can both pick the clash, and the second must find the pair already done.
spn_hide_field_clash_effect = {
	save_scope_as = spn_fc_champ
	if = {
		limit = { exists = scope:spn_fc_foe }
		clear_saved_scope = spn_fc_foe
	}
	scope:combat_side.enemy_side = {
		if = {
			limit = { side_commander ?= { spn_hide_clash_foe_trigger = { CHAMP = scope:spn_fc_champ } } }
			side_commander = { save_scope_as = spn_fc_foe }
		}
		else = {
			random_side_knight = {
				limit = { spn_hide_clash_foe_trigger = { CHAMP = scope:spn_fc_champ } }
				save_scope_as = spn_fc_foe
			}
		}
	}
	if = {
		limit = {
			is_alive = yes
			exists = scope:spn_fc_foe
		}
		# 1. The pair, first: once a battle, the feat unblocked for both, and the aftermath finds them.
		add_to_variable_list = { name = spn_clash_done_with target = scope:spn_fc_foe days = 60 }
		scope:spn_fc_foe = { add_to_variable_list = { name = spn_clash_done_with target = scope:spn_fc_champ days = 60 } }
		set_variable = { name = spn_feat_province value = scope:combat_side.combat.location days = 60 }
		scope:spn_fc_foe = { set_variable = { name = spn_feat_province value = scope:combat_side.combat.location days = 60 } }
		add_character_flag = { flag = spn_feat_seen days = 60 }
		scope:spn_fc_foe = { add_character_flag = { flag = spn_feat_seen days = 60 } }
		# 2. The clash.
		spn_creature_clash_effect = {
			A = scope:spn_fc_champ
			B = scope:spn_fc_foe
			THEN = spn_hide_field_clash_after_effect
			FORM = no
		}
	}
}

# 3. The loser's wounds are read BEFORE one is added: a wolf already badly hurt dies of this one.
spn_hide_field_clash_after_effect = {
	# Posted on the winner's side, so "ours" is always the one who came out on top.
	if = {
		limit = { scope:spn_clash_winner = scope:spn_fc_champ }
		scope:combat_side = { spn_hide_clash_battle_event_effect = yes }
	}
	else = {
		scope:combat_side.enemy_side = { spn_hide_clash_battle_event_effect = yes }
	}
	if = {
		limit = {
			scope:spn_clash_loser = {
				OR = {
					has_trait_rank = {
						trait = wounded
						rank = 2
					}
					has_trait_rank = {
						trait = wounded
						rank = 3
					}
					has_trait = maimed
				}
			}
		}
		scope:spn_clash_winner = {
			if = {
				limit = { NOT = { has_character_flag = spn_feat_weighed } }
				add_character_flag = { flag = spn_feat_weighed days = 60 }
			}
			spn_creature_act_fulfillment_effect = { ACT = kill }
		}
		scope:spn_clash_loser = {
			death = {
				death_reason = spn_death_torn_in_battle
				killer = scope:spn_clash_winner
			}
		}
	}
	else = {
		scope:spn_clash_loser = { increase_wounds_no_death_effect = { REASON = fight } }
	}
}
