# Supernatural 2.41 (WORK-ORDER-2.41 §7.1) -- the demon-blooded wolf's lean. scope:spn_tn_child is the child.
spn_two_natures_lean_effect = {
	scope:spn_tn_child = {
		if = {
			limit = { NOT = { exists = var:spn_two_natures_lean } }
			set_variable = { name = spn_two_natures_lean value = 0 }
		}
		change_variable = { name = spn_two_natures_lean add = $ADD$ }
	}
}

# CHARACTER scope: the demon-blooded wolf child, on a birthday. Fires the beat to whoever is asked.
spn_two_natures_beat_effect = {
	save_scope_as = spn_tn_child
	if = {
		limit = { is_ai = no }
		save_scope_as = spn_tn_viewer
		trigger_event = $EVENT$
	}
	else_if = {
		limit = { any_parent = { is_alive = yes is_ai = no } }
		random_parent = {
			limit = { is_alive = yes is_ai = no }
			save_scope_as = spn_tn_viewer
			trigger_event = $EVENT$
		}
	}
}
