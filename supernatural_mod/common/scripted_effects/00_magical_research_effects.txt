gain_magical_research_progress_effect = {

		limit = { scope:activity = { exists = var:activity_special_type_progression } }
		if = {
			limit = {
				$P$ = 1
			}
			custom_tooltip = gain_magical_research_progress_1_txt
		}
		else_if = {
			limit = {
				$P$ = 2
			}
			custom_tooltip = gain_magical_research_progress_2_txt
		}
		else_if = {
			limit = {
				$P$ = 3
			}
			custom_tooltip = gain_magical_research_progress_3_txt
		}
		else_if = {
			limit = {
				$P$ = 5
			}
			custom_tooltip = gain_magical_research_progress_5_txt
		}
		else_if = {
			limit = {
				$P$ = 7
			}
			custom_tooltip = gain_magical_research_progress_7_txt
		}
		else_if = {
			limit = {
				$P$ = 10
			}
			custom_tooltip = gain_magical_research_progress_10_txt
		}
		else_if = {
			limit = {
				$P$ = 15
			}
			custom_tooltip = gain_magical_research_progress_15_txt
		}
		else_if = {
			limit = {
				$P$ = 20
			}
			custom_tooltip = gain_magical_research_progress_20_txt
		}
		else_if = {
			limit = {
				$P$ = 100
			}
			custom_tooltip = gain_magical_research_progress_100_txt
		}

	}
}
mr_save_as_potential_assistant_effect = {
	if = {
		limit = {
			exists = this
			is_alive = yes
		}
		if = {
			limit = {
				NOT = {
					exists = scope:assistant
				}
			}
			save_scope_as = assistant
		}
		else_if = {
			limit = {
				NOR = {
					exists = scope:assistant2
					this = scope:assistant
				}
			}
			save_scope_as = assistant2
		}
		else_if = {
			limit = {
				NOR = {
					exists = scope:assistant3
					this = scope:assistant
					this = scope:assistant2
				}
			}
			save_scope_as = assistant3
		}
	}
}
mr_give_reward_for_assistant_effect = {
	hidden_effect = {
		if = {
			limit = {
				ROOT = {
					any_secret = { type = secret_witch NOT = { is_known_by = PREV } }
				}
			}
			ROOT = {
				random_secret = {
					limit = {
						secret_type = secret_witch
					}
					reveal_to = PREV
				}
			}
		}
		if = {
			limit = {
				has_character_modifier = mr_magic_assistant_modifier
			}
			remove_character_modifier = mr_magic_assistant_modifier
		}
		send_interface_toast = {
			title = magical_research
			left_icon = THIS
			right_icon = ROOT
			show_as_tooltip = {
				add_character_modifier = {
					modifier = mr_magic_assistant_modifier
					years = 20
				}
			}
		}
		save_temporary_scope_as = my_assistant
		ROOT = {
			set_variable = {
				name = mr_assistant
				value = scope:my_assistant
			}
		}
	}
	if = {
		limit = {
			is_true_mage = yes
			has_trait_xp = {
				trait = archmage
				track = secret_knowledge
				value < 100
			}
		}
		gain_sc_xp2_effect = yes
	}
	else = {
		add_learning_skill = 1
	}
	
	add_character_modifier = {
		modifier = mr_magic_assistant_modifier
		years = 20
	}
	add_opinion = {
		target = ROOT
		modifier = grateful_opinion
		opinion = 20
	}
	if = {
		limit = {
			is_true_mage = yes
			has_trait_xp = {
				trait = archmage
				track = secret_knowledge
				value > 74
			}
		}
		ROOT = {
			gain_magical_research_progress_effect = { P = 5 }
		}
	}
	else_if = {
		limit = {
			is_true_mage = yes
			has_trait_xp = {
				trait = archmage
				track = secret_knowledge
				value > 24
			}
		}
		ROOT = {
			gain_magical_research_progress_effect = { P = 3 }
		}
	}
	else_if = {
		limit = {
			is_true_mage = yes
		}
		ROOT = {
			gain_magical_research_progress_effect = { P = 2 }
		}
	}
	else_if = {
		limit = {
			OR = {
				has_trait = true_witch
				is_a_demon_trigger = yes
				has_trait = scholar
			}
		}
		ROOT = {
			gain_magical_research_progress_effect = { P = 1 }
		}
	}
	else = {
		ROOT = {
			gain_magical_research_progress_effect = { P = 0 }
		}
	}
}
