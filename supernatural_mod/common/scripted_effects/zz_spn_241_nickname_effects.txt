# Supernatural 2.41 (WORK-ORDER-2.41 §4) -- repair for saves from 2.38-2.40. Under those versions the Keeper's seat
# handed out mage3_nickname. The sitting Keeper who wears it gets the office's own nickname instead. Anyone else who
# wears mage3_nickname keeps it: its text is "the Lore Master" again, so nothing is stripped from any save.
# One character at most (the global the seat already keeps); cheap enough for every load and every year.
spn_keeper_nickname_repair_effect = {
	global_var:current_loremaster_global ?= {
		if = {
			limit = {
				is_alive = yes
				has_character_modifier = mc_loremaster_modifier
				has_nickname = mage3_nickname
			}
			give_nickname = spn_keeper_of_the_reliquary_nickname
		}
	}
}
