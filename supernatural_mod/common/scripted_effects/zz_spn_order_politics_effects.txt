# House rules: bank first, anything irreversible (removal, a seat or chair emptied, a temple dissolved) last; every delayed
# step re-checks its own situation; nothing here walks the world (the watch list is a short global list).

#   global spn_invocare_called (2 years: a motion waiting for its Conclave), spn_invocare_cd (10 years: one Invocare a

#     marks a chair call kept behind a waiting Invocare).

#     the activity).

#     the call they answer; cleared at the next call); spn_iv_power, spn_iv_bound, spn_iv_paid (1 year, set when the motion

#     (50 years), spn_order_on_trial (21 days), spn_order_trial_told (30 days); variables spn_order_temples_swept

# An AI Grand Magus chooses his defence at the call, by his nature (a player answers in The Chair Is Questioned). What
# it costs is taken when the motion is heard (spn_invocare_defence_effect); the speech is made now.
spn_invocare_ai_defend_effect = {
	random_list = {
		# Then they will remember what they owe me.
		0 = {
			trigger = {
				any_hooked_character = { spn_conclave_voter_trigger = yes }
			}
			modifier = { add = 100 }
			modifier = {
				add = 20
				has_trait = ambitious
			}
			add_character_flag = { flag = spn_iv_def_hooks years = 2 }
		}
		# Open the council's coffers.
		30 = {
			trigger = {
				mc_gold_global_avaiable_trigger = yes
				NOT = { global_var:current_admin_global ?= scope:spn_iv_caller }
			}
			modifier = {
				add = -20
				has_trait = greedy
			}
			modifier = {
				add = 20
				has_trait = generous
			}
			add_character_flag = { flag = spn_iv_def_coffers years = 2 }
		}
		# Let me remind them what the Order has done under me.
		30 = {
			modifier = {
				add = 20
				learning >= 15
			}
			modifier = {
				add = 20
				has_trait = gregarious
			}
			modifier = {
				add = -20
				has_trait = shy
			}
			random = {
				chance = spn_order_coin_chance_learning_value
				add_character_flag = { flag = spn_iv_spoke_well years = 2 }
			}
		}
		# A show of power.
		20 = {
			trigger = { can_spend_huge_ps_trigger = yes }
			modifier = {
				add = 20
				has_trait = arrogant
			}

		30 = {

			modifier = {
				add = 20

					has_trait = wrathful

			}
			modifier = {
				add = -30
				has_trait = compassionate
			}
			add_character_flag = { flag = spn_iv_feared years = 2 }
		}

		0 = {

				add = 40
				has_trait = content
			}
			modifier = {
				add = 30
				has_trait = humble
			}
			modifier = {
				add = 20
				has_trait = craven
			}

				factor = 0

					has_trait = arrogant
					has_trait = stubborn

	}

# Activity scope, at the top of The Reckoning, before anyone is judged (red team M3). If the motion can be heard, the two
# charges the masks will weigh are banked on the activity (spn_iv_charges: 2 = broke the rules, +1 = favouritism), and the
# Grand Magus is saved as spn_iv_held so the judging block passes him by: the vote is his judgment.
spn_invocare_bank_effect = {
	if = {
		limit = { spn_invocare_can_open_trigger = yes }
		var:spn_iv_magus = { save_scope_as = spn_iv_held }
		set_variable = { name = spn_iv_charges value = 0 }
		if = {
			limit = { scope:spn_iv_held = { has_character_flag = spn_order_broke_rules } }
			change_variable = { name = spn_iv_charges add = 2 }
		}
		if = {
			limit = { scope:spn_iv_held = { has_character_flag = spn_order_favouritism } }
			change_variable = { name = spn_iv_charges add = 1 }
		}
	}
}

# Host scope, at the end of The Reckoning (after the judging and the favouritism blocks). The motion is heard, or lapses.
# The call is spent either way. Heard: the price is taken now (the caller's 20 standing, the 10-year world lock, the Grand
# Magus's chosen defence or nothing if he can no longer pay for it); then every voter in the room casts.
spn_invocare_open_effect = {
	if = {
		limit = { scope:activity = { var:spn_cv_reason ?= flag:invocare } }
		if = {
			limit = { has_global_variable = spn_invocare_called }
			remove_global_variable = spn_invocare_called
		}
		if = {
			limit = { scope:activity = { spn_invocare_can_open_trigger = yes } }
			scope:activity = {
				var:spn_iv_magus = { save_scope_as = spn_iv_magus }
				var:spn_iv_caller = { save_scope_as = spn_iv_caller }
			}
			# The price.
			set_global_variable = { name = spn_invocare_cd value = yes years = 10 }
			scope:spn_iv_caller = { spn_order_add_standing_effect = { VALUE = -20 } }
			# Review L-5: a voter bought or bound for an earlier motion is free again before this one's defence is paid.
			scope:activity = {
				every_attending_character = {
					limit = {
						OR = {
							has_character_flag = spn_iv_bound
							has_character_flag = spn_iv_paid
						}
					}
					if = {
						limit = { has_character_flag = spn_iv_bound }
						remove_character_flag = spn_iv_bound
					}
					if = {
						limit = { has_character_flag = spn_iv_paid }
						remove_character_flag = spn_iv_paid
					}
				}
			}
			scope:spn_iv_magus = { spn_invocare_defence_effect = yes }
			# He stood down before the vote: nothing to count, and The Vote below simply fills the chair.
			if = {
				limit = { scope:activity = { has_variable = spn_iv_yielded } }
				save_scope_value_as = { name = spn_iv_result value = flag:yielded }
				spn_invocare_tell_effect = yes
			}
			else = {
				scope:activity = {
					set_variable = { name = spn_iv_yes value = 0 }
					set_variable = { name = spn_iv_no value = 0 }
					every_attending_character = {
						limit = { spn_conclave_voter_trigger = yes }
						if = {
							limit = { this = scope:spn_iv_caller }
							scope:activity = { change_variable = { name = spn_iv_yes add = 1 } }
						}
						else_if = {
							limit = { this = scope:spn_iv_magus }
							scope:activity = { change_variable = { name = spn_iv_no add = 1 } }
						}
						else_if = {
							limit = { is_ai = no }
							trigger_event = spn_conclave.0250
						}
						else_if = {
							limit = { has_character_flag = spn_iv_bound }
							scope:activity = { change_variable = { name = spn_iv_no add = 1 } }
						}
						else = {
							spn_invocare_ai_vote_effect = yes
						}
					}
				}
			}
		}
		# Review L-9: it lapsed (the Grand Magus left the chair or took Imperium, or the caller left the Order). The caller
		# and the Grand Magus heard it called, so they hear it lapse, if they are players.
		else = {
			scope:activity = {
				var:spn_iv_caller ?= {
					if = {
						limit = {
							is_alive = yes
							is_ai = no
						}
						send_interface_toast = {
							type = event_toast_text_neutral
							title = spn_invocare_toast_lapsed
							desc = spn_invocare_toast_lapsed_desc
							left_icon = this
						}
					}
				}
				var:spn_iv_magus ?= {
					if = {
						limit = {
							is_alive = yes
							is_ai = no
						}
						send_interface_toast = {
							type = event_toast_text_neutral
							title = spn_invocare_toast_lapsed
							desc = spn_invocare_toast_lapsed_desc
							left_icon = this
						}
					}
				}
			}
		}
	}
}

# The Grand Magus's defence, paid now or lapsed (red team L25). His scope; scope:spn_iv_magus, scope:spn_iv_caller.
spn_invocare_defence_effect = {
	if = {
		limit = { has_character_flag = spn_iv_def_hooks }
		# Every voter in the room he holds a usable hook on is called in, and votes to keep him.
		scope:activity = {
			every_attending_character = {
				limit = {
					spn_conclave_voter_trigger = yes
					NOT = { this = scope:spn_iv_magus }
					NOT = { this = scope:spn_iv_caller }
					scope:spn_iv_magus = { has_usable_hook = prev }
				}
				scope:spn_iv_magus = { use_hook = prev }
				add_character_flag = { flag = spn_iv_bound years = 1 }
			}
		}
	}
	else_if = {
		limit = {
			has_character_flag = spn_iv_def_coffers
			mc_gold_global_avaiable_trigger = yes
			NOT = { global_var:current_admin_global ?= scope:spn_iv_caller }
		}
		# 100 gold from the council's funds, to the three masks most open to him who are not already bound.
		decrease_mc_funds_effect = { VALUE = 100 }
		scope:activity = {
			ordered_attending_character = {
				limit = {
					spn_conclave_voter_trigger = yes
					NOT = { this = scope:spn_iv_magus }
					NOT = { this = scope:spn_iv_caller }
					NOT = { has_character_flag = spn_iv_bound }
					is_ai = yes # 2.39 (final pass, 17 Oct): only an AI vote can be bought (spn_invocare_yes_chance_value)
				}
				order_by = spn_invocare_waver_value # 2.39 (final pass, 17 Oct): the wavering votes (F1), was his warmest friends
				max = 3
				check_range_bounds = no
				add_character_flag = { flag = spn_iv_paid years = 1 }
			}
		}
	}
	else_if = {
		limit = {
			has_character_flag = spn_iv_def_power
			can_spend_huge_ps_trigger = yes
		}
		spend_huge_ps_effect = yes
		add_character_flag = { flag = spn_iv_power years = 1 }
	}
	else_if = {
		limit = { has_character_flag = spn_iv_def_yield }
		# He steps down with dignity: standing first, the chair last.
		spn_order_add_standing_effect = { VALUE = 10 }
		scope:activity = { set_variable = spn_iv_yielded }
		add_character_flag = { flag = spn_order_recalled years = 1 } # 2.39 (final pass, 17 Oct): not in the election that replaces him (D4)
		if = { limit = { has_character_modifier = mage_council_support_modifier } remove_character_modifier = mage_council_support_modifier } # 2.39 (WORK-ORDER-2.39 F12): Imperium ends with the chair on the Order's own routes
		remove_from_mc_effect = yes
		spn_order_master_refresh_effect = yes
		spn_order_toast_members_effect = { TITLE = spn_order_toast_chair_vacated }
	}
}

# One AI voter casts (voter scope). The roll is hidden: Aye at spn_invocare_yes_chance_value, Nay at the rest.
spn_invocare_ai_vote_effect = {
	random_list = {
		0 = {
			modifier = {
				add = { value = spn_invocare_yes_chance_value }
			}
			scope:activity = { change_variable = { name = spn_iv_yes add = 1 } }
		}
		0 = {
			modifier = {
				add = { value = spn_invocare_no_chance_value }
			}
			scope:activity = { change_variable = { name = spn_iv_no add = 1 } }
		}
	}
}

# Host scope, the first line of The Vote's opening (spn_conclave_vote_open_effect), so the chair is empty, if it falls,
# before the election below it runs. Counts once. A tie keeps him (Witchcraft's own rule: ms_vote_yes > ms_vote_no).
spn_conclave_invocare_close_effect = {
	if = {
		limit = {
			scope:activity = {
				exists = var:spn_iv_yes
				NOT = { exists = var:spn_iv_counted }
			}
		}
		scope:activity = {
			set_variable = spn_iv_counted
			var:spn_iv_magus ?= { save_scope_as = spn_iv_magus }
			var:spn_iv_caller ?= { save_scope_as = spn_iv_caller }
		}
		if = {
			limit = {
				scope:spn_iv_magus ?= { is_alive = yes }
				global_var:current_archmage_global ?= scope:spn_iv_magus
				scope:activity = { var:spn_iv_yes > var:spn_iv_no }
			}
			# Passed. Bank the flag and the standing, then the chair falls empty (last).
			save_scope_value_as = { name = spn_iv_result value = flag:passed }
			scope:spn_iv_magus = {
				add_character_flag = { flag = spn_order_recalled years = 1 }
				spn_order_add_standing_effect = { VALUE = -25 }
			}
			scope:activity = {
				add_activity_log_entry = {
					key = spn_conclave_log_invocare_passed
					show_in_conclusion = yes
					character = scope:spn_iv_magus
				}
			}
			spn_invocare_tell_effect = yes
			scope:spn_iv_magus = {
				if = { limit = { has_character_modifier = mage_council_support_modifier } remove_character_modifier = mage_council_support_modifier } # 2.39 (WORK-ORDER-2.39 F12): Imperium ends with the chair on the Order's own routes
				remove_from_mc_effect = yes
			}
			spn_order_master_refresh_effect = yes
			spn_order_toast_members_effect = { TITLE = spn_order_toast_chair_vacated }
		}
		else_if = {
			limit = {
				scope:spn_iv_magus ?= { is_alive = yes }
				global_var:current_archmage_global ?= scope:spn_iv_magus
			}
			# Failed. The Grand Magus gains standing and a grudge; the caller pays again, and his seat goes last.
			save_scope_value_as = { name = spn_iv_result value = flag:kept }
			scope:spn_iv_magus = {
				spn_order_add_standing_effect = { VALUE = 10 }
				if = {
					limit = { scope:spn_iv_caller ?= { is_alive = yes } }
					add_opinion = {
						target = scope:spn_iv_caller
						modifier = spn_order_invocare_opinion
					}
				}
			}
			scope:activity = {
				add_activity_log_entry = {
					key = spn_conclave_log_invocare_kept
					show_in_conclusion = yes
					character = scope:spn_iv_magus
				}
			}
			spn_invocare_tell_effect = yes
			scope:spn_iv_caller ?= {
				spn_order_add_standing_effect = { VALUE = -20 }
				if = {
					limit = {
						has_variable = mc_member_var
						NOT = { global_var:current_archmage_global ?= this }
					}
					remove_from_mc_effect = yes
				}
			}
		}
	}
}

# The Slips Are Counted, to the caller and the Grand Magus if they are players. scope:spn_iv_result says which ending.
spn_invocare_tell_effect = {
	scope:spn_iv_caller ?= {
		if = {
			limit = {
				is_alive = yes
				is_ai = no
			}
			trigger_event = spn_conclave.0260
		}
	}
	scope:spn_iv_magus ?= {
		if = {
			limit = {
				is_alive = yes
				is_ai = no
			}
			trigger_event = spn_conclave.0260
		}
	}
}

# The call is spent without being heard (The Masks Must Meet found the Grand Magus gone, or the Conclave fell apart).
# Host scope. Nobody pays for a motion that was never heard.
spn_invocare_clear_call_effect = {
	if = {
		limit = { has_character_flag = spn_conclave_may_call }
		remove_character_flag = spn_conclave_may_call
	}
	if = {
		limit = { has_variable = spn_conclave_reason }
		remove_variable = spn_conclave_reason
	}
	if = {
		limit = { has_variable = spn_conclave_iv_magus }
		remove_variable = spn_conclave_iv_magus
	}
	if = {
		limit = { has_variable = spn_conclave_iv_caller }
		remove_variable = spn_conclave_iv_caller
	}
	if = {
		limit = { has_variable = spn_conclave_chair_after }
		remove_variable = spn_conclave_chair_after
	}
	if = {
		limit = { has_global_variable = spn_invocare_called }
		remove_global_variable = spn_invocare_called
	}
}

# F14 (D15): the masks raise favouritism, a lesser charge, at The Reckoning. Host scope. Every attending member with the
# flag loses 10 standing and the flag goes; only its own flag is removed (P4 sets it on junior and patron, 10 years). A
# Grand Magus under Invocare had his banked first (spn_invocare_bank_effect), so the vote still weighs it.
spn_conclave_favouritism_effect = {
	scope:activity = {
		every_attending_character = {
			limit = {
				spn_is_order_member_trigger = yes
				has_character_flag = spn_order_favouritism
			}
			spn_order_add_standing_effect = { VALUE = -10 }
			remove_character_flag = spn_order_favouritism
			if = {
				limit = { is_ai = no }
				send_interface_toast = {
					type = event_toast_effect_bad
					title = spn_conclave_toast_favouritism
					desc = spn_conclave_toast_favouritism_desc
					left_icon = this
				}
			}
		}
	}
}

# The Break itself (the hidden step, spn_order.0984, after every re-check). Root: the breaker. Everything is read and banked
# first; then, in this order: unmask, the treasury, the council, the temples last. Every removal goes through the existing
# primitives (remove_from_mc_effect, spn_order_dissolve_effect), which already handle loans, pledges, mirrors, tutors,
# heirs and, from 2.39, the Collegium (F15). Every writer of the seats and the chapters was enumerated for this
# (findings/order_politics.md §2.1).
spn_order_break_effect = {
	save_scope_as = spn_bk_breaker
	# 1. BANK: every temple head, every living member, the chair and the seats, the treasury.
	every_in_global_list = {
		variable = spn_order_chapters
		limit = { exists = var:spn_order_chapter }
		add_to_temporary_list = spn_bk_heads
		var:spn_order_chapter = {
			every_in_list = {
				variable = spn_order_members
				limit = {
					is_alive = yes
					spn_is_order_member_trigger = yes
				}
				add_to_temporary_list = spn_bk_members
			}
		}
	}
	global_var:current_archmage_global ?= { add_to_temporary_list = spn_bk_council }
	global_var:current_loremaster_global ?= {
		if = {
			limit = { NOT = { is_in_list = spn_bk_council } }
			add_to_temporary_list = spn_bk_council
		}
	}
	global_var:current_black_mage_global ?= {
		if = {
			limit = { NOT = { is_in_list = spn_bk_council } }
			add_to_temporary_list = spn_bk_council
		}
	}
	global_var:current_admin_global ?= {
		if = {
			limit = { NOT = { is_in_list = spn_bk_council } }
			add_to_temporary_list = spn_bk_council
		}
	}
	if = {
		limit = {
			mc_gold_global_exists_trigger = yes
			global_var:mc_gold_global > 0
		}
		if = {
