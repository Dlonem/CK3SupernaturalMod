# Supernatural 2.39a -- the one-path warning, where a player can see it (tool/pkgs/PB/HANDOFF.md §14).
# Hovering a magic capstone shows its trait, not the perk's effect, so the warning 2.39 put in the three capstone effects
# (archmage_perk_tt, warlock_perk_tt, true_witch_perk_tt) never reaches the player. These tooltips go on the perk before
# each capstone (both texts) and on each tree's root (the "closed" text only). Tooltip only: no effect runs.
# The tests mirror the capstones' own can_be_picked (spn_capstone_another_path): a character who already holds the
# capstone sees nothing; one who holds another path is told this tree's last perk is closed.

# The Mage (mage_p4_perk), after Forbidden Ritual.
spn_capstone_next_mage_tt_effect = {
	if = {
		limit = {
			NOT = { has_trait = archmage }
			OR = {
				has_trait = warlock
				has_trait = true_witch
			}
		}
		custom_tooltip = spn_capstone_closed_mage_tt
	}
	else_if = {
		limit = { NOT = { has_trait = archmage } }
		custom_tooltip = spn_capstone_next_mage_tt
	}
}

# The Warlock (warlock_perk), after Summoner.
spn_capstone_next_warlock_tt_effect = {
	if = {
		limit = {
			NOT = { has_trait = warlock }
			OR = {
				is_true_mage = yes
				has_trait = true_witch
			}
		}
		custom_tooltip = spn_capstone_closed_warlock_tt
	}
	else_if = {
		limit = { NOT = { has_trait = warlock } }
		custom_tooltip = spn_capstone_next_warlock_tt
	}
}
