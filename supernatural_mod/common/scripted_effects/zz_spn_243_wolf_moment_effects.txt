# Supernatural 2.43 -- picks this year's werewolf moment (the pattern of spn_moment_pick_effect, 2.41).
# CHARACTER scope: the player werewolf. Never the same moment twice running; each has a 15-year cooldown.
spn_wolf_moment_xp_effect = {
	if = {
		limit = { has_trait = supernatural_werewolf }
		add_trait_xp = { trait = supernatural_werewolf track = lycanthropy value = $XP$ }
	}
}
spn_wolf_moment_pick_effect = {
	random_courtier = {
		limit = { is_adult = yes is_ai = yes is_imprisoned = no NOT = { is_spouse_of = root } }
		save_scope_as = spn_wm_servant
	}
	random_courtier = {
		limit = { is_adult = no age >= 5 }
		save_scope_as = spn_wm_child
	}
	random_list = {
		10 = {
			trigger = {
				NOT = { var:spn_last_wolf_moment ?= flag:hounds }
				NOT = { has_character_flag = spn_wm_cd_hounds }
				exists = scope:spn_wm_servant
			}
			set_variable = { name = spn_last_wolf_moment value = flag:hounds }
			add_character_flag = { flag = spn_wm_cd_hounds years = 15 }
			trigger_event = { id = spn_moment.0026 days = { 1 60 } }
		}
		10 = {
			trigger = {
				NOT = { var:spn_last_wolf_moment ?= flag:daybefore }
				NOT = { has_character_flag = spn_wm_cd_daybefore }
			}
			set_variable = { name = spn_last_wolf_moment value = flag:daybefore }
			add_character_flag = { flag = spn_wm_cd_daybefore years = 15 }
			trigger_event = { id = spn_moment.0027 days = { 1 60 } }
		}
		10 = {
			trigger = {
				NOT = { var:spn_last_wolf_moment ?= flag:venison }
				NOT = { has_character_flag = spn_wm_cd_venison }
			}
			set_variable = { name = spn_last_wolf_moment value = flag:venison }
			add_character_flag = { flag = spn_wm_cd_venison years = 15 }
			trigger_event = { id = spn_moment.0028 days = { 1 60 } }
		}
		10 = {
			trigger = {
				NOT = { var:spn_last_wolf_moment ?= flag:stairs }
				NOT = { has_character_flag = spn_wm_cd_stairs }
				exists = scope:spn_wm_servant
			}
			set_variable = { name = spn_last_wolf_moment value = flag:stairs }
			add_character_flag = { flag = spn_wm_cd_stairs years = 15 }
			trigger_event = { id = spn_moment.0029 days = { 1 60 } }
		}
		10 = {
			trigger = {
				NOT = { var:spn_last_wolf_moment ?= flag:rain }
				NOT = { has_character_flag = spn_wm_cd_rain }
			}
			set_variable = { name = spn_last_wolf_moment value = flag:rain }
			add_character_flag = { flag = spn_wm_cd_rain years = 15 }
			trigger_event = { id = spn_moment.0030 days = { 1 60 } }
		}
		10 = {
			trigger = {
				NOT = { var:spn_last_wolf_moment ?= flag:voice }
				NOT = { has_character_flag = spn_wm_cd_voice }
			}
			set_variable = { name = spn_last_wolf_moment value = flag:voice }
			add_character_flag = { flag = spn_wm_cd_voice years = 15 }
			trigger_event = { id = spn_moment.0031 days = { 1 60 } }
		}
		10 = {
			trigger = {
				NOT = { var:spn_last_wolf_moment ?= flag:favourite }
				NOT = { has_character_flag = spn_wm_cd_favourite }
				exists = scope:spn_wm_child
			}
			set_variable = { name = spn_last_wolf_moment value = flag:favourite }
			add_character_flag = { flag = spn_wm_cd_favourite years = 15 }
			trigger_event = { id = spn_moment.0032 days = { 1 60 } }
		}
		10 = {
			trigger = {
				NOT = { var:spn_last_wolf_moment ?= flag:silver }
				NOT = { has_character_flag = spn_wm_cd_silver }
			}
			set_variable = { name = spn_last_wolf_moment value = flag:silver }
			add_character_flag = { flag = spn_wm_cd_silver years = 15 }
			trigger_event = { id = spn_moment.0033 days = { 1 60 } }
		}
	}
}
