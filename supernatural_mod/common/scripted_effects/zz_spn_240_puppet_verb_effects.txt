# Companion of spn_puppet_shared_verb_type_trigger (zz_spn_240_puppet_verb_triggers.txt).

# Called as the LAST line of the on_accept of every shared puppet verb (Seize and Imprison, Strike Them Down, Loom Over
# Them, Keep Them Company, Seduce, Romance, Order a Murder, Hand It Over, Send Me Your Gold). Interaction scope:
# scope:actor = the puppeteer, scope:puppet_or_actor = the puppet, scope:recipient = the target. Package QC (COMPEL)
# inserts after the marker line, through QC/edits.py, what a one-command puppet type does once its command is given
# (vanilla clear_puppet_status). The golem and the Sire Bond need nothing here: both last.
spn_puppet_shared_verb_done_effect = {
	# 2.40 COMPEL DONE
	# 2.40 (package QC, boot 4): a Compelled puppet obeys once, and this was its command: the compulsion ends
	# (clear_puppet_status, last) and someone close to them speaks up some days later (spn_compel.0001).
	if = {
		limit = { scope:actor.puppet:spn_compel_puppet ?= scope:puppet_or_actor }
		spn_compel_command_given_effect = { MASTER = scope:actor PUPPET = scope:puppet_or_actor }
	}
}
