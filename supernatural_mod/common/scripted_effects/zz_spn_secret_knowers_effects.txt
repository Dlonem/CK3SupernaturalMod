# Supernatural 2.34a -- THE PEOPLE WHO MADE YOU KNOW WHAT YOU ARE.
#
# THE BUG (Mertehan, Discord, 4 Sep 2026: "my literal sons are werewolf and i am a werewolf ... yet still
# scrying tells me the grass is green"). The base game's Find Secrets and Witchcraft's scrying both pick a
# random secret in your court that the ENGINE says you do not know, and vanilla weights a family member's
# secret three times over. Supernatural was creating its four secrets with no knower at all:
#   born      -- 10_spn_birthday_on_actions.txt tells the parents at BIRTH, but the child has no secret
#                yet; it is created ~16 years later by spn_initiate.1001/2001, which tells nobody.
#   embrace   -- spn_embrace.1004 (and 1005, 1006, 1009) reads the new vampire's secret in the same option
#                that queues spn_initiate.0001, so the secret does not exist yet and the sire learns nothing.
#   hunters   -- 10_spn_train_hunter.txt, same shape around spn_initiate.3001.
# The confer interaction alone got this right, by firing spn_initiate.9001 a day later.
#
# THE FIX. One effect, called from the four spn_give_*_secret_or_trait_effect the instant the secret
# exists, so every road into a creature trait is covered by the same line: the maker
# (spn_conferred_sire -- confer, embrace, blood donor, sired bite), the mother-sire of vampire blood
# (spn_mother_sire), the hunter's trainer (spn_hunter_trainer), and, for the born, a living mother or
# real father who is supernatural themselves -- the same people the birth code meant to tell.
# reveal_to is what the mod already uses; these four secret types declare no on_discover, so nothing fires.
#
# Every arm is guarded: the person must exist, be alive, not be the owner, and not already know. Scopes
# are saved INSIDE the guarded block, right before use, so the game-start repair sweep below can call this
# thousands of times in one chain without one character's maker leaking into the next.
#
# $TYPE$ = the secret type just added. Scope: the secret's owner.
spn_reveal_secret_to_those_who_made_me_effect = {
	save_temporary_scope_as = spn_fresh_owner
	random_secret = {
		limit = { secret_type = $TYPE$ }
		save_temporary_scope_as = spn_fresh_secret
		scope:spn_fresh_owner = {
			# The maker. Not for hunters: a hunter has a trainer, not a sire.
			if = {
				limit = { NOT = { scope:spn_fresh_secret = { secret_type = spn_secret_hunter } } }
				var:spn_conferred_sire ?= {
					save_temporary_scope_as = spn_secret_maker
					if = {
						limit = {
							is_alive = yes
							NOT = { this = scope:spn_fresh_owner }
							NOT = { scope:spn_fresh_secret = { is_known_by = scope:spn_secret_maker } }
						}
						scope:spn_fresh_secret = { reveal_to = scope:spn_secret_maker }
					}
				}
				var:spn_mother_sire ?= {
					save_temporary_scope_as = spn_secret_maker
					if = {
						limit = {
							is_alive = yes
							NOT = { this = scope:spn_fresh_owner }
							NOT = { scope:spn_fresh_secret = { is_known_by = scope:spn_secret_maker } }
						}
						scope:spn_fresh_secret = { reveal_to = scope:spn_secret_maker }
					}
				}
				# The born: a supernatural parent was told at birth, and now has something to know.
				mother ?= {
					save_temporary_scope_as = spn_secret_maker
					if = {
						limit = {
							is_alive = yes
							spn_has_supernatural_trait = yes
							NOT = { this = scope:spn_fresh_owner }
							NOT = { scope:spn_fresh_secret = { is_known_by = scope:spn_secret_maker } }
						}
						scope:spn_fresh_secret = { reveal_to = scope:spn_secret_maker }
					}
				}
				real_father ?= {
					save_temporary_scope_as = spn_secret_maker
					if = {
						limit = {
							is_alive = yes
							spn_has_supernatural_trait = yes
							NOT = { this = scope:spn_fresh_owner }
							NOT = { scope:spn_fresh_secret = { is_known_by = scope:spn_secret_maker } }
						}
						scope:spn_fresh_secret = { reveal_to = scope:spn_secret_maker }
					}
				}
			}
			# The trainer. Only for hunters.
			else = {
				var:spn_hunter_trainer ?= {
					save_temporary_scope_as = spn_secret_maker
					if = {
						limit = {
							is_alive = yes
							NOT = { this = scope:spn_fresh_owner }
							NOT = { scope:spn_fresh_secret = { is_known_by = scope:spn_secret_maker } }
						}
						scope:spn_fresh_secret = { reveal_to = scope:spn_secret_maker }
					}
				}
			}
		}
	}
}
