# Supernatural 2.38 -- "The Blue Rose": the Hermetic Order of the Blue Rose (WORK-ORDER-2.38 §5).
#
# THE DATA MODEL (C20: membership is data, never a trait).
#   On a member:  var:spn_order_member (set at initiation and seeding), var:spn_order_chapter (the chapter story; a neophyte
#                 has it too), var:spn_order_rank (flag:neophyte|acolyte|medicum|magistratus), var:spn_order_standing
#                 (0-100), var:spn_order_tutor, list spn_order_tutees, var:spn_order_pledge_step, var:spn_order_pledge_since.
#   On the story: spn_order_head, spn_order_seat (the seat county), spn_order_founded, spn_order_cap, spn_order_size,
#                 spn_order_way, the lists spn_order_members (everyone, the head and the neophytes too),
#                 spn_order_magistrati / _medici / _acolytes / _neophytes (the panel's rank lists, without the head),
#                 spn_order_reliquary, spn_order_reliquary_count, spn_order_reliquary_magic.
#   Global:       spn_order_chapters (every chapter's HEAD, kept by spn_order_set_head_effect), spn_order_master.
#
# REASON for leaving (§5.1): flag:death | flag:forget (Pulveris Memoria; strips a member's magic, T30) | flag:defect
# (keeps memory and magic; marked a defector) | flag:quiet (no cost).
#
# House rules: bank before anything irreversible; every create_story / create_artifact is followed by an exists guard
# before its saved scope is read (the 2.37 tooltip lesson); lists are copied to a temporary list before anyone is removed
# from them; a set or removal of a hidden scripted relation is inside hidden_effect (2.37e).

# ==================================================================================================
# MEMBERSHIP
# ==================================================================================================

# The ONE place membership and the secret are given (§5.5). Initiation and seeding call it; so do the chair's masks.
spn_order_grant_membership_effect = {
	set_variable = spn_order_member
	if = {
		limit = { NOT = { exists = var:spn_order_standing } }
		set_variable = { name = spn_order_standing value = 5 }
	}
	if = {
		limit = { NOT = { any_secret = { secret_type = spn_secret_blue_rose } } }
		add_secret = { type = spn_secret_blue_rose }
	}
	save_scope_as = spn_grant_who
	if = {
		limit = { var:spn_order_chapter ?= { NOT = { story_owner = scope:spn_grant_who } } }
		var:spn_order_chapter.story_owner = { save_scope_as = spn_grant_head }
		random_secret = {
			limit = {
				secret_type = spn_secret_blue_rose
				NOT = { is_known_by = scope:spn_grant_head }
			}
			reveal_to = scope:spn_grant_head
		}
	}
	spn_order_studies_apply_effect = yes
}

# Add $WHO$ to $CHAPTER$ (a story) at $RANK$ (a flag literal). A neophyte is a candidate: no membership, no secret.
spn_order_add_member_effect = {
	$WHO$ = { save_scope_as = spn_add_who }
	$CHAPTER$ = { save_scope_as = spn_add_ch }
	scope:spn_add_who = {
		set_variable = { name = spn_order_chapter value = scope:spn_add_ch }
		set_variable = { name = spn_order_rank value = $RANK$ }
		if = {
			# 2.38a2: ?= (a tooltip builds this without the variable set: 7,275 errors while A Note and a Blue Rose was open)
			limit = { var:spn_order_rank ?= flag:neophyte }
			set_variable = { name = spn_order_pledge_since value = current_year }
		}
	}
	scope:spn_add_ch = {
		if = {
			limit = { NOT = { is_target_in_variable_list = { name = spn_order_members target = scope:spn_add_who } } }
			add_to_variable_list = { name = spn_order_members target = scope:spn_add_who }
		}
		spn_order_refresh_effect = yes
	}
}

# Set $WHO$'s rank ($RANK$, a flag literal) and refresh his chapter.
spn_order_set_rank_effect = {
	$WHO$ = {
		set_variable = { name = spn_order_rank value = $RANK$ }
		var:spn_order_chapter ?= { spn_order_refresh_effect = yes }
	}
}

# Remove $WHO$ for $REASON$ (flag:death | flag:forget | flag:defect | flag:quiet). §5.1's order: bank first.
spn_order_remove_member_effect = {
	$WHO$ = { save_scope_as = spn_rem_who }
	save_scope_value_as = { name = spn_rem_reason value = $REASON$ }
	scope:spn_rem_who = {
		if = {
			limit = { exists = var:spn_order_chapter }
			var:spn_order_chapter = { save_scope_as = spn_rem_ch }
			if = {
				limit = { spn_is_order_member_trigger = yes }
				save_scope_value_as = { name = spn_rem_was_member value = yes }
			}
			# A head who leaves alive passes the temple on (the 2.37 alpha lesson): banked here, run LAST, after he is off
			# the list and his data is gone, so a succession that dissolves the chapter can never reach him again (review,
			# 9 Oct: it recursed). A dead head is left to the story's own on_owner_death. Never while the chapter is
			# already dissolving.
			if = {
				limit = {
					scope:spn_rem_ch = {
						story_owner = scope:spn_rem_who
						NOT = { has_variable = spn_order_dissolving }
					}
					NOT = { scope:spn_rem_reason = flag:death }
				}
				scope:spn_rem_ch = { save_scope_as = spn_rem_succ_ch }
			}
		}
		# Off the chapter's list.
		if = {
			limit = { exists = scope:spn_rem_ch }
			scope:spn_rem_ch = {
				if = {
					limit = { is_target_in_variable_list = { name = spn_order_members target = scope:spn_rem_who } }
					remove_list_variable = { name = spn_order_members target = scope:spn_rem_who }
				}
				if = {
					limit = { var:spn_order_heir ?= scope:spn_rem_who }
					remove_variable = spn_order_heir
				}
			}
		}
		# His tutor's list, and his own tutees (re-assigned at the chapter's next year).
		if = {
			limit = { exists = var:spn_order_tutor }
			var:spn_order_tutor = {
				if = {
					limit = { is_target_in_variable_list = { name = spn_order_tutees target = scope:spn_rem_who } }
					remove_list_variable = { name = spn_order_tutees target = scope:spn_rem_who }
				}
				hidden_effect = {
					if = {
						limit = { has_relation_spn_order_tutor_of = scope:spn_rem_who }
						remove_relation_spn_order_tutor_of = scope:spn_rem_who
					}
				}
			}
			remove_variable = spn_order_tutor
		}
		if = {
			limit = { has_variable_list = spn_order_tutees }
			every_in_list = {
				variable = spn_order_tutees
				add_to_temporary_list = spn_rem_tutees
			}
			every_in_list = {
				list = spn_rem_tutees
				if = {
					limit = { var:spn_order_tutor ?= scope:spn_rem_who }
					remove_variable = spn_order_tutor
				}
				hidden_effect = {
					if = {
						limit = { scope:spn_rem_who = { has_relation_spn_order_tutor_of = prev } }
						scope:spn_rem_who = { remove_relation_spn_order_tutor_of = prev }
					}
				}
			}
			clear_variable_list = spn_order_tutees
		}
		# The data (C20) and its one visible trace.
		remove_variable = spn_order_chapter
		remove_variable = spn_order_rank
		remove_variable = spn_order_member
		remove_variable = spn_order_pledge_step
		remove_variable = spn_order_pledge_since
		remove_variable = spn_order_taught_years
		# A transfer to a new chapter (a charter, a daughter, a founding) keeps who he is in the Order (review, 9 Oct).
		if = {
			limit = { NOT = { scope:spn_rem_reason = flag:transfer } }
			remove_variable = spn_order_standing
			remove_variable = spn_order_adeptus
			if = {
				limit = { has_character_flag = spn_order_aspires_chair }
				remove_character_flag = spn_order_aspires_chair
			}
			if = {
				limit = { has_character_flag = spn_order_tutor_strict }
				remove_character_flag = spn_order_tutor_strict
			}
			if = {
				limit = { has_character_flag = spn_order_tutor_patient }
				remove_character_flag = spn_order_tutor_patient
			}
		}
		if = {
			limit = { has_character_flag = spn_order_summit_errand }
			remove_character_flag = spn_order_summit_errand
		}
		if = {
			limit = { has_character_flag = spn_order_seat_declined }
			remove_character_flag = spn_order_seat_declined
		}
		spn_order_studies_remove_effect = yes
		if = {
			limit = { has_character_modifier = spn_order_tutor_modifier }
			remove_character_modifier = spn_order_tutor_modifier
		}
		if = {
			limit = { has_character_modifier = spn_reliquary_modifier }
			remove_character_modifier = spn_reliquary_modifier
		}
		remove_variable = spn_reliquary_cached
		if = {
			limit = { NOT = { scope:spn_rem_reason = flag:transfer } }
			every_secret = {
				limit = { secret_type = spn_secret_blue_rose }
				remove_secret = yes
			}
		}
		# His pledges to that chapter, and any loan he holds.
		if = {
			limit = { exists = scope:spn_rem_ch }
			every_character_artifact = {
				limit = { var:spn_pledged_to ?= scope:spn_rem_ch }
				add_to_temporary_list = spn_rem_pledges
			}
			every_in_list = {
				list = spn_rem_pledges
				spn_relic_unpledge_effect = { ART = this }
			}
		}
		spn_relic_return_loans_of_effect = { WHO = scope:spn_rem_who }
		# His mirror.
		random_owned_story = {
			limit = { story_type = story_spn_order_mirror }
			end_story = yes
		}
	}
	if = {
		limit = { exists = scope:spn_rem_ch }
		scope:spn_rem_ch = { spn_order_refresh_effect = yes }
	}
	# 2.38b: the council is the Order's (DESIGN-2.38b §3). Leaving the Order leaves the council: not for a transfer, and
	# not for a death (Witchcraft clears a dead man's seat itself, on_action_mc_remover).
	if = {
		limit = {
			NOR = {
				scope:spn_rem_reason = flag:transfer
				scope:spn_rem_reason = flag:death
			}
			scope:spn_rem_who = { has_variable = mc_member_var }
		}
		scope:spn_rem_who = { remove_from_mc_effect = yes }
	}
	# What the reason costs (§5.10). Last: the bank is done.
	if = {
		limit = { scope:spn_rem_reason = flag:forget }
		scope:spn_rem_who = { spn_order_pulveris_effect = yes }
	}
	else_if = {
		limit = { scope:spn_rem_reason = flag:defect }
		scope:spn_rem_who = {
			add_character_flag = spn_order_defector
			if = {
				limit = { exists = scope:spn_rem_ch }
				set_variable = { name = spn_order_defected_from value = scope:spn_rem_ch.story_owner years = 20 }
			}
			if = {
				limit = { NOT = { has_character_flag = spn_order_left_alone } }
				random = {
					chance = 40
					trigger_event = {
						id = spn_order.0193
						days = { 730 1825 }
					}
				}
			}
		}
	}
	# The succession, last (see above). The banked chapter is moved to a second name first, so a member removed by a
	# dissolve inside it never sees it.
	if = {
		limit = { exists = scope:spn_rem_succ_ch }
		scope:spn_rem_succ_ch = { save_scope_as = spn_rem_succ_now }
		clear_saved_scope = spn_rem_succ_ch
		scope:spn_rem_succ_now = {
			if = {
				limit = { NOT = { has_variable = spn_order_dissolving } }
				spn_order_succession_effect = yes
			}
		}
	}
}

# The chapter's studies modifier, chosen by its way (§5.22). Member scope. years = 2, re-applied every chapter year.
spn_order_studies_remove_effect = {
	if = {
		limit = { has_character_modifier = spn_order_studies_modifier }
		remove_character_modifier = spn_order_studies_modifier
	}
	if = {
		limit = { has_character_modifier = spn_order_studies_silence_modifier }
		remove_character_modifier = spn_order_studies_silence_modifier
	}
	if = {
		limit = { has_character_modifier = spn_order_studies_subservience_modifier }
		remove_character_modifier = spn_order_studies_subservience_modifier
	}
	if = {
		limit = { has_character_modifier = spn_order_studies_observance_modifier }
		remove_character_modifier = spn_order_studies_observance_modifier
	}
	if = {
		limit = { has_character_modifier = spn_order_way_silence_modifier }
		remove_character_modifier = spn_order_way_silence_modifier
	}
	if = {
		limit = { has_character_modifier = spn_order_way_observance_modifier }
		remove_character_modifier = spn_order_way_observance_modifier
	}
}
spn_order_studies_apply_effect = {
	spn_order_studies_remove_effect = yes
	if = {
		limit = { spn_is_order_member_trigger = yes }
		if = {
			limit = { var:spn_order_chapter ?= { var:spn_order_way ?= flag:silence } }
			add_character_modifier = { modifier = spn_order_studies_silence_modifier years = 2 }
			add_character_modifier = { modifier = spn_order_way_silence_modifier years = 2 }
		}
		else_if = {
			limit = { var:spn_order_chapter ?= { var:spn_order_way ?= flag:subservience } }
			add_character_modifier = { modifier = spn_order_studies_subservience_modifier years = 2 }
		}
		else_if = {
			limit = { var:spn_order_chapter ?= { var:spn_order_way ?= flag:observance } }
			add_character_modifier = { modifier = spn_order_studies_observance_modifier years = 2 }
			add_character_modifier = { modifier = spn_order_way_observance_modifier years = 2 }
		}
		else = {
			add_character_modifier = { modifier = spn_order_studies_modifier years = 2 }
		}
	}
}

# Story scope. Rebuild the panel's rank lists and the size from spn_order_members, then the mirrors.
spn_order_refresh_effect = {
	save_scope_as = spn_ref_ch
	set_variable = { name = spn_order_head value = story_owner }
	clear_variable_list = spn_order_magistrati
	clear_variable_list = spn_order_medici
	clear_variable_list = spn_order_acolytes
	clear_variable_list = spn_order_neophytes
	set_variable = { name = spn_order_size value = 0 }
	every_in_list = {
		variable = spn_order_members
		limit = { is_alive = yes }
		scope:spn_ref_ch = { change_variable = { name = spn_order_size add = 1 } }
		if = {
			limit = { NOT = { this = scope:spn_ref_ch.story_owner } }
			if = {
				limit = { var:spn_order_rank ?= flag:magistratus }
				scope:spn_ref_ch = { add_to_variable_list = { name = spn_order_magistrati target = prev } }
			}
			else_if = {
				limit = { var:spn_order_rank ?= flag:medicum }
				scope:spn_ref_ch = { add_to_variable_list = { name = spn_order_medici target = prev } }
			}
			else_if = {
				limit = { var:spn_order_rank ?= flag:acolyte }
				scope:spn_ref_ch = { add_to_variable_list = { name = spn_order_acolytes target = prev } }
			}
			else = {
				scope:spn_ref_ch = { add_to_variable_list = { name = spn_order_neophytes target = prev } }
			}
		}
	}
	spn_order_sync_mirrors_effect = yes
}

# Story scope (§5.7). Every PLAYER member who is not the head owns a mirror whose variables are copies (C2).
spn_order_sync_mirrors_effect = {
	save_scope_as = spn_mir_ch
	every_in_list = {
		variable = spn_order_members
		limit = {
			is_ai = no
			is_alive = yes
			spn_is_order_member_trigger = yes
			NOT = { this = scope:spn_mir_ch.story_owner }
		}
		if = {
			limit = { NOT = { any_owned_story = { story_type = story_spn_order_mirror } } }
			create_story = story_spn_order_mirror
		}
		random_owned_story = {
			limit = { story_type = story_spn_order_mirror }
			set_variable = { name = spn_mirror_of value = scope:spn_mir_ch }
			set_variable = { name = spn_order_head value = scope:spn_mir_ch.story_owner }
			set_variable = { name = spn_order_seat value = scope:spn_mir_ch.var:spn_order_seat }
			# 2.39 (WORK-ORDER-2.39 F15): the hall, for the member's panel line.
			if = {
				limit = { scope:spn_mir_ch = { exists = var:spn_order_hall } }
				set_variable = { name = spn_order_hall value = scope:spn_mir_ch.var:spn_order_hall }
			}
			else_if = {
				limit = { has_variable = spn_order_hall }
				remove_variable = spn_order_hall
			}
			set_variable = { name = spn_order_founded value = scope:spn_mir_ch.var:spn_order_founded }
			set_variable = { name = spn_order_cap value = scope:spn_mir_ch.var:spn_order_cap }
			set_variable = { name = spn_order_size value = scope:spn_mir_ch.var:spn_order_size }
			if = {
				limit = { scope:spn_mir_ch = { exists = var:spn_order_way } }
				set_variable = { name = spn_order_way value = scope:spn_mir_ch.var:spn_order_way }
			}
			else = {
				remove_variable = spn_order_way
			}
			if = {
				limit = { scope:spn_mir_ch = { exists = var:spn_order_reliquary_count } }
				set_variable = { name = spn_order_reliquary_count value = scope:spn_mir_ch.var:spn_order_reliquary_count }
			}
			clear_variable_list = spn_order_magistrati
			clear_variable_list = spn_order_medici
			clear_variable_list = spn_order_acolytes
			clear_variable_list = spn_order_neophytes
			clear_variable_list = spn_order_reliquary
			save_scope_as = spn_mir_story
			scope:spn_mir_ch = {
				every_in_list = {
					variable = spn_order_magistrati
					scope:spn_mir_story = { add_to_variable_list = { name = spn_order_magistrati target = prev } }
				}
				every_in_list = {
					variable = spn_order_medici
					scope:spn_mir_story = { add_to_variable_list = { name = spn_order_medici target = prev } }
				}
				every_in_list = {
					variable = spn_order_acolytes
					scope:spn_mir_story = { add_to_variable_list = { name = spn_order_acolytes target = prev } }
				}
				every_in_list = {
					variable = spn_order_neophytes
					scope:spn_mir_story = { add_to_variable_list = { name = spn_order_neophytes target = prev } }
				}
				every_in_list = {
					variable = spn_order_reliquary
					scope:spn_mir_story = { add_to_variable_list = { name = spn_order_reliquary target = prev } }
				}
			}
		}
	}
}

# Initiation (Will You Heed the Call?, §5.5). Root: the neophyte.
spn_order_initiate_effect = {
	spn_order_grant_membership_effect = yes
	# 2.39 (WORK-ORDER-2.39 F14 §2): a favour earned as a neophyte is kept; otherwise he starts at 5, as before.
	if = {
		limit = {
			OR = {
				NOT = { exists = var:spn_order_standing }
				var:spn_order_standing < 5
			}
		}
		set_variable = { name = spn_order_standing value = 5 }
	}
	remove_variable = spn_order_pledge_step
	remove_variable = spn_order_pledge_since
	spn_order_set_rank_effect = { WHO = this RANK = flag:acolyte }
	spn_order_assign_tutor_effect = { WHO = this }
	if = {
		limit = { NOT = { any_memory = { memory_type = spn_order_initiated_memory } } }
		create_character_memory = { type = spn_order_initiated_memory }
	}
}

# A tutor for $WHO$ (an acolyte): the Medicum or higher of his chapter with the fewest tutees (§5.6 step 4).
spn_order_assign_tutor_effect = {
	$WHO$ = { save_scope_as = spn_tut_who }
	scope:spn_tut_who.var:spn_order_chapter ?= {
		ordered_in_list = {
			variable = spn_order_members
			limit = {
				is_alive = yes
				NOT = { this = scope:spn_tut_who }
				spn_order_medicum_plus_trigger = yes
				is_imprisoned = no
			}
			order_by = spn_order_tutor_pick_value
			save_scope_as = spn_tut_tutor
		}
	}
	if = {
		limit = { exists = scope:spn_tut_tutor }
		scope:spn_tut_who = { set_variable = { name = spn_order_tutor value = scope:spn_tut_tutor } }
		scope:spn_tut_tutor = {
			add_to_variable_list = { name = spn_order_tutees target = scope:spn_tut_who }
			hidden_effect = {
				if = {
					limit = { NOT = { has_relation_spn_order_tutor_of = scope:spn_tut_who } }
					set_relation_spn_order_tutor_of = scope:spn_tut_who
				}
			}
		}
		scope:spn_tut_who = {
			if = {
				limit = { NOT = { any_memory = { memory_type = spn_order_tutor_memory } } }
				create_character_memory = {
					type = spn_order_tutor_memory
					participants = { tutor = scope:spn_tut_tutor }
				}
			}
			if = {
				limit = { is_ai = no }
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = spn_order_toast_tutor_got
					left_icon = this
					right_icon = scope:spn_tut_tutor
				}
			}
		}
		scope:spn_tut_tutor = {
			if = {
				limit = { is_ai = no }
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = spn_order_toast_tutee_got
					left_icon = this
					right_icon = scope:spn_tut_who
				}
			}
		}
		# 2.38b: The First Lesson (DESIGN-2.38b §7), for whichever of the two is a player.
		scope:spn_tut_tutor = { save_scope_as = spn_tp_tutor }
		scope:spn_tut_who = { save_scope_as = spn_tp_tutee }
		if = {

# Story scope. The one place a chapter's head changes (§5.8; the 2.37 set-alpha shape).

			limit = { NOT = { is_target_in_global_variable_list = { name = spn_order_chapters target = scope:spn_sh_new } } }
			add_to_global_variable_list = { name = spn_order_chapters target = scope:spn_sh_new }
		}

			limit = { var:spn_order_heir ?= scope:spn_sh_new }
			remove_variable = spn_order_heir
		}
		scope:spn_sh_new = {

				limit = { story_type = story_spn_order_mirror }

			}

			}

				if = {
					limit = { NOT = { scope:spn_sh_ch = { exists = var:spn_order_way } } }
					spn_order_ai_pick_way_effect = yes
				}
			}
		}
		spn_order_refresh_effect = yes
		spn_order_master_refresh_effect = yes
	}
}

# on_owner_death, or a head who leaves alive (§5.8). Story scope. The named heir first, then the best Magistratus, then
# Medicum, then acolyte (members only); nobody -> the chapter dissolves.
spn_order_succession_effect = {
	save_scope_as = spn_su_ch

		limit = {
			var:spn_order_heir ?= {
				is_alive = yes
				NOT = { this = scope:spn_su_old }
				var:spn_order_chapter ?= scope:spn_su_ch
				var:spn_order_rank ?= flag:magistratus
			}
		}
		var:spn_order_heir = { save_scope_as = spn_su_new }
	}
	else = {
		ordered_in_list = {
			variable = spn_order_members
			limit = {
				is_alive = yes
				NOT = { this = scope:spn_su_old }
				spn_is_order_member_trigger = yes
			}
			order_by = spn_order_succession_score_value
			save_scope_as = spn_su_new
		}
	}
	if = {
		limit = { exists = scope:spn_su_new }
		spn_order_set_head_effect = { NEW = scope:spn_su_new }
	}
	else = {
		spn_order_dissolve_effect = yes
	}
}

# $WHO$ moves up one rank (§5.6 step 6). Standing is never reduced by a promotion.
spn_order_promote_effect = {
	$WHO$ = {
		save_scope_as = spn_pr_who
		if = {
			limit = { var:spn_order_rank ?= flag:acolyte }
			spn_order_set_rank_effect = { WHO = this RANK = flag:medicum }
			# A Medicum has no tutor of his own any more.
			if = {
				limit = { exists = var:spn_order_tutor }
				var:spn_order_tutor = {
					if = {
						limit = { is_target_in_variable_list = { name = spn_order_tutees target = scope:spn_pr_who } }
						remove_list_variable = { name = spn_order_tutees target = scope:spn_pr_who }
					}
					hidden_effect = {
						if = {
							limit = { has_relation_spn_order_tutor_of = scope:spn_pr_who }
							remove_relation_spn_order_tutor_of = scope:spn_pr_who
						}
					}
				}
				remove_variable = spn_order_tutor
			}
			if = {
				limit = { is_ai = no }
				trigger_event = { id = spn_order.0141 days = 1 }
			}
			else = {
				random_list = {
					40 = {
						modifier = { add = 30 has_trait = wrathful }
						modifier = { add = 20 has_trait = arrogant }
						add_character_flag = spn_order_tutor_strict
					}
					60 = {
						modifier = { add = 30 has_trait = compassionate }
						modifier = { add = 20 has_trait = patient }
						add_character_flag = spn_order_tutor_patient
					}
				}
			}
		}
		else_if = {
			limit = { var:spn_order_rank ?= flag:medicum }
			spn_order_set_rank_effect = { WHO = this RANK = flag:magistratus }
			if = {
				limit = { is_ai = no }
				send_interface_toast = {
					type = event_toast_effect_good
					title = spn_order_toast_rank_magistratus
					left_icon = this
					custom_tooltip = spn_order_toast_rank_magistratus_tt
				}
			}
		}
	}
}

# Standing is 0-100 (§4.2). $VALUE$ may be a number or a saved scope value.

	if = {
		limit = { NOT = { exists = var:spn_order_standing } }

	if = {
		limit = { var:spn_order_standing > 100 }
		set_variable = { name = spn_order_standing value = 100 }
	}
	else_if = {
		limit = { var:spn_order_standing < 0 }
		set_variable = { name = spn_order_standing value = 0 }
	}
}

spn_order_chair_yearly_effect = {
	# The Chair Is Empty: once, the year the formed council has no chair.
	if = {
		limit = {
			exists = global_var:mc_has_been_formed
			NOT = { exists = global_var:current_archmage_global }
			NOT = { exists = global_var:spn_order_chair_empty_told }
		}
		set_global_variable = spn_order_chair_empty_told
		spn_order_toast_members_effect = { TITLE = spn_order_toast_chair_empty }
	}
	if = {
		limit = { exists = global_var:current_archmage_global }
		remove_global_variable = spn_order_chair_empty_told
	}
	# A new chair: the masks are offered once to a Mage who is not a member; one who cannot join leaves the Order with
	# no chair of its own.
	if = {
		limit = {
			global_var:current_archmage_global ?= {
				is_alive = yes
				spn_is_order_member_trigger = no
			}
			NOT = { global_var:spn_order_chair_seen ?= global_var:current_archmage_global }
			any_in_global_list = {
				variable = spn_order_chapters
				is_alive = yes
			}
		}
		set_global_variable = { name = spn_order_chair_seen value = global_var:current_archmage_global }
		global_var:current_archmage_global = {
			if = {
				limit = { spn_order_can_join_trigger = yes }
				if = {
					limit = { is_ai = yes }
					spn_order_masks_bow_effect = yes
				}
				else = {
					trigger_event = spn_order.0960
				}
			}
			# 2.38b: the chair is the Order's. One who cannot join leaves it empty for the temples to fill.
			else = {
				remove_from_mc_effect = yes
				spn_order_toast_members_effect = { TITLE = spn_order_toast_chair_vacated }
			}
		}
	}
	# 2.39 (WORK-ORDER-2.39 F1): a new face in the chair is dated (Invocare's three years). Identity state, no expiry.
	# Review L-2: the first face this pulse ever sees in an older save (three years or more after its start) has sat there
	# longer than anyone can tell, so he counts as settled. Every later face, and the first one of a new game, is dated to
	# this year. spn_order_chair_looked (identity state, no expiry) marks that the pulse has looked at the chair once.
	if = {
		limit = {
			exists = global_var:current_archmage_global
			NOT = { global_var:spn_order_chair_dated ?= global_var:current_archmage_global }
		}
		if = {
			limit = {
				NOT = { exists = global_var:spn_order_chair_looked }
				years_from_game_start >= 3
			}
			set_global_variable = { name = spn_order_chair_year value = spn_order_three_years_ago_value }
		}
		else = {
			set_global_variable = { name = spn_order_chair_year value = current_year }
		}
		set_global_variable = { name = spn_order_chair_dated value = global_var:current_archmage_global }
	}
	if = {
		limit = { NOT = { exists = global_var:spn_order_chair_looked } }
		set_global_variable = { name = spn_order_chair_looked value = yes }
	}
	# A member in the chair, once per man: The Chair of the Grand Magus.
	if = {
		limit = {
			global_var:current_archmage_global ?= {
				is_alive = yes
				spn_is_order_member_trigger = yes
				NOT = { has_character_flag = spn_order_chair_moment }
			}
		}
		global_var:current_archmage_global = {
			add_character_flag = spn_order_chair_moment
			if = {
				limit = { is_ai = no }
				trigger_event = spn_order.0963
			}
		}
	}
	# The Council Calls You: an AI member chair, an empty seat, the Order's best candidate.
	if = {
		limit = {
			global_var:current_archmage_global ?= {
				is_alive = yes
				is_ai = yes
				spn_is_order_member_trigger = yes
				is_imprisoned = no
			}
			OR = {
				NOT = { exists = global_var:current_loremaster_global }
				NOT = { exists = global_var:current_black_mage_global }
				NOT = { exists = global_var:current_admin_global }
			}
		}
		global_var:current_archmage_global = { save_scope_as = spn_cc_chair }
		ordered_in_global_list = {
			variable = spn_order_chapters
			limit = { spn_order_seat_candidate_trigger = { CHAIR = scope:spn_cc_chair } }
			order_by = var:spn_order_standing
			save_scope_as = spn_cc_pick
		}
		if = {
			limit = { NOT = { exists = scope:spn_cc_pick } }
			scope:spn_cc_chair.var:spn_order_chapter ?= {
				ordered_in_list = {
					variable = spn_order_members
					limit = {
						var:spn_order_rank ?= flag:magistratus
						spn_order_seat_candidate_trigger = { CHAIR = scope:spn_cc_chair }
					}
					order_by = var:spn_order_standing
					save_scope_as = spn_cc_pick
				}
			}
		}
		if = {
			limit = { exists = scope:spn_cc_pick }
			scope:spn_cc_pick = { trigger_event = spn_order.0970 }
		}
	}
}

# 2.38b: THE TUTOR AND THE ACOLYTE (DESIGN-2.38b §7). Root: the tutee; var:spn_order_tutor the tutor. The event goes to
# the player of the two (to the tutee if both are players).

spn_order_tutor_year_effect = {
	save_scope_as = spn_tp_tutee

	if = {
		limit = { scope:spn_tp_tutee = { is_ai = no } }

	}
	else = {
		scope:spn_tp_tutor = { save_scope_as = spn_tp_player }
	}
	# 2.39 (WORK-ORDER-2.39 F14 V2): one tutor moment a year for a PLAYER, not one per pair (a tutor with three acolytes
	# rolled three times), and none while the flavor gate is closed. The branch that sends sets spn_tp_moment_year.
	random = {
		chance = {
			value = 50
			if = {
				limit = {
					scope:spn_tp_player = {
						OR = {
							has_character_flag = spn_tp_moment_year
							spn_flavor_ready_trigger = no
						}
					}
				}
				value = 0
			}
		}
		random_list = {
			# The Errand: the acolyte is sent somewhere he should not go.
			10 = {
				trigger = {
					scope:spn_tp_tutee = { is_ai = no }
					NOT = { scope:spn_tp_tutee = { has_character_flag = spn_tp_seen_errand } }
				}
				scope:spn_tp_tutee = { trigger_event = { id = spn_order.0321 days = { 10 120 } } } # 2.39: seen on arrival (.0321)

						scope:spn_tp_tutor = { involved_activity ?= { has_activity_type = magical_research } }
					}

				scope:spn_tp_player = { # 2.39: seen on arrival (.0323)
					trigger_event = { id = spn_order.0323 days = { 10 120 } }
					add_character_flag = { flag = spn_tp_moment_year months = 11 }
				}
			}
			# Something more: grown adults only (18 or older, both), not family, once for a pair.
			5 = {
				trigger = {

						NOT = { has_character_flag = spn_tp_seen_close }

									}

# 2.38c: the Order teaches its Mages (Thomas, 9 Oct: "are there any opportunities as a mage to grow within the order?").
# 2.38e: one point a year, no more, on a path drawn at random (Thomas's 73-year run: ten members at Secret Knowledge 75-100,
# no outsider there; "I'd have it randomize one XP from the different mage trait tracks"). The temple's way doubles the
# chance of its own path: Observance Secret Knowledge, Silence Alchemy, Subservience Combat Magic; Treat with the Pit
# doubles Demonology. A finished path (100) is never drawn. Character scope, from the chapter's year.
spn_order_mage_growth_effect = {
	if = {
		limit = {
			has_trait = archmage
			spn_is_order_member_trigger = yes
			OR = {
				has_trait_xp = { trait = archmage track = secret_knowledge value < 100 }
				has_trait_xp = { trait = archmage track = combat_magic value < 100 }
				has_trait_xp = { trait = archmage track = demonology value < 100 }
				has_trait_xp = { trait = archmage track = alchemy value < 100 }
			}
		}
		random_list = {
			10 = {
				trigger = { has_trait_xp = { trait = archmage track = secret_knowledge value < 100 } }
				modifier = {
					add = 10
					var:spn_order_chapter ?= { var:spn_order_way ?= flag:observance }
				}
				add_trait_xp = { trait = archmage track = secret_knowledge value = 1 }
			}
			10 = {
				trigger = { has_trait_xp = { trait = archmage track = combat_magic value < 100 } }
				modifier = {
					add = 10
					var:spn_order_chapter ?= { var:spn_order_way ?= flag:subservience }
				}
				add_trait_xp = { trait = archmage track = combat_magic value = 1 }
			}
			10 = {
				trigger = { has_trait_xp = { trait = archmage track = demonology value < 100 } }
				modifier = {
					add = 10
					spn_order_prerogative_is_trigger = { P = flag:pit }
				}
				add_trait_xp = { trait = archmage track = demonology value = 1 }
			}
			10 = {
				trigger = { has_trait_xp = { trait = archmage track = alchemy value < 100 } }
				modifier = {
					add = 10
					var:spn_order_chapter ?= { var:spn_order_way ?= flag:silence }
				}
				add_trait_xp = { trait = archmage track = alchemy value = 1 }
			}
		}
	}
}
