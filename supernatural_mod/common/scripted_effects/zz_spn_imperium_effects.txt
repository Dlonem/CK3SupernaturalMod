# Supernatural 2.39 -- Imperium and the council's works (WORK-ORDER-2.39 F12, F13). The state model is at the top of
# common/scripted_triggers/zz_spn_imperium_triggers.txt. Extends Witchcraft's own meeting ("the second part",
# magic_ev10.85), its council support modifier (mage_council_support_modifier, renamed "Imperium" in the replace loc),
# F1's yes/no rule (Aye must beat Nay; a tie refuses), the Order's crime (spn_order_broke_rules_effect) and Respondeo's
# own three effects. Not a second vote system, not a second support modifier, not a second memory system.
#
# Order of a motion: open (bank the voters) -> levers (a player Grand Magus picks one in Before the Council Votes; an AI
# picks by nature; none in an emergency session) -> start (the AI seats vote at once, the fallback count is queued) ->
# next (each player seat in Witchcraft's turn order) -> count (once: spn_imp_counted) -> the result.

# ==================================================================================================
# THE MOTION
# ==================================================================================================
# Open a vote of the council on this Grand Magus's motion. $MOTION$ = flag:imperium | flag:respondeo; $CRISIS$ = flag:none
# for a motion he brings, or flag:death | flag:rift | flag:plague | flag:hunters for an emergency session (no levers, no
# meeting). Grand Magus scope.
spn_imperium_open_effect = {
	save_scope_as = spn_imp_gm
	# The vote's number. global_var:spn_imp_serial only ever counts up; it is bookkeeping (identity state), so it has no
	# expiry. A ballot carries it, so a window left open from an earlier vote never counts in this one.
	if = {
		limit = { NOT = { exists = global_var:spn_imp_serial } }
		set_global_variable = { name = spn_imp_serial value = 0 }
	}
	change_global_variable = { name = spn_imp_serial add = 1 }
	set_variable = { name = spn_imp_vote_id value = global_var:spn_imp_serial days = 60 }
	save_scope_value_as = { name = spn_imp_ballot value = global_var:spn_imp_serial }
	set_variable = { name = spn_imp_open value = yes days = 60 }
	set_variable = { name = spn_imp_motion value = $MOTION$ days = 60 }
	set_variable = { name = spn_imp_crisis value = $CRISIS$ days = 60 }
	set_variable = { name = spn_imp_yes value = 0 days = 60 }
	set_variable = { name = spn_imp_no value = 0 days = 60 }
	set_variable = { name = spn_imp_pledges value = 0 days = 60 }
	set_variable = { name = spn_imp_pledge_slips value = 0 days = 60 }
	if = {
		limit = { has_variable = spn_imp_counted }
		remove_variable = spn_imp_counted
	}
	if = {
		limit = { has_variable = spn_imp_struck }
		remove_variable = spn_imp_struck
	}
	if = {
		limit = { has_variable = spn_imp_rm_target }
		remove_variable = spn_imp_rm_target
	}
	if = {
		limit = { has_variable_list = spn_imp_voters }
		clear_variable_list = spn_imp_voters
	}
	if = {
		limit = { has_variable_list = spn_imp_nays }
		clear_variable_list = spn_imp_nays
	}
	if = {
		limit = { has_variable_list = spn_imp_pledgers }
		clear_variable_list = spn_imp_pledgers
	}
	# The councillor a staged rift struck (The Ground Splits saves him) weighs Aye in the session it brings.
	if = {
		limit = {
			exists = scope:spn_imp_struck
			var:spn_imp_crisis = flag:rift
		}
		set_variable = { name = spn_imp_struck value = scope:spn_imp_struck days = 60 }
	}
	# An emergency session has no meeting to end, and it leaves spn_imp_in_meeting alone: that marker belongs to the
	# meeting's own motion (I1), and only the count of a motion he brought moves it into its result (review LOW-7).
	# What the Great Respondeo is aimed at (the yearly pulse's cache, banked for this vote).
	if = {
		limit = { var:spn_imp_motion = flag:respondeo }
		global_var:spn_respondeo_magna_target ?= { save_scope_as = spn_rm_ruler }
		if = {
			limit = { exists = scope:spn_rm_ruler }
			set_variable = { name = spn_imp_rm_target value = scope:spn_rm_ruler days = 60 }
		}
	}
	# The voters: the three seat holders in Witchcraft's own turn order, kept as a list on the Grand Magus (not saved
	# scopes), so one who dies mid-vote cannot stall it. The Grand Magus never votes on his own motion.
	global_var:current_black_mage_global ?= {
		if = {
			limit = { spn_imperium_voter_ok_trigger = yes }
			spn_imperium_add_voter_effect = { SEAT = 3 }
		}
	}
	global_var:current_loremaster_global ?= {
		if = {
			limit = { spn_imperium_voter_ok_trigger = yes }
			spn_imperium_add_voter_effect = { SEAT = 2 }
		}
	}
	global_var:current_admin_global ?= {
		if = {
			limit = { spn_imperium_voter_ok_trigger = yes }
			spn_imperium_add_voter_effect = { SEAT = 1 }
		}
	}
	# Before the vote: a motion for Imperium he brought himself has its levers first.
	if = {
		limit = {
			var:spn_imp_motion = flag:imperium
			var:spn_imp_crisis = flag:none
		}
		if = {
			limit = { is_ai = no }
			trigger_event = spn_order.1001 # Before the Council Votes; its option starts the vote
		}
		else = {
			spn_imperium_ai_lever_effect = yes
			spn_imperium_start_effect = yes
		}
	}
	else = {
		spn_imperium_start_effect = yes
	}
}

# One voter onto the list, with his seat for the turn order, and nothing left over from an earlier vote. Voter scope.
spn_imperium_add_voter_effect = {
	save_temporary_scope_as = spn_imp_v
	scope:spn_imp_gm = {
		add_to_variable_list = { name = spn_imp_voters target = scope:spn_imp_v days = 60 }
	}
	set_variable = { name = spn_imp_seat value = $SEAT$ days = 60 }
	if = {
		limit = { has_character_flag = spn_imp_voted }
		remove_character_flag = spn_imp_voted
	}
	if = {
		limit = { has_character_flag = spn_imp_bound }
		remove_character_flag = spn_imp_bound
	}
	if = {
		limit = { has_character_flag = spn_imp_favour }
		remove_character_flag = spn_imp_favour
	}
	if = {
		limit = { has_character_flag = spn_imp_paid }
		remove_character_flag = spn_imp_paid
	}
	if = {
		limit = { has_character_flag = spn_imp_spoke }
		remove_character_flag = spn_imp_spoke
	}
}

# The AI seats vote at once; the hidden fallback count is queued; the first player seat is asked. Grand Magus scope
# (scope:spn_imp_gm and scope:spn_imp_ballot saved by the open effect).
spn_imperium_start_effect = {
	# The vote's state lives 60 days from the day the vote STARTS, not the day the motion opened: a player Grand Magus may
	# answer Before the Council Votes weeks later, and the state must outlive the 30-day fallback count queued below
	# (review LOW-2). Re-set with its own value, as vanilla renews a timed variable (00_hunt_effects.txt:3398). Nothing
	# has been cast yet, so the counters are 0 again, and the nay and pledger lists are still empty.
	set_variable = { name = spn_imp_open value = yes days = 60 }
	set_variable = { name = spn_imp_vote_id value = var:spn_imp_vote_id days = 60 }
	set_variable = { name = spn_imp_motion value = var:spn_imp_motion days = 60 }
	set_variable = { name = spn_imp_crisis value = var:spn_imp_crisis days = 60 }
	set_variable = { name = spn_imp_yes value = 0 days = 60 }
	set_variable = { name = spn_imp_no value = 0 days = 60 }
	set_variable = { name = spn_imp_pledges value = 0 days = 60 }
	set_variable = { name = spn_imp_pledge_slips value = 0 days = 60 }
	if = {
		limit = {
			has_variable = spn_imp_in_meeting
			var:spn_imp_crisis = flag:none
		}
		set_variable = { name = spn_imp_in_meeting value = yes days = 60 }
	}
	if = {
		limit = { has_variable = spn_imp_struck }
		set_variable = { name = spn_imp_struck value = var:spn_imp_struck days = 60 }
	}
	if = {
		limit = { has_variable = spn_imp_rm_target }
		set_variable = { name = spn_imp_rm_target value = var:spn_imp_rm_target days = 60 }
	}
	# The voters and their seats, the same way: the list is rebuilt, so every entry runs 60 days from now.
	if = {
		limit = { has_variable_list = spn_imp_voters }
		every_in_list = {
			variable = spn_imp_voters
			add_to_temporary_list = spn_imp_renew
		}
		clear_variable_list = spn_imp_voters
		every_in_list = {
			list = spn_imp_renew
			save_temporary_scope_as = spn_imp_v
			scope:spn_imp_gm = {
				add_to_variable_list = { name = spn_imp_voters target = scope:spn_imp_v days = 60 }
			}
			if = {
				limit = { has_variable = spn_imp_seat }
				set_variable = { name = spn_imp_seat value = var:spn_imp_seat days = 60 }
			}
		}
	}
	every_in_list = {
		variable = spn_imp_voters
		limit = {
			is_ai = yes
			is_alive = yes
		}
		spn_imperium_ai_vote_effect = yes
	}
	trigger_event = { id = spn_order.1009 days = 30 } # counts whatever was cast, once (a player seat may never answer)
	spn_imperium_next_effect = yes
}

# The next player seat in turn, or the count. Any scope; reads scope:spn_imp_gm.
spn_imperium_next_effect = {
	scope:spn_imp_gm = {
		if = {
			limit = {
				spn_imperium_vote_running_trigger = yes
				any_in_list = {
					variable = spn_imp_voters
					spn_imperium_awaits_trigger = yes
				}
			}
			ordered_in_list = {
				variable = spn_imp_voters
				limit = { spn_imperium_awaits_trigger = yes }
				order_by = spn_imperium_turn_value
				if = {
					limit = { scope:spn_imp_gm = { var:spn_imp_motion = flag:respondeo } }
					trigger_event = spn_order.1004 # The Council Pledges
				}
				else = {
					trigger_event = spn_order.1002 # The Grand Magus Asks for Imperium
				}
			}
		}
		else_if = {
			limit = { spn_imperium_vote_running_trigger = yes }
			spn_imperium_count_effect = yes
		}
	}
}

# A player's slip from his vote window: counted only if the ballot is still live (the option's own trigger says the same;
# this is the guard if a window is answered after the count). Then the next seat. Voter scope.
spn_imperium_player_vote_effect = {
	if = {
		limit = { spn_imperium_ballot_live_trigger = yes }
		spn_imperium_cast_effect = { VOTE = $VOTE$ }
		spn_imperium_next_effect = yes
	}
}

# A player Grand Magus's lever from Before the Council Votes, then the vote starts (same guard). A lever picked after
# the motion lapsed does what the window's own lapsed option does. Grand Magus scope.
spn_imperium_player_lever_effect = {
	if = {
		limit = { spn_imperium_ballot_live_trigger = yes }
		spn_imperium_lever_effect = { LEVER = $LEVER$ }
		spn_imperium_start_effect = yes
	}
	else = {
		spn_imperium_lapse_effect = yes
	}
}

# Before the Council Votes was answered after the motion's 60 days ran out (review LOW-2): the vote never started and
# nothing was counted. The meeting it was brought to (scope:amage exists only in a meeting's chain) ends now, as it
# would have after the count, if he still holds the chair; his own next meeting is four years off (Witchcraft's
# decision cooldown), so no later meeting can be the one ended. Grand Magus scope.
spn_imperium_lapse_effect = {
	if = {
		limit = {
			exists = scope:amage
			is_alive = yes
			global_var:current_archmage_global ?= this
		}
		scope:amage = { trigger_event = magic_ev10.92 }
	}
}

# One slip. $VOTE$ = flag:aye | flag:nay | flag:blank | flag:pledge (an Aye that pays if the work is carried). Voter scope.
spn_imperium_cast_effect = {
	save_temporary_scope_as = spn_imp_caster
	save_temporary_scope_value_as = { name = spn_imp_slip value = $VOTE$ }
	add_character_flag = { flag = spn_imp_voted days = 60 }
	if = {
		limit = {
			OR = {
				scope:spn_imp_slip = flag:aye
				scope:spn_imp_slip = flag:pledge
			}
		}
		scope:spn_imp_gm = { change_variable = { name = spn_imp_yes add = 1 } }
		if = {
			limit = { scope:spn_imp_slip = flag:pledge }
			scope:spn_imp_gm = {
				change_variable = { name = spn_imp_pledge_slips add = 1 }
				add_to_variable_list = { name = spn_imp_pledgers target = scope:spn_imp_caster days = 60 }
			}
		}
	}
	else_if = {
		limit = { scope:spn_imp_slip = flag:nay }
		scope:spn_imp_gm = {
			change_variable = { name = spn_imp_no add = 1 }
			add_to_variable_list = { name = spn_imp_nays target = scope:spn_imp_caster days = 60 }
		}
	}
}

# An AI seat votes. The staged-rift secret wins over a hook: a voter who knows it votes Nay even if he is bound
# (red team L24). A voter the Grand Magus called a hook on votes Aye. Otherwise the hidden roll. An AI never leaves a
# blank slip. On the Great Respondeo an AI seat always votes Aye, and pledges by nature if it can pay. Voter scope.
spn_imperium_ai_vote_effect = {
	if = {
		limit = { scope:spn_imp_gm = { var:spn_imp_motion = flag:respondeo } }
		if = {
			limit = { spn_respondeo_magna_can_pledge_trigger = yes }
			random = {
				chance = spn_respondeo_magna_pledge_chance_value
				spn_imperium_cast_effect = { VOTE = flag:pledge }
			}
		}
		if = {
			limit = { NOT = { has_character_flag = spn_imp_voted } }
			spn_imperium_cast_effect = { VOTE = flag:aye }
		}
	}
	else_if = {
		limit = { spn_imperium_knows_rift_trigger = yes }
		spn_imperium_cast_effect = { VOTE = flag:nay }
	}
	else_if = {
		limit = { has_character_flag = spn_imp_bound }
		spn_imperium_cast_effect = { VOTE = flag:aye }
	}
	else = {
		random = {
			chance = spn_imperium_aye_chance_value
			spn_imperium_cast_effect = { VOTE = flag:aye }
		}
		if = {
			limit = { NOT = { has_character_flag = spn_imp_voted } }
			spn_imperium_cast_effect = { VOTE = flag:nay }
		}
	}
}

# ==================================================================================================
# BEFORE THE VOTE: THE LEVERS (Witchcraft's Seeking Support shape, magic_ev7.111)
# ==================================================================================================
# $LEVER$ = flag:hooks | flag:favour | flag:gold | flag:talk | flag:none. Grand Magus scope (scope:spn_imp_gm).
spn_imperium_lever_effect = {
	save_temporary_scope_value_as = { name = spn_imp_lv value = $LEVER$ }
	if = {
		limit = { scope:spn_imp_lv = flag:hooks }
		every_in_list = {
			variable = spn_imp_voters
			save_temporary_scope_as = spn_imp_v
			if = {
				limit = { scope:spn_imp_gm = { has_usable_hook = scope:spn_imp_v } }
				scope:spn_imp_gm = { use_hook = scope:spn_imp_v }
				add_character_flag = { flag = spn_imp_bound days = 60 }
			}
		}
	}
	else_if = {
		limit = { scope:spn_imp_lv = flag:favour }
		every_in_list = {
			variable = spn_imp_voters
			add_hook = {
				type = favor_hook
				target = scope:spn_imp_gm
			}
			add_character_flag = { flag = spn_imp_favour days = 60 }
		}
	}
	else_if = {
		limit = { scope:spn_imp_lv = flag:gold }
		every_in_list = {
			variable = spn_imp_voters
			save_temporary_scope_as = spn_imp_v
			scope:spn_imp_gm = {
				pay_short_term_gold = {
					target = scope:spn_imp_v
					gold = medium_gold_value_check
				}
			}
			add_character_flag = { flag = spn_imp_paid days = 60 }
		}
	}
	else_if = {
		limit = { scope:spn_imp_lv = flag:talk }
		# The shape of spn_conclave.0120's duel, rolled out of sight: the odds are never shown.
		duel = {
			skill = diplomacy
			value = 14
			60 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3
				}
				every_in_list = {
					variable = spn_imp_voters
					add_character_flag = { flag = spn_imp_spoke days = 60 }
				}
				if = {
					limit = { is_ai = no }
					send_interface_toast = {
						type = event_toast_effect_good
						title = spn_imperium_toast_spoke_well
						left_icon = this
					}
				}
			}
			40 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3
				}
				if = {
					limit = { is_ai = no }
					send_interface_toast = {
						type = event_toast_effect_neutral
						title = spn_imperium_toast_spoke_badly
						left_icon = this
					}
				}
			}
		}
	}
}

# An AI Grand Magus picks one lever by nature (findings/imperium_endgame.md §4.3). Grand Magus scope.
spn_imperium_ai_lever_effect = {
	random_list = {
		100 = {
			trigger = {
				any_in_list = {
					variable = spn_imp_voters
					scope:spn_imp_gm = { has_usable_hook = prev }
				}
			}
			spn_imperium_lever_effect = { LEVER = flag:hooks }
		}
		30 = {
			modifier = {
				add = 20
				has_trait = generous
			}
			modifier = {
				add = -20
				has_trait = greedy
			}
			spn_imperium_lever_effect = { LEVER = flag:favour }
		}
		20 = {
			trigger = { gold >= spn_imperium_gold_need_value }
			modifier = {
				add = 20
				gold > 500
			}
			modifier = {
				add = -30
				has_trait = greedy
			}
			modifier = {
				factor = 0
				has_trait = greedy
				gold <= 500
			}
			spn_imperium_lever_effect = { LEVER = flag:gold }
		}
		30 = {
			modifier = {
				add = 20
				has_trait = diplomat
			}
			modifier = {
				add = 20
				has_trait = gregarious
			}
			modifier = {
				add = -20
				has_trait = shy
			}
			spn_imperium_lever_effect = { LEVER = flag:talk }
		}
		20 = {
			spn_imperium_lever_effect = { LEVER = flag:none }
		}
	}
}

# Once only (spn_imp_counted). Any scope; reads scope:spn_imp_gm.
spn_imperium_count_effect = {
	scope:spn_imp_gm = {
		set_variable = { name = spn_imp_counted value = yes days = 60 }
		if = {
			limit = { has_variable = spn_imp_open }
			remove_variable = spn_imp_open
		}
		# Bank the count before any result is sent (review LOW-3). The Council Has Spoken reads these saved values, never
		# the Grand Magus's variables, which run out after 60 days and which a later vote resets.
		save_scope_value_as = { name = spn_imp_ayes value = var:spn_imp_yes }
		save_scope_value_as = { name = spn_imp_nays_n value = var:spn_imp_no }
		# 2.39 (Boot 4, 16 Oct): banked as one of two flag literals (from the variable it was refused; The Council Has Spoken
		# reads only whether there was a crisis)
		if = {
			limit = {
				exists = var:spn_imp_crisis
				NOT = { var:spn_imp_crisis = flag:none }
			}
			save_scope_value_as = { name = spn_imp_was_crisis value = flag:crisis }
		}
		else = {
			save_scope_value_as = { name = spn_imp_was_crisis value = flag:none }
		}
		# The meeting this motion was brought to (I1) ends after its result (review LOW-7). The marker moves from the Grand
		# Magus into this vote's own chain now, so an emergency session that opens while The Council Has Spoken is still
		# unanswered cannot take the meeting's end from it. An emergency session never ends a meeting. Saved every count,
		# so a later vote in the same chain never reads an earlier one's.
		if = {
			limit = {
				has_variable = spn_imp_in_meeting
				var:spn_imp_crisis = flag:none
			}
			remove_variable = spn_imp_in_meeting
			save_scope_value_as = { name = spn_imp_ends_meeting value = yes }
		}
		else = {
			save_scope_value_as = { name = spn_imp_ends_meeting value = no }
		}
		if = {
			limit = { var:spn_imp_motion = flag:respondeo }
			spn_respondeo_magna_count_effect = yes
		}
		else = {
			spn_imperium_result_effect = yes
		}
	}
}

# Imperium: more Ayes than Nays grants it; a tie or an empty slate refuses (Witchcraft's Summit rule, F1's rule). A
# motion whose Grand Magus has left the chair simply lapses. Grand Magus scope.
spn_imperium_result_effect = {
	# Saved scopes travel down an event chain: nothing from an earlier vote may be read as this one's.
	clear_saved_scope = spn_imp_result
	clear_saved_scope = spn_imp_dissenter
	if = {
		limit = {
			is_alive = yes
			spn_order_on_trigger = yes
			global_var:current_archmage_global ?= this
			spn_is_order_member_trigger = yes
			var:spn_imp_yes > var:spn_imp_no
		}
		save_scope_value_as = { name = spn_imp_result value = flag:granted }
		spn_imperium_grant_effect = yes
	}
	else_if = {
		limit = {
			is_alive = yes
			global_var:current_archmage_global ?= this
		}
		save_scope_value_as = { name = spn_imp_result value = flag:refused }
		# The record (3 years). Only a motion he brought leaves one: an emergency session is the reconsideration itself,
		# so a refusal is reconsidered once, and the dark road opens once per refusal.
		if = {
			limit = { var:spn_imp_crisis = flag:none }
			set_global_variable = { name = spn_imperium_refused value = this years = 3 }
		}
		# The most opposed Nay voter, named by the result and the one a staged rift would strike.
		if = {
