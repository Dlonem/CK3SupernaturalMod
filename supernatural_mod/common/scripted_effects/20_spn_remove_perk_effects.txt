spn_former_werewolf_effect = {
	if = {
		limit = {
			has_trait = supernatural_hybrid
			OR = {
				has_perk = curse_wolf_perk
				has_perk = wolf_politics_perk
				has_perk = pack_mentality_perk
				has_perk = pack_redistribution_perk
				has_perk = natural_selection_perk
				has_perk = the_prowl_perk
				has_perk = berserker_perk
				has_perk = true_werewolf_perk
			}
		}
		# 2.35h -- GUARDED, one line each. The OR above only proves the character has AT LEAST ONE of the
		# eight; every one they do not have logged "remove_perk effect [ <name> already lacks the perk
		# <perk> ]". A hybrid losing its wolf half almost never holds all eight, so this was a steady
		# drizzle in every long session -- error.log 9 Sep 16:38:57 and every log before it.
		if = { limit = { has_perk = curse_wolf_perk } remove_perk = curse_wolf_perk }
		if = { limit = { has_perk = wolf_politics_perk } remove_perk = wolf_politics_perk }
		if = { limit = { has_perk = pack_mentality_perk } remove_perk = pack_mentality_perk }
		if = { limit = { has_perk = pack_redistribution_perk } remove_perk = pack_redistribution_perk }
		if = { limit = { has_perk = natural_selection_perk } remove_perk = natural_selection_perk }
		if = { limit = { has_perk = the_prowl_perk } remove_perk = the_prowl_perk }
		if = { limit = { has_perk = berserker_perk } remove_perk = berserker_perk }
		if = { limit = { has_perk = true_werewolf_perk } remove_perk = true_werewolf_perk }
	}
}

# 2.32 -- the perk list, extracted so it is written once. spn_former_werewolf_effect above is
# gated on has_trait = supernatural_hybrid (a hybrid losing its wolf half), which meant a CURED
# werewolf -- spn_initiate.1002 -- hit none of it: they kept spn_true_werewolf and all eight
# perks forever. `remove_trait = spn_true_werewolf` appeared NOWHERE in the mod.
#
# spn_true_werewolf is granted by true_werewolf_perk's own effect, so removing the perk does not
# take the trait with it. Both have to go, and the trait carries prowess +5, life_expectancy +20,
# +2 knight/vassal/domain limits and +1 MAA limit.
#
# 2.35h -- and the same guard as above. remove_trait is NOT guarded: CK3 does not complain about
# removing a trait the character lacks, only a perk.
spn_strip_werewolf_perks_effect = {
	remove_trait = spn_true_werewolf
	if = { limit = { has_perk = curse_wolf_perk } remove_perk = curse_wolf_perk }
	if = { limit = { has_perk = wolf_politics_perk } remove_perk = wolf_politics_perk }
	if = { limit = { has_perk = pack_mentality_perk } remove_perk = pack_mentality_perk }
	if = { limit = { has_perk = pack_redistribution_perk } remove_perk = pack_redistribution_perk }
	if = { limit = { has_perk = natural_selection_perk } remove_perk = natural_selection_perk }
	if = { limit = { has_perk = the_prowl_perk } remove_perk = the_prowl_perk }
	if = { limit = { has_perk = berserker_perk } remove_perk = berserker_perk }
	if = { limit = { has_perk = true_werewolf_perk } remove_perk = true_werewolf_perk }
}
