# Supernatural 2.38 -- the Order's secret (WORK-ORDER-2.38 §4.5). The shape of spn_secret_vampire (20_spn_secret_types.txt).
# A scandal, not a crime: the witch secret a Mage already carries holds the crime. Added only by
# spn_order_grant_membership_effect, and known at once by the member's Temple Magus.
spn_secret_blue_rose = {
	category = deviancy

	is_valid = {
		scope:secret_owner = { spn_is_order_member_trigger = yes }
	}

	is_shunned = {
		always = no
	}

	is_criminal = {
		always = no
	}

	on_expose = {
		save_scope_as = secret
		secret_exposed_notification_effect = yes
		scope:secret_owner = {
			spn_order_exposed_effect = yes
		}
	}
}

# 2.39 (WORK-ORDER-2.39 F12): a Grand Magus who staged a rift where a councillor who refused him Imperium stood. Added only
# by The Ground Splits (spn_order.1005), at the moment the rift opens, so a rift that never fires leaves no secret.
spn_secret_staged_rift = {
	category = murder

	is_valid = {
		scope:secret_owner = { is_alive = yes }
	}

	is_shunned = {
		always = yes
	}

	# A crime under the Order's law: a member who knows it sees a crime (and so can hold a strong hook).
	is_criminal = {
		scope:target ?= { spn_is_order_member_trigger = yes }
	}

	on_expose = {
		save_scope_as = secret
		secret_exposed_notification_effect = yes
		scope:secret_owner = {
			spn_imperium_exposed_effect = { VICTIM = scope:secret_target }
		}
	}
}
