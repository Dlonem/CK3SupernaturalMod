# Supernatural 2.40 (WORK-ORDER-2.40 D6) -- the hunt's text counts the hunters. A host may bring up to four guests
# (hunters or apprentices); each adds to the fight, and one hunter among them can be hurt when it goes badly
# (zz_spn_hunt_effects.txt). The texts were written for one man. Root = the hunt (type = activity), read through the
# saved scope: [spn_hunt_activity.Custom('...')] from the creature's side, [activity.Custom('...')] in the hunt window.
# The host attends his own hunt, so "two or more attending" means at least one guest.

# Someone Is Asking Questions: "[hunter] <this> [he] did not come to ask."
SpnHuntAtTheDoor = {
	type = activity
	text = {
		trigger = { any_attending_character = { count >= 3 } }
		localization_key = spn_hunt_cl_door_many
	}
	text = {
		trigger = { any_attending_character = { count >= 2 } }
		localization_key = spn_hunt_cl_door_one
	}
	text = {
		localization_key = spn_hunt_cl_door_alone
	}
}

# Someone Is Asking Questions, "Let them come": who else is in the fight.
SpnHuntAtTheDoorFight = {
	type = activity
	text = {
		trigger = { any_attending_character = { count >= 3 } }
		localization_key = spn_hunt_cl_fight_many
	}
	text = {
		trigger = { any_attending_character = { count >= 2 } }
		localization_key = spn_hunt_cl_fight_one
	}
	text = {
		localization_key = spn_hunt_cl_none
	}
}

# Overmatched: one hunter who rode with him is hurt too (spn_hunt_defender_won_effect wounds one guest hunter).
SpnHuntWithYouDown = {
	type = activity
	text = {
		trigger = { any_attending_character = { has_trait = supernatural_hunter count >= 3 } }
		localization_key = spn_hunt_cl_down_many
	}
	text = {
		trigger = { any_attending_character = { has_trait = supernatural_hunter count >= 2 } }
		localization_key = spn_hunt_cl_down_one
	}
	text = {
		localization_key = spn_hunt_cl_none
	}
}

# Turned Back: the others remember nothing either.
SpnHuntTurnedBackCompany = {

	}

	}
}
