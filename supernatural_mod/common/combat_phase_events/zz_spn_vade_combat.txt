# Supernatural 2.38 -- "The Blue Rose": The Blow Turned Aside (WORK-ORDER-2.38 §12.7). A man bound to the Vade Maecum
# Infernal does not fall in battle.
#
# The 2.33 idiom (10_spn_combat.txt, spn_blood_took_the_blow): base = 100000 while valid takes the bound man out of the
# combat lottery, so his pulses come here instead of to knight_killed / commander_killed. Nothing happens on most pulses;
# about one in thirty, the line says the blow found the man beside him.
# THE TRADE-OFF, as in 2.33: while valid, he also draws none of vanilla's knight or commander events (no wounds, no
# berserker, no prowess gain from a kill). Being bound is for life (§12.7), so this lasts as long as he does.
# Loc: localization/english/spn_vade_l_english.yml (spn_vade_blow_turned_aside_friendly / _enemy).

spn_vade_blow_turned_aside_knight = {
	type = knight
	is_valid = {
		is_alive = yes
		has_trait = spn_vade_bound
		spn_can_be_battle_taken_trigger = yes
	}
	chance = {
		base = 100000
	}
	effect = {
		save_scope_as = spn_vb_knight
		random = {
			chance = 3
			scope:combat_side = {
				battle_event = {
					key = "spn_vade_blow_turned_aside"
					left_portrait = scope:spn_vb_knight
					type = wound
				}
			}
		}
	}
}
