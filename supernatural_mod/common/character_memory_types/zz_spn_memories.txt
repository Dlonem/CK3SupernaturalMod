# Supernatural 2.34 -- "The Blood Remembers Who". Character memories for the turnings, the bonds, the first
# hunter, demon blood and the mage's oath. WORK-ORDER-2.34-v3-BUILD §2.
#
# House rules (each earned from a verified finding, see the work order):
#  * three perspective blocks on every type -- a missing one shows a raw key on someone else's page;
#  * duration = { years = 1000 } on every type -- memory_default_duration is FIVE years for an unlanded AI,
#    which would delete most of these before the maker was even dead;
#  * icons only from gfx/interface/icons/memory_types/ (vanilla inventory, verified on disk);
#  * participants only for real characters (maker, clawer, trainer, progeny); never for a generated
#    confrontation hunter, who is disposed of when the story closes;
#  * never `positive` together with martial/battle/war -- that combination makes CE1 legend material;
#  * `private` throughout: vanilla flavour treats it as "a stranger will not retell this";
#  * descriptions lower-case, past tense, no trailing period -- the GUI uppercases and punctuates.
#
# Every firing site guards with any_memory = { memory_type = ... }, never a character flag, so curing and
# re-embracing (four reachable routes) can never write a second copy.
#
# Memory variables: `spn_forced` (flag:forced) is set on the new memory when the turning was compelled with a
# hook or by force; the descriptions read it with exists = var:spn_forced (vanilla idiom, bp2_memories.txt:764).

# points at a real character who is not the owner. `maker` = that character.

	categories = { negative private supernatural }
	icon = "blood_brother.dds"

	description = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = var:spn_forced }
				desc = spn_became_vampire_memory_desc_forced
			}
			desc = spn_became_vampire_memory_desc
		}
	}
	second_perspective_description = {
		first_valid = {
			triggered_desc = {

			desc = spn_became_vampire_memory_desc_second_perspective
		}

}

# The maker's side of spn_became_vampire_memory. `progeny` = the new vampire. The relationship word comes from
# [owner.Custom2('RelationToMeShort', progeny)] -- the same call vanilla's relative_died uses -- so "my friend",
# "my lover", "my son" stays correct for as long as the memory lives. Written in spn_initiate.0001 on the maker.
spn_made_a_vampire_memory = {
	categories = { private supernatural }
	icon = "blood_brother.dds"

	# The `_named` variants carry the relationship word; the plain ones only the name, so nobody is ever "my acquaintance".
	description = {
		first_valid = {
			triggered_desc = {
				trigger = { memory_owner = { spn_relation_word_exists_trigger = { OTHER = root.memory_participant:progeny } } }
				desc = spn_made_a_vampire_memory_desc_named
			}
			desc = spn_made_a_vampire_memory_desc
		}
	}
	second_perspective_description = {
