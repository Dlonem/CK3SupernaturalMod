	# WHICH of the twenty commands: spn_bond_ai_choice_value, read by the starts_enabled on each option above.
	# The eight-year cooldown_against_recipient means a maker gets one command per child per lifetime-ish, so
	# the cascade leads with the things worth waiting eight years for -- land, a faith, a marriage -- and ends
	# with the thing a bored monster does to its own blood.

		# 2.39 (WORK-ORDER-2.39 B6, D2): a SECRET witch. The menu never vanishes on a secret (is_shown hides only an open
		# witch), so the attempt is made, and it fails HERE, in the effect: never in ai_accept, whose breakdown would print
		# the reason before you send. The failure is what tells it: her secret to the vampire (vanilla reveal_to, the mod's
		# own pattern in zz_spn_order_effects.txt), and his to her, as on_decline does for anyone who resists.
		# TVD 6x06: Alaric compels Jo, "it didn't take", and that is how they learn she is a witch.
		# All of it HIDDEN, and the normal body below gated on a 1-day flag set in this hidden branch, never on an if/else on
		# the witch test or a saved scope: the send window previews on_accept, and the preview walks into hidden_effect and
		# runs its scope saves (vanilla relies on it: 00_revoke_title_interaction.txt with 00_interaction_effects.txt,
		# 30_mongol_invasion_decisions.txt). Either would preview the failure for a secret witch, which is the very leak B6
		# closes. A tooltip never writes game state, so the preview never sees the flag and always shows the ordinary
		# command. The flag is removed at the end of on_accept (2.39 review H1).
		hidden_effect = {
		if = {
			limit = { scope:recipient = { is_witch_trigger = yes } }
			# A flag, not a saved scope: the tooltip preview runs scope saves inside hidden_effect, but never writes game
			# state, so the send window can never see this.
			scope:recipient = { add_character_flag = { flag = spn_cm_did_not_take days = 1 } }
			scope:recipient = {
				every_secret = {
					limit = {
						secret_type = secret_witch
						NOT = { is_known_by = scope:actor }
					}
					reveal_to = scope:actor
				}
			}
			scope:actor = {
				if = {
					limit = { any_secret = { type = spn_secret_vampire NOT = { is_known_by = scope:recipient } } }
					spn_give_knowledge_of_vampire_secret = { CHARACTER = scope:recipient }
				}
				if = {
					limit = { any_secret = { type = spn_secret_hybrid NOT = { is_known_by = scope:recipient } } }
					spn_give_knowledge_of_hybrid_secret = { CHARACTER = scope:recipient }
				}
			}
			# F16 step 7 (red team L26): a Release Me that fails here costs the cell, as a refusal does.
			if = {
				limit = {
					scope:actor = {
						is_imprisoned_by = scope:recipient
						is_in_prison_type = house_arrest
					}
				}
				scope:actor = { hidden_effect = { change_prison_type = dungeon } }
			}
			scope:actor = {
				send_interface_message = {
					type = event_generic_neutral
					title = spn_compel_did_not_take_actor
					right_icon = scope:recipient
					show_as_tooltip = {
						scope:recipient = { custom_tooltip = spn_msg_compel_did_not_take_actor }
					}
				}
			}
			scope:recipient = {
				send_interface_message = {
					type = event_generic_neutral
					title = spn_compel_did_not_take_recipient
					right_icon = scope:actor
					show_as_tooltip = {
						scope:actor = { custom_tooltip = spn_msg_compel_did_not_take_recipient }
					}
				}
			}
		}
		}
		# Everything below is the compulsion taking hold (2.35-2.38, unchanged), and 2.39's Release Me last of all.
		if = {
			limit = { scope:recipient = { NOT = { has_character_flag = spn_cm_did_not_take } } }
		# 2.36 -- End Your Life is a kill: the vampire's own Rite weighs it (zz_spn_fulfillment_effects.txt). Here, before the
		# death inside the toast below, and outside that toast so a player gets the faith's own toast, not one nested in it.
		if = {
			limit = { scope:compel_mortal_suicide = yes }
			scope:actor = { spn_creature_act_fulfillment_effect = { ACT = kill } }
		}
		# Drive all visible effects through a toast so both sides render correctly
		scope:actor = {
			send_interface_message = {
				type = event_toast_effect_neutral
				title = compulsion_interaction
				left_icon  = scope:actor
				right_icon = scope:recipient

				# ---- DEMAND GOLD ----
				if = {
					limit = { scope:compel_mortal_goold = yes }
					# Recipient pays, you receive (vanilla helper prints proper UI line)
					scope:recipient = { pay_short_term_gold = { gold = 100 target = scope:actor } }

					# XP cost
					if      = { limit = { has_trait = supernatural_vampire } add_trait_xp = { trait = supernatural_vampire track = vampirism        value = -15 } }
					else_if = { limit = { has_trait = supernatural_hybrid  } add_trait_xp = { trait = supernatural_hybrid  track = hybrid_vampirism value = -15 } }
				}

				# ---- REVEAL SECRETS ----
				else_if = {
					limit = { scope:compel_mortal_secrets = yes }
					# Reveal any unknown secrets to you
					hidden_effect = {
						scope:recipient = {
							every_secret = {
								limit = { NOT = { is_known_by = scope:actor } }
								# 2.35g -- INSIDE every_secret, `this` IS THE SECRET, and reveal_to is a secret effect.
								# PREV here steps back OUT to scope:recipient and hands reveal_to a character:
								# "reveal_to effect [ Wrong scope: character, expected secret ]", 10_spn_interactions.txt
								# line 1835, error.log 9 Sep 13:xx. Vanilla writes it bare inside the iterator
								# (00_test_interactions.txt:259-260), which is all this ever needed to be. The four
								# PREV-wrapped reveal_to calls up in Command Progeny are a DIFFERENT shape and correct:
								# there PREV steps back from scope:actor's send_interface_toast INTO the secret.
								reveal_to = scope:actor
							}
						}
					}
					if      = { limit = { has_trait = supernatural_vampire } add_trait_xp = { trait = supernatural_vampire track = vampirism        value = -20 } }
					else_if = { limit = { has_trait = supernatural_hybrid  } add_trait_xp = { trait = supernatural_hybrid  track = hybrid_vampirism value = -20 } }
				}

				# ---- JOIN MY COURT ----
				else_if = {
					limit = { scope:compel_mortal_join_me = yes }
					# Use the vanilla effect that prints "joins your court" in the toast
					scope:actor = { add_courtier = scope:recipient }
					if      = { limit = { has_trait = supernatural_vampire } add_trait_xp = { trait = supernatural_vampire track = vampirism        value = -15 } }
					else_if = { limit = { has_trait = supernatural_hybrid  } add_trait_xp = { trait = supernatural_hybrid  track = hybrid_vampirism value = -15 } }
				}

				# ---- BE OBEDIENT (weak hook) ----
				else_if = {
					limit = { scope:compel_mortal_obedient = yes }
					add_hook = { type = loyalty_hook target = scope:recipient }
					if      = { limit = { has_trait = supernatural_vampire } add_trait_xp = { trait = supernatural_vampire track = vampirism        value = -35 } }
					else_if = { limit = { has_trait = supernatural_hybrid  } add_trait_xp = { trait = supernatural_hybrid  track = hybrid_vampirism value = -35 } }
				}

				# ---- BECOME MY LOVER ----
				else_if = {
					limit = { scope:compel_mortal_lover = yes }
					scope:recipient = {
						if = { limit = { has_relation_lover    = scope:actor } remove_relation_lover    = scope:actor }
						if = { limit = { has_relation_soulmate = scope:actor } remove_relation_soulmate = scope:actor }
						# 2.39 (WORK-ORDER-2.39 F14, red team M14): a compelled lover is no romance. Both carry the 3-day flag BEFORE
						# the relation is set, so the Order's romance hook (on_set_relation_lover, either side) passes it by.
						hidden_effect = { add_character_flag = { flag = spn_compelled_lover days = 3 } }
						scope:actor = { hidden_effect = { add_character_flag = { flag = spn_compelled_lover days = 3 } } }
						set_relation_lover = scope:actor
					}
					if      = { limit = { has_trait = supernatural_vampire } add_trait_xp = { trait = supernatural_vampire track = vampirism        value = -15 } }
					else_if = { limit = { has_trait = supernatural_hybrid  } add_trait_xp = { trait = supernatural_hybrid  track = hybrid_vampirism value = -15 } }
				}

				# ---- BECOME MY PATRON (contact/strong relationship) ----
				else_if = {
					limit = { scope:compel_mortal_patron = yes }
					add_contact = scope:recipient
					if      = { limit = { has_trait = supernatural_vampire } add_trait_xp = { trait = supernatural_vampire track = vampirism        value = -10 } }
					else_if = { limit = { has_trait = supernatural_hybrid  } add_trait_xp = { trait = supernatural_hybrid  track = hybrid_vampirism value = -10 } }
				}

				# ---- RELEASE FROM PRISON ----
				else_if = {
					limit = { scope:compel_mortal_release = yes }
					scope:recipient = { release_from_prison = yes }
					if      = { limit = { has_trait = supernatural_vampire } add_trait_xp = { trait = supernatural_vampire track = vampirism        value = -5 } }
					else_if = { limit = { has_trait = supernatural_hybrid  } add_trait_xp = { trait = supernatural_hybrid  track = hybrid_vampirism value = -5 } }
				}

				# ---- RELEASE ME (2.39, WORK-ORDER-2.39 F16). The release itself is the last line of on_accept. ----
				else_if = {
					limit = { scope:compel_mortal_free_me = yes }
					custom_tooltip = spn_compel_free_me_tt
					if      = { limit = { has_trait = supernatural_vampire } add_trait_xp = { trait = supernatural_vampire track = vampirism        value = -15 } }
					else_if = { limit = { has_trait = supernatural_hybrid  } add_trait_xp = { trait = supernatural_hybrid  track = hybrid_vampirism value = -15 } }
				}

				# ---- END YOUR LIFE ----
				else_if = {
					limit = { scope:compel_mortal_suicide = yes }
					scope:recipient = { death = { death_reason = death_suicide } }
					if      = { limit = { has_trait = supernatural_vampire } add_trait_xp = { trait = supernatural_vampire track = vampirism        value = -85 } }
					else_if = { limit = { has_trait = supernatural_hybrid  } add_trait_xp = { trait = supernatural_hybrid  track = hybrid_vampirism value = -85 } }
				}

				# ---- Werewolf surcharge (applies after any successful command) ----
				if = {
					limit = { scope:recipient = { has_trait = supernatural_werewolf } }
					if      = { limit = { has_trait = supernatural_vampire } add_trait_xp = { trait = supernatural_vampire track = vampirism        value = -5 } }
					else_if = { limit = { has_trait = supernatural_hybrid  } add_trait_xp = { trait = supernatural_hybrid  track = hybrid_vampirism value = -5 } }
				}
			}
		}

		# 2.35 -- W5: what the compulsion left behind. Outside the toast on purpose -- this is a hidden roll
		# and it must not print a line under a command the player already read. See
		# zz_spn_consequence_effects.txt for the odds and the reasoning.
		hidden_effect = {
			scope:actor = { spn_compel_leaves_a_trace_effect = yes }
		}

		# Notify both sides that the compulsion took hold (4001 handles actor + recipient)
		scope:actor = { trigger_event = { id = spn_message.4001 } } # 2.37: an interaction's on_accept/on_decline has no character in scope

		# 2.39 (WORK-ORDER-2.39 F16): Release Me, LAST. The release is the irreversible step, after the trace and the message.
		#   quiet: vanilla's own release (ransom_me's on_accept, 00_prison_interactions.txt); the gaoler believes he chose to;
		#   loud:  vanilla's escape (opinion, memory, release); he comes to himself in front of an empty cell.
		# "Older blood leaves more quietly": spn_compel_cell_quiet_chance_value, rolled and banked at the top. Hidden: the
		# toast line above already says he walks out, and a preview (which never rolls) must not print the loud version.
		hidden_effect = {
			if = {
				limit = {
					scope:compel_mortal_free_me = yes
					scope:actor = { is_imprisoned_by = scope:recipient }
				}
				if = {
					limit = { exists = scope:spn_cell_quiet }
					scope:actor = {
						if = {
							limit = { has_character_modifier = allowed_to_go_outside }
							remove_character_modifier = allowed_to_go_outside
						}
						if = {
							limit = { has_character_modifier = moldy_gruel_diet }
							remove_character_modifier = moldy_gruel_diet
						}
						release_from_prison = yes
					}
				}
				else = {
					scope:actor = { escape_from_prison_effect = yes }
				}
			}
		}
		}
		# 2.39 (WORK-ORDER-2.39 B6, review H1): B6's failure flag has done its work; nothing after this reads it.
		hidden_effect = {
			scope:recipient = {
				if = {
					limit = { has_character_flag = spn_cm_did_not_take }
					remove_character_flag = spn_cm_did_not_take
				}
			}
		}
	}


	# ------------------------------------------------------------------------------------------------------
	# 2.35 -- THE AI CAN DO THIS NOW. It never could: there was no ai_frequency, no ai_targets and no
	# ai_will_do, so in every campaign ever played the only vampire who compelled anybody was the player.
	#
	# WHICH COMMAND it sends is decided by spn_compel_ai_choice_value (zz_spn_ai_command_values.txt), read by
	# the starts_enabled on each send_option above -- vanilla's own idiom for an AI-capable interaction with
	# exclusive options (mpo_offer_submission_or_ruin, 09_mpo_interactions.txt). WHETHER it sends at all is
	# decided here.
	#
	# THE RESERVE IS THE DECISION. Every command is paid for out of the vampirism track and a feeding activity
	# is what puts it back, so this reads as a hunger clock: a creature with a full reserve spends freely, one
	# under fifteen spends nothing at all and goes to eat instead. Thomas's number, 8 Sep 2026.
	#
	# WHOSE TOES. ai_targets is deliberately close to home -- a court, a council, a cell, a guest list -- so an
	# AI vampire works on the people around it rather than reaching into strangers' households. Rulers carry a
	# penalty for the same reason. And because auto_accept is `no` on this interaction, the engine will not
	# even offer it to an AI whose target would refuse (_character_interactions.info: "interactions that need
	# acceptance cannot be sent to the ai unless it will accept") -- so ai_accept below, and the whole
	# resistance ladder in spn_compulsion_margin_accept, is the second gate on all of this for free.
	# ------------------------------------------------------------------------------------------------------
	ai_targets = {
		ai_recipients = courtiers
	}
	ai_targets = {
		ai_recipients = guests
	}
	ai_targets = {
		ai_recipients = councillors
	}
	ai_targets = {
		ai_recipients = prisoners
	}
	ai_targets = {
		ai_recipients = activity_guests
	}
	# 2.39 (WORK-ORDER-2.39 F16): a jailed vampire's gaoler, through the redirect (vanilla ransom_me's AI target).
	ai_targets = {
		ai_recipients = self
	}

	ai_target_quick_trigger = {
		adult = yes
	}

	# CADENCE BY WHAT YOU ARE. The global cooldown on this interaction is six months, so every number here is
	# at least twice that: nobody is doing this the moment the clock allows it.
	#
	# A king is surrounded by people whose week nobody audits and has had a lifetime of being obeyed; a
	# baron has a hall of six and knows all their names. The gradient is that, not power -- power is already
	# priced into the reserve.
	#
	# NOTE ON duchy: a LANDLESS ADVENTURER reads as duchy tier, because a camp is a duchy-tier title -- see
	# vanilla's own landless-only interactions, which set duchy live and every other tier to 0
	# (purchase_land_interaction, contact_list_request_interaction, 06_ep3_laamp_interactions.txt). So this
	# row carries both a duke with a real court and a creature moving through strangers, which is why it is
	# the busiest of the middle ranks and why ai_will_do below separates them by hand.
	ai_frequency_by_tier = {
		barony = 24
		county = 18
		duchy = 14
		kingdom = 12
		empire = 12
		hegemony = 12
	}

	ai_potential = {
		# 2.39 (WORK-ORDER-2.39 F16): is_imprisoned = no is gone: from a cell the AI may send Release Me (is_shown allows only that).
		is_incapable = no
		OR = {
			has_trait = supernatural_vampire
			has_trait = supernatural_hybrid
		}
		# Nothing to spend. Go and feed.
		spn_vampiric_reserve_value >= 15
	}

	ai_will_do = {
		base = 0

		# --- THE RESERVE. Four rungs, so a full creature is roughly four times as willing as a lean one.
		modifier = { add = 12  scope:actor = { spn_vampiric_reserve_value >= 25 } }
		modifier = { add = 12  scope:actor = { spn_vampiric_reserve_value >= 40 } }
		modifier = { add = 12  scope:actor = { spn_vampiric_reserve_value >= 60 } }
		modifier = { add = 14  scope:actor = { spn_vampiric_reserve_value >= 80 } }
		# 2.39 (WORK-ORDER-2.39 F16): a creature in a cell wants out more than it wants anything else. Before the two
		# factor-0 lines below, so a hungry creature still hoards.
		modifier = { add = 60  scope:actor = { is_imprisoned_by = scope:recipient } }
		# Hungry. Hoard it.
		modifier = { factor = 0  scope:actor = { spn_vampiric_reserve_value < 15 } }
		# The cascade found no command worth the price for this pair.
		modifier = { factor = 0  spn_compel_ai_choice_value = 0 }

		# --- TEMPERAMENT. Who reaches for other people's minds as a first resort.
		modifier = { add = 15  scope:actor = { has_trait = ambitious } }
		modifier = { add = 15  scope:actor = { has_trait = deceitful } }
		modifier = { add = 12  scope:actor = { has_trait = arrogant } }
		modifier = { add = 12  scope:actor = { has_trait = arbitrary } }
		modifier = { add = 10  scope:actor = { has_trait = sadistic } }
		modifier = { add = 8   scope:actor = { has_trait = impatient } }
		modifier = { add = -20 scope:actor = { has_trait = just } }
		modifier = { add = -15 scope:actor = { has_trait = compassionate } }
		modifier = { add = -15 scope:actor = { has_trait = humble } }
		modifier = { add = -12 scope:actor = { has_trait = content } }
		modifier = { add = -10 scope:actor = { has_trait = patient } }
		ai_value_modifier = { ai_honor = -0.5 }
		ai_value_modifier = { ai_boldness = 0.4 }

		# --- WHO, AND WHAT IT COSTS IF IT GOES WRONG
		# Somebody in your own hall. Nobody outside it has any reason to look.
		modifier = { add = 15  scope:recipient = { is_courtier_of = scope:actor } }
		# A ruler who starts behaving strangely is a ruler three realms talk about.
		modifier = { add = -25 scope:recipient = { is_ruler = yes } }
		# Leaning on a hunter is how creatures end up on walls -- unless you are strong enough that he will
		# never be sure it happened, which is exactly the kind of arrogance an elder has.
		modifier = { add = -60 scope:recipient = { has_trait = supernatural_hunter } }
		modifier = {
			add = 40
			scope:recipient = { has_trait = supernatural_hunter }
			scope:actor = { spn_vampiric_reserve_value >= 80 }
		}
		# Somebody who already knows what you are is the worst possible person to give an order to.
		modifier = {
			add = -30
			scope:recipient = { spn_knows_about_character_being_vampire_trigger = { CHARACTER = scope:actor } }
		}
		# A creature everybody already watches should be spending less of this, not more.
		modifier = { add = -20 scope:actor = { spn_hunt_is_known_creature_trigger = yes } }
		# --- WHAT YOU ARE. Both ends of the ladder reach for this more than the middle does, for opposite
		# reasons: a king has the habit of being obeyed and a hall full of people nobody audits, and a
		# landless creature has no realm, no name to protect and the road out of town. A count in his own
		# county has neighbours who have known him thirty years.
		modifier = { add = 8  scope:actor = { highest_held_title_tier >= tier_kingdom } }
		modifier = { add = 8  scope:actor = { is_landless_adventurer = yes } }
		# Busy.
		modifier = { add = -20 scope:actor = { is_at_war = yes } }
	}

	# AI acceptance driven by the scripted value (utility file)
	ai_accept = {
		base = 0
		modifier = { add = { value = spn_compulsion_margin_accept } desc = DS_COMPULSION_MARGIN }

		# −30 if they know what you are
		modifier = {
			add = -30
			desc = DS_COMPULSION_KNOWS_YOU
			OR = {
				scope:actor = { any_secret = { type = spn_secret_vampire is_known_by = scope:recipient } }
				scope:actor = { any_secret = { type = spn_secret_hybrid  is_known_by = scope:recipient } }
				scope:actor = { has_trait = spn_known_vampire }
				scope:actor = { has_trait = spn_known_hybrid }
			}
		}

		# +50 if the recipient is your prisoner
		modifier = {
			add = 50
			desc = DS_COMPULSION_PRISONER
			scope:recipient = { is_imprisoned_by = scope:actor }
		}

		# +10 if the recipient is of your dynasty (handles lowborn safely)
		modifier = {
			add = 10
			desc = DS_COMPULSION_SAME_DYNASTY
			exists = scope:actor.dynasty
			exists = scope:recipient.dynasty
			scope:actor.dynasty = { this = scope:recipient.dynasty }
		}
		
		# Exactly Duke (duchy or higher, but NOT king+)
		modifier = {
			add = -10
			desc = DS_COMPULSION_TARGET_DUKE
			scope:recipient = { highest_held_title_tier >= tier_duchy }
			NOT = { scope:recipient = { highest_held_title_tier >= tier_kingdom } }
		}

		# Exactly King (kingdom or higher, but NOT emperor)
		modifier = {
			add = -15
			desc = DS_COMPULSION_TARGET_KING
			scope:recipient = { highest_held_title_tier >= tier_kingdom }
			NOT = { scope:recipient = { highest_held_title_tier >= tier_empire } }
		}

		# Emperor (highest tier)
		modifier = {
			add = -25
			desc = DS_COMPULSION_TARGET_EMPEROR
			scope:recipient = { highest_held_title_tier >= tier_empire }
		}

		# +10 if recipient is your vassal (or below your realm)
		modifier = {
			add = 10
			desc = DS_COMPULSION_VASSAL
			scope:recipient = { is_vassal_or_below_of = scope:actor }
		}
		
		# Hunter training resists Compulsion (scales with Monster-Hunting XP)
		modifier = {
			add = { value = spn_compulsion_hunter_resist }
			desc = DS_COMPULSION_HUNTER_RESIST
			scope:recipient = { has_trait = supernatural_hunter }
		}

	}


	auto_accept = no
