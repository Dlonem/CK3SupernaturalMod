	# Be Obedient (weak hook on target) (cost 35)
	send_option = {
		is_valid = {
			custom_description = {
				text = invoke_bond_requires_vampiric_strength_35
				scope:actor = {
					OR = {
						has_trait_xp = { trait = supernatural_vampire track = vampirism        value >= 35 }
						has_trait_xp = { trait = supernatural_hybrid  track = hybrid_vampirism value >= 35 }
					}
				}
			}
		}
		flag = compel_mortal_obedient
		localization = COMPEL_MORTAL_11
	}

	# Become My Patron (strong hook on target) (cost 20)
	send_option = {
	  is_shown = {
		scope:actor = { is_landless_adventurer = yes }
		scope:recipient = { is_landed_or_landless_administrative = yes NOT = { is_contact_of = scope:actor } }
	  }
	  is_valid = {
		custom_description = {
		  text = invoke_bond_requires_vampiric_strength_10
		  scope:actor = {
			OR = {
			  has_trait_xp = { trait = supernatural_vampire track = vampirism        value >= 10 }
			  has_trait_xp = { trait = supernatural_hybrid  track = hybrid_vampirism value >= 10 }
			}
		  }
		}
	  }
	  flag = compel_mortal_patron
	  localization = COMPEL_MORTAL_O9   # "(Cost: 10 Vampirism)"
	}

	# Release From Prison (NEW) (cost 20)
	send_option = {
		is_shown = { scope:recipient = { is_imprisoned = yes } }
		is_valid = {
			custom_description = {
				text = invoke_bond_requires_vampiric_strength_20
				scope:actor = {
					OR = {
						has_trait_xp = { trait = supernatural_vampire track = vampirism        value >= 20 }
						has_trait_xp = { trait = supernatural_hybrid  track = hybrid_vampirism value >= 20 }
					}
				}
			}
		}
		flag = compel_mortal_release
		localization = COMPEL_MORTAL_RELEASE
	}

	# End Your Life (cost 85)
	send_option = {
		is_valid = {
			custom_description = {
				text = invoke_bond_requires_vampiric_strength_75
				scope:actor = {
					OR = {
						has_trait_xp = { trait = supernatural_vampire track = vampirism        value >= 85 }
						has_trait_xp = { trait = supernatural_hybrid  track = hybrid_vampirism value >= 85 }
					}
				}
			}
		}
		flag = compel_mortal_suicide
		localization = COMPEL_MORTAL_03
	}

	# ---- Decline / Accept ----
	on_decline = {
		# Reveal ONLY if the actor has a vampire secret the recipient doesn't know
		scope:actor = {
			if = {
				limit = {
					any_secret = {
						type = spn_secret_vampire
						NOT = { is_known_by = scope:recipient }
					}
				}
				spn_give_knowledge_of_vampire_secret = { CHARACTER = scope:recipient }
			}
		}

		# Recipient resisted: notify both sides (the actor's nature is exposed just above)
		trigger_event = { id = spn_message.4002 }
	}


	on_accept = {
		# Drive all visible effects through a toast so both sides render correctly
		scope:actor = {
			send_interface_message = {
				type = event_toast_effect_neutral
				title = compulsion_interaction
				left_icon  = scope:actor
				right_icon = scope:recipient

				# ---- DEMAND GOLD ----
				if = {
					limit = { scope:compel_mortal_goold = yes }
					# Recipient pays, you receive (vanilla helper prints proper UI line)
					scope:recipient = { pay_short_term_gold = { gold = 100 target = scope:actor } }

					# XP cost
					if      = { limit = { has_trait = supernatural_vampire } add_trait_xp = { trait = supernatural_vampire track = vampirism        value = -15 } }
					else_if = { limit = { has_trait = supernatural_hybrid  } add_trait_xp = { trait = supernatural_hybrid  track = hybrid_vampirism value = -15 } }
				}

				# ---- REVEAL SECRETS ----
				else_if = {
					limit = { scope:compel_mortal_secrets = yes }
					# Reveal any unknown secrets to you
					hidden_effect = {
						scope:recipient = {
							every_secret = {
								limit = { NOT = { is_known_by = scope:actor } }
								PREV = { reveal_to = scope:actor }
							}
						}
					}
					if      = { limit = { has_trait = supernatural_vampire } add_trait_xp = { trait = supernatural_vampire track = vampirism        value = -20 } }
					else_if = { limit = { has_trait = supernatural_hybrid  } add_trait_xp = { trait = supernatural_hybrid  track = hybrid_vampirism value = -20 } }
				}

				# ---- JOIN MY COURT ----
				else_if = {
					limit = { scope:compel_mortal_join_me = yes }
					# Use the vanilla effect that prints "joins your court" in the toast
					scope:actor = { add_courtier = scope:recipient }
					if      = { limit = { has_trait = supernatural_vampire } add_trait_xp = { trait = supernatural_vampire track = vampirism        value = -15 } }
					else_if = { limit = { has_trait = supernatural_hybrid  } add_trait_xp = { trait = supernatural_hybrid  track = hybrid_vampirism value = -15 } }
				}

				# ---- BE OBEDIENT (weak hook) ----
				else_if = {
					limit = { scope:compel_mortal_obedient = yes }
					add_hook = { type = loyalty_hook target = scope:recipient }
					if      = { limit = { has_trait = supernatural_vampire } add_trait_xp = { trait = supernatural_vampire track = vampirism        value = -35 } }
					else_if = { limit = { has_trait = supernatural_hybrid  } add_trait_xp = { trait = supernatural_hybrid  track = hybrid_vampirism value = -35 } }
				}

				# ---- BECOME MY LOVER ----
				else_if = {
					limit = { scope:compel_mortal_lover = yes }
					scope:recipient = {
						if = { limit = { has_relation_lover    = scope:actor } remove_relation_lover    = scope:actor }
						if = { limit = { has_relation_soulmate = scope:actor } remove_relation_soulmate = scope:actor }
						set_relation_lover = scope:actor
					}
					if      = { limit = { has_trait = supernatural_vampire } add_trait_xp = { trait = supernatural_vampire track = vampirism        value = -15 } }
					else_if = { limit = { has_trait = supernatural_hybrid  } add_trait_xp = { trait = supernatural_hybrid  track = hybrid_vampirism value = -15 } }
				}

				# ---- BECOME MY PATRON (contact/strong relationship) ----
				else_if = {
					limit = { scope:compel_mortal_patron = yes }
					add_contact = scope:recipient
					if      = { limit = { has_trait = supernatural_vampire } add_trait_xp = { trait = supernatural_vampire track = vampirism        value = -10 } }
					else_if = { limit = { has_trait = supernatural_hybrid  } add_trait_xp = { trait = supernatural_hybrid  track = hybrid_vampirism value = -10 } }
				}

				# ---- RELEASE FROM PRISON ----
				else_if = {
					limit = { scope:compel_mortal_release = yes }
					scope:recipient = { release_from_prison = yes }
					if      = { limit = { has_trait = supernatural_vampire } add_trait_xp = { trait = supernatural_vampire track = vampirism        value = -5 } }
					else_if = { limit = { has_trait = supernatural_hybrid  } add_trait_xp = { trait = supernatural_hybrid  track = hybrid_vampirism value = -5 } }
				}

				# ---- END YOUR LIFE ----
				else_if = {
					limit = { scope:compel_mortal_suicide = yes }
					scope:recipient = { death = { death_reason = death_suicide } }
					if      = { limit = { has_trait = supernatural_vampire } add_trait_xp = { trait = supernatural_vampire track = vampirism        value = -85 } }
					else_if = { limit = { has_trait = supernatural_hybrid  } add_trait_xp = { trait = supernatural_hybrid  track = hybrid_vampirism value = -85 } }
				}

				# ---- Werewolf surcharge (applies after any successful command) ----
				if = {
					limit = { scope:recipient = { has_trait = supernatural_werewolf } }
					if      = { limit = { has_trait = supernatural_vampire } add_trait_xp = { trait = supernatural_vampire track = vampirism        value = -5 } }
					else_if = { limit = { has_trait = supernatural_hybrid  } add_trait_xp = { trait = supernatural_hybrid  track = hybrid_vampirism value = -5 } }
				}
			}
		}

		# Notify both sides that the compulsion took hold (4001 handles actor + recipient)
		trigger_event = { id = spn_message.4001 }
	}


	# AI acceptance driven by the scripted value (utility file)
	ai_accept = {
		base = 0
		modifier = { add = { value = spn_compulsion_margin_accept } desc = DS_COMPULSION_MARGIN }

		# −30 if they know what you are
		modifier = {
			add = -30
			desc = DS_COMPULSION_KNOWS_YOU
			OR = {
				scope:actor = { any_secret = { type = spn_secret_vampire is_known_by = scope:recipient } }
				scope:actor = { any_secret = { type = spn_secret_hybrid  is_known_by = scope:recipient } }
				scope:actor = { has_trait = spn_known_vampire }
				scope:actor = { has_trait = spn_known_hybrid }
			}
		}

		# +50 if the recipient is your prisoner
		modifier = {
			add = 50
			desc = DS_COMPULSION_PRISONER
			scope:recipient = { is_imprisoned_by = scope:actor }
		}

		# +10 if the recipient is of your dynasty (handles lowborn safely)
		modifier = {
			add = 10
			desc = DS_COMPULSION_SAME_DYNASTY
			exists = scope:actor.dynasty
			exists = scope:recipient.dynasty
			scope:actor.dynasty = { this = scope:recipient.dynasty }
		}
		
		# Exactly Duke (duchy or higher, but NOT king+)
		modifier = {
			add = -10
			desc = DS_COMPULSION_TARGET_DUKE
			scope:recipient = { highest_held_title_tier >= tier_duchy }
			NOT = { scope:recipient = { highest_held_title_tier >= tier_kingdom } }
		}

		# Exactly King (kingdom or higher, but NOT emperor)
		modifier = {
			add = -15
			desc = DS_COMPULSION_TARGET_KING
			scope:recipient = { highest_held_title_tier >= tier_kingdom }
			NOT = { scope:recipient = { highest_held_title_tier >= tier_empire } }
		}

		# Emperor (highest tier)
		modifier = {
			add = -25
			desc = DS_COMPULSION_TARGET_EMPEROR
			scope:recipient = { highest_held_title_tier >= tier_empire }
		}

		# +10 if recipient is your vassal (or below your realm)
		modifier = {
			add = 10
			desc = DS_COMPULSION_VASSAL
			scope:recipient = { is_vassal_or_below_of = scope:actor }
		}
		
		# Hunter training resists Compulsion (scales with Monster-Hunting XP)
		modifier = {
			add = { value = spn_compulsion_hunter_resist }
			desc = DS_COMPULSION_HUNTER_RESIST
			scope:recipient = { has_trait = supernatural_hunter }
		}

	}


	auto_accept = no
}
