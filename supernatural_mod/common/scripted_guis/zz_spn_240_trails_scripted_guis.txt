# Supernatural 2.40 boot 3 (TRAILS): the scripted guis of the Supernatural Trails view.

# The Trails button (gui/zz_spn_hud_buttons.gui, spn_hud_trails_button) runs this first, then shows the Supernatural
# Trails window (gui/zz_spn_240_trails_window.gui). Root = the local player. It fills the six lists the window shows
# (spn_trails_fill_view_effect, zz_spn_240_trails_effects.txt). Vanilla's own pattern for a list a button opens:
# funeral_fill_available_targets_gui (ce1_funeral_scripted_guis.txt), run the same way from
# funeral_deceased_selection_button.gui:27. It changes nothing but the player's own list cache, so it is safe in
# multiplayer: each player fills his own lists, nobody waits on anybody.
spn_trails_fill_gui = {
	scope = character

	effect = {
		spn_trails_fill_view_effect = yes
	}
}
